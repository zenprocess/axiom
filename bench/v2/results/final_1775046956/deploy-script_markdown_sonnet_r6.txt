```bash
#!/usr/bin/env bash
# deploy.sh — build, push, tag, deploy, and rollback support for axiom-rdl
# Usage:
#   ./deploy.sh [--env <prod|staging>] [--tag <version>] [--rollback <tag>]
# Env vars (required unless set in env):
#   REGISTRY        — e.g. ghcr.io/zenprocess
#   DEPLOY_HOST     — SSH target, e.g. user@prod.example.com
#   DEPLOY_DIR      — remote dir containing docker-compose.yml, e.g. /srv/axiom
set -euo pipefail

IMAGE_NAME="axiom-rdl"
ENV="${DEPLOY_ENV:-prod}"
REGISTRY="${REGISTRY:-ghcr.io/zenprocess}"
DEPLOY_HOST="${DEPLOY_HOST:-}"
DEPLOY_DIR="${DEPLOY_DIR:-/srv/axiom}"
ROLLBACK_TAG=""
VERSION_FILE="pyproject.toml"

RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; NC='\033[0m'
info()    { echo -e "${GREEN}[deploy]${NC} $*"; }
warn()    { echo -e "${YELLOW}[warn]${NC}  $*"; }
die()     { echo -e "${RED}[error]${NC} $*" >&2; exit 1; }

while [[ $# -gt 0 ]]; do
  case "$1" in
    --env)      ENV="$2";          shift 2 ;;
    --tag)      EXPLICIT_TAG="$2"; shift 2 ;;
    --rollback) ROLLBACK_TAG="$2"; shift 2 ;;
    *)          die "Unknown argument: $1" ;;
  esac
done

# Resolve version from pyproject.toml unless overridden
if [[ -n "${ROLLBACK_TAG:-}" ]]; then
  TAG="$ROLLBACK_TAG"
elif [[ -n "${EXPLICIT_TAG:-}" ]]; then
  TAG="$EXPLICIT_TAG"
else
  TAG="$(grep '^version' "$VERSION_FILE" | head -1 | sed 's/.*= *"\(.*\)"/\1/')"
  [[ -z "$TAG" ]] && die "Could not parse version from $VERSION_FILE"
fi

FULL_IMAGE="${REGISTRY}/${IMAGE_NAME}:${TAG}"
LATEST_IMAGE="${REGISTRY}/${IMAGE_NAME}:latest"

preflight() {
  info "Preflight checks..."
  command -v docker  >/dev/null 2>&1 || die "docker not found"
  command -v git     >/dev/null 2>&1 || die "git not found"
  [[ -f "Dockerfile" ]]              || die "Dockerfile not found in $(pwd)"
  [[ -n "$DEPLOY_HOST" ]]            || die "DEPLOY_HOST is not set"

  local dirty
  dirty=$(git status --porcelain)
  if [[ -n "$dirty" && -z "${FORCE:-}" ]]; then
    warn "Working tree is dirty. Set FORCE=1 to deploy anyway."
    git status --short
    exit 1
  fi
  info "Preflight OK — deploying ${FULL_IMAGE} to ${ENV}"
}

# 1. Build
build_image() {
  info "Building Docker image..."
  docker build \
    --build-arg VERSION="$TAG" \
    --build-arg BUILD_DATE="$(date -u +%Y-%m-%dT%H:%M:%SZ)" \
    --build-arg GIT_SHA="$(git rev-parse --short HEAD 2>/dev/null || echo unknown)" \
    --label "org.opencontainers.image.version=${TAG}" \
    --label "org.opencontainers.image.source=https://github.com/zenprocess/axiom" \
    -t "${FULL_IMAGE}" \
    -t "${LATEST_IMAGE}" \
    .
  info "Build complete: ${FULL_IMAGE}"
}

# 2. Push
push_image() {
  info "Pushing to registry..."
  docker push "${FULL_IMAGE}"
  docker push "${LATEST_IMAGE}"
  info "Push complete"
}

# 3. Git tag
tag_release() {
  local git_tag="v${TAG}"
  if git rev-parse "$git_tag" >/dev/null 2>&1; then
    warn "Git tag ${git_tag} already exists — skipping"
    return
  fi
  info "Tagging release ${git_tag}..."
  git tag -a "$git_tag" -m "Release ${git_tag}"
  git push origin "$git_tag"
  info "Tagged and pushed ${git_tag}"
}

# 4. Deploy (with automatic rollback on health-check failure)
deploy_remote() {
  info "Deploying ${FULL_IMAGE} to ${DEPLOY_HOST}:${DEPLOY_DIR}..."

  # Capture currently running tag before we replace it
  local previous_tag
  previous_tag=$(ssh "$DEPLOY_HOST" \
    "docker inspect --format '{{index .Config.Image}}' ${IMAGE_NAME} 2>/dev/null || echo ''" \
    | sed 's/.*://')

  ssh "$DEPLOY_HOST" bash -s -- "$FULL_IMAGE" "$DEPLOY_DIR" <<'REMOTE'
    set -euo pipefail
    IMAGE="$1"; DIR="$2"
    cd "$DIR"
    echo "IMAGE_TAG=${IMAGE}" > .env.deploy
    docker compose --env-file .env.deploy pull
    docker compose --env-file .env.deploy up -d --remove-orphans
    echo "Waiting for health check..."
    sleep 5
    docker compose ps
REMOTE

  # Smoke-test: container still running after 10s?
  sleep 10
  local status
  status=$(ssh "$DEPLOY_HOST" \
    "docker inspect --format '{{.State.Status}}' ${IMAGE_NAME} 2>/dev/null || echo missing")
  if [[ "$status" != "running" ]]; then
    warn "Container status: ${status}. Auto-rolling back to ${previous_tag}..."
    _do_rollback "$previous_tag"
    die "Deploy failed — rolled back to ${previous_tag}"
  fi
  info "Deploy successful."
}

_do_rollback() {
  local prev="$1"
  [[ -z "$prev" ]] && { warn "No previous tag available for rollback."; return; }
  info "Rolling back to ${REGISTRY}/${IMAGE_NAME}:${prev}..."
  ssh "$DEPLOY_HOST" bash -s -- "${REGISTRY}/${IMAGE_NAME}:${prev}" "$DEPLOY_DIR" <<'REMOTE'
    set -euo pipefail
    IMAGE="$1"; DIR="$2"
    cd "$DIR"
    echo "IMAGE_TAG=${IMAGE}" > .env.deploy
    docker compose --env-file .env.deploy pull
    docker compose --env-file .env.deploy up -d --remove-orphans
REMOTE
  warn "Rollback complete."
}

main() {
  preflight
  if [[ -n "$ROLLBACK_TAG" ]]; then
    _do_rollback "$ROLLBACK_TAG"
  else
    build_image
    push_image
    tag_release
    deploy_remote
  fi
  info "Done."
}

main
```

---

**How it works:**

| Step | What happens |
|---|---|
| Preflight | Checks for `docker`, `git`, `Dockerfile`; blocks on dirty working tree (override with `FORCE=1`) |
| Build | Tags with version from `pyproject.toml` + OCI labels (version, source, git SHA) |
| Push | Pushes both `:<version>` and `:latest` |
| Git tag | Creates annotated `v<version>` tag and pushes it; skips if tag already exists |
| Deploy | SSH into `DEPLOY_HOST`, writes `.env.deploy`, pulls and restarts via `docker compose` |
| Auto-rollback | Records the running image tag before deploy; if the container isn't running after 10s, redeploys the previous tag automatically |

**Usage examples:**

```bash
# Normal deploy (version read from pyproject.toml → 0.6.0)
REGISTRY=ghcr.io/zenprocess DEPLOY_HOST=user@prod.example.com ./deploy.sh

# Deploy to staging with an explicit tag
./deploy.sh --env staging --tag 0.5.9

# Manual rollback to a prior version
./deploy.sh --rollback 0.5.8

# Force deploy despite dirty working tree
FORCE=1 ./deploy.sh
```
