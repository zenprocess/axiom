```python
"""REST API client for fetching user data."""

import asyncio
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel

logger = structlog.get_logger(__name__)

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """User address information."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User data returned from the API."""

    id: int
    name: str
    email: str
    address: Optional[Address] = None
    phone: Optional[str] = None


class UserAPIError(Exception):
    """Raised when the user API returns an error."""

    def __init__(self, status_code: int, message: str) -> None:
        self.status_code = status_code
        super().__init__(f"API error {status_code}: {message}")


class UserClient:
    """HTTP client for the user REST API."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = 10.0) -> None:
        """Initialize the client with a base URL and timeout."""
        self._base_url = base_url
        self._timeout = timeout

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Args:
            user_id: The integer ID of the user to fetch.

        Returns:
            A User instance populated with API response data.

        Raises:
            UserAPIError: If the API returns a non-2xx status code.
            httpx.HTTPError: On network-level errors.
        """
        url = f"{self._base_url}/users/{user_id}"
        log = logger.bind(user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)

            if response.status_code == 404:
                raise UserAPIError(404, f"User {user_id} not found")

            if not response.is_success:
                raise UserAPIError(response.status_code, response.text)

            user = User.model_validate(response.json())
            log.info("user_fetched", name=user.name, email=user.email)
            return user

        except UserAPIError:
            raise
        except httpx.TimeoutException:
            log.error("request_timeout")
            raise
        except httpx.HTTPError as exc:
            log.error("http_error", error=str(exc))
            raise


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.get_user(42)
    logger.info("result", user_id=user.id, name=user.name, email=user.email)


if __name__ == "__main__":
    asyncio.run(main())
```

**Key design decisions:**

- **`User` / `Address`** — Pydantic `BaseModel`s; `model_validate()` parses the JSON response with automatic type coercion and validation.
- **`UserAPIError`** — typed exception carrying `status_code` so callers can branch on HTTP semantics without re-parsing.
- **`UserClient.get_user`** — `async def` with `httpx.AsyncClient`; `try/except` separates 404, other HTTP errors, timeouts, and generic network failures.
- **`structlog`** — all observability goes through `logger.bind(...)` / `log.info/error`; no `print()` or `logging` anywhere.
- **`main`** — thin `async def` entry point, runs via `asyncio.run`.

Dependencies: `httpx`, `pydantic`, `structlog`.
