```python
"""HTTP client for fetching user data from the REST API."""

from typing import Optional

import httpx
import structlog
from pydantic import BaseModel

logger = structlog.get_logger(__name__)

API_BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """User address data."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User response model from the REST API."""

    id: int
    name: str
    email: str
    username: str
    phone: Optional[str] = None
    address: Optional[Address] = None


class UserNotFoundError(Exception):
    """Raised when the requested user does not exist."""


class UserAPIError(Exception):
    """Raised when the API returns an unexpected error."""


class UserClient:
    """Client for the user REST API."""

    def __init__(self, base_url: str = API_BASE_URL, timeout: float = 10.0) -> None:
        """Initialize the client with a base URL and timeout."""
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Args:
            user_id: The integer ID of the user to fetch.

        Returns:
            A User instance populated with the API response data.

        Raises:
            UserNotFoundError: If the user does not exist (HTTP 404).
            UserAPIError: If the API returns any other error.
        """
        url = f"{self._base_url}/users/{user_id}"
        log = logger.bind(user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
        except httpx.TimeoutException as exc:
            log.error("request_timeout", error=str(exc))
            raise UserAPIError(f"Request timed out for user {user_id}") from exc
        except httpx.RequestError as exc:
            log.error("request_error", error=str(exc))
            raise UserAPIError(f"Network error fetching user {user_id}") from exc

        if response.status_code == 404:
            log.warning("user_not_found")
            raise UserNotFoundError(f"User {user_id} not found")

        if response.status_code != 200:
            log.error("unexpected_status_code", status_code=response.status_code, body=response.text)
            raise UserAPIError(
                f"Unexpected status {response.status_code} fetching user {user_id}"
            )

        try:
            user = User.model_validate(response.json())
        except Exception as exc:
            log.error("response_parse_error", error=str(exc))
            raise UserAPIError(f"Failed to parse response for user {user_id}") from exc

        log.info("user_fetched", name=user.name, email=user.email)
        return user


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        logger.info("fetched_user", id=user.id, name=user.name, email=user.email)
    except UserNotFoundError:
        logger.warning("user_not_found", user_id=42)
    except UserAPIError as exc:
        logger.error("api_error", error=str(exc))


if __name__ == "__main__":
    import asyncio

    asyncio.run(main())
```

**Key design decisions:**

- **`User` / `Address`** — Pydantic `BaseModel` with typed fields; `phone` and `address` are `Optional` since real APIs often omit them.
- **`UserClient.get_user`** — `async def` using `httpx.AsyncClient`. Two separate `except` branches distinguish timeout vs. network errors before inspecting the status code.
- **404 vs other errors** — `UserNotFoundError` is a distinct exception so callers can handle "missing" separately from "broken API."
- **Response parsing** — `model_validate` is wrapped in its own `try/except` so a malformed payload produces a clear `UserAPIError` rather than a raw Pydantic crash.
- **Logging** — `structlog` throughout; no `print()` or `logging` imports.

Dependencies: `httpx`, `pydantic`, `structlog`.
