```python
# user_client.py
"""Client module for fetching user data from the example REST API."""

import asyncio
import os

import httpx
import structlog
from pydantic import BaseModel, EmailStr

log = structlog.get_logger()

BASE_URL = os.environ.get("API_BASE_URL", "https://api.example.com")
DEFAULT_TIMEOUT = 30


class Address(BaseModel):
    street: str
    city: str
    zipcode: str
    country: str


class User(BaseModel):
    id: int
    name: str
    email: EmailStr
    username: str
    address: Address | None = None
    phone: str | None = None


class UserNotFoundError(Exception):
    """Raised when the requested user does not exist."""

class UserAPIError(Exception):
    """Raised when the API returns an unexpected error."""


class UserClient:
    """Client for the /users endpoint of the example REST API."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = DEFAULT_TIMEOUT) -> None:
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout

    async def fetch_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Args:
            user_id: The numeric user ID to retrieve.
        Returns:
            A User instance populated with the API response data.
        Raises:
            UserNotFoundError: If the API returns 404.
            UserAPIError: If the API returns any other error status.
            httpx.RequestError: On network-level failures.
        """
        url = f"{self._base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
        except httpx.TimeoutException as exc:
            log.error("request_timeout", user_id=user_id, url=url)
            raise httpx.RequestError(f"Request to {url} timed out") from exc
        except httpx.RequestError:
            log.error("request_failed", user_id=user_id, url=url)
            raise

        if response.status_code == 404:
            log.warning("user_not_found", user_id=user_id)
            raise UserNotFoundError(f"User {user_id} not found")

        if response.status_code != 200:
            log.error("unexpected_status", user_id=user_id, status_code=response.status_code)
            raise UserAPIError(f"API returned {response.status_code} for user {user_id}")

        user = User.model_validate(response.json())
        log.info("user_fetched", user_id=user_id, username=user.username)
        return user


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.fetch_user(42)
        log.info("result", id=user.id, name=user.name, email=user.email)
    except UserNotFoundError:
        log.warning("user_does_not_exist", user_id=42)
    except UserAPIError as exc:
        log.error("api_error", error=str(exc))
    except httpx.RequestError as exc:
        log.error("network_error", error=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```

**Key design decisions:**

- **`httpx` + `async def`** — all I/O is async per the project rules; `httpx.AsyncClient` is the idiomatic choice
- **Pydantic `BaseModel`** — `User` and `Address` give runtime validation and a clear schema; `EmailStr` validates email format
- **`structlog`** — all logging uses structured key-value pairs, no `print()` or stdlib `logging`
- **Domain exceptions** — `UserNotFoundError` (404) and `UserAPIError` (other HTTP errors) let callers handle cases without inspecting status codes
- **`API_BASE_URL` env var** — base URL is configurable without touching code, keeping secrets out of source

**Dependencies:** `httpx`, `pydantic[email]`, `structlog`
