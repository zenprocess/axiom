```python
# user_client.py
"""Client for fetching user data from the example REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel, EmailStr

log = structlog.get_logger()

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """A user's address."""
    street: str
    city: str
    zip_code: str


class User(BaseModel):
    """User resource returned by the API."""
    id: int
    name: str
    email: EmailStr
    address: Address | None = None


class UserClient:
    """HTTP client for the /users endpoint."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = 10.0) -> None:
        """Initialize the client.

        Args:
            base_url: Root URL of the API (no trailing slash).
            timeout: Request timeout in seconds.
        """
        self._base_url = base_url
        self._timeout = timeout

    async def fetch_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Args:
            user_id: Numeric user identifier.

        Returns:
            Parsed User model.

        Raises:
            httpx.HTTPStatusError: On 4xx/5xx responses.
            httpx.RequestError: On connection or timeout failures.
        """
        url = f"{self._base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
                response.raise_for_status()
        except httpx.HTTPStatusError as exc:
            log.error(
                "user_fetch_http_error",
                user_id=user_id,
                status_code=exc.response.status_code,
                detail=exc.response.text,
            )
            raise
        except httpx.RequestError as exc:
            log.error("user_fetch_request_error", user_id=user_id, error=str(exc))
            raise

        user = User.model_validate(response.json())
        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.fetch_user(42)
    log.info("result", user=user.model_dump())


if __name__ == "__main__":
    asyncio.run(main())
```

**Key design decisions following project rules:**

| Rule | Applied as |
|---|---|
| `async-io` | `fetch_user` is `async def`, uses `httpx.AsyncClient` |
| `pydantic-models` | `User` and `Address` are `BaseModel` subclasses |
| `structlog-only` | All logging via `structlog.get_logger()`, no `print()` |
| `type-hints` | All signatures fully annotated |
| `error-handling` | Separate `except` blocks for `HTTPStatusError` and `RequestError`, both log context then re-raise |
| `docstrings` | All public classes and methods documented |

**Dependencies needed:** `httpx`, `structlog`, `pydantic[email]`
