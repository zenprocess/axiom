```python
# user_client.py
"""Client module for fetching user data from the example REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel, EmailStr

log = structlog.get_logger()

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """Represents a user's address."""
    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """Represents a user returned by the API."""
    id: int
    name: str
    email: EmailStr
    address: Address | None = None
    phone: str | None = None


class UserNotFoundError(Exception):
    """Raised when the requested user does not exist."""


class UserClient:
    """Async HTTP client for the /users endpoint."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = 10.0) -> None:
        """
        Args:
            base_url: Root URL of the API (no trailing slash).
            timeout: Request timeout in seconds.
        """
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Raises:
            UserNotFoundError: 404 from the server.
            httpx.HTTPStatusError: Other 4xx/5xx responses.
            httpx.RequestError: Network-level failures.
        """
        url = f"{self._base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
                if response.status_code == 404:
                    log.warning("user_not_found", user_id=user_id)
                    raise UserNotFoundError(f"User {user_id} not found")
                response.raise_for_status()
        except httpx.TimeoutException as exc:
            log.error("request_timeout", user_id=user_id, url=url)
            raise httpx.RequestError(f"Request timed out for user {user_id}") from exc
        except httpx.RequestError:
            log.error("network_error", user_id=user_id, url=url)
            raise

        user = User.model_validate(response.json())
        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        log.info("result", id=user.id, name=user.name, email=user.email)
    except UserNotFoundError as exc:
        log.error("user_not_found", error=str(exc))
    except httpx.HTTPStatusError as exc:
        log.error("http_error", status=exc.response.status_code, error=str(exc))
    except httpx.RequestError as exc:
        log.error("request_failed", error=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```

Key design decisions:

- **`User` / `Address`** — Pydantic `BaseModel` with `EmailStr` validation; `address` and `phone` are optional since real APIs often omit fields.
- **`UserClient.get_user`** — `async def` + `httpx.AsyncClient` so it never blocks the event loop.
- **Error hierarchy** — 404 → domain `UserNotFoundError`; other 4xx/5xx → `httpx.HTTPStatusError`; network failures → `httpx.RequestError`. Each is caught and logged separately in `main`.
- **`structlog`** — all log calls use structured key=value pairs, no `print` or stdlib `logging`.
- **No hardcoded secrets** — URL is a constant pointing at a public example host; credentials would come from env vars if needed.
