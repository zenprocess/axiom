```python
"""HTTP client for fetching user data from the example REST API."""

import asyncio
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel, Field

logger = structlog.get_logger(__name__)

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    street: str
    city: str
    state: str
    zip_code: str = Field(alias="zipCode")

    model_config = {"populate_by_name": True}


class User(BaseModel):
    id: int
    name: str
    email: str
    username: str
    address: Optional[Address] = None
    phone: Optional[str] = None


class UserNotFoundError(Exception):
    pass


class UserAPIError(Exception):
    pass


class UserClient:
    """Client for the user REST API."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = 10.0) -> None:
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Raises:
            UserNotFoundError: If the user does not exist (404).
            UserAPIError: For any other non-2xx response or network failure.
        """
        url = f"{self._base_url}/users/{user_id}"
        log = logger.bind(user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
        except httpx.TransportError as exc:
            log.error("network_error", error=str(exc))
            raise UserAPIError(f"Network error fetching user {user_id}: {exc}") from exc

        if response.status_code == 404:
            log.warning("user_not_found")
            raise UserNotFoundError(f"User {user_id} not found")

        if response.status_code != 200:
            log.error("api_error", status_code=response.status_code, body=response.text)
            raise UserAPIError(
                f"API returned {response.status_code} for user {user_id}"
            )

        log.info("user_fetched")
        return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and print the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        print(f"id={user.id}  name={user.name}  email={user.email}")
    except UserNotFoundError as exc:
        logger.warning("not_found", error=str(exc))
    except UserAPIError as exc:
        logger.error("fetch_failed", error=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```

**Key decisions:**

- **`httpx.AsyncClient`** — async I/O per project rules; also supports HTTP/2 and connection pooling
- **`structlog`** — structured logging per rules (no `import logging`)
- **Pydantic `BaseModel`** — typed models with `Field(alias=...)` to handle camelCase API fields
- **Two exception types** — `UserNotFoundError` for 404 (often handled differently by callers) vs `UserAPIError` for everything else
- **`httpx.TransportError`** — catches DNS failures, timeouts, and connection resets at the transport layer

Dependencies needed: `httpx`, `structlog`, `pydantic`.
