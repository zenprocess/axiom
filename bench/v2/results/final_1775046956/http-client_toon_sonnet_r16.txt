```python
"""Client for fetching user data from the example REST API."""

import asyncio
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel, HttpUrl

log = structlog.get_logger(__name__)

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    id: int
    name: str
    email: str
    username: str
    phone: Optional[str] = None
    website: Optional[HttpUrl] = None
    address: Optional[Address] = None


class UserNotFoundError(Exception):
    """Raised when the requested user does not exist."""


class UserAPIError(Exception):
    """Raised on unexpected API errors."""


class UserClient:
    """Async client for the user REST API."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = 10.0) -> None:
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Args:
            user_id: The numeric user ID to retrieve.

        Returns:
            A User model populated with the API response data.

        Raises:
            UserNotFoundError: If the API returns 404.
            UserAPIError: On any other non-2xx response or network error.
        """
        url = f"{self._base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
        except httpx.RequestError as exc:
            log.error("request_failed", user_id=user_id, error=str(exc))
            raise UserAPIError(f"Network error fetching user {user_id}: {exc}") from exc

        if response.status_code == 404:
            log.warning("user_not_found", user_id=user_id)
            raise UserNotFoundError(f"User {user_id} not found")

        if response.status_code != 200:
            log.error(
                "unexpected_status",
                user_id=user_id,
                status_code=response.status_code,
                body=response.text[:200],
            )
            raise UserAPIError(
                f"Unexpected status {response.status_code} fetching user {user_id}"
            )

        user = User.model_validate(response.json())
        log.info("user_fetched", user_id=user_id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and print their details."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        print(f"ID:       {user.id}")
        print(f"Name:     {user.name}")
        print(f"Email:    {user.email}")
        print(f"Username: {user.username}")
        if user.phone:
            print(f"Phone:    {user.phone}")
        if user.address:
            addr = user.address
            print(f"Address:  {addr.street}, {addr.city}, {addr.state} {addr.zip_code}")
    except UserNotFoundError as exc:
        log.error("user_not_found", error=str(exc))
    except UserAPIError as exc:
        log.error("api_error", error=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```

**Key design decisions:**

- **Models**: `User` and `Address` are Pydantic `BaseModel`s — `model_validate()` handles parsing and validation from the JSON response.
- **Error hierarchy**: `UserNotFoundError` (404) is separated from the general `UserAPIError` so callers can handle "not found" distinctly from "server broken".
- **Async I/O**: `get_user` is `async def` using `httpx.AsyncClient`, which is non-blocking and plays well with any async framework.
- **Error handling**: `httpx.RequestError` covers DNS failures, timeouts, and connection errors; non-200 status codes are caught explicitly.
- **Logging**: `structlog` with structured key-value fields (`user_id=`, `status_code=`) rather than `logging` or `print`.

Dependencies needed: `httpx`, `pydantic`, `structlog`.
