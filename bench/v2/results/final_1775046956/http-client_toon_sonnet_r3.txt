```python
"""HTTP client for fetching user data from the REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    street: str
    city: str
    zip: str


class UserResponse(BaseModel):
    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Client for the users endpoint of the REST API."""

    def __init__(self, base_url: str = BASE_URL) -> None:
        self._base_url = base_url

    async def fetch_user(self, user_id: int) -> UserResponse:
        """Fetch a single user by ID.

        Raises:
            httpx.HTTPStatusError: On 4xx/5xx responses.
            httpx.RequestError: On network-level failures.
        """
        url = f"{self._base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=10.0)
                response.raise_for_status()
        except httpx.HTTPStatusError as exc:
            log.error("http_error", user_id=user_id,
                      status_code=exc.response.status_code,
                      detail=exc.response.text)
            raise
        except httpx.RequestError as exc:
            log.error("request_error", user_id=user_id, error=str(exc))
            raise

        user = UserResponse.model_validate(response.json())
        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and print the result."""
    client = UserClient()
    user = await client.fetch_user(42)
    print(user.model_dump_json(indent=2))


if __name__ == "__main__":
    asyncio.run(main())
```

Key design decisions per your project rules:
- **`async def fetch_user`** — I/O is async with `httpx.AsyncClient` (`async-io` rule)
- **`structlog`** — structured logging at info/error levels, no `import logging` (`structlog-only` rule)
- **Pydantic `BaseModel`** — `UserResponse` + nested `Address` for type-safe deserialization (`pydantic-models` rule)
- **`try/except`** — separates HTTP status errors from network errors, logs both before re-raising (`error-handling` rule)
- **Full type hints** on all signatures (`type-hints` rule)
