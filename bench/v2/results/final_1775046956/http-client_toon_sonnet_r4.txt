```python
"""User API client — fetches user data from https://api.example.com/users/{id}."""

import asyncio
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel, HttpUrl

log = structlog.get_logger(__name__)

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    id: int
    name: str
    email: str
    username: str
    phone: Optional[str] = None
    website: Optional[HttpUrl] = None
    address: Optional[Address] = None


class UserNotFoundError(Exception):
    pass


class UserAPIError(Exception):
    pass


class UserClient:
    """Client for the user REST API."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = 10.0) -> None:
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Raises:
            UserNotFoundError: if the server returns 404.
            UserAPIError: for any other HTTP or network error.
        """
        url = f"{self._base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)

            if response.status_code == 404:
                raise UserNotFoundError(f"User {user_id} not found")

            response.raise_for_status()
            user = User.model_validate(response.json())
            log.info("user_fetched", user_id=user_id, name=user.name)
            return user

        except UserNotFoundError:
            raise
        except httpx.TimeoutException as exc:
            log.error("request_timeout", user_id=user_id, error=str(exc))
            raise UserAPIError(f"Request timed out for user {user_id}") from exc
        except httpx.HTTPStatusError as exc:
            log.error("http_error", user_id=user_id, status=exc.response.status_code)
            raise UserAPIError(
                f"HTTP {exc.response.status_code} fetching user {user_id}"
            ) from exc
        except httpx.RequestError as exc:
            log.error("request_error", user_id=user_id, error=str(exc))
            raise UserAPIError(f"Network error fetching user {user_id}") from exc


async def main() -> None:
    """Fetch user #42 and print the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        print(f"ID:       {user.id}")
        print(f"Name:     {user.name}")
        print(f"Email:    {user.email}")
        print(f"Username: {user.username}")
        if user.phone:
            print(f"Phone:    {user.phone}")
        if user.address:
            print(f"City:     {user.address.city}, {user.address.state}")
    except UserNotFoundError as exc:
        log.warning("user_not_found", error=str(exc))
    except UserAPIError as exc:
        log.error("api_error", error=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```

Key design decisions:

- **`httpx` + `async def`** — all I/O is async per project rules; `httpx.AsyncClient` used as a context manager so connections are properly closed
- **Two exception types** — `UserNotFoundError` (404, caller may want to handle specially) vs `UserAPIError` (all other failures)
- **Pydantic models** — `User` and `Address` with `model_validate()` for strict parsing; optional fields default to `None` rather than requiring the API to always return them
- **`structlog`** — structured logging throughout, never `print()` or stdlib `logging`
- **`raise ... from exc`** — preserves the original traceback chain on re-raises
