```python
"""Client for fetching user data from the example REST API."""

import asyncio
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel, HttpUrl

log = structlog.get_logger(__name__)


class Address(BaseModel):
    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    id: int
    name: str
    email: str
    username: str
    phone: Optional[str] = None
    address: Optional[Address] = None
    website: Optional[HttpUrl] = None


class UserNotFoundError(Exception):
    """Raised when the requested user does not exist."""


class UserAPIError(Exception):
    """Raised when the API returns an unexpected error."""


class UserClient:
    """Fetches user data from the REST API at api.example.com."""

    BASE_URL = "https://api.example.com"

    def __init__(self, timeout: float = 10.0) -> None:
        self._client = httpx.AsyncClient(
            base_url=self.BASE_URL,
            timeout=timeout,
            headers={"Accept": "application/json"},
        )

    async def __aenter__(self) -> "UserClient":
        return self

    async def __aexit__(self, *_: object) -> None:
        await self._client.aclose()

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Args:
            user_id: The numeric user identifier.

        Returns:
            A User instance populated from the API response.

        Raises:
            UserNotFoundError: If the user does not exist (404).
            UserAPIError: If the API returns any other error.
        """
        log.info("fetching user", user_id=user_id)
        try:
            response = await self._client.get(f"/users/{user_id}")
        except httpx.TimeoutException as exc:
            raise UserAPIError(f"Request timed out for user {user_id}") from exc
        except httpx.RequestError as exc:
            raise UserAPIError(f"Network error fetching user {user_id}: {exc}") from exc

        if response.status_code == 404:
            raise UserNotFoundError(f"User {user_id} not found")

        if response.status_code != 200:
            raise UserAPIError(
                f"Unexpected status {response.status_code} for user {user_id}: {response.text}"
            )

        log.info("user fetched successfully", user_id=user_id)
        return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and print their details."""
    async with UserClient() as client:
        try:
            user = await client.get_user(42)
            log.info("user details", id=user.id, name=user.name, email=user.email)
        except UserNotFoundError as exc:
            log.error("user not found", error=str(exc))
        except UserAPIError as exc:
            log.error("api error", error=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```

**Key design decisions:**

- **`httpx.AsyncClient`** — async HTTP, used as a context manager so the connection pool is always closed cleanly
- **Two custom exceptions** — `UserNotFoundError` (404) vs `UserAPIError` (everything else), so callers can handle "doesn't exist" differently from "network broken"
- **Pydantic `User` / `Address`** — validates and coerces the JSON response; `Address` and `website` are optional since real APIs often return partial data
- **`structlog`** — structured logging per project rules, no `print()` or `logging`
- **`main()`** uses the context manager and catches both exception types, logging each distinctly

Dependencies: `httpx`, `pydantic`, `structlog`.
