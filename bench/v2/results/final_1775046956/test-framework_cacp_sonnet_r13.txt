**`testkit/framework.py`**
```python
"""Core test framework: assertions, fixtures, TestCase, and test runner."""

from __future__ import annotations

import inspect
import importlib.util
import sys
import time
import traceback
from contextlib import contextmanager
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Callable, Generator, Iterator, Optional, Type


# ---------------------------------------------------------------------------
# Assertions
# ---------------------------------------------------------------------------

def assert_equal(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert that actual == expected."""
    if actual != expected:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected {expected!r}, got {actual!r}{detail}")


def assert_not_equal(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert that actual != expected."""
    if actual == expected:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected values to differ, both are {actual!r}{detail}")


def assert_true(expr: Any, msg: str = "") -> None:
    """Assert that expr is truthy."""
    if not expr:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected truthy, got {expr!r}{detail}")


def assert_false(expr: Any, msg: str = "") -> None:
    """Assert that expr is falsy."""
    if expr:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected falsy, got {expr!r}{detail}")


def assert_in(item: Any, container: Any, msg: str = "") -> None:
    """Assert that item is in container."""
    if item not in container:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected {item!r} to be in {container!r}{detail}")


def assert_is_none(value: Any, msg: str = "") -> None:
    """Assert that value is None."""
    if value is not None:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected None, got {value!r}{detail}")


def assert_is_not_none(value: Any, msg: str = "") -> None:
    """Assert that value is not None."""
    if value is None:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected non-None value{detail}")


@contextmanager
def assert_raises(exc_type: Type[BaseException], msg: str = "") -> Iterator[None]:
    """Context manager: assert that the block raises exc_type."""
    try:
        yield
    except exc_type:
        return
    except Exception as e:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(
            f"Expected {exc_type.__name__} but got {type(e).__name__}: {e}{detail}"
        ) from e
    else:
        detail = f" — {msg}" if msg else ""
        raise AssertionError(f"Expected {exc_type.__name__} to be raised{detail}")


# ---------------------------------------------------------------------------
# Fixtures
# ---------------------------------------------------------------------------

_FIXTURE_ATTR = "_testkit_fixture"


def fixture(fn: Callable) -> Callable:
    """Decorator: marks a method as a fixture (setup/teardown via yield)."""
    setattr(fn, _FIXTURE_ATTR, True)
    return fn


# ---------------------------------------------------------------------------
# Result types
# ---------------------------------------------------------------------------

@dataclass
class TestResult:
    name: str
    passed: bool
    error: Optional[str] = None
    duration_ms: float = 0.0


@dataclass
class SuiteResult:
    suite_name: str
    results: list[TestResult] = field(default_factory=list)

    @property
    def passed(self) -> int:
        return sum(1 for r in self.results if r.passed)

    @property
    def failed(self) -> int:
        return sum(1 for r in self.results if not r.passed)

    @property
    def total(self) -> int:
        return len(self.results)


# ---------------------------------------------------------------------------
# TestCase base class
# ---------------------------------------------------------------------------

class TestCase:
    """Base class for test suites. Methods named test_* are auto-discovered."""

    def setup(self) -> None:
        """Called before each test method. Override to prepare state."""

    def teardown(self) -> None:
        """Called after each test method. Override to clean up state."""

    # Convenience assertion shortcuts
    assert_equal = staticmethod(assert_equal)
    assert_not_equal = staticmethod(assert_not_equal)
    assert_true = staticmethod(assert_true)
    assert_false = staticmethod(assert_false)
    assert_in = staticmethod(assert_in)
    assert_is_none = staticmethod(assert_is_none)
    assert_is_not_none = staticmethod(assert_is_not_none)
    assert_raises = staticmethod(assert_raises)


# ---------------------------------------------------------------------------
# Runner
# ---------------------------------------------------------------------------

def _collect_test_methods(case: TestCase) -> list[tuple[str, Callable]]:
    """Return all test_* methods on a TestCase instance, sorted by name."""
    methods = []
    for name in sorted(dir(case)):
        if name.startswith("test_"):
            fn = getattr(case, name)
            if callable(fn):
                methods.append((name, fn))
    return methods


def _collect_fixtures(case: TestCase) -> list[Callable]:
    """Return all @fixture-decorated methods on a TestCase instance."""
    fixtures = []
    for name in dir(case):
        fn = getattr(case, name)
        if callable(fn) and getattr(fn, _FIXTURE_ATTR, False):
            fixtures.append(fn)
    return fixtures


def _run_suite(case: TestCase) -> SuiteResult:
    """Execute all tests in a single TestCase instance."""
    suite_name = type(case).__name__
    result = SuiteResult(suite_name=suite_name)

    # Activate fixtures (generators) — each fixture yields once
    active_fixtures: list[Generator] = []
    for fx in _collect_fixtures(case):
        gen = fx()
        if inspect.isgenerator(gen):
            try:
                next(gen)
                active_fixtures.append(gen)
            except StopIteration:
                pass

    for name, method in _collect_test_methods(case):
        case.setup()
        start = time.perf_counter()
        error: Optional[str] = None
        passed = True
        try:
            method()
        except AssertionError as e:
            passed = False
            error = str(e)
        except Exception as e:
            passed = False
            error = traceback.format_exc().strip()
        finally:
            case.teardown()

        duration_ms = (time.perf_counter() - start) * 1000
        result.results.append(TestResult(name, passed, error, duration_ms))

    # Tear down fixtures in reverse order
    for gen in reversed(active_fixtures):
        try:
            next(gen)
        except StopIteration:
            pass

    return result


def _print_suite_result(suite: SuiteResult) -> None:
    """Print formatted results for a single suite."""
    print(f"\n{'='*60}")
    print(f"  {suite.suite_name}")
    print(f"{'='*60}")
    for r in suite.results:
        status = "PASS" if r.passed else "FAIL"
        mark = "✓" if r.passed else "✗"
        print(f"  {mark} [{status}] {r.name}  ({r.duration_ms:.1f}ms)")
        if r.error:
            for line in r.error.splitlines():
                print(f"         {line}")
    print(f"\n  Results: {suite.passed}/{suite.total} passed", end="")
    if suite.failed:
        print(f", {suite.failed} failed", end="")
    print()


def run(*cases: TestCase) -> bool:
    """Run one or more TestCase instances and print results. Returns True if all passed."""
    all_passed = True
    for case in cases:
        suite = _run_suite(case)
        _print_suite_result(suite)
        if suite.failed:
            all_passed = False

    return all_passed


# ---------------------------------------------------------------------------
# Discovery
# ---------------------------------------------------------------------------

def _load_module_from_path(path: Path):
    """Import a Python file as a module."""
    spec = importlib.util.spec_from_file_location(path.stem, path)
    mod = importlib.util.module_from_spec(spec)
    sys.modules[path.stem] = mod
    spec.loader.exec_module(mod)
    return mod


def discover_and_run(directory: str = ".", pattern: str = "test_*.py") -> bool:
    """
    Discover test files matching pattern under directory, run all TestCase subclasses.

    Returns True if every test passed.
    """
    root = Path(directory).resolve()
    files = sorted(root.rglob(pattern))

    if not files:
        print(f"No test files matching '{pattern}' found under {root}")
        return True

    print(f"Discovered {len(files)} test file(s) under {root}")

    all_passed = True
    seen_bases = set()

    for path in files:
        mod = _load_module_from_path(path)
        cases: list[TestCase] = []
        for attr_name in dir(mod):
            cls = getattr(mod, attr_name)
            if (
                isinstance(cls, type)
                and issubclass(cls, TestCase)
                and cls is not TestCase
                and id(cls) not in seen_bases
            ):
                seen_bases.add(id(cls))
                cases.append(cls())

        if cases:
            if not run(*cases):
                all_passed = False
        else:
            print(f"  (no TestCase subclasses in {path.name})")

    # Summary
    print(f"\n{'='*60}")
    print(f"  Overall: {'ALL PASSED' if all_passed else 'FAILURES DETECTED'}")
    print(f"{'='*60}")
    return all_passed
```

**`testkit/example_tests.py`** — usage example
```python
"""Example: demonstrates TestCase, fixtures, and all assertion helpers."""

from testkit import (
    TestCase, fixture,
    assert_equal, assert_raises,
)


class TestMath(TestCase):
    def test_addition(self):
        self.assert_equal(1 + 1, 2)

    def test_subtraction(self):
        assert_equal(10 - 3, 7)

    def test_division_by_zero(self):
        with assert_raises(ZeroDivisionError):
            _ = 1 / 0

    def test_string_in_list(self):
        self.assert_in("b", ["a", "b", "c"])


class TestWithFixture(TestCase):
    data: list

    @fixture
    def _setup_data(self):
        self.data = [1, 2, 3]
        yield                  # test runs here
        self.data = []         # teardown

    def test_list_length(self):
        self.assert_equal(len(self.data), 3)

    def test_first_element(self):
        self.assert_equal(self.data[0], 1)


if __name__ == "__main__":
    from testkit import run
    run(TestMath(), TestWithFixture())
```

---

Key design decisions:
- **Fixtures** use `yield`-based generators (like pytest) — code before `yield` is setup, after is teardown
- **Discovery** uses `importlib` to load `test_*.py` files and finds `TestCase` subclasses automatically
- **`setup()`/`teardown()`** run around every test method; `@fixture` runs once per suite
- Assertions are available both as standalone functions and as `self.assert_*` methods on `TestCase`
