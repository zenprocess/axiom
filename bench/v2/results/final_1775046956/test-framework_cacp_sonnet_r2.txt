```python
# minitest.py — minimal Python test framework
# NOTE: print() is used intentionally for test runner output, per your request.
#       The project rules prohibit print() for debugging/logging — this is output.

from __future__ import annotations

import inspect
import sys
import time
import traceback
from collections.abc import Callable, Generator
from contextlib import contextmanager
from dataclasses import dataclass, field
from pathlib import Path
from types import ModuleType
from typing import Any

from pydantic import BaseModel


# ── Models ──────────────────────────────────────────────────────────────────


class TestResult(BaseModel):
    """Outcome of a single test function."""
    name: str
    passed: bool
    duration_ms: float
    error: str | None = None
    skipped: bool = False
    skip_reason: str | None = None


class SuiteResult(BaseModel):
    """Aggregated results for a test suite."""
    suite_name: str
    results: list[TestResult] = []

    @property
    def passed(self) -> int:
        return sum(1 for r in self.results if r.passed and not r.skipped)

    @property
    def failed(self) -> int:
        return sum(1 for r in self.results if not r.passed and not r.skipped)

    @property
    def skipped(self) -> int:
        return sum(1 for r in self.results if r.skipped)

    @property
    def total(self) -> int:
        return len(self.results)


# ── Exceptions ───────────────────────────────────────────────────────────────


class AssertionFailure(AssertionError):
    """Raised when a minitest assertion fails."""


class SkipTest(Exception):
    """Raise to skip the current test."""
    def __init__(self, reason: str = "") -> None:
        super().__init__(reason)
        self.reason = reason


# ── Assertions ───────────────────────────────────────────────────────────────


def assert_equal(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert actual == expected."""
    if actual != expected:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}{actual!r} != {expected!r}")


def assert_not_equal(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert actual != expected."""
    if actual == expected:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}{actual!r} == {expected!r}")


def assert_true(value: Any, msg: str = "") -> None:
    """Assert value is truthy."""
    if not value:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}expected truthy, got {value!r}")


def assert_false(value: Any, msg: str = "") -> None:
    """Assert value is falsy."""
    if value:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}expected falsy, got {value!r}")


def assert_is_none(value: Any, msg: str = "") -> None:
    """Assert value is None."""
    if value is not None:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}{value!r} is not None")


def assert_is_not_none(value: Any, msg: str = "") -> None:
    """Assert value is not None."""
    if value is None:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}unexpected None")


def assert_in(item: Any, container: Any, msg: str = "") -> None:
    """Assert item in container."""
    if item not in container:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}{item!r} not in {container!r}")


def assert_not_in(item: Any, container: Any, msg: str = "") -> None:
    """Assert item not in container."""
    if item in container:
        raise AssertionFailure(f"{msg + ': ' if msg else ''}{item!r} in {container!r}")


@contextmanager
def assert_raises(exc_type: type[BaseException], msg: str = "") -> Generator[None, None, None]:
    """Context manager: assert that exc_type is raised inside the block.

    Usage::

        with assert_raises(ValueError):
            int("oops")
    """
    try:
        yield
    except exc_type:
        return
    except Exception as exc:
        raise AssertionFailure(
            f"{msg + ': ' if msg else ''}expected {exc_type.__name__}, "
            f"got {type(exc).__name__}: {exc}"
        ) from exc
    else:
        raise AssertionFailure(
            f"{msg + ': ' if msg else ''}{exc_type.__name__} was not raised"
        )


def skip(reason: str = "") -> None:
    """Skip the current test."""
    raise SkipTest(reason)


# ── Fixtures ─────────────────────────────────────────────────────────────────


@dataclass
class Fixture:
    """A named setup / teardown pair."""
    name: str
    setup: Callable[[], Any] | None = None
    teardown: Callable[[Any], None] | None = None

    def run_setup(self) -> Any:
        """Run setup and return the fixture value."""
        return self.setup() if self.setup else None

    def run_teardown(self, value: Any) -> None:
        """Run teardown with the fixture value."""
        if self.teardown:
            self.teardown(value)


_REGISTRY: dict[str, Fixture] = {}


def fixture(
    name: str,
    setup: Callable[[], Any] | None = None,
    teardown: Callable[[Any], None] | None = None,
) -> Fixture:
    """Register and return a named fixture."""
    f = Fixture(name=name, setup=setup, teardown=teardown)
    _REGISTRY[name] = f
    return f


# ── Test suite ───────────────────────────────────────────────────────────────


@dataclass
class TestSuite:
    """A named collection of test functions."""
    name: str
    tests: list[Callable[..., Any]] = field(default_factory=list)
    fixture_names: list[str] = field(default_factory=list)

    def add(self, fn: Callable[..., Any]) -> Callable[..., Any]:
        """Register a test (usable as a decorator)."""
        self.tests.append(fn)
        return fn

    def use_fixture(self, name: str) -> None:
        """Declare a suite-level fixture requirement."""
        self.fixture_names.append(name)

    def run(self) -> SuiteResult:
        """Execute all tests; run suite-level fixtures around them."""
        suite_result = SuiteResult(suite_name=self.name)
        active: list[tuple[Fixture, Any]] = []

        for fname in self.fixture_names:
            f = _REGISTRY.get(fname)
            if f is None:
                raise RuntimeError(f"Fixture '{fname}' not registered")
            active.append((f, f.run_setup()))

        for fn in self.tests:
            suite_result.results.append(_run_test(fn))

        for f, value in reversed(active):
            f.run_teardown(value)

        return suite_result


# ── Runner internals ─────────────────────────────────────────────────────────


def _run_test(fn: Callable[..., Any]) -> TestResult:
    """Execute one test function and return a TestResult."""
    start = time.perf_counter()
    try:
        if inspect.iscoroutinefunction(fn):
            import asyncio
            asyncio.get_event_loop().run_until_complete(fn())
        else:
            fn()
        ms = round((time.perf_counter() - start) * 1000, 2)
        return TestResult(name=fn.__name__, passed=True, duration_ms=ms)
    except SkipTest as exc:
        ms = round((time.perf_counter() - start) * 1000, 2)
        return TestResult(
            name=fn.__name__, passed=False, duration_ms=ms,
            skipped=True, skip_reason=exc.reason or None,
        )
    except Exception:
        ms = round((time.perf_counter() - start) * 1000, 2)
        return TestResult(
            name=fn.__name__, passed=False, duration_ms=ms,
            error=traceback.format_exc(),
        )


# ── Discovery ────────────────────────────────────────────────────────────────


def discover_tests(module: ModuleType) -> TestSuite:
    """Build a TestSuite from all test_* callables in *module*."""
    suite = TestSuite(name=getattr(module, "__name__", repr(module)))
    for attr in dir(module):
        if attr.startswith("test_"):
            obj = getattr(module, attr)
            if callable(obj):
                suite.add(obj)
    return suite


def discover_from_path(path: str | Path) -> list[TestSuite]:
    """Recursively import test_*.py files under *path* and return suites."""
    import importlib.util

    suites: list[TestSuite] = []
    for test_file in sorted(Path(path).resolve().rglob("test_*.py")):
        spec = importlib.util.spec_from_file_location(test_file.stem, test_file)
        if not spec or not spec.loader:
            continue
        module = importlib.util.module_from_spec(spec)
        try:
            spec.loader.exec_module(module)   # type: ignore[union-attr]
        except Exception:
            continue
        suites.append(discover_tests(module))
    return suites


# ── Output / runner ──────────────────────────────────────────────────────────


def run_suite(suite: TestSuite, *, verbose: bool = False) -> SuiteResult:
    """Run one suite and print per-test results."""
    print(f"\n{'=' * 60}")
    print(f"Suite: {suite.name}")
    print("=" * 60)

    result = suite.run()

    for r in result.results:
        if r.skipped:
            extra = f"  ({r.skip_reason})" if r.skip_reason else ""
            print(f"  [SKIP] {r.name}{extra}  ({r.duration_ms:.1f}ms)")
        elif r.passed:
            print(f"  [PASS] {r.name}  ({r.duration_ms:.1f}ms)")
        else:
            print(f"  [FAIL] {r.name}  ({r.duration_ms:.1f}ms)")
            if r.error and verbose:
                for line in r.error.strip().splitlines():
                    print(f"         {line}")

    print("-" * 60)
    print(
        f"  {result.total} tests  |  {result.passed} passed  |  "
        f"{result.failed} failed  |  {result.skipped} skipped"
    )
    return result


def run_suites(suites: list[TestSuite], *, verbose: bool = False) -> int:
    """Run all suites, print totals, return exit code (0=ok, 1=failures)."""
    all_results = [run_suite(s, verbose=verbose) for s in suites]

    total   = sum(r.total   for r in all_results)
    passed  = sum(r.passed  for r in all_results)
    failed  = sum(r.failed  for r in all_results)
    skipped = sum(r.skipped for r in all_results)

    print(f"\n{'=' * 60}")
    print("TOTAL")
    print("=" * 60)
    print(f"  {total} tests  |  {passed} passed  |  {failed} failed  |  {skipped} skipped")
    return 0 if failed == 0 else 1


# ── CLI ──────────────────────────────────────────────────────────────────────


def main() -> None:
    """CLI entry point: discover and run tests from a directory."""
    import argparse

    parser = argparse.ArgumentParser(description="minitest runner")
    parser.add_argument("path", nargs="?", default=".", help="root directory")
    parser.add_argument("-v", "--verbose", action="store_true", help="show tracebacks")
    args = parser.parse_args()

    suites = discover_from_path(args.path)
    if not suites:
        print(f"No test files found under '{args.path}'")
        sys.exit(0)

    sys.exit(run_suites(suites, verbose=args.verbose))


if __name__ == "__main__":
    main()
```

**Usage — inline suite:**
```python
from minitest import TestSuite, assert_equal, assert_raises, fixture, skip, run_suite

db_fixture = fixture("db", setup=lambda: {"users": []}, teardown=lambda v: v.clear())

suite = TestSuite("math")
suite.use_fixture("db")

@suite.add
def test_addition() -> None:
    assert_equal(1 + 1, 2)

@suite.add
def test_divide_by_zero() -> None:
    with assert_raises(ZeroDivisionError):
        _ = 1 / 0

@suite.add
def test_wip() -> None:
    skip("not ready yet")

run_suite(suite, verbose=True)
```

**Key design decisions:**

| Feature | Approach |
|---|---|
| Data models | `TestResult`, `SuiteResult` via Pydantic `BaseModel` |
| Assertions | Standalone functions raising `AssertionFailure`; `assert_raises` is a context manager |
| Fixtures | Named `Fixture` objects in a global registry; declared per-suite via `use_fixture()` |
| Discovery | `discover_tests(module)` scans `test_*` callables; `discover_from_path()` walks the FS |
| Async tests | `inspect.iscoroutinefunction` detection + `asyncio.run` |
| Skip | `skip()` raises `SkipTest` — clean separation from failures |
| Output | `print()` used only for structured runner output, per your explicit request |
