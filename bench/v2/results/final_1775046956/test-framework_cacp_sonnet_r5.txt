## `testframework.py` — the framework

```python
"""A simple Python test framework with assertions, fixtures, and async test runner."""

import asyncio
import inspect
import time
import traceback
from collections.abc import Callable, Generator
from contextlib import contextmanager
from typing import Any

import structlog
from pydantic import BaseModel

logger = structlog.get_logger()


# ---------------------------------------------------------------------------
# Data models
# ---------------------------------------------------------------------------


class TestResult(BaseModel):
    """Result of a single test function execution."""

    name: str
    passed: bool
    error: str | None = None
    duration_s: float = 0.0


class TestSuiteResult(BaseModel):
    """Aggregated results for a full test run."""

    name: str
    results: list[TestResult] = []

    @property
    def passed(self) -> int:
        """Number of passing tests."""
        return sum(1 for r in self.results if r.passed)

    @property
    def failed(self) -> int:
        """Number of failing tests."""
        return sum(1 for r in self.results if not r.passed)

    @property
    def total(self) -> int:
        """Total number of tests."""
        return len(self.results)


# ---------------------------------------------------------------------------
# Assertions
# ---------------------------------------------------------------------------


class AssertionFailure(Exception):
    """Raised when a test assertion fails."""


def assert_equal(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert that actual == expected."""
    if actual != expected:
        raise AssertionFailure(msg or f"assert_equal: {actual!r} != {expected!r}")


def assert_not_equal(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert that actual != expected."""
    if actual == expected:
        raise AssertionFailure(msg or f"assert_not_equal: both are {actual!r}")


def assert_true(value: Any, msg: str = "") -> None:
    """Assert that value is truthy."""
    if not value:
        raise AssertionFailure(msg or f"assert_true: {value!r} is not truthy")


def assert_false(value: Any, msg: str = "") -> None:
    """Assert that value is falsy."""
    if value:
        raise AssertionFailure(msg or f"assert_false: {value!r} is not falsy")


def assert_none(value: Any, msg: str = "") -> None:
    """Assert that value is None."""
    if value is not None:
        raise AssertionFailure(msg or f"assert_none: got {value!r}")


def assert_not_none(value: Any, msg: str = "") -> None:
    """Assert that value is not None."""
    if value is None:
        raise AssertionFailure(msg or "assert_not_none: got None")


def assert_in(item: Any, container: Any, msg: str = "") -> None:
    """Assert that item is in container."""
    if item not in container:
        raise AssertionFailure(msg or f"assert_in: {item!r} not in {container!r}")


def assert_not_in(item: Any, container: Any, msg: str = "") -> None:
    """Assert that item is not in container."""
    if item in container:
        raise AssertionFailure(msg or f"assert_not_in: {item!r} found in {container!r}")


def assert_isinstance(value: Any, expected_type: type | tuple[type, ...], msg: str = "") -> None:
    """Assert that value is an instance of expected_type."""
    if not isinstance(value, expected_type):
        raise AssertionFailure(
            msg or f"assert_isinstance: {type(value).__name__} is not {expected_type}"
        )


@contextmanager
def assert_raises(exc_type: type[Exception], msg: str = "") -> Generator[None, None, None]:
    """Context manager asserting the enclosed block raises exc_type."""
    try:
        yield
    except exc_type:
        return
    except Exception as e:
        raise AssertionFailure(
            msg or f"assert_raises: expected {exc_type.__name__}, got {type(e).__name__}"
        ) from e
    raise AssertionFailure(msg or f"assert_raises: {exc_type.__name__} was not raised")


# ---------------------------------------------------------------------------
# Fixture registry
# ---------------------------------------------------------------------------

_fixtures: dict[str, Callable[[], Any]] = {}


def fixture(func: Callable[[], Any]) -> Callable[[], Any]:
    """Decorator that registers a callable as a named fixture."""
    _fixtures[func.__name__] = func
    return func


# ---------------------------------------------------------------------------
# Test registry
# ---------------------------------------------------------------------------

_tests: list[Callable[..., Any]] = []


def test(func: Callable[..., Any]) -> Callable[..., Any]:
    """Decorator that registers a function as a test case."""
    _tests.append(func)
    return func


def discover_tests(module: Any) -> list[Callable[..., Any]]:
    """Discover functions whose names start with 'test_' in a module."""
    return [
        obj
        for name, obj in inspect.getmembers(module, inspect.isfunction)
        if name.startswith("test_")
    ]


# ---------------------------------------------------------------------------
# Runner internals
# ---------------------------------------------------------------------------


async def _resolve_fixtures(func: Callable[..., Any]) -> dict[str, Any]:
    """Build the kwargs dict by resolving each parameter against the fixture registry."""
    kwargs: dict[str, Any] = {}
    for param_name in inspect.signature(func).parameters:
        if param_name not in _fixtures:
            logger.warning("fixture_not_found", param=param_name, test=func.__name__)
            continue
        fn = _fixtures[param_name]
        kwargs[param_name] = await fn() if inspect.iscoroutinefunction(fn) else fn()
    return kwargs


async def _run_single(func: Callable[..., Any]) -> TestResult:
    """Execute one test and return a TestResult."""
    kwargs = await _resolve_fixtures(func)
    start = time.perf_counter()
    try:
        if inspect.iscoroutinefunction(func):
            await func(**kwargs)
        else:
            func(**kwargs)
        return TestResult(name=func.__name__, passed=True, duration_s=time.perf_counter() - start)
    except AssertionFailure as exc:
        return TestResult(
            name=func.__name__,
            passed=False,
            error=str(exc),
            duration_s=time.perf_counter() - start,
        )
    except Exception as exc:
        return TestResult(
            name=func.__name__,
            passed=False,
            error=f"{type(exc).__name__}: {exc}\n{traceback.format_exc()}",
            duration_s=time.perf_counter() - start,
        )


def _print_result(result: TestResult) -> None:
    """Print one test result line plus optional error detail."""
    status = "PASS" if result.passed else "FAIL"
    print(f"  [{status}] {result.name} ({result.duration_s:.3f}s)")
    if not result.passed and result.error:
        for line in result.error.splitlines():
            print(f"         {line}")


def _print_summary(suite: TestSuiteResult) -> None:
    """Print the pass/fail totals."""
    bar = "=" * 44
    print(f"\n{bar}")
    print(f"  {suite.passed}/{suite.total} passed  |  {suite.failed} failed")
    if suite.failed == 0:
        print("  All tests passed.")
    print(f"{bar}\n")


# ---------------------------------------------------------------------------
# Public runner
# ---------------------------------------------------------------------------


async def run_tests(
    tests: list[Callable[..., Any]] | None = None,
    suite_name: str = "Test Suite",
) -> TestSuiteResult:
    """Run all registered (or explicitly provided) tests and print results."""
    cases = tests if tests is not None else _tests
    suite = TestSuiteResult(name=suite_name)

    print(f"\n=== {suite.name} ===\n")
    for func in cases:
        result = await _run_single(func)
        suite.results.append(result)
        _print_result(result)

    _print_summary(suite)
    return suite


async def main() -> None:
    """Entry point — runs all @test-decorated functions in this process."""
    await run_tests()


if __name__ == "__main__":
    asyncio.run(main())
```

## `test_example.py` — usage demo

```python
"""Example tests demonstrating the testframework API."""

import asyncio

from testframework import (
    assert_equal,
    assert_false,
    assert_in,
    assert_isinstance,
    assert_not_none,
    assert_raises,
    assert_true,
    fixture,
    run_tests,
    test,
)


# ---------------------------------------------------------------------------
# Fixtures
# ---------------------------------------------------------------------------


@fixture
def sample_list() -> list[int]:
    """Provide a small list of integers."""
    return [1, 2, 3, 4, 5]


@fixture
async def async_config() -> dict[str, str]:
    """Provide config loaded asynchronously (simulated)."""
    await asyncio.sleep(0)          # represents an async fetch
    return {"env": "test", "version": "1.0"}


# ---------------------------------------------------------------------------
# Tests
# ---------------------------------------------------------------------------


@test
def test_addition() -> None:
    """Basic arithmetic passes."""
    assert_equal(1 + 1, 2)
    assert_equal(10 - 3, 7)


@test
def test_string_ops() -> None:
    """String membership and truthiness."""
    assert_in("hello", "hello world")
    assert_true("non-empty")
    assert_false("")


@test
def test_type_check() -> None:
    """isinstance assertion."""
    assert_isinstance(42, int)
    assert_isinstance("hi", (str, bytes))


@test
def test_raises_on_bad_division() -> None:
    """assert_raises catches ZeroDivisionError."""
    with assert_raises(ZeroDivisionError):
        _ = 1 / 0


@test
def test_fixture_injection(sample_list: list[int]) -> None:
    """Fixture is injected by parameter name."""
    assert_equal(len(sample_list), 5)
    assert_in(3, sample_list)
    assert_not_none(sample_list)


@test
async def test_async_fixture(async_config: dict[str, str]) -> None:
    """Async fixture is awaited before injection."""
    assert_equal(async_config["env"], "test")
    assert_in("version", async_config)


@test
def test_intentional_failure() -> None:
    """This test is expected to fail — demonstrates failure output."""
    assert_equal(1, 2, "one should equal two (intentional failure)")


# ---------------------------------------------------------------------------
# Run
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    asyncio.run(run_tests(suite_name="Example Tests"))
```

## Key design decisions

| Concern | Approach |
|---|---|
| **Assertions** | Standalone functions raising `AssertionFailure`; `assert_raises` is a context manager |
| **Fixtures** | `@fixture` registers by function name; injected by matching test parameter names |
| **Async support** | Both fixtures and test functions can be `async def` — runner detects and awaits |
| **Discovery** | `@test` decorator for explicit registration; `discover_tests(module)` for name-convention scanning |
| **Models** | `TestResult` / `TestSuiteResult` are Pydantic `BaseModel` — easy to serialize or extend |
| **Logging** | Internal warnings use `structlog`; test output uses `print` as requested |
