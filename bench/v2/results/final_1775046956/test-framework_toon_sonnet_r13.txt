**`testkit.py`**

```python
"""Simple Python test framework with assertions, fixtures, and test discovery."""

from __future__ import annotations

import inspect
import sys
import time
import traceback
from collections.abc import Callable
from dataclasses import dataclass, field
from typing import Any


# ---------------------------------------------------------------------------
# Assertions
# ---------------------------------------------------------------------------

class AssertionFailed(Exception):
    pass


def assert_eq(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert that actual == expected."""
    if actual != expected:
        detail = f"{actual!r} != {expected!r}"
        raise AssertionFailed(f"assert_eq failed: {detail}" + (f" — {msg}" if msg else ""))


def assert_ne(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert that actual != expected."""
    if actual == expected:
        raise AssertionFailed(f"assert_ne failed: both are {actual!r}" + (f" — {msg}" if msg else ""))


def assert_true(value: Any, msg: str = "") -> None:
    """Assert that value is truthy."""
    if not value:
        raise AssertionFailed(f"assert_true failed: {value!r}" + (f" — {msg}" if msg else ""))


def assert_false(value: Any, msg: str = "") -> None:
    """Assert that value is falsy."""
    if value:
        raise AssertionFailed(f"assert_false failed: {value!r}" + (f" — {msg}" if msg else ""))


def assert_in(item: Any, container: Any, msg: str = "") -> None:
    """Assert that item is in container."""
    if item not in container:
        raise AssertionFailed(f"assert_in failed: {item!r} not in {container!r}" + (f" — {msg}" if msg else ""))


def assert_raises(exc_type: type[BaseException], fn: Callable, *args: Any, **kwargs: Any) -> BaseException:
    """Assert fn(*args, **kwargs) raises exc_type. Returns the exception."""
    try:
        fn(*args, **kwargs)
    except exc_type as e:
        return e
    except Exception as e:
        raise AssertionFailed(
            f"assert_raises: expected {exc_type.__name__}, got {type(e).__name__}: {e}"
        ) from None
    raise AssertionFailed(f"assert_raises: {exc_type.__name__} was not raised")


# ---------------------------------------------------------------------------
# Fixtures
# ---------------------------------------------------------------------------

_fixtures: dict[str, Callable] = {}


def fixture(fn: Callable) -> Callable:
    """Decorator to register a fixture by name."""
    _fixtures[fn.__name__] = fn
    return fn


def _resolve_fixtures(fn: Callable) -> dict[str, Any]:
    """Resolve fixtures for a test function's parameters."""
    sig = inspect.signature(fn)
    resolved: dict[str, Any] = {}
    for name in sig.parameters:
        if name not in _fixtures:
            raise RuntimeError(f"No fixture named '{name}' for test '{fn.__name__}'")
        resolved[name] = _fixtures[name]()
    return resolved


# ---------------------------------------------------------------------------
# Registration
# ---------------------------------------------------------------------------

_tests: list[Callable] = []


def test(fn: Callable) -> Callable:
    """Decorator to register a test function."""
    _tests.append(fn)
    return fn


def discover(module: Any) -> list[Callable]:
    """Discover test functions in a module (names starting with 'test_')."""
    found = []
    for name, obj in inspect.getmembers(module, inspect.isfunction):
        if name.startswith("test_") and obj not in _tests:
            found.append(obj)
    return found


# ---------------------------------------------------------------------------
# Result types
# ---------------------------------------------------------------------------

@dataclass
class TestResult:
    name: str
    passed: bool
    duration_ms: float
    error: str = ""


@dataclass
class SuiteResult:
    results: list[TestResult] = field(default_factory=list)

    @property
    def passed(self) -> int:
        return sum(1 for r in self.results if r.passed)

    @property
    def failed(self) -> int:
        return sum(1 for r in self.results if not r.passed)

    @property
    def total(self) -> int:
        return len(self.results)


# ---------------------------------------------------------------------------
# Runner
# ---------------------------------------------------------------------------

def _run_one(fn: Callable) -> TestResult:
    name = fn.__name__
    start = time.perf_counter()
    try:
        kwargs = _resolve_fixtures(fn)
        fn(**kwargs)
        duration = (time.perf_counter() - start) * 1000
        return TestResult(name=name, passed=True, duration_ms=duration)
    except (AssertionFailed, AssertionError) as e:
        duration = (time.perf_counter() - start) * 1000
        return TestResult(name=name, passed=False, duration_ms=duration, error=str(e))
    except Exception as e:
        duration = (time.perf_counter() - start) * 1000
        return TestResult(name=name, passed=False, duration_ms=duration, error=traceback.format_exc().strip())


def run(tests: list[Callable] | None = None) -> SuiteResult:
    """Run tests and print results. Uses registered tests if none provided."""
    suite = tests if tests is not None else list(_tests)
    results = SuiteResult()

    print(f"\nRunning {len(suite)} test(s)...\n")

    for fn in suite:
        result = _run_one(fn)
        results.results.append(result)
        status = "PASS" if result.passed else "FAIL"
        print(f"  [{status}] {result.name}  ({result.duration_ms:.1f}ms)")
        if not result.passed:
            for line in result.error.splitlines():
                print(f"         {line}")

    print(f"\n{'=' * 40}")
    print(f"  {results.passed}/{results.total} passed", end="")
    if results.failed:
        print(f"  |  {results.failed} failed", end="")
    print(f"\n{'=' * 40}\n")

    return results


def main(module: Any | None = None) -> None:
    """Entry point: discover + run all tests. Call with if __name__ == '__main__'."""
    if module is None:
        frame = inspect.stack()[1]
        import importlib.util
        spec = importlib.util.spec_from_file_location("__test_module__", frame.filename)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)

    discovered = discover(module)
    suite = run(list(_tests) + discovered)
    sys.exit(0 if suite.failed == 0 else 1)
```

**Example usage (`test_example.py`):**

```python
import testkit as t

# --- Fixtures ---

@t.fixture
def numbers() -> list[int]:
    return [1, 2, 3]

@t.fixture
def empty_dict() -> dict:
    return {}

# --- Tests via @test decorator ---

@t.test
def test_basic_math() -> None:
    t.assert_eq(1 + 1, 2)
    t.assert_ne(1 + 1, 3)

@t.test
def test_with_fixture(numbers) -> None:
    t.assert_eq(len(numbers), 3)
    t.assert_in(2, numbers)

@t.test
def test_exception() -> None:
    exc = t.assert_raises(ZeroDivisionError, lambda: 1 / 0)
    t.assert_in("division", str(exc))

@t.test
def test_intentional_failure() -> None:
    t.assert_eq("foo", "bar", "strings should match")

# --- Tests via naming convention (discovered automatically) ---

def test_discovered_automatically() -> None:
    t.assert_true(isinstance({}, dict))

def test_fixture_in_discovered(empty_dict) -> None:
    t.assert_eq(len(empty_dict), 0)


if __name__ == "__main__":
    t.main()
```

**Design summary:**

| Feature | How it works |
|---|---|
| **Assertions** | `assert_eq`, `assert_ne`, `assert_true`, `assert_false`, `assert_in`, `assert_raises` — all raise `AssertionFailed` |
| **Fixtures** | `@fixture` decorator registers by name; injected via parameter name matching |
| **Discovery** | `discover(module)` finds any function named `test_*` not already registered |
| **Registration** | `@test` decorator for explicit registration |
| **Runner** | `run()` executes all tests, catches errors, prints `[PASS]`/`[FAIL]` with timing |
| **Entry point** | `main()` auto-discovers from the calling module and exits with code 0/1 |
