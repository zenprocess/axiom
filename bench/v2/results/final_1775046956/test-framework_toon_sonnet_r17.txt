```python
"""minitest — a minimal Python test framework."""
from __future__ import annotations

import inspect
import sys
import time
import traceback
from collections.abc import Callable, Generator
from contextlib import contextmanager
from dataclasses import dataclass, field
from typing import Any


# ---------------------------------------------------------------------------
# Assertions
# ---------------------------------------------------------------------------

class AssertionFailed(Exception):
    pass


def assert_equal(actual: Any, expected: Any, msg: str = "") -> None:
    """Assert that actual == expected."""
    if actual != expected:
        detail = f"{actual!r} != {expected!r}"
        raise AssertionFailed(f"assert_equal failed: {detail}" + (f" — {msg}" if msg else ""))


def assert_not_equal(actual: Any, expected: Any, msg: str = "") -> None:
    if actual == expected:
        raise AssertionFailed(f"assert_not_equal failed: both are {actual!r}" + (f" — {msg}" if msg else ""))


def assert_true(value: Any, msg: str = "") -> None:
    if not value:
        raise AssertionFailed(f"assert_true failed: {value!r} is falsy" + (f" — {msg}" if msg else ""))


def assert_false(value: Any, msg: str = "") -> None:
    if value:
        raise AssertionFailed(f"assert_false failed: {value!r} is truthy" + (f" — {msg}" if msg else ""))


def assert_is_none(value: Any, msg: str = "") -> None:
    if value is not None:
        raise AssertionFailed(f"assert_is_none failed: {value!r} is not None" + (f" — {msg}" if msg else ""))


def assert_is_not_none(value: Any, msg: str = "") -> None:
    if value is None:
        raise AssertionFailed("assert_is_not_none failed: value is None" + (f" — {msg}" if msg else ""))


def assert_in(item: Any, container: Any, msg: str = "") -> None:
    if item not in container:
        raise AssertionFailed(f"assert_in failed: {item!r} not in {container!r}" + (f" — {msg}" if msg else ""))


def assert_not_in(item: Any, container: Any, msg: str = "") -> None:
    if item in container:
        raise AssertionFailed(f"assert_not_in failed: {item!r} in {container!r}" + (f" — {msg}" if msg else ""))


def assert_raises(exc_type: type[Exception], func: Callable, *args: Any, **kwargs: Any) -> Exception:
    """Assert func raises exc_type. Returns the exception."""
    try:
        func(*args, **kwargs)
    except exc_type as e:
        return e
    except Exception as e:
        raise AssertionFailed(
            f"assert_raises: expected {exc_type.__name__}, got {type(e).__name__}: {e}"
        )
    raise AssertionFailed(f"assert_raises: {exc_type.__name__} was not raised")


@contextmanager
def assert_raises_ctx(exc_type: type[Exception]) -> Generator[None, None, None]:
    """Context manager: `with assert_raises_ctx(ValueError): ...`"""
    try:
        yield
    except exc_type:
        return
    except Exception as e:
        raise AssertionFailed(
            f"assert_raises_ctx: expected {exc_type.__name__}, got {type(e).__name__}: {e}"
        )
    raise AssertionFailed(f"assert_raises_ctx: {exc_type.__name__} was not raised")


# ---------------------------------------------------------------------------
# Fixtures
# ---------------------------------------------------------------------------

_FIXTURES: dict[str, Callable] = {}


def fixture(func: Callable) -> Callable:
    """Decorator: register a named fixture. Generator fixtures support teardown."""
    _FIXTURES[func.__name__] = func
    return func


def _resolve_fixtures(func: Callable) -> dict[str, tuple[Any, Any]]:
    sig = inspect.signature(func)
    result: dict[str, tuple[Any, Any]] = {}
    for name in sig.parameters:
        if name in _FIXTURES:
            fn = _FIXTURES[name]
            if inspect.isgeneratorfunction(fn):
                gen = fn()
                result[name] = (next(gen), gen)   # (value, generator for teardown)
            else:
                result[name] = (fn(), None)
    return result


# ---------------------------------------------------------------------------
# Registry & decorator
# ---------------------------------------------------------------------------

@dataclass
class TestCase:
    name: str
    func: Callable
    module: str = ""


_REGISTRY: list[TestCase] = []


def test(func: Callable) -> Callable:
    """Decorator: register a function as a test case."""
    _REGISTRY.append(TestCase(name=func.__name__, func=func, module=func.__module__ or ""))
    return func


# ---------------------------------------------------------------------------
# Results
# ---------------------------------------------------------------------------

@dataclass
class TestResult:
    name: str
    passed: bool
    error: str = ""
    duration_ms: float = 0.0


@dataclass
class RunSummary:
    results: list[TestResult] = field(default_factory=list)

    @property
    def passed(self) -> int:
        return sum(1 for r in self.results if r.passed)

    @property
    def failed(self) -> int:
        return sum(1 for r in self.results if not r.passed)

    @property
    def total(self) -> int:
        return len(self.results)


# ---------------------------------------------------------------------------
# Runner
# ---------------------------------------------------------------------------

def _run_one(case: TestCase) -> TestResult:
    fixture_map = _resolve_fixtures(case.func)
    injected = {k: v for k, (v, _) in fixture_map.items()}
    generators = {k: g for k, (_, g) in fixture_map.items() if g is not None}

    start = time.perf_counter()
    passed, error = True, ""
    try:
        case.func(**injected)
    except AssertionFailed as e:
        passed, error = False, str(e)
    except Exception as e:
        passed, error = False, "".join(traceback.format_exception(type(e), e, e.__traceback__))
    finally:
        for gen in generators.values():
            try:
                next(gen)
            except StopIteration:
                pass

    return TestResult(
        name=case.name,
        passed=passed,
        error=error,
        duration_ms=(time.perf_counter() - start) * 1000,
    )


def run(pattern: str = "", stop_on_fail: bool = False) -> RunSummary:
    """Run all registered tests, optionally filtered by name substring."""
    cases = [c for c in _REGISTRY if not pattern or pattern in c.name]

    if not cases:
        print("No tests found." + (f" (filter: {pattern!r})" if pattern else ""))
        return RunSummary()

    print(f"Running {len(cases)} test(s)...\n")
    summary = RunSummary()

    for case in cases:
        result = _run_one(case)
        summary.results.append(result)

        timing = f"({result.duration_ms:.1f}ms)"
        if result.passed:
            print(f"  [PASS] {case.name} {timing}")
        else:
            print(f"  [FAIL] {case.name} {timing}")
            for line in result.error.strip().splitlines():
                print(f"         {line}")

        if stop_on_fail and not result.passed:
            print("\nStopped on first failure.")
            break

    print(f"\n{'='*50}")
    print(f"Results: {summary.passed}/{summary.total} passed", end="")
    if summary.failed:
        print(f", {summary.failed} failed", end="")
    print(f"\n{'='*50}")
    return summary


# ---------------------------------------------------------------------------
# Discovery
# ---------------------------------------------------------------------------

def discover(paths: list[str] | None = None) -> None:
    """Walk paths, import test_*.py files so @test decorators register."""
    import importlib.util
    import os

    for base in (paths or ["."]):
        files = [base] if os.path.isfile(base) else [
            os.path.join(root, f)
            for root, _, names in os.walk(base)
            for f in names
            if f.startswith("test_") and f.endswith(".py")
        ]
        for fpath in sorted(files):
            spec = importlib.util.spec_from_file_location(fpath, fpath)
            if spec and spec.loader:
                mod = importlib.util.module_from_spec(spec)
                spec.loader.exec_module(mod)  # type: ignore[union-attr]


# ---------------------------------------------------------------------------
# CLI
# ---------------------------------------------------------------------------

def main() -> None:
    import argparse
    parser = argparse.ArgumentParser(description="minitest runner")
    parser.add_argument("paths", nargs="*", default=["."])
    parser.add_argument("-k", "--filter", default="")
    parser.add_argument("-x", "--stop-on-fail", action="store_true")
    args = parser.parse_args()

    discover(args.paths)
    summary = run(pattern=args.filter, stop_on_fail=args.stop_on_fail)
    sys.exit(0 if summary.failed == 0 else 1)


if __name__ == "__main__":
    main()
```

**Usage example** — `test_math.py`:

```python
from minitest import fixture, test, assert_equal, assert_raises_ctx

@fixture
def numbers():
    data = [1, 2, 3, 4, 5]
    yield data          # setup
    data.clear()        # teardown (runs after test)

@test
def test_sum(numbers):
    assert_equal(sum(numbers), 15)

@test
def test_division_by_zero():
    with assert_raises_ctx(ZeroDivisionError):
        _ = 1 / 0

@test
def test_will_fail():
    assert_equal(1 + 1, 3)   # intentional failure
```

**Key design decisions:**

| Feature | How it works |
|---|---|
| **Assertions** | Raise `AssertionFailed` with descriptive messages; `assert_raises` / `assert_raises_ctx` for exceptions |
| **Fixtures** | `@fixture` decorator + parameter name matching; generator fixtures yield the value and run teardown after `yield` |
| **Discovery** | `discover()` walks directories for `test_*.py`, imports them so `@test` decorators self-register |
| **Runner** | `run()` iterates `_REGISTRY`, resolves fixtures, captures errors, prints per-test results + summary |
| **CLI** | `-k` for name filter, `-x` to stop on first failure; exits 1 if any test failed |
