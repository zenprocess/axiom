```bash
#!/usr/bin/env bash
set -euo pipefail

# ---------------------------------------------------------------------------
# deploy.sh — Build, push, tag, and deploy a Docker image with rollback.
#
# Usage:
#   ./deploy.sh <version>              Deploy a new version
#   ./deploy.sh --rollback [version]   Roll back to previous (or given) tag
#
# Required env vars:
#   REGISTRY        Container registry URL  (e.g. ghcr.io/zenprocess)
#   IMAGE_NAME      Image name              (e.g. myapp)
#   DEPLOY_HOST     SSH target              (e.g. deploy@prod.example.com)
#   COMPOSE_FILE    Remote compose path     (e.g. /opt/app/docker-compose.yml)
#
# Optional:
#   HEALTH_URL      Post-deploy health check endpoint
#   HEALTH_TIMEOUT  Seconds to wait (default: 60)
# ---------------------------------------------------------------------------

readonly SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
readonly HEALTH_TIMEOUT="${HEALTH_TIMEOUT:-60}"

log()   { printf '\033[1;34m[deploy]\033[0m %s\n' "$*"; }
err()   { printf '\033[1;31m[deploy]\033[0m %s\n' "$*" >&2; }
abort() { err "$@"; exit 1; }

require_var() {
    local name="$1"
    [[ -n "${!name:-}" ]] || abort "Missing required env var: ${name}"
}

for var in REGISTRY IMAGE_NAME DEPLOY_HOST COMPOSE_FILE; do
    require_var "$var"
done

for cmd in docker git ssh curl; do
    command -v "$cmd" &>/dev/null || abort "Required command not found: $cmd"
done

FULL_IMAGE="${REGISTRY}/${IMAGE_NAME}"

# ── parse args -------------------------------------------------------------

ROLLBACK=false
VERSION=""

case "${1:-}" in
    --rollback) ROLLBACK=true; VERSION="${2:-}" ;;
    -h|--help|"") sed -n '/^# Usage:/,/^# --/{ /^# --/d; s/^# \?//; p; }' "$0"; exit 1 ;;
    *) VERSION="$1" ;;
esac

# ── health check -----------------------------------------------------------

health_check() {
    [[ -n "${HEALTH_URL:-}" ]] || return 0
    log "Health check: ${HEALTH_URL} (timeout: ${HEALTH_TIMEOUT}s)"
    local deadline=$((SECONDS + HEALTH_TIMEOUT))
    while (( SECONDS < deadline )); do
        curl -sf --max-time 5 "${HEALTH_URL}" >/dev/null 2>&1 && { log "Healthy"; return 0; }
        sleep 2
    done
    err "Health check failed after ${HEALTH_TIMEOUT}s"
    return 1
}

# ── rollback ---------------------------------------------------------------

do_rollback() {
    local target="$1"
    if [[ -z "$target" ]]; then
        target="$(git tag --sort=-version:refname | sed -n '2p')"
        [[ -n "$target" ]] || abort "No previous tag found to roll back to"
    fi
    log "Rolling back to ${target}"
    ssh "${DEPLOY_HOST}" \
        "IMAGE=${FULL_IMAGE}:${target} docker compose -f ${COMPOSE_FILE} up -d --pull always"
    health_check
    log "Rollback to ${target} complete"
}

if "$ROLLBACK"; then
    do_rollback "$VERSION"
    exit 0
fi

# ── deploy -----------------------------------------------------------------

[[ -n "$(git status --porcelain)" ]] && abort "Working tree is dirty — commit or stash first"

PREVIOUS_TAG="$(git tag --sort=-version:refname | head -n1)"
TAG="v${VERSION}"

log "Deploying ${FULL_IMAGE}:${TAG}"

# 1) Build
log "Building image …"
docker build \
    --build-arg "VERSION=${VERSION}" \
    --tag "${FULL_IMAGE}:${TAG}" \
    --tag "${FULL_IMAGE}:latest" \
    "${SCRIPT_DIR}"

# 2) Push
log "Pushing to registry …"
docker push "${FULL_IMAGE}:${TAG}"
docker push "${FULL_IMAGE}:latest"

# 3) Git tag
log "Tagging ${TAG} …"
git tag -a "${TAG}" -m "Release ${TAG}"
git push origin "${TAG}"

# 4) Deploy via SSH + docker compose
log "Deploying on ${DEPLOY_HOST} …"
ssh "${DEPLOY_HOST}" \
    "IMAGE=${FULL_IMAGE}:${TAG} docker compose -f ${COMPOSE_FILE} up -d --pull always"

# 5) Health check → auto-rollback on failure
if ! health_check; then
    err "Deploy failed health check — auto-rolling back"
    if [[ -n "${PREVIOUS_TAG:-}" ]]; then
        do_rollback "${PREVIOUS_TAG}"
        abort "Rolled back to ${PREVIOUS_TAG}. Investigate before retrying."
    else
        abort "No previous tag to roll back to. Manual intervention required."
    fi
fi

log "Successfully deployed ${TAG}"
```

**Key features:**

- **`./deploy.sh 1.2.0`** — builds, pushes `v1.2.0` to registry, creates a git tag, deploys via SSH + docker compose
- **`./deploy.sh --rollback`** — rolls back to the previous git version tag
- **`./deploy.sh --rollback v1.1.0`** — rolls back to a specific version
- **Auto-rollback** — if the health check fails after deploy, it automatically reverts to the previous tag
- **Safety checks** — requires clean working tree, validates all env vars and commands upfront, uses `set -euo pipefail`

Configure it via env vars (or wrap in a `.env` file):
```bash
export REGISTRY=ghcr.io/myorg
export IMAGE_NAME=myapp
export DEPLOY_HOST=deploy@prod.example.com
export COMPOSE_FILE=/opt/myapp/docker-compose.yml
export HEALTH_URL=https://myapp.example.com/health
```
