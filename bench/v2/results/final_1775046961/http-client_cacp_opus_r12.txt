```python
"""Client for fetching user data from the REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class User(BaseModel):
    """User data returned by the API."""

    id: int
    name: str
    email: str
    username: str


async def fetch_user(client: httpx.AsyncClient, user_id: int) -> User | None:
    """Fetch a single user by ID from the REST API."""
    url = f"https://api.example.com/users/{user_id}"
    try:
        response = await client.get(url)
        response.raise_for_status()
    except httpx.HTTPStatusError as exc:
        log.error("api_http_error", user_id=user_id, status=exc.response.status_code)
        return None
    except httpx.RequestError as exc:
        log.error("api_request_error", user_id=user_id, error=str(exc))
        return None

    return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    async with httpx.AsyncClient() as client:
        user = await fetch_user(client, user_id=42)
        if user:
            log.info("user_fetched", user=user.model_dump())
        else:
            log.warning("user_not_found", user_id=42)


if __name__ == "__main__":
    asyncio.run(main())
```

Key compliance points:
- **async I/O** — `fetch_user` and `main` are both `async def`
- **Pydantic** — `User` model instead of raw dicts
- **structlog** — all output via `log.info`/`log.error`/`log.warning`, no `print()`
- **Type hints** — all function signatures fully annotated
- **Error handling** — `HTTPStatusError` and `RequestError` caught separately
- **Docstrings** — all public functions documented
- **No star imports, no secrets** — clean imports, URL is a public endpoint pattern
