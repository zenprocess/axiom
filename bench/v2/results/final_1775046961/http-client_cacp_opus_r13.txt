"""Fetch user data from a REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class Address(BaseModel):
    """User address data."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User profile returned by the API."""

    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Async client for the users REST API."""

    def __init__(self, base_url: str = "https://api.example.com") -> None:
        self._base_url = base_url

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self._base_url}/users/{user_id}"
        async with httpx.AsyncClient() as client:
            try:
                response = await client.get(url)
                response.raise_for_status()
            except httpx.HTTPStatusError as exc:
                await log.aerror(
                    "api_http_error",
                    status_code=exc.response.status_code,
                    user_id=user_id,
                )
                raise
            except httpx.RequestError as exc:
                await log.aerror(
                    "api_request_error",
                    url=url,
                    detail=str(exc),
                )
                raise

        return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        await log.ainfo("user_fetched", user=user.model_dump())
    except (httpx.HTTPStatusError, httpx.RequestError):
        await log.aerror("failed_to_fetch_user", user_id=42)


if __name__ == "__main__":
    asyncio.run(main())
