```python
"""Fetch user data from a REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class Address(BaseModel):
    """A user's address."""

    street: str
    city: str
    zipcode: str


class User(BaseModel):
    """User data returned by the API."""

    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Client for fetching user data from the REST API."""

    BASE_URL = "https://api.example.com"

    def __init__(self, base_url: str | None = None) -> None:
        self.base_url = base_url or self.BASE_URL

    async def get_user(self, user_id: int) -> User:
        """Fetch a user by ID.

        Args:
            user_id: The ID of the user to fetch.

        Returns:
            The user data.

        Raises:
            httpx.HTTPStatusError: If the API returns a non-2xx status.
            httpx.RequestError: If the request fails (network error, timeout).
        """
        url = f"{self.base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=10.0)
                response.raise_for_status()
        except httpx.HTTPStatusError:
            log.error("api_error", user_id=user_id, status=response.status_code)
            raise
        except httpx.RequestError as exc:
            log.error("request_failed", user_id=user_id, error=str(exc))
            raise

        data = response.json()
        user = User.model_validate(data)
        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.get_user(42)
    log.info("result", user=user.model_dump())


if __name__ == "__main__":
    asyncio.run(main())
```

- **async I/O** — `get_user` and `main` are `async def`
- **structlog only** — no `print()` or `logging` imports
- **Pydantic models** — `User` and `Address` use `BaseModel`
- **Type hints** — all function signatures annotated
- **Error handling** — API call wrapped in `try/except` for both HTTP and network errors
- **Docstrings** — all public classes and functions documented
- **No star imports, no hardcoded secrets**
