```python
"""Client for fetching user data from the example REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class User(BaseModel):
    """Represents a user returned by the API."""

    id: int
    name: str
    email: str
    username: str


class UserClient:
    """Async client for the users API."""

    def __init__(self, base_url: str = "https://api.example.com") -> None:
        self.base_url = base_url

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self.base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url)
                response.raise_for_status()
        except httpx.HTTPStatusError as exc:
            log.error("api_http_error", user_id=user_id, status=exc.response.status_code)
            raise
        except httpx.RequestError as exc:
            log.error("api_request_error", user_id=user_id, url=url, detail=str(exc))
            raise

        user = User.model_validate(response.json())
        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.get_user(42)
    log.info("result", user=user.model_dump())


if __name__ == "__main__":
    asyncio.run(main())
```
