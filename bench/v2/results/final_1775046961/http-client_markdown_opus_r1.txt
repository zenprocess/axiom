```python
"""Client for fetching user data from the REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class User(BaseModel):
    """A user returned by the API."""

    id: int
    name: str
    email: str


async def fetch_user(client: httpx.AsyncClient, user_id: int) -> User:
    """Fetch a single user by ID.

    Args:
        client: An async HTTP client.
        user_id: The ID of the user to fetch.

    Returns:
        The requested User.

    Raises:
        httpx.HTTPStatusError: If the API returns a non-2xx status.
        httpx.RequestError: If a connection or timeout error occurs.
    """
    url = f"https://api.example.com/users/{user_id}"
    try:
        resp = await client.get(url, timeout=30)
        resp.raise_for_status()
    except httpx.RequestError as exc:
        log.error("request_failed", url=url, error=str(exc))
        raise
    except httpx.HTTPStatusError as exc:
        log.error("bad_status", url=url, status=exc.response.status_code)
        raise
    return User.model_validate(resp.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    async with httpx.AsyncClient() as client:
        user = await fetch_user(client, 42)
        log.info("user_fetched", user_id=user.id, name=user.name, email=user.email)


if __name__ == "__main__":
    asyncio.run(main())
```

Key design choices per project rules:

- **`async def`** — all I/O uses `httpx.AsyncClient` (async-io rule)
- **Pydantic `BaseModel`** — `User` model with typed fields (pydantic-models rule)
- **`structlog`** — structured logging, no `print()` or stdlib `logging` (structlog-only, no-print-debug rules)
- **Error handling** — `try/except` around the HTTP call catching `RequestError` and `HTTPStatusError` separately (error-handling rule)
- **Full type hints** on all functions (type-hints rule)
- **Docstrings** on all public functions and classes (docstrings rule)
