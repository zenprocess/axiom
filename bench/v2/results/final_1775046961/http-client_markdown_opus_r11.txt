```python
"""Async client for fetching user data from a REST API."""

import asyncio
import os

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()

API_BASE_URL = os.environ.get("API_BASE_URL", "https://api.example.com")


class Address(BaseModel):
    """A user's mailing address."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User profile returned by the API."""

    id: int
    name: str
    email: str
    address: Address | None = None


class UserFetchError(Exception):
    """Raised when a user cannot be fetched from the API."""

    def __init__(self, user_id: int, reason: str) -> None:
        self.user_id = user_id
        self.reason = reason
        super().__init__(f"Failed to fetch user {user_id}: {reason}")


class UserClient:
    """Async HTTP client for the /users endpoint."""

    def __init__(self, base_url: str = API_BASE_URL, timeout: float = 30.0) -> None:
        self._base_url = base_url
        self._timeout = timeout

    async def fetch_user(self, user_id: int) -> User:
        """Fetch a single user by ID.

        Args:
            user_id: The numeric user ID to look up.

        Returns:
            A validated User model.

        Raises:
            UserFetchError: If the request fails or returns a non-200 status.
        """
        url = f"{self._base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
                response.raise_for_status()
        except httpx.TimeoutException:
            log.error("user_fetch_timeout", user_id=user_id, url=url)
            raise UserFetchError(user_id, "request timed out")
        except httpx.HTTPStatusError as exc:
            log.error(
                "user_fetch_http_error",
                user_id=user_id,
                status=exc.response.status_code,
            )
            raise UserFetchError(user_id, f"HTTP {exc.response.status_code}")
        except httpx.RequestError as exc:
            log.error("user_fetch_request_error", user_id=user_id, error=str(exc))
            raise UserFetchError(user_id, str(exc))

        return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.fetch_user(42)
        log.info("user_fetched", user_id=user.id, name=user.name, email=user.email)
    except UserFetchError:
        log.error("failed_to_fetch_user_42")


if __name__ == "__main__":
    asyncio.run(main())
```
