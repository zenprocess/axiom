"""Fetch user data from a REST API with async I/O, structured logging, and validation."""

import asyncio
import os
import sys

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()

API_BASE_URL = os.environ.get("API_BASE_URL", "https://api.example.com")


class Address(BaseModel):
    """A user's mailing address."""
    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User profile returned by the API."""
    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Async client for the users REST API."""

    def __init__(self, base_url: str = API_BASE_URL, timeout: float = 30.0) -> None:
        self._base_url = base_url.rstrip("/")
        self._timeout = timeout

    async def fetch_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self._base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)
                response.raise_for_status()
        except httpx.HTTPStatusError as exc:
            log.error("api_http_error", url=url, status_code=exc.response.status_code)
            raise
        except httpx.RequestError as exc:
            log.error("api_request_failed", url=url, error=str(exc))
            raise

        user = User.model_validate(response.json())
        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and display the result."""
    client = UserClient()
    try:
        user = await client.fetch_user(42)
        log.info("result", user=user.model_dump())
    except (httpx.HTTPStatusError, httpx.RequestError):
        sys.exit(1)


if __name__ == "__main__":
    asyncio.run(main())
