```python
"""Client for fetching user data from a REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class Address(BaseModel):
    """User's address."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User data returned by the API."""

    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Async client for the users REST API."""

    def __init__(self, base_url: str = "https://api.example.com") -> None:
        """Initialize the client with a base URL."""
        self.base_url = base_url

    async def fetch_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self.base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url)
                response.raise_for_status()
        except httpx.HTTPStatusError as exc:
            log.error("api_error", status=exc.response.status_code, user_id=user_id)
            raise
        except httpx.RequestError as exc:
            log.error("request_failed", url=url, error=str(exc))
            raise

        user = User.model_validate(response.json())
        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.fetch_user(42)
    log.info("result", user=user.model_dump())


if __name__ == "__main__":
    asyncio.run(main())
```

Key design choices:

- **Pydantic models** (`User`, `Address`) for response validation
- **Async I/O** via `httpx.AsyncClient` for the HTTP call
- **structlog** for all logging (no `print` or stdlib `logging`)
- **Error handling** catches `HTTPStatusError` (4xx/5xx) and `RequestError` (connection failures) separately, logs context, and re-raises
- **Type hints** on all signatures
