```python
"""Client for fetching user data from the REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

logger = structlog.get_logger()

BASE_URL = "https://api.example.com"


class User(BaseModel):
    """User data returned by the API."""

    id: int
    name: str
    email: str
    username: str


class UserClient:
    """Async client for the users API."""

    def __init__(self, base_url: str = BASE_URL) -> None:
        self.base_url = base_url

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self.base_url}/users/{user_id}"
        async with httpx.AsyncClient() as client:
            try:
                response = await client.get(url)
                response.raise_for_status()
            except httpx.HTTPStatusError as exc:
                logger.error(
                    "api_error",
                    status_code=exc.response.status_code,
                    user_id=user_id,
                )
                raise
            except httpx.RequestError as exc:
                logger.error("request_failed", url=url, detail=str(exc))
                raise
        return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.get_user(42)
    logger.info("user_fetched", user=user.model_dump())


if __name__ == "__main__":
    asyncio.run(main())
```

Key points:
- **`User` Pydantic model** validates the API response
- **`get_user` is async** since it performs I/O
- **Error handling** catches HTTP errors (4xx/5xx) and connection-level failures separately
- **structlog** for structured logging (no `print()` or stdlib `logging`)
- **`main()`** fetches user #42 and logs the result
