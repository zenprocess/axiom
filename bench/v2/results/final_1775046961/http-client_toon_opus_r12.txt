"""Client for fetching user data from the example REST API."""

import asyncio

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """User's address."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User data returned by the API."""

    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Async client for the /users endpoint."""

    def __init__(self, base_url: str = BASE_URL) -> None:
        """Initialize with the API base URL."""
        self._base_url = base_url

    async def fetch_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self._base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url)
                response.raise_for_status()
        except httpx.HTTPStatusError as exc:
            log.error("api_http_error", user_id=user_id, status=exc.response.status_code)
            raise
        except httpx.RequestError as exc:
            log.error("api_request_error", user_id=user_id, url=url, detail=str(exc))
            raise

        return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.fetch_user(42)
    log.info("user_fetched", user=user.model_dump())


if __name__ == "__main__":
    asyncio.run(main())
