```python
"""Async client for fetching user data from a REST API."""

import asyncio
import sys

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """User's address."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User data returned by the API."""

    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Async client for the /users endpoint."""

    def __init__(self, base_url: str = BASE_URL) -> None:
        self.base_url = base_url

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self.base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=10.0)
                response.raise_for_status()
        except httpx.TimeoutException:
            await log.aerror("request_timeout", url=url)
            raise
        except httpx.HTTPStatusError as exc:
            await log.aerror(
                "http_error",
                url=url,
                status_code=exc.response.status_code,
            )
            raise
        except httpx.RequestError as exc:
            await log.aerror("request_failed", url=url, error=str(exc))
            raise

        return User.model_validate(response.json())


async def main() -> int:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
    except httpx.HTTPError:
        return 1

    await log.ainfo("user_fetched", user_id=user.id, name=user.name, email=user.email)
    return 0


if __name__ == "__main__":
    sys.exit(asyncio.run(main()))
```

Key design choices:

- **Pydantic models** (`User`, `Address`) for validated response data
- **`async def`** for all I/O (`get_user`, `main`)
- **`structlog`** for structured logging (no `print()` or `import logging`)
- **Type hints** on all signatures
- **`try/except`** around the `httpx` call catching timeout, HTTP status errors, and general request errors separately
- **`main()`** returns an exit code so errors propagate to the shell
