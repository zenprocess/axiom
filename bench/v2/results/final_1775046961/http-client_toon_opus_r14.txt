```python
"""Client for fetching user data from the REST API."""

import asyncio
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel

logger = structlog.get_logger()

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """User address."""
    street: str
    city: str
    state: str
    zipcode: str


class User(BaseModel):
    """User data returned by the API."""
    id: int
    name: str
    email: str
    phone: Optional[str] = None
    address: Optional[Address] = None


class UserClientError(Exception):
    """Raised when a user API request fails."""


class UserClient:
    """Async client for the users API."""

    def __init__(self, base_url: str = BASE_URL) -> None:
        self.base_url = base_url

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self.base_url}/users/{user_id}"
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url)
                response.raise_for_status()
        except httpx.HTTPStatusError as exc:
            logger.error("api_http_error", user_id=user_id, status=exc.response.status_code)
            raise UserClientError(
                f"HTTP {exc.response.status_code} for user {user_id}"
            ) from exc
        except httpx.RequestError as exc:
            logger.error("api_request_error", user_id=user_id, error=str(exc))
            raise UserClientError(f"Request failed for user {user_id}") from exc

        return User.model_validate(response.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        logger.info("user_fetched", user_id=user.id, name=user.name, email=user.email)
    except UserClientError:
        logger.error("failed_to_fetch_user", user_id=42)


if __name__ == "__main__":
    asyncio.run(main())
```

Key design choices:

- **Async I/O** with `httpx.AsyncClient` for non-blocking HTTP
- **Pydantic models** (`User`, `Address`) for validated response parsing
- **structlog** for structured logging (no `print` or stdlib `logging`)
- **Error handling** — catches both HTTP errors (4xx/5xx) and connection-level failures, wraps them in `UserClientError`
- **Type hints** on all signatures
