```python
"""Client for fetching user data from a REST API."""

from __future__ import annotations

import asyncio

import httpx
import structlog
from pydantic import BaseModel

logger = structlog.get_logger()

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    """User address."""

    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    """User returned by the API."""

    id: int
    name: str
    email: str
    address: Address | None = None


class UserClient:
    """Async client for the /users endpoint."""

    def __init__(self, base_url: str = BASE_URL) -> None:
        self.base_url = base_url

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self.base_url}/users/{user_id}"
        async with httpx.AsyncClient() as client:
            try:
                resp = await client.get(url)
                resp.raise_for_status()
            except httpx.HTTPStatusError as exc:
                await logger.aerror(
                    "api_error",
                    status=exc.response.status_code,
                    url=url,
                )
                raise
            except httpx.RequestError as exc:
                await logger.aerror("request_failed", url=url, detail=str(exc))
                raise

        return User.model_validate(resp.json())


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    user = await client.get_user(42)
    await logger.ainfo("fetched_user", user=user.model_dump())


if __name__ == "__main__":
    asyncio.run(main())
```

Key design choices:

- **`async def`** for all I/O (`get_user`, `main`) per project rules
- **Pydantic `BaseModel`** for `User` and `Address` instead of raw dicts
- **`structlog`** for logging (no `print()` or `import logging`)
- **`try/except`** around the `httpx` call catching both `HTTPStatusError` (4xx/5xx) and `RequestError` (network failures)
- **Full type hints** on all signatures
- **Docstrings** on all public classes/functions
