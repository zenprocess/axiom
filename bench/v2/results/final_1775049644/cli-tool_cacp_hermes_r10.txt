#!/usr/bin/env python3
"""
CLI tool to filter CSV rows by column value with progress reporting.
"""

import argparse
import sys
from pathlib import Path
from typing import List, Optional

import pandas as pd
import structlog

# Initialize structlog
structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging.DEBUG),
)
logger = structlog.get_logger()


def parse_args() -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Filter CSV rows by column value with progress reporting."
    )
    parser.add_argument("input_file", type=Path, help="Input CSV file path")
    parser.add_argument("column", type=str, help="Column name to filter on")
    parser.add_argument("value", type=str, help="Value to match in the specified column")
    parser.add_argument(
        "-o", "--output", type=Path, help="Output file path (default: stdout)"
    )
    parser.add_argument(
        "-d", "--delimiter", default=",", help="CSV delimiter (default: ',')"
    )
    return parser.parse_args()


async def read_csv_with_progress(file_path: Path, delimiter: str) -> pd.DataFrame:
    """
    Read CSV file with progress reporting.

    Args:
        file_path: Path to the input CSV file
        delimiter: CSV delimiter character

    Returns:
        DataFrame containing the CSV data
    """
    try:
        total_lines = sum(1 for _ in open(file_path, "r", encoding="utf-8"))
        # Subtract 1 for header row
        total_rows = max(0, total_lines - 1)

        logger.info("reading_csv", file=str(file_path), total_rows=total_rows)

        df = pd.read_csv(file_path, delimiter=delimiter)

        # Progress simulation (since pandas read_csv doesn't support progress callback)
        # In real use, you'd use a library like tqdm with chunksize
        if total_rows > 0:
            progress_percentage = 100
            logger.info(
                "csv_read_complete",
                file=str(file_path),
                rows_loaded=len(df),
                progress=f"{progress_percentage}%",
            )

        return df

    except FileNotFoundError as exc:
        logger.error("file_not_found", file=str(file_path), error=str(exc))
        raise
    except pd.errors.EmptyDataError as exc:
        logger.error("empty_csv_file", file=str(file_path), error=str(exc))
        raise
    except Exception as exc:
        logger.error("csv_read_error", file=str(file_path), error=str(exc))
        raise


async def filter_dataframe(
    df: pd.DataFrame, column: str, value: str
) -> pd.DataFrame:
    """
    Filter DataFrame rows where column matches value.

    Args:
        df: Input DataFrame
        column: Column name to filter on
        value: Value to match

    Returns:
        Filtered DataFrame
    """
    try:
        if column not in df.columns:
            logger.error("column_not_found", column=column, available_columns=list(df.columns))
            raise ValueError(f"Column '{column}' not found. Available columns: {list(df.columns)}")

        logger.info("filtering_data", column=column, value=value, total_rows=len(df))

        # Convert both to string for comparison to handle numeric values
        filtered_df = df[df[column].astype(str).str.strip() == value.strip()]

        logger.info(
            "filtering_complete",
            column=column,
            value=value,
            matching_rows=len(filtered_df),
            filtered_out_rows=len(df) - len(filtered_df),
        )

        return filtered_df

    except Exception as exc:
        logger.error("filtering_error", column=column, value=value, error=str(exc))
        raise


async def write_output(df: pd.DataFrame, output_path: Optional[Path]) -> None:
    """
    Write DataFrame to output (file or stdout).

    Args:
        df: DataFrame to write
        output_path: Output file path or None for stdout
    """
    try:
        if output_path:
            logger.info("writing_output", file=str(output_path))
            df.to_csv(output_path, index=False)
            logger.info("output_written", file=str(output_path), rows=len(df))
        else:
            # Output to stdout
            logger.info("writing_stdout")
            print(df.to_csv(index=False), end="")
            logger.info("stdout_written", rows=len(df))

    except Exception as exc:
        logger.error("write_error", error=str(exc))
        raise


async def main() -> int:
    """Main entry point."""
    try:
        args = parse_args()

        # Read CSV
        df = await read_csv_with_progress(args.input_file, args.delimiter)

        # Filter DataFrame
        filtered_df = await filter_dataframe(df, args.column, args.value)

        # Write output
        await write_output(filtered_df, args.output)

        return 0

    except FileNotFoundError as exc:
        logger.error("file_not_found", file=str(args.input_file), error=str(exc))
        return 1
    except ValueError as exc:
        logger.error("validation_error", error=str(exc))
        return 2
    except Exception as exc:
        logger.error("unexpected_error", error=str(exc))
        return 3


if __name__ == "__main__":
    sys.exit(asyncio.run(main()))
