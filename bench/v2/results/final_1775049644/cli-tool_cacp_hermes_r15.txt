```python
#!/usr/bin/env python3
"""
CSV Filter CLI Tool

Reads a CSV file, filters rows by a specified column value, and outputs the results.
"""

import argparse
import sys
from pathlib import Path
from typing import List, Optional

import pandas as pd
import structlog

# Initialize structlog
structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
)
logger = structlog.get_logger()


class FilterConfig:
    """Configuration model for CSV filter operation."""

    def __init__(
        self,
        input_path: Path,
        output_path: Optional[Path],
        column_name: str,
        column_value: str,
    ) -> None:
        self.input_path = input_path
        self.output_path = output_path
        self.column_name = column_name
        self.column_value = column_value


def parse_args() -> FilterConfig:
    """Parse command-line arguments and return configuration."""
    parser = argparse.ArgumentParser(
        description="Filter CSV rows by column value."
    )
    parser.add_argument(
        "input",
        type=Path,
        help="Input CSV file path",
    )
    parser.add_argument(
        "column_name",
        type=str,
        help="Name of the column to filter on",
    )
    parser.add_argument(
        "column_value",
        type=str,
        help="Value to match in the specified column",
    )
    parser.add_argument(
        "-o",
        "--output",
        type=Path,
        default=None,
        help="Output CSV file path (default: stdout)",
    )
    parser.add_argument(
        "-v",
        "--verbose",
        action="store_true",
        help="Enable verbose progress output",
    )
    args = parser.parse_args()

    return FilterConfig(
        input_path=args.input,
        output_path=args.output,
        column_name=args.column_name,
        column_value=args.column_value,
    )


async def read_csv_with_progress(input_path: Path) -> pd.DataFrame:
    """Read CSV file with progress indication."""
    try:
        total_rows = sum(1 for _ in open(input_path, "r", encoding="utf-8")) - 1
        if total_rows < 0:
            total_rows = 0
    except OSError as exc:
        logger.error("failed_to_count_rows", path=input_path, error=str(exc))
        raise

    logger.info("reading_csv", path=input_path, total_rows=total_rows)

    # Read in chunks to simulate progress
    chunk_size = 1000
    chunks: List[pd.DataFrame] = []
    processed_rows = 0

    try:
        for chunk in pd.read_csv(
            input_path, chunksize=chunk_size, encoding="utf-8"
        ):
            chunks.append(chunk)
            processed_rows += len(chunk)
            if processed_rows % 1000 == 0:
                logger.info(
                    "progress", rows_read=processed_rows, total_rows=total_rows
                )
        logger.info("reading_complete", rows_read=processed_rows)
    except pd.errors.EmptyDataError:
        logger.error("empty_csv", path=input_path)
        raise
    except pd.errors.ParserError as exc:
        logger.error("csv_parse_error", path=input_path, error=str(exc))
        raise

    return pd.concat(chunks, ignore_index=True)


async def filter_csv(
    df: pd.DataFrame, column_name: str, column_value: str
) -> pd.DataFrame:
    """Filter DataFrame rows by column value."""
    if column_name not in df.columns:
        logger.error(
            "column_not_found",
            column=column_name,
            available_columns=list(df.columns),
        )
        raise KeyError(f"Column '{column_name}' not found in CSV")

    logger.info(
        "filtering",
        column=column_name,
        value=column_value,
        total_rows=len(df),
    )

    filtered_df = df[df[column_name].astype(str) == column_value]

    logger.info(
        "filtering_complete",
        matched_rows=len(filtered_df),
        total_rows=len(df),
    )

    return filtered_df


async def write_csv(
    df: pd.DataFrame, output_path: Optional[Path]
) -> None:
    """Write DataFrame to CSV or stdout."""
    try:
        if output_path is None:
            sys.stdout.write(df.to_csv(index=False))
        else:
            df.to_csv(output_path, index=False)
            logger.info("output_written", path=output_path)
    except OSError as exc:
        logger.error("write_error", path=output_path, error=str(exc))
        raise


async def main() -> int:
    """Main entry point."""
    try:
        config = parse_args()
    except SystemExit as exc:
        return exc.code or 1

    try:
        # Read CSV with progress
        df = await read_csv_with_progress(config.input_path)

        # Filter rows
        filtered_df = await filter_csv(
            df, config.column_name, config.column_value
        )

        # Write output
        await write_csv(filtered_df, config.output_path)

        return 0
    except (KeyError, pd.errors.EmptyDataError, pd.errors.ParserError) as exc:
        logger.error("operation_failed", error=str(exc))
        return 1
    except OSError as exc:
        logger.error("io_error", error=str(exc))
        return 2
    except Exception as exc:
        logger.error("unexpected_error", error=str(exc))
        return 3


if __name__ == "__main__":
    import asyncio
    import logging  # required for structlog filtering

    sys.exit(asyncio.run(main()))
```
