Here's a Python CLI tool that meets all your requirements:

```python
#!/usr/bin/env python3
"""CLI tool to filter CSV rows by column value with progress output."""

import argparse
import sys
from pathlib import Path
from typing import Optional

import pandas as pd
import structlog

# Configure structlog for structured logging
structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging.WARNING),
)
logger = structlog.get_logger()


def parse_args() -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Filter rows from a CSV file based on a column value."
    )
    parser.add_argument("input_file", type=Path, help="Path to input CSV file")
    parser.add_argument("output_file", type=Path, help="Path to output CSV file")
    parser.add_argument(
        "--column", "-c", required=True, help="Column name to filter on"
    )
    parser.add_argument(
        "--value", "-v", required=True, help="Value to match in the specified column"
    )
    parser.add_argument(
        "--case-sensitive",
        action="store_true",
        help="Perform case-sensitive matching (default: case-insensitive)",
    )
    return parser.parse_args()


async def process_csv(
    input_path: Path, output_path: Path, column: str, value: str, case_sensitive: bool
) -> None:
    """Process CSV file with progress output and write filtered results."""
    try:
        if not input_path.exists():
            logger.error("input_file_not_found", path=str(input_path))
            sys.exit(1)

        logger.info("reading_csv", path=str(input_path))
        df = pd.read_csv(input_path)

        if column not in df.columns:
            logger.error("column_not_found", column=column, available_columns=list(df.columns))
            sys.exit(1)

        total_rows = len(df)
        logger.info("filtering_rows", column=column, value=value, total_rows=total_rows)

        # Prepare filter condition
        if case_sensitive:
            mask = df[column] == value
        else:
            mask = df[column].astype(str).str.lower() == str(value).lower()

        filtered_df = df[mask]
        filtered_count = len(filtered_df)

        logger.info(
            "filtering_complete",
            matched_rows=filtered_count,
            total_rows=total_rows,
            match_rate=f"{(filtered_count / total_rows * 100):.2f}%" if total_rows > 0 else "N/A",
        )

        logger.info("writing_output", path=str(output_path))
        filtered_df.to_csv(output_path, index=False)

        logger.info("processing_complete", output_rows=filtered_count)

    except pd.errors.EmptyDataError:
        logger.error("empty_csv_file", path=str(input_path))
        sys.exit(1)
    except pd.errors.ParserError as e:
        logger.error("csv_parsing_error", error=str(e))
        sys.exit(1)
    except Exception as e:
        logger.error("unexpected_error", error=str(e))
        sys.exit(1)


def main() -> None:
    """Main entry point."""
    args = parse_args()

    # Run async function in main event loop
    import asyncio

    asyncio.run(
        process_csv(
            input_path=args.input_file,
            output_path=args.output_file,
            column=args.column,
            value=args.value,
            case_sensitive=args.case_sensitive,
        )
    )


if __name__ == "__main__":
    main()
```
