```python
#!/usr/bin/env python3
"""
CLI tool to filter CSV rows by a column value.

This module provides functionality to read a CSV file, filter rows based on
a specified column's value, and output the filtered results.
"""

import argparse
import csv
import sys
from pathlib import Path
from typing import List, Optional

import structlog
from pydantic import BaseModel, Field


structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
)

logger = structlog.get_logger()


class FilterOptions(BaseModel):
    """Configuration for CSV filtering operation."""
    input_file: Path = Field(..., description="Path to input CSV file")
    output_file: Optional[Path] = Field(None, description="Path to output CSV file")
    column_name: str = Field(..., description="Column name to filter on")
    column_value: str = Field(..., description="Value to match in the specified column")
    delimiter: str = Field(",", description="CSV delimiter character")


def parse_arguments() -> FilterOptions:
    """Parse command-line arguments into FilterOptions model."""
    parser = argparse.ArgumentParser(
        description="Filter CSV rows by a column value"
    )
    parser.add_argument(
        "input_file",
        type=Path,
        help="Path to input CSV file"
    )
    parser.add_argument(
        "-o", "--output",
        type=Path,
        help="Path to output CSV file (default: stdout)"
    )
    parser.add_argument(
        "-c", "--column",
        required=True,
        help="Column name to filter on"
    )
    parser.add_argument(
        "-v", "--value",
        required=True,
        help="Value to match in the specified column"
    )
    parser.add_argument(
        "-d", "--delimiter",
        default=",",
        help="CSV delimiter character (default: ,)"
    )

    args = parser.parse_args()
    
    try:
        return FilterOptions(
            input_file=args.input_file,
            output_file=args.output,
            column_name=args.column,
            column_value=args.value,
            delimiter=args.delimiter
        )
    except ValueError as e:
        logger.error("invalid_argument", error=str(e))
        sys.exit(1)


def read_csv_with_progress(file_path: Path, delimiter: str) -> List[dict]:
    """Read CSV file with progress logging.
    
    Args:
        file_path: Path to the CSV file
        delimiter: CSV delimiter character
        
    Returns:
        List of dictionaries representing CSV rows
    """
    try:
        with file_path.open(newline="") as csvfile:
            reader = csv.DictReader(csvfile, delimiter=delimiter)
            rows = list(reader)
            logger.info("csv_read_complete", row_count=len(rows))
            return rows
    except FileNotFoundError:
        logger.error("file_not_found", path=str(file_path))
        sys.exit(1)
    except PermissionError:
        logger.error("permission_denied", path=str(file_path))
        sys.exit(1)
    except csv.Error as e:
        logger.error("csv_parse_error", path=str(file_path), error=str(e))
        sys.exit(1)


def filter_rows(rows: List[dict], column_name: str, column_value: str) -> List[dict]:
    """Filter rows by column value.
    
    Args:
        rows: List of CSV row dictionaries
        column_name: Name of column to filter on
        column_value: Value to match
        
    Returns:
        Filtered list of rows
    """
    filtered = []
    for i, row in enumerate(rows):
        if row.get(column_name) == column_value:
            filtered.append(row)
            
        # Log progress every 1000 rows
        if (i + 1) % 1000 == 0:
            logger.info("filter_progress", processed=i + 1, filtered=len(filtered))
    
    logger.info("filter_complete", total=len(rows), filtered=len(filtered))
    return filtered


def write_csv_output(rows: List[dict], output_path: Optional[Path]) -> None:
    """Write filtered rows to output.
    
    Args:
        rows: List of filtered row dictionaries
        output_path: Path to output file or None for stdout
    """
    if not rows:
        logger.warning("no_rows_to_write")
        return
    
    fieldnames = list(rows[0].keys())
    
    try:
        if output_path is None:
            writer = csv.DictWriter(sys.stdout, fieldnames=fieldnames)
            writer.writeheader()
            for row in rows:
                writer.writerow(row)
            logger.info("output_written", destination="stdout", row_count=len(rows))
        else:
            with output_path.open("w", newline="") as csvfile:
                writer = csv.DictWriter(csvfile, fieldnames=fieldnames)
                writer.writeheader()
                for row in rows:
                    writer.writerow(row)
            logger.info("output_written", destination=str(output_path), row_count=len(rows))
    except PermissionError:
        logger.error("permission_denied", path=str(output_path))
        sys.exit(1)
    except OSError as e:
        logger.error("output_error", path=str(output_path), error=str(e))
        sys.exit(1)


def main() -> None:
    """Main entry point for the CLI tool."""
    try:
        options = parse_arguments()
        
        logger.info("starting_filter", 
                   input_file=str(options.input_file),
                   column=options.column_name,
                   value=options.column_value)
        
        rows = read_csv_with_progress(options.input_file, options.delimiter)
        filtered_rows = filter_rows(rows, options.column_name, options.column_value)
        write_csv_output(filtered_rows, options.output_file)
        
    except KeyboardInterrupt:
        logger.warning("operation_interrupted")
        sys.exit(130)
    except Exception as e:
        logger.error("unexpected_error", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()
```
