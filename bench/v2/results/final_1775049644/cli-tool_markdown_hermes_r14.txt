```python
#!/usr/bin/env python3
"""
A CLI tool to filter CSV rows by a column value.

Usage:
    python csv_filter.py input.csv column_name value
"""

import argparse
import csv
import sys
from pathlib import Path
from typing import List, Dict, Any

import structlog

# Configure structlog
structlog.configure(
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.StackInfoRenderer(),
        structlog.dev.set_exc_info,
        structlog.dev.ConsoleRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=False,
)

log = structlog.get_logger()


def read_csv_file(file_path: Path) -> List[Dict[str, str]]:
    """Read a CSV file and return a list of dictionaries."""
    try:
        with open(file_path, 'r', newline='', encoding='utf-8') as csvfile:
            reader = csv.DictReader(csvfile)
            return list(reader)
    except FileNotFoundError:
        log.error("file_not_found", path=str(file_path))
        sys.exit(1)
    except PermissionError:
        log.error("permission_denied", path=str(file_path))
        sys.exit(1)
    except csv.Error as e:
        log.error("csv_parsing_error", path=str(file_path), error=str(e))
        sys.exit(1)


def filter_rows(rows: List[Dict[str, str]], column: str, value: str) -> List[Dict[str, str]]:
    """Filter rows where the specified column matches the given value."""
    filtered = []
    total = len(rows)
    
    for i, row in enumerate(rows):
        # Progress output every 1000 rows
        if (i + 1) % 1000 == 0:
            log.info("progress", processed=i + 1, total=total, percentage=((i + 1) / total * 100))
        
        # Check if column exists and matches value
        if column in row and row[column] == value:
            filtered.append(row)
    
    return filtered


def output_results(filtered_rows: List[Dict[str, str]]) -> None:
    """Output the filtered rows as CSV to stdout."""
    if not filtered_rows:
        log.info("no_matching_rows")
        return
    
    # Get fieldnames from the first row
    fieldnames = list(filtered_rows[0].keys())
    
    # Write CSV to stdout
    writer = csv.DictWriter(sys.stdout, fieldnames=fieldnames)
    writer.writeheader()
    
    for row in filtered_rows:
        writer.writerow(row)


def parse_args() -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Filter CSV rows by a column value",
        formatter_class=argparse.RawDescriptionHelpFormatter,
        epilog="""
Examples:
  %(prog)s data.csv status active
  %(prog)s data.csv category "Electronics & Gadgets"
        """
    )
    
    parser.add_argument(
        'input_file',
        type=Path,
        help='Path to the input CSV file'
    )
    
    parser.add_argument(
        'column',
        type=str,
        help='Column name to filter on'
    )
    
    parser.add_argument(
        'value',
        type=str,
        help='Value to match in the specified column'
    )
    
    parser.add_argument(
        '-v', '--verbose',
        action='store_true',
        help='Enable verbose output'
    )
    
    return parser.parse_args()


def main() -> None:
    """Main entry point for the CSV filter tool."""
    args = parse_args()
    
    # Adjust logging level based on verbose flag
    if args.verbose:
        structlog.configure(
            processors=[
                structlog.processors.add_log_level,
                structlog.processors.TimeStamper(fmt="iso"),
                structlog.processors.StackInfoRenderer(),
                structlog.dev.set_exc_info,
                structlog.dev.ConsoleRenderer(),
            ],
            wrapper_class=structlog.make_filtering_bound_logger(logging.DEBUG),
            context_class=dict,
            logger_factory=structlog.PrintLoggerFactory(),
            cache_logger_on_first_use=False,
        )
        log = structlog.get_logger()
        log.debug("verbose_mode_enabled")
    
    # Check if input file exists
    if not args.input_file.exists():
        log.error("file_not_found", path=str(args.input_file))
        sys.exit(1)
    
    # Read the CSV file
    log.info("reading_csv", file=str(args.input_file))
    rows = read_csv_file(args.input_file)
    log.info("read_rows", count=len(rows))
    
    # Filter the rows
    log.info("filtering_rows", column=args.column, value=args.value)
    filtered_rows = filter_rows(rows, args.column, args.value)
    
    # Report results
    log.info("filtering_complete", total=len(rows), filtered=len(filtered_rows))
    
    # Output results
    output_results(filtered_rows)


if __name__ == "__main__":
    main()
```
