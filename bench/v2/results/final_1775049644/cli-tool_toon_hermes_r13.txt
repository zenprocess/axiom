Here's a robust Python CLI tool that meets all the specified rules:

```python
#!/usr/bin/env python3
"""
A CLI tool to filter rows from a CSV file based on a column value.
"""

import argparse
import csv
import sys
from pathlib import Path
from typing import List, Optional, Tuple

import structlog

# Configure structlog
structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
)
logger = structlog.get_logger()


def parse_args() -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Filter rows from a CSV file based on a column value."
    )
    parser.add_argument("input_file", type=Path, help="Input CSV file path")
    parser.add_argument(
        "column_name", type=str, help="Name of the column to filter on"
    )
    parser.add_argument("match_value", type=str, help="Value to match in the column")
    parser.add_argument(
        "-o",
        "--output",
        type=Path,
        default=None,
        help="Output file path (default: stdout)",
    )
    parser.add_argument(
        "-d",
        "--delimiter",
        type=str,
        default=",",
        help="CSV delimiter (default: ',')",
    )
    parser.add_argument(
        "-q",
        "--quiet",
        action="store_true",
        help="Suppress progress messages",
    )
    return parser.parse_args()


def read_csv(
    file_path: Path, delimiter: str, quiet: bool = False
) -> Tuple[List[dict], Optional[str]]:
    """
    Read CSV file and return list of rows as dictionaries.

    Args:
        file_path: Path to the CSV file
        delimiter: Field delimiter
        quiet: If True, suppress progress messages

    Returns:
        Tuple of (rows, error_message)
    """
    try:
        if not file_path.exists():
            return [], f"Error: File '{file_path}' not found."
        if not file_path.is_file():
            return [], f"Error: '{file_path}' is not a file."

        with open(file_path, "r", newline="", encoding="utf-8") as f:
            reader = csv.DictReader(f, delimiter=delimiter)
            rows = list(reader)

        if not quiet:
            logger.info("read_complete", rows_count=len(rows))

        return rows, None

    except PermissionError as e:
        return [], f"Error: Permission denied to read '{file_path}': {e}"
    except UnicodeDecodeError as e:
        return [], f"Error: Encoding error while reading '{file_path}': {e}"
    except csv.Error as e:
        return [], f"Error: Invalid CSV format in '{file_path}': {e}"
    except Exception as e:
        return [], f"Unexpected error reading '{file_path}': {e}"


def filter_rows(
    rows: List[dict], column_name: str, match_value: str, quiet: bool = False
) -> Tuple[List[dict], Optional[str]]:
    """
    Filter rows where column_name equals match_value.

    Args:
        rows: List of row dictionaries
        column_name: Column to filter on
        match_value: Value to match
        quiet: If True, suppress progress messages

    Returns:
        Tuple of (filtered_rows, error_message)
    """
    try:
        if not rows:
            return [], "Error: No rows to filter."

        if column_name not in rows[0]:
            return [], (
                f"Error: Column '{column_name}' not found. "
                f"Available columns: {list(rows[0].keys())}"
            )

        filtered = [row for row in rows if row.get(column_name) == match_value]

        if not quiet:
            logger.info(
                "filter_complete",
                total_rows=len(rows),
                matched_rows=len(filtered),
                column=column_name,
                value=match_value,
            )

        return filtered, None

    except Exception as e:
        return [], f"Unexpected error filtering rows: {e}"


def write_output(
    rows: List[dict],
    output_path: Optional[Path],
    delimiter: str,
    quiet: bool = False,
) -> Optional[str]:
    """
    Write filtered rows to output (file or stdout).

    Args:
        rows: List of row dictionaries to write
        output_path: Output file path or None for stdout
        delimiter: Field delimiter for output
        quiet: If True, suppress progress messages

    Returns:
        Error message if any, None otherwise
    """
    try:
        if not quiet:
            logger.info("writing_output", rows_count=len(rows))

        if output_path is None:
            # Write to stdout
            fieldnames = rows[0].keys() if rows else []
            writer = csv.DictWriter(sys.stdout, fieldnames=fieldnames, delimiter=delimiter)
            writer.writeheader()
            writer.writerows(rows)
        else:
            # Write to file
            with open(output_path, "w", newline="", encoding="utf-8") as f:
                fieldnames = rows[0].keys() if rows else []
                writer = csv.DictWriter(f, fieldnames=fieldnames, delimiter=delimiter)
                writer.writeheader()
                writer.writerows(rows)

        if not quiet:
            logger.info(
                "write_complete",
                output=str(output_path or "stdout"),
                rows_written=len(rows),
            )

        return None

    except PermissionError as e:
        return f"Error: Permission denied writing to '{output_path}': {e}"
    except Exception as e:
        return f"Unexpected error writing output: {e}"


def main() -> int:
    """Main entry point for the CLI tool."""
    args = parse_args()

    # Read CSV file
    rows, error = read_csv(args.input_file, args.delimiter, args.quiet)
    if error:
        logger.error("read_failed", message=error)
        return 1

    # Filter rows
    filtered_rows, error = filter_rows(rows, args.column_name, args.match_value, args.quiet)
    if error:
        logger.error("filter_failed", message=error)
        return 1

    # Write output
    error = write_output(filtered_rows, args.output, args.delimiter, args.quiet)
    if error:
        logger.error("write_failed", message=error)
        return 1

    logger.info("success", message="Filtering completed successfully")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

### Usage Examples:

```bash
# Basic usage: filter rows where "status" equals "active"
python filter_csv.py data.csv status active

# Output to file
python filter_csv.py data.csv status active -o filtered.csv

# Custom delimiter (tab-separated)
python filter_csv.py data.csv name "John Doe" -d $'\t'

# Suppress progress messages
python filter_csv.py data.csv status active -q
```

The tool gracefully handles file not found, permission errors, encoding issues, missing columns, and other common problems with clear error messages.
