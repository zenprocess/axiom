```python
#!/usr/bin/env python3
"""
CLI tool to filter rows from a CSV file based on a column value.

Usage:
    python csv_filter.py input.csv column_name value [--output output.csv]
"""

import argparse
import csv
import sys
from pathlib import Path
from typing import List, Optional

import structlog

# Configure structlog
structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(
        min_level=structlog.logging_levels.INFO
    ),
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.dev.ConsoleRenderer(colors=True),
    ],
)
logger = structlog.get_logger()


class CSVFilterError(Exception):
    """Custom exception for CSV filter errors."""
    pass


def parse_args() -> argparse.Namespace:
    """Parse command-line arguments."""
    parser = argparse.ArgumentParser(
        description="Filter rows from a CSV file based on a column value."
    )
    parser.add_argument(
        "input_file",
        type=str,
        help="Input CSV file path",
    )
    parser.add_argument(
        "column_name",
        type=str,
        help="Column name to filter on",
    )
    parser.add_argument(
        "value",
        type=str,
        help="Value to match in the specified column",
    )
    parser.add_argument(
        "-o",
        "--output",
        type=str,
        default=None,
        help="Output file path (default: stdout)",
    )
    parser.add_argument(
        "-d",
        "--delimiter",
        type=str,
        default=",",
        help="CSV delimiter (default: comma)",
    )
    parser.add_argument(
        "-v",
        "--verbose",
        action="store_true",
        help="Enable verbose output",
    )
    
    return parser.parse_args()


def read_csv_rows(input_path: Path, delimiter: str) -> List[dict]:
    """Read all rows from a CSV file."""
    rows = []
    try:
        with input_path.open("r", newline="", encoding="utf-8") as csvfile:
            reader = csv.DictReader(csvfile, delimiter=delimiter)
            for row in reader:
                rows.append(row)
        return rows
    except FileNotFoundError:
        raise CSVFilterError(f"Input file not found: {input_path}")
    except csv.Error as e:
        raise CSVFilterError(f"Error parsing CSV: {e}")


def filter_rows(
    rows: List[dict],
    column_name: str,
    value: str,
    verbose: bool = False,
) -> List[dict]:
    """Filter rows where column_name matches value."""
    filtered = []
    matched = 0
    total = len(rows)
    
    for i, row in enumerate(rows):
        # Progress output
        if verbose and i % max(1, total // 10) == 0:
            logger.info("Progress", current=i, total=total, percentage=i / total * 100)
        
        # Check if column exists and matches
        if column_name in row and row[column_name] == value:
            filtered.append(row)
            matched += 1
    
    if verbose:
        logger.info(
            "Filtering complete",
            total=total,
            matched=matched,
            percentage=matched / total * 100 if total > 0 else 0,
        )
    
    return filtered


def write_output(
    rows: List[dict],
    output_path: Optional[Path],
    delimiter: str,
) -> None:
    """Write filtered rows to output."""
    if not rows:
        logger.warning("No rows matched the filter criteria")
        return
    
    # Get fieldnames from first row
    fieldnames = list(rows[0].keys())
    
    # Handle output destination
    if output_path:
        try:
            with output_path.open("w", newline="", encoding="utf-8") as csvfile:
                writer = csv.DictWriter(csvfile, fieldnames=fieldnames, delimiter=delimiter)
                writer.writeheader()
                writer.writerows(rows)
            logger.info("Output written", path=str(output_path), row_count=len(rows))
        except IOError as e:
            raise CSVFilterError(f"Error writing to output file: {e}")
    else:
        # Write to stdout
        writer = csv.DictWriter(sys.stdout, fieldnames=fieldnames, delimiter=delimiter)
        writer.writeheader()
        writer.writerows(rows)
        logger.info("Output written to stdout", row_count=len(rows))


def main() -> None:
    """Main entry point for the CLI tool."""
    args = parse_args()
    
    # Set up logging level
    if args.verbose:
        structlog.configure(
            wrapper_class=structlog.make_filtering_bound_logger(
                min_level=structlog.logging_levels.DEBUG
            ),
        )
    
    # Validate input file
    input_path = Path(args.input_file)
    if not input_path.exists():
        logger.error("Input file not found", path=args.input_file)
        sys.exit(1)
    
    try:
        # Read CSV
        logger.info("Reading CSV", file=str(input_path))
        rows = read_csv_rows(input_path, args.delimiter)
        logger.info("CSV loaded", row_count=len(rows))
        
        # Filter rows
        logger.info(
            "Filtering rows",
            column=args.column_name,
            value=args.value,
        )
        filtered_rows = filter_rows(
            rows,
            args.column_name,
            args.value,
            args.verbose,
        )
        
        # Write output
        output_path = Path(args.output) if args.output else None
        write_output(filtered_rows, output_path, args.delimiter)
        
    except CSVFilterError as e:
        logger.error("Error", message=str(e))
        sys.exit(1)
    except Exception as e:
        logger.error("Unexpected error", exception=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()
```
