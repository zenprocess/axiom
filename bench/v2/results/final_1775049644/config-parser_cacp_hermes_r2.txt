```python
import os
import sys
from pathlib import Path
from typing import Any, Dict, Optional, Type

import yaml
import structlog
from pydantic import BaseModel, BaseSettings, Field, ValidationError
from dotenv import load_dotenv


# Configure structlog for logging
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.processors.StackInfoRenderer(),
        structlog.dev.set_exc_info,
        structlog.dev.ConsoleRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.WARNING),
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=False,
)

logger = structlog.get_logger()


class Config(BaseModel):
    """Application configuration model."""
    app_name: str = Field(default="myapp", description="Application name")
    debug: bool = Field(default=False, description="Enable debug mode")
    database_url: str = Field(..., description="Database connection URL")
    api_key: Optional[str] = Field(default=None, description="API key for external services")
    log_level: str = Field(default="INFO", description="Logging level")
    max_connections: int = Field(default=10, ge=1, le=100, description="Maximum database connections")


class ConfigLoader:
    """Loads and merges configuration from multiple sources with priority."""
    
    def __init__(self, config_path: Optional[str] = None):
        """Initialize the config loader.
        
        Args:
            config_path: Path to the YAML config file. Defaults to None.
        """
        self.config_path = config_path
        self._config: Optional[Config] = None
    
    async def load(self) -> Config:
        """Load and merge configuration from all sources.
        
        Priority (lowest to highest):
        1. Default values in Config model
        2. Environment variables (prefixed with APP_)
        3. YAML config file
        4. Command-line arguments
        
        Returns:
            Merged configuration object.
        """
        # Start with defaults
        config_data: Dict[str, Any] = {}
        
        # Load from YAML file if specified
        if self.config_path:
            try:
                yaml_data = await self._load_yaml(self.config_path)
                config_data.update(yaml_data)
            except Exception as e:
                logger.warning("yaml_load_failed", path=self.config_path, error=str(e))
        
        # Load from environment variables (APP_* prefix)
        env_data = self._load_env_vars()
        config_data.update(env_data)
        
        # Load from command-line arguments
        cli_data = self._load_cli_args()
        config_data.update(cli_data)
        
        # Validate and create final config object
        try:
            self._config = Config(**config_data)
            return self._config
        except ValidationError as e:
            logger.error("config_validation_error", errors=e.errors())
            raise
    
    async def _load_yaml(self, path: str) -> Dict[str, Any]:
        """Load configuration from YAML file.
        
        Args:
            path: Path to the YAML file.
            
        Returns:
            Dictionary of configuration values.
        """
        yaml_path = Path(path)
        if not yaml_path.exists():
            logger.warning("yaml_file_not_found", path=path)
            return {}
        
        try:
            async with await open_file(yaml_path, "r") as f:
                content = await f.read()
                data = yaml.safe_load(content) or {}
                logger.info("yaml_loaded", path=path, keys=list(data.keys()))
                return data
        except Exception as e:
            logger.error("yaml_read_failed", path=path, error=str(e))
            return {}
    
    def _load_env_vars(self) -> Dict[str, Any]:
        """Load configuration from environment variables.
        
        Only environment variables prefixed with APP_ are considered.
        The prefix is stripped and the rest is converted to snake_case.
        
        Returns:
            Dictionary of environment configuration values.
        """
        env_data = {}
        prefix = "APP_"
        
        for key, value in os.environ.items():
            if key.startswith(prefix):
                # Convert APP_VAR_NAME to var_name
                config_key = key[len(prefix):].lower().replace("_", "_")
                env_data[config_key] = value
                logger.debug("env_var_loaded", key=key, value=value)
        
        return env_data
    
    def _load_cli_args(self) -> Dict[str, Any]:
        """Load configuration from command-line arguments.
        
        CLI arguments override all other sources.
        
        Returns:
            Dictionary of CLI configuration values.
        """
        cli_data = {}
        args = sys.argv[1:]
        
        i = 0
        while i < len(args):
            arg = args[i]
            if arg.startswith("--"):
                # Handle --key=value format
                if "=" in arg:
                    key, value = arg[2:].split("=", 1)
                    cli_data[key] = value
                # Handle --key value format
                elif i + 1 < len(args) and not args[i + 1].startswith("--"):
                    key = arg[2:]
                    value = args[i + 1]
                    cli_data[key] = value
                    i += 1
                else:
                    key = arg[2:]
                    cli_data[key] = True
            i += 1
        
        if cli_data:
            logger.info("cli_args_loaded", keys=list(cli_data.keys()))
        
        return cli_data
    
    @property
    def config(self) -> Config:
        """Get the loaded configuration.
        
        Returns:
            The loaded configuration object.
            
        Raises:
            RuntimeError: If config hasn't been loaded yet.
        """
        if self._config is None:
            raise RuntimeError("Configuration not loaded. Call load() first.")
        return self._config


async def open_file(path: Path, mode: str):
    """Async file opener using asyncio."""
    import asyncio
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(None, open, path, mode)


async def main():
    """Main function to demonstrate config loading."""
    # Load .env file if present
    load_dotenv()
    
    # Create config loader with optional YAML path
    config_loader = ConfigLoader(config_path="config.yaml")
    
    try:
        config = await config_loader.load()
        
        # Log final config for debugging (using structlog)
        logger.info("config_loaded", config=config.dict())
        
        return config
    except Exception as e:
        logger.error("config_load_failed", error=str(e))
        raise


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
