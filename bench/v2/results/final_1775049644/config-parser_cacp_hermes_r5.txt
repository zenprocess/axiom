Here's a Python config loader that follows your strict coding standards:

```python
import os
import sys
from typing import Any, Dict, Optional

import yaml
import structlog
from pydantic import BaseModel, Field, ValidationError
from dotenv import load_dotenv


structlog.configure(
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.dev.ConsoleRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.WARNING),
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=False,
)

log = structlog.get_logger()


class Config(BaseModel):
    """Application configuration model."""

    app_name: str = Field(default="my_app", description="Name of the application")
    debug: bool = Field(default=False, description="Enable debug mode")
    log_level: str = Field(default="INFO", description="Logging level (DEBUG, INFO, WARNING, ERROR)")
    database_url: str = Field(default="sqlite:///app.db", description="Database connection URL")
    api_key: Optional[str] = Field(default=None, description="API key for external services")


def load_yaml_config(file_path: str) -> Dict[str, Any]:
    """
    Load configuration from a YAML file.

    Args:
        file_path: Path to the YAML configuration file.

    Returns:
        Dictionary of configuration values.
    """
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return yaml.safe_load(f) or {}
    except FileNotFoundError:
        log.warning("yaml_config_not_found", file_path=file_path)
        return {}
    except yaml.YAMLError as e:
        log.error("yaml_parse_error", file_path=file_path, error=str(e))
        return {}


def load_env_config() -> Dict[str, Any]:
    """
    Load configuration from environment variables.

    Returns:
        Dictionary of environment-based configuration values.
    """
    load_dotenv()
    env_config = {
        "app_name": os.getenv("APP_NAME"),
        "debug": os.getenv("DEBUG", "false").lower() == "true",
        "log_level": os.getenv("LOG_LEVEL", "INFO").upper(),
        "database_url": os.getenv("DATABASE_URL"),
        "api_key": os.getenv("API_KEY"),
    }
    # Filter out None values
    return {k: v for k, v in env_config.items() if v is not None}


def load_cli_config() -> Dict[str, Any]:
    """
    Load configuration from command-line arguments.

    Returns:
        Dictionary of CLI-based configuration values.
    """
    cli_config = {}
    args = sys.argv[1:]
    
    i = 0
    while i < len(args):
        arg = args[i]
        if arg.startswith("--"):
            key = arg[2:].replace("-", "_")
            if i + 1 < len(args) and not args[i + 1].startswith("--"):
                value = args[i + 1]
                # Try to parse as bool/int/float, otherwise keep as string
                if value.lower() == "true":
                    value = True
                elif value.lower() == "false":
                    value = False
                elif value.isdigit():
                    value = int(value)
                elif value.replace(".", "", 1).isdigit():
                    value = float(value)
                cli_config[key] = value
                i += 2
            else:
                # Boolean flag without value
                cli_config[key] = True
                i += 1
        else:
            i += 1
    
    return cli_config


def merge_configs(*configs: Dict[str, Any]) -> Dict[str, Any]:
    """
    Merge multiple configuration dictionaries with priority:
    last argument has highest priority.

    Args:
        *configs: Configuration dictionaries to merge.

    Returns:
        Merged configuration dictionary.
    """
    merged = {}
    for config in reversed(configs):
        for key, value in config.items():
            if key not in merged:
                merged[key] = value
    return merged


def validate_config(raw_config: Dict[str, Any]) -> Config:
    """
    Validate and convert raw configuration to Pydantic model.

    Args:
        raw_config: Raw configuration dictionary.

    Returns:
        Validated Config model instance.
    """
    try:
        return Config(**raw_config)
    except ValidationError as e:
        log.error("config_validation_error", error=str(e))
        raise


def load_config(
    yaml_path: str = "config.yaml",
) -> Config:
    """
    Load configuration from YAML file, environment variables, and CLI args.

    Priority (lowest to highest):
    1. YAML file
    2. Environment variables
    3. Command-line arguments

    Args:
        yaml_path: Path to the YAML configuration file.

    Returns:
        Validated Config model instance.
    """
    yaml_config = load_yaml_config(yaml_path)
    env_config = load_env_config()
    cli_config = load_cli_config()
    
    merged_config = merge_configs(yaml_config, env_config, cli_config)
    validated_config = validate_config(merged_config)
    
    # Log final config (without secrets)
    safe_config = validated_config.model_dump()
    if "api_key" in safe_config:
        safe_config["api_key"] = "••••••••" if safe_config["api_key"] else None
    log.info("final_config", config=safe_config)
    
    return validated_config
```

Key features:
- Uses Pydantic `BaseModel` for config validation
- Uses `structlog` for logging (no print statements)
- All I/O is synchronous (no async I/O needed for config loading)
- No hardcoded secrets or API keys
- CLI flags are parsed manually without external libraries
- Priority merging: CLI > ENV > YAML
- Safe logging of final config (redacts secrets)
- Proper error handling and validation
- Type hints on all functions
- Docstrings on public functions
- No star imports
- No `print()` or `logging` imports (uses structlog only)

The `log.warning` and `log.error` calls are safe since they're using structlog, not print statements. The final config is logged with secrets redacted for safety.
