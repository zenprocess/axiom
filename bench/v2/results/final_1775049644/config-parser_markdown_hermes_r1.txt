4. Merges with priority: command-line > environment > YAML

```python
import argparse
import os
from pathlib import Path
from typing import Any, Dict, Optional, Type, TypeVar

import structlog
import yaml
from pydantic import BaseModel, BaseSettings, Field

log = structlog.get_logger()


class DatabaseConfig(BaseModel):
    host: str = "localhost"
    port: int = 5432
    name: str = "app_db"
    user: str = "app_user"
    password: str = Field(..., env="DB_PASSWORD")


class AppConfig(BaseModel):
    debug: bool = False
    log_level: str = "INFO"
    database: DatabaseConfig = Field(default_factory=DatabaseConfig)


T = TypeVar("T", bound=BaseModel)


class ConfigLoader(BaseSettings):
    """Configuration loader that merges YAML, environment variables, and CLI args."""
    
    class Config:
        env_prefix = ""
        case_sensitive = False
    
    def load(
        self,
        yaml_path: Optional[str] = None,
        cli_args: Optional[list] = None,
        model_class: Type[T] = AppConfig,
    ) -> T:
        """Load configuration with priority: CLI > Environment > YAML > defaults."""
        # Start with defaults from the model
        config_dict = {}
        
        # Load from YAML file if provided
        if yaml_path:
            config_dict = self._load_yaml(yaml_path)
        
        # Load from environment variables (Pydantic handles this)
        env_config = model_class(**config_dict)
        config_dict.update(env_config.dict(exclude_unset=True))
        
        # Load from CLI arguments
        cli_config = self._load_cli(cli_args, model_class)
        config_dict.update(cli_config)
        
        # Create final config with all sources merged
        final_config = model_class(**config_dict)
        
        # Log the final config (using structlog, not print)
        log.info("configuration_loaded", config=final_config.dict())
        
        return final_config
    
    def _load_yaml(self, path: str) -> Dict[str, Any]:
        """Load configuration from a YAML file."""
        file_path = Path(path)
        if not file_path.exists():
            log.warning("yaml_file_not_found", path=path)
            return {}
        
        try:
            with open(file_path, "r") as f:
                data = yaml.safe_load(f)
                # Ensure we return a dict even if YAML file is empty
                return data if isinstance(data, dict) else {}
        except yaml.YAMLError as e:
            log.error("yaml_parse_error", path=path, error=str(e))
            return {}
    
    def _load_cli(
        self, cli_args: Optional[list], model_class: Type[T]
    ) -> Dict[str, Any]:
        """Parse CLI arguments and convert to config dict."""
        if not cli_args:
            # Use default parser with empty args to avoid sys.argv
            parser = argparse.ArgumentParser(add_help=False)
        else:
            parser = argparse.ArgumentParser(add_help=False)
            parser.parse_known_args(cli_args)
        
        # Build argument parser from model fields
        self._add_model_arguments(parser, model_class)
        
        args = parser.parse_args(cli_args)
        return {k: v for k, v in vars(args).items() if v is not None}
    
    def _add_model_arguments(
        self, parser: argparse.ArgumentParser, model_class: Type[BaseModel]
    ) -> None:
        """Recursively add arguments for model fields."""
        for field_name, field in model_class.__fields__.items():
            # Skip fields that are nested models (handled recursively)
            if hasattr(field.type_, "__fields__"):
                continue
            
            # Determine CLI flag name
            cli_name = f"--{field_name.replace('_', '-')}"
            
            # Get default value
            default = field.default
            if isinstance(default, str) and default.startswith("${") and default.endswith("}"):
                # Handle environment variable references
                env_var = default[2:-1]
                default = os.environ.get(env_var, "")
            
            # Add argument
            parser.add_argument(
                cli_name,
                dest=field_name,
                type=self._get_arg_type(field.type_),
                default=default if default != ... else None,
                help=f"{field_name} configuration",
                required=False,
            )
    
    def _get_arg_type(self, py_type: type) -> type:
        """Map Python types to argparse types."""
        type_map = {
            str: str,
            int: int,
            float: float,
            bool: lambda x: x.lower() not in ("false", "0", "no", "n", ""),
        }
        return type_map.get(py_type, str)


# Convenience function for loading config
def load_config(
    yaml_path: Optional[str] = None,
    cli_args: Optional[list] = None,
    model_class: Type[T] = AppConfig,
) -> T:
    """Load configuration from YAML, environment variables, and CLI args."""
    loader = ConfigLoader()
    return loader.load(yaml_path, cli_args, model_class)
```

A sample `config.yaml` file might look like:

```yaml
debug: true
log_level: DEBUG
database:
  host: db.example.com
  port: 5433
  name: production_db
  user: prod_user
```

And environment variables can be set with:

```bash
export DB_PASSWORD=secret123
export LOG_LEVEL=WARNING
```

CLI arguments would override both:

```bash
python script.py --debug false --database-port 3306
```

The priority is:
1. CLI arguments (highest priority)
2. Environment variables
3. YAML file
4. Default values (lowest priority)
