import os
import argparse
import sys
from pathlib import Path
from typing import Any, Dict, Optional

import yaml
import structlog
from pydantic import BaseModel, Field, ValidationError

log = structlog.get_logger()


class Config(BaseModel):
    """Application configuration model."""
    app_name: str = Field(default="my_app", description="Application name")
    debug: bool = Field(default=False, description="Enable debug mode")
    log_level: str = Field(default="INFO", description="Logging level")
    database_url: str = Field(default="sqlite:///app.db", description="Database connection URL")
    api_key: Optional[str] = Field(default=None, description="API key for external services")
    max_workers: int = Field(default=4, description="Maximum number of workers")


class ConfigLoader:
    """Configuration loader that merges YAML files, environment variables, and CLI flags."""
    
    def __init__(self, config_paths: list[str] = None):
        """Initialize the config loader.
        
        Args:
            config_paths: List of paths to YAML config files to load
        """
        self.config_paths = config_paths or []
        self._raw_config = {}
    
    def load(self) -> Config:
        """Load and merge configuration from all sources.
        
        Returns:
            Config model with all values merged with priority:
            CLI flags > environment variables > YAML files > defaults
        """
        # Start with defaults
        merged_config = {}
        
        # Load YAML files (in order, later files override earlier ones)
        for path in self.config_paths:
            try:
                file_config = self._load_yaml(path)
                merged_config.update(file_config)
            except FileNotFoundError:
                log.warning("config_file_not_found", path=path)
            except yaml.YAMLError as e:
                log.error("config_file_parse_error", path=path, error=str(e))
                raise
        
        # Load environment variables (override YAML)
        env_config = self._load_env_vars()
        merged_config.update(env_config)
        
        # Load CLI arguments (highest priority)
        cli_config = self._load_cli_args()
        merged_config.update(cli_config)
        
        # Validate and create config model
        try:
            config = Config(**merged_config)
        except ValidationError as e:
            log.error("config_validation_error", error=str(e))
            raise
        
        return config
    
    def _load_yaml(self, path: str) -> Dict[str, Any]:
        """Load configuration from a YAML file."""
        with open(path, 'r') as f:
            return yaml.safe_load(f) or {}
    
    def _load_env_vars(self) -> Dict[str, Any]:
        """Load configuration from environment variables.
        
        Converts environment variables to config keys:
        - Converts to lowercase
        - Replaces underscores with dots
        - Removes common prefixes like APP_, CONFIG_
        """
        env_config = {}
        
        # Common prefixes to remove
        prefixes = ['APP_', 'CONFIG_', 'MYAPP_']
        
        for key, value in os.environ.items():
            # Skip non-config environment variables
            if not any(key.startswith(prefix) for prefix in prefixes):
                continue
            
            # Convert key to config format
            config_key = key.lower()
            for prefix in prefixes:
                if config_key.startswith(prefix.lower()):
                    config_key = config_key[len(prefix):]
                    break
            
            # Convert underscores to dots
            config_key = config_key.replace('_', '.')
            
            # Try to convert value to appropriate type
            try:
                if value.lower() in ('true', 'false'):
                    value = value.lower() == 'true'
                elif value.isdigit():
                    value = int(value)
                else:
                    # Try float
                    try:
                        value = float(value)
                    except ValueError:
                        pass  # Keep as string
            except AttributeError:
                pass  # Keep original value
            
            env_config[config_key] = value
        
        return env_config
    
    def _load_cli_args(self) -> Dict[str, Any]:
        """Load configuration from command-line arguments."""
        parser = argparse.ArgumentParser(description="Application configuration")
        
        parser.add_argument(
            "--app-name", 
            type=str, 
            help="Application name"
        )
        parser.add_argument(
            "--debug", 
            action="store_true", 
            help="Enable debug mode"
        )
        parser.add_argument(
            "--log-level", 
            type=str, 
            choices=["DEBUG", "INFO", "WARNING", "ERROR", "CRITICAL"],
            help="Logging level"
        )
        parser.add_argument(
            "--database-url", 
            type=str, 
            help="Database connection URL"
        )
        parser.add_argument(
            "--api-key", 
            type=str, 
            help="API key for external services"
        )
        parser.add_argument(
            "--max-workers", 
            type=int, 
            help="Maximum number of workers"
        )
        
        # Parse known args to avoid issues with other CLI tools
        args, _ = parser.parse_known_args()
        
        # Convert to config format (replace hyphens with underscores)
        config = {}
        for key, value in vars(args).items():
            if value is not None:
                config[key.replace('_', '-')] = value
        
        return config


def main():
    """Main entry point demonstrating config loading."""
    # Set up structlog
    structlog.configure(
        processors=[
            structlog.processors.add_log_level,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.dev.ConsoleRenderer(),
        ],
        wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=False,
    )
    
    # Set up logging level based on environment
    log_level = os.environ.get("LOG_LEVEL", "INFO").upper()
    structlog.configure(
        wrapper_class=structlog.make_filtering_bound_logger(getattr(logging, log_level, logging.INFO)),
    )
    
    # Define config paths (later files override earlier ones)
    config_paths = [
        "config.default.yaml",
        "config.yaml",
        "config.local.yaml",
    ]
    
    # Load configuration
    loader = ConfigLoader(config_paths)
    config = loader.load()
    
    # Log final configuration (without sensitive data)
    safe_config = config.model_dump()
    if "api_key" in safe_config:
        safe_config["api_key"] = "••••••••" if safe_config["api_key"] else None
    
    log.info("config_loaded", config=safe_config)
    
    # Return config for application use
    return config


# Add missing import for logging
import logging


if __name__ == "__main__":
    config = main()
