```python
import argparse
import os
from typing import Any, Dict, Optional

import structlog
import yaml
from pydantic import BaseModel

log = structlog.get_logger()


class Config(BaseModel):
    """Application configuration model."""
    debug: bool = False
    database_url: str
    log_level: str = "INFO"
    max_connections: int = 10
    timeout: float = 30.0


def load_yaml_config(file_path: str) -> Dict[str, Any]:
    """Load configuration from a YAML file."""
    if not os.path.exists(file_path):
        log.warning("yaml_file_not_found", file_path=file_path)
        return {}
    
    with open(file_path, 'r') as f:
        try:
            config = yaml.safe_load(f)
            log.info("yaml_config_loaded", file_path=file_path, config_keys=list(config.keys()) if config else [])
            return config or {}
        except yaml.YAMLError as e:
            log.error("yaml_parse_error", file_path=file_path, error=str(e))
            return {}


def get_env_config() -> Dict[str, Any]:
    """Load configuration from environment variables."""
    env_config = {}
    
    # Map environment variables to config keys
    env_mappings = {
        'DEBUG': 'debug',
        'DATABASE_URL': 'database_url',
        'LOG_LEVEL': 'log_level',
        'MAX_CONNECTIONS': 'max_connections',
        'TIMEOUT': 'timeout'
    }
    
    for env_name, config_key in env_mappings.items():
        value = os.environ.get(env_name)
        if value is not None:
            # Convert string values to appropriate types
            if config_key == 'debug':
                env_config[config_key] = value.lower() in ('true', '1', 'yes')
            elif config_key in ('max_connections', 'timeout'):
                env_config[config_key] = int(value) if config_key == 'max_connections' else float(value)
            else:
                env_config[config_key] = value
    
    log.info("env_config_loaded", keys=list(env_config.keys()))
    return env_config


def get_cli_config() -> Dict[str, Any]:
    """Load configuration from command-line arguments."""
    parser = argparse.ArgumentParser(description='Application configuration')
    parser.add_argument('--debug', action='store_true', help='Enable debug mode')
    parser.add_argument('--database-url', dest='database_url', help='Database connection URL')
    parser.add_argument('--log-level', dest='log_level', help='Logging level')
    parser.add_argument('--max-connections', type=int, dest='max_connections', help='Maximum connections')
    parser.add_argument('--timeout', type=float, dest='timeout', help='Request timeout in seconds')
    
    args = parser.parse_args()
    cli_config = {}
    
    # Convert parsed args to config dict
    for key, value in vars(args).items():
        if value is not None:
            cli_config[key] = value
    
    log.info("cli_config_loaded", keys=list(cli_config.keys()))
    return cli_config


def merge_configs(*configs: Dict[str, Any]) -> Dict[str, Any]:
    """Merge multiple configuration sources with priority (last wins)."""
    merged = {}
    
    for config in configs:
        if not config:
            continue
        for key, value in config.items():
            merged[key] = value
    
    log.info("config_merged", keys=list(merged.keys()))
    return merged


def load_config(yaml_path: Optional[str] = None) -> Config:
    """Load configuration from YAML, environment, and CLI with priority merging."""
    # Priority order (highest to lowest): CLI > Environment > YAML
    yaml_config = load_yaml_config(yaml_path) if yaml_path else {}
    env_config = get_env_config()
    cli_config = get_cli_config()
    
    # Merge configs with priority (last source wins)
    merged_config = merge_configs(yaml_config, env_config, cli_config)
    
    # Validate and create Config model instance
    config = Config(**merged_config)
    
    # Log the final config (without secrets)
    log.info("final_config", 
             debug=config.debug,
             database_url="***" if config.database_url else None,
             log_level=config.log_level,
             max_connections=config.max_connections,
             timeout=config.timeout)
    
    return config


# Example usage (would not be in production code)
if __name__ == "__main__":
    # In real code, you'd pass the config file path from a CLI argument
    config = load_config("config.yaml")
    print(f"Loaded config: {config}")
```
