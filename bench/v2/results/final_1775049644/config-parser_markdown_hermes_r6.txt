I'll create a Python config loader that reads from YAML files, environment variables, and command-line flags with priority merging, following all the development rules.

Key considerations:
- Use `structlog` for logging instead of `print()`
- Use `async def` for I/O operations (file reading)
- Use Pydantic models for data structures
- Include proper type hints
- Handle errors appropriately
- Include docstrings

```python
import os
import sys
import argparse
from typing import Any, Dict, Optional, Type, TypeVar, Union
from pathlib import Path
import yaml
import structlog
from pydantic import BaseModel, ValidationError, Field
from pydantic.fields import FieldInfo

log = structlog.get_logger()


T = TypeVar('T', bound=BaseModel)


class ConfigError(Exception):
    """Custom exception for configuration errors."""
    pass


class AppConfig(BaseModel):
    """Application configuration model."""
    app_name: str = Field(default="MyApp", description="Application name")
    debug: bool = Field(default=False, description="Enable debug mode")
    database_url: str = Field(default="sqlite:///app.db", description="Database connection URL")
    log_level: str = Field(default="INFO", description="Logging level")
    api_key: Optional[str] = Field(default=None, description="API key for external services")


class ConfigLoader:
    """Configuration loader that merges settings from YAML, environment variables, and CLI flags."""
    
    def __init__(self, config_path: Optional[str] = None, model_class: Type[T] = AppConfig):
        """
        Initialize the config loader.
        
        Args:
            config_path: Optional path to the YAML config file
            model_class: Pydantic model class for validation
        """
        self.config_path = config_path
        self.model_class = model_class
        self._raw_config: Dict[str, Any] = {}
    
    async def load(self) -> T:
        """
        Load configuration from all sources and merge with priority.
        
        Priority (highest to lowest):
        1. Command-line arguments
        2. Environment variables
        3. YAML config file
        4. Default values from model
        
        Returns:
            Validated configuration model instance
            
        Raises:
            ConfigError: If configuration validation fails
        """
        # Load from YAML file if path provided
        if self.config_path:
            await self._load_yaml_config(self.config_path)
        
        # Load from environment variables
        self._load_env_vars()
        
        # Merge with CLI args (highest priority)
        cli_config = self._load_cli_args()
        if cli_config:
            self._raw_config.update(cli_config)
        
        # Validate and create model instance
        try:
            config = self.model_class(**self._raw_config)
        except ValidationError as e:
            log.error('config_validation_failed', errors=str(e))
            raise ConfigError(f"Configuration validation failed: {e}")
        
        log.info('config_loaded', config=config.model_dump())
        return config
    
    async def _load_yaml_config(self, path: str) -> None:
        """
        Load configuration from a YAML file.
        
        Args:
            path: Path to the YAML configuration file
            
        Raises:
            ConfigError: If the file cannot be read or parsed
        """
        try:
            file_path = Path(path)
            if not file_path.exists():
                log.warning('config_file_not_found', path=path)
                return
            
            content = await self._read_file_async(str(file_path))
            config_data = yaml.safe_load(content)
            
            if config_data and isinstance(config_data, dict):
                self._raw_config.update(config_data)
                log.info('yaml_config_loaded', path=path, keys=list(config_data.keys()))
            else:
                log.warning('yaml_config_empty', path=path)
                
        except yaml.YAMLError as e:
            log.error('yaml_parse_error', path=path, error=str(e))
            raise ConfigError(f"Failed to parse YAML config file {path}: {e}")
        except Exception as e:
            log.error('config_file_error', path=path, error=str(e))
            raise ConfigError(f"Failed to read config file {path}: {e}")
    
    async def _read_file_async(self, path: str) -> str:
        """
        Read file content asynchronously.
        
        Args:
            path: Path to the file
            
        Returns:
            File content as string
        """
        import asyncio
        loop = asyncio.get_event_loop()
        with open(path, 'r') as f:
            return await loop.run_in_executor(None, f.read)
    
    def _load_env_vars(self) -> None:
        """Load configuration from environment variables."""
        # Map model fields to environment variable names
        env_config = {}
        
        for field_name, field_info in self.model_class.model_fields.items():
            env_var_name = field_name.upper()
            
            if env_var_name in os.environ:
                env_value = os.environ[env_var_name]
                # Convert to appropriate type
                field_type = field_info.annotation
                try:
                    converted_value = self._convert_env_value(env_value, field_type)
                    env_config[field_name] = converted_value
                    log.debug('env_config_loaded', field=field_name, value=converted_value)
                except ValueError as e:
                    log.warning('env_config_conversion_failed', field=field_name, value=env_value, error=str(e))
        
        if env_config:
            self._raw_config.update(env_config)
    
    def _convert_env_value(self, value: str, target_type: Type) -> Any:
        """
        Convert environment variable string to appropriate type.
        
        Args:
            value: String value from environment variable
            target_type: Target type for conversion
            
        Returns:
            Converted value
            
        Raises:
            ValueError: If conversion fails
        """
        # Handle optional types
        if hasattr(target_type, '__origin__') and target_type.__origin__ is Union:
            # Get the non-None type
            args = [arg for arg in target_type.__args__ if arg is not type(None)]
            if args:
                target_type = args[0]
        
        if target_type is bool:
            return value.lower() in ('true', '1', 'yes')
        elif target_type is int:
            return int(value)
        elif target_type is float:
            return float(value)
        else:
            return value
    
    def _load_cli_args(self) -> Dict[str, Any]:
        """
        Load configuration from command-line arguments.
        
        Returns:
            Dictionary of CLI arguments
        """
        parser = argparse.ArgumentParser(description='Application configuration')
        
        # Add arguments for each config field
        for field_name, field_info in self.model_class.model_fields.items():
            arg_name = f'--{field_name.replace("_", "-")}'
            help_text = field_info.description or f'{field_name} configuration'
            
            # Determine argument type
            field_type = field_info.annotation
            if hasattr(field_type, '__origin__') and field_type.__origin__ is Union:
                args = [arg for arg in field_type.__args__ if arg is not type(None)]
                if args:
                    field_type = args[0]
            
            if field_type is bool:
                parser.add_argument(arg_name, dest=field_name, action='store_true', help=help_text)
            else:
                parser.add_argument(arg_name, type=field_type, help=help_text)
        
        # Parse known args to avoid errors with extra arguments
        args, _ = parser.parse_known_args()
        
        # Convert to dictionary, filtering None values
        cli_config = {k: v for k, v in vars(args).items() if v is not None}
        
        if cli_config:
            log.info('cli_args_loaded', config=cli_config)
        
        return cli_config
```

Example usage:

```python
async def main():
    # Example 1: Using default config with CLI overrides
    loader = ConfigLoader(config_path="config.yaml")
    config = await loader.load()
    print(f"Running {config.app_name} in {'debug' if config.debug else 'production'} mode")
    
    # Example 2: Custom config model
    class CustomConfig(BaseModel):
        server_host: str = "localhost"
        server_port: int = 8080
        enable_tls: bool = False
    
    custom_loader = ConfigLoader(model_class=CustomConfig)
    custom_config = await custom_loader.load()
    print(f"Server running on {custom_config.server_host}:{custom_config.server_port}")

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```

This implementation:
- Uses `structlog` for structured logging (no print statements)
- Implements async file reading with `async def`
- Uses Pydantic models for configuration validation
- Includes comprehensive type hints
- Handles errors gracefully with proper exception handling
- Provides docstrings for all public methods
- Follows the priority order: CLI > Environment > YAML > Defaults
- Converts environment variables to appropriate types
- Logs configuration loading for debugging purposes
