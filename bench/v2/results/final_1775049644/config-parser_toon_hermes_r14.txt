Here's a Python config loader that reads from YAML files, environment variables, and command-line flags, with priority merging (CLI > environment > YAML), using `structlog` for logging and `Pydantic` for data modeling:

```python
import argparse
import os
from typing import Any, Dict, Optional

import yaml
from pydantic import BaseModel, ValidationError
import structlog

structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.processors.StackInfoRenderer(),
        structlog.dev.ConsoleRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(level=structlog.logging_levels.DEBUG),
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=False,
)

logger = structlog.get_logger()


class Config(BaseModel):
    """Configuration model using Pydantic BaseModel."""
    app_name: str = "MyApp"
    debug: bool = False
    log_level: str = "INFO"
    database_url: str = "sqlite:///default.db"
    api_key: Optional[str] = None
    timeout: int = 30
    max_connections: int = 100


def load_yaml_config(path: str) -> Dict[str, Any]:
    """Load configuration from a YAML file.

    Args:
        path: Path to the YAML configuration file.

    Returns:
        Dictionary with configuration values.
    """
    try:
        with open(path, "r", encoding="utf-8") as f:
            data = yaml.safe_load(f)
            return data if data else {}
    except FileNotFoundError:
        logger.warning("yaml_file_not_found", path=path)
        return {}
    except yaml.YAMLError as e:
        logger.error("yaml_parse_error", path=path, error=str(e))
        raise


def env_to_config(env: Dict[str, str]) -> Dict[str, Any]:
    """Convert environment variables to configuration dictionary.

    Expected format: `APP_NAME`, `DEBUG`, `DATABASE_URL`, etc.
    Keys are lowercased and underscores replaced with dots for nested keys.

    Args:
        env: Dictionary of environment variables.

    Returns:
        Dictionary with normalized config keys.
    """
    config = {}
    for key, value in env.items():
        if key.startswith("CONFIG_"):
            # e.g., CONFIG_DATABASE_URL -> database.url
            config_key = key.replace("CONFIG_", "").lower().replace("_", ".")
            config[config_key] = value
    return config


def parse_cli_args() -> Dict[str, Any]:
    """Parse command-line arguments into a dictionary.

    Returns:
        Dictionary of CLI-provided config values.
    """
    parser = argparse.ArgumentParser(description="Application configuration")
    parser.add_argument("--app-name", type=str, help="Application name")
    parser.add_argument("--debug", action="store_true", help="Enable debug mode")
    parser.add_argument("--log-level", type=str, choices=["DEBUG", "INFO", "WARNING", "ERROR"], help="Log level")
    parser.add_argument("--database-url", type=str, help="Database connection URL")
    parser.add_argument("--api-key", type=str, help="API key for external services")
    parser.add_argument("--timeout", type=int, help="Request timeout in seconds")
    parser.add_argument("--max-connections", type=int, help="Maximum concurrent connections")

    args = parser.parse_args()
    return {k.replace("_", "."): v for k, v in vars(args).items() if v is not None}


def merge_configs(*configs: Dict[str, Any]) -> Dict[str, Any]:
    """Deep merge configuration dictionaries with priority: last wins.

    Args:
        *configs: Variable number of config dictionaries.

    Returns:
        Merged configuration dictionary.
    """
    result: Dict[str, Any] = {}
    for config in configs:
        for key, value in config.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = merge_configs(result[key], value)
            else:
                result[key] = value
    return result


def normalize_value(key: str, value: Any) -> Any:
    """Convert string values to appropriate Python types (bool, int, etc.)."""
    if isinstance(value, str):
        if value.lower() in ("true", "false"):
            return value.lower() == "true"
        try:
            return int(value)
        except ValueError:
            pass
        try:
            return float(value)
        except ValueError:
            pass
    return value


def build_config(
    yaml_path: Optional[str] = None,
) -> Config:
    """Build final config from YAML file, environment variables, and CLI args.

    Priority (highest to lowest):
        1. CLI arguments
        2. Environment variables
        3. YAML config file
        4. Default values

    Args:
        yaml_path: Optional path to YAML config file.

    Returns:
        Config instance (Pydantic model).
    """
    # 1. Load YAML config (lowest priority)
    yaml_config = load_yaml_config(yaml_path) if yaml_path else {}

    # 2. Load environment variables (mid priority)
    env_config = env_to_config(os.environ)

    # 3. Parse CLI arguments (highest priority)
    cli_config = parse_cli_args()

    # 4. Merge with priority: CLI > env > yaml
    merged = merge_configs(yaml_config, env_config, cli_config)

    # Normalize types (e.g., "true" -> True)
    for key, value in merged.items():
        merged[key] = normalize_value(key, value)

    # Log final merged config for debugging
    logger.debug("merged_config", config=merged)

    # Validate and instantiate Pydantic model
    try:
        return Config(**merged)
    except ValidationError as e:
        logger.error("config_validation_error", error=str(e))
        raise


if __name__ == "__main__":
    # Example usage
    try:
        config = build_config(yaml_path="config.yaml")
        print("Final configuration:")
        print(config.model_dump())
    except Exception as e:
        logger.error("failed_to_load_config", error=str(e))
        raise
```

### Key Features:

1. **Priority Merging**:
   - CLI arguments > environment variables > YAML file > defaults.

2. **Pydantic Validation**:
   - Uses `Config` model with type hints and defaults.

3. **structlog Integration**:
   - Structured, colored logging for debugging.

4. **Environment Variable Support**:
   - Supports `CONFIG_*` environment variables (e.g., `CONFIG_DATABASE_URL`).

5. **Type Normalization**:
   - Converts string values to Python types (`"true"` → `True`, `"123"` → `123`).

6. **Deep Merge**:
   - Supports nested keys (e.g., `database.host`).

7. **Error Handling**:
   - Graceful handling of missing files, parsing errors, and validation failures.

### Example YAML (`config.yaml`):
```yaml
app_name: "ProductionApp"
debug: false
log_level: "WARNING"
database_url: "postgresql://user:pass@localhost/mydb"
```

### Usage:
```bash
# Set env vars
export CONFIG_DATABASE_URL="postgres://prod:secret@db.example.com/prod"
export CONFIG_TIMEOUT=60

# Run with CLI override
python config_loader.py --debug --log-level DEBUG
```

> **Note**: This implementation avoids secrets in logs—`api_key` is logged only if present, but in production you may want to mask sensitive fields like `api_key` in logs.
