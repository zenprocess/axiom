```python
"""Configuration loader that merges YAML files, environment variables, and CLI flags."""

import os
from pathlib import Path
from typing import Any, Dict, Optional, Type, TypeVar
import yaml

import structlog
from pydantic import BaseModel, BaseSettings, Field, ValidationError
from pydantic.env_settings import SettingsSourceConfig

logger = structlog.get_logger(__name__)

T = TypeVar("T", bound=BaseModel)


class ConfigLoader:
    """Loads configuration from multiple sources with priority merging.
    
    Priority order (highest to lowest):
    1. Command-line arguments (parsed separately and merged)
    2. Environment variables
    3. YAML configuration files
    """
    
    def __init__(self, config_path: Optional[str] = None):
        """Initialize the config loader.
        
        Args:
            config_path: Optional path to the primary YAML config file
        """
        self.config_path = config_path
        self._yaml_config: Optional[Dict[str, Any]] = None
    
    def load_yaml(self, path: Optional[str] = None) -> Dict[str, Any]:
        """Load configuration from a YAML file.
        
        Args:
            path: Path to YAML file. Uses instance config_path if not provided.
            
        Returns:
            Dictionary with configuration values
        """
        effective_path = path or self.config_path
        if not effective_path:
            return {}
            
        try:
            path_obj = Path(effective_path)
            if not path_obj.exists():
                logger.warning("config_file_not_found", path=str(path_obj))
                return {}
                
            with open(path_obj, "r") as f:
                config = yaml.safe_load(f)
                
            logger.info("config_loaded_from_yaml", path=str(path_obj))
            return config or {}
            
        except yaml.YAMLError as e:
            logger.error("yaml_parse_error", path=str(effective_path), error=str(e))
            raise
        except OSError as e:
            logger.error("file_read_error", path=str(effective_path), error=str(e))
            raise
    
    def load_env_vars(self, prefix: str = "") -> Dict[str, Any]:
        """Load configuration from environment variables.
        
        Args:
            prefix: Prefix to filter environment variables (e.g., "MYAPP_")
            
        Returns:
            Dictionary with environment variable values
        """
        env_config = {}
        
        # Process all environment variables
        for key, value in os.environ.items():
            if prefix and not key.startswith(prefix):
                continue
                
            # Clean up the key name
            clean_key = key[len(prefix):] if prefix else key
            # Convert to nested structure using dots as separators
            self._set_nested_value(env_config, clean_key.lower(), value)
            
        logger.info("config_loaded_from_env", prefix=prefix, count=len(env_config))
        return env_config
    
    def _set_nested_value(self, d: Dict[str, Any], key: str, value: Any) -> None:
        """Set a value in a nested dictionary using dot notation."""
        keys = key.split("__")  # Double underscore for nesting
        for k in keys[:-1]:
            if k not in d:
                d[k] = {}
            d = d[k]
        d[keys[-1]] = value
    
    def merge_configs(
        self, 
        *configs: Optional[Dict[str, Any]], 
        priority_order: str = "last_wins"
    ) -> Dict[str, Any]:
        """Merge multiple configuration dictionaries.
        
        Args:
            *configs: Configuration dictionaries to merge
            priority_order: "last_wins" (later configs override earlier ones)
            
        Returns:
            Merged configuration dictionary
        """
        if not configs:
            return {}
            
        result = {}
        
        for config in configs:
            if config is None:
                continue
                
            self._deep_merge(result, config, priority_order == "last_wins")
            
        return result
    
    def _deep_merge(
        self, 
        base: Dict[str, Any], 
        override: Dict[str, Any], 
        override_enabled: bool = True
    ) -> None:
        """Recursively merge two dictionaries."""
        for key, value in override.items():
            if key in base and isinstance(base[key], dict) and isinstance(value, dict):
                self._deep_merge(base[key], value, override_enabled)
            elif override_enabled or key not in base:
                base[key] = value
    
    def parse_cli_args(self, cli_args: Dict[str, Any]) -> Dict[str, Any]:
        """Parse command-line arguments into configuration.
        
        Args:
            cli_args: Dictionary of CLI argument names to values
            
        Returns:
            Configuration dictionary from CLI arguments
        """
        if not cli_args:
            return {}
            
        # Convert CLI args to nested structure (assuming dot notation in keys)
        cli_config = {}
        for key, value in cli_args.items():
            if value is not None:  # Only include explicitly set values
                self._set_nested_value(cli_config, key.lower(), value)
                
        logger.info("config_loaded_from_cli", count=len(cli_config))
        return cli_config
    
    def load_config(
        self,
        yaml_path: Optional[str] = None,
        env_prefix: str = "",
        cli_args: Optional[Dict[str, Any]] = None,
        model_class: Optional[Type[T]] = None
    ) -> T:
        """Load and validate configuration with Pydantic model.
        
        Args:
            yaml_path: Path to YAML config file
            env_prefix: Prefix for environment variables
            cli_args: Dictionary of CLI arguments
            model_class: Pydantic model class for validation
            
        Returns:
            Validated configuration model instance
        """
        # Load each source
        yaml_config = self.load_yaml(yaml_path)
        env_config = self.load_env_vars(env_prefix)
        cli_config = self.parse_cli_args(cli_args or {})
        
        # Merge with priority: CLI > env > yaml
        merged_config = self.merge_configs(
            yaml_config, 
            env_config, 
            cli_config,
            priority_order="last_wins"
        )
        
        # Log final config for debugging
        logger.info("final_config", config=merged_config)
        
        # Validate with Pydantic if model provided
        if model_class:
            try:
                return model_class(**merged_config)
            except ValidationError as e:
                logger.error("config_validation_error", error=str(e))
                raise
        else:
            return merged_config  # type: ignore


# Example Pydantic model for configuration
class AppConfig(BaseModel):
    """Application configuration model."""
    
    app_name: str = Field(default="MyApp", description="Application name")
    debug: bool = Field(default=False, description="Enable debug mode")
    database: Dict[str, Any] = Field(
        default_factory=dict, 
        description="Database configuration"
    )
    log_level: str = Field(default="INFO", description="Logging level")
    
    class Config:
        env_prefix = ""
        env_nested_delimiter = "__"


# Example usage function
def get_application_config(
    yaml_path: Optional[str] = "config.yaml",
    env_prefix: str = "MYAPP_",
    cli_args: Optional[Dict[str, Any]] = None,
    model_class: Type[AppConfig] = AppConfig
) -> AppConfig:
    """Get the application configuration with proper merging and validation.
    
    Args:
        yaml_path: Path to the YAML configuration file
        env_prefix: Prefix for environment variables
        cli_args: Command-line arguments as dictionary
        model_class: Pydantic model class to use for validation
        
    Returns:
        Validated application configuration
    """
    loader = ConfigLoader(yaml_path)
    return loader.load_config(
        yaml_path=yaml_path,
        env_prefix=env_prefix,
        cli_args=cli_args,
        model_class=model_class
    )
```

1. **Uses structlog** for all logging (no print or standard logging)
2. **Pydantic models** for configuration validation with proper type hints
3. **Proper error handling** with try/except blocks
4. **Priority merging** with CLI > env > YAML
5. **Deep merging** of nested dictionaries
6. **Comprehensive docstrings** for public methods
7. **Type hints** throughout the codebase
8. **No hardcoded secrets** - all sensitive values should come from env vars or secure sources
