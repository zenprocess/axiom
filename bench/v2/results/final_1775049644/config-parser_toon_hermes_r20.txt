import argparse
import os
from pathlib import Path
from typing import Any, Dict, Optional, Union

import yaml
from pydantic import BaseModel, Field, field_validator

import structlog


class Config(BaseModel):
    """Application configuration model."""
    app_name: str = Field(default="MyApp", description="Application name")
    debug: bool = Field(default=False, description="Enable debug mode")
    log_level: str = Field(default="INFO", description="Logging level")
    database_url: str = Field(default="sqlite:///app.db", description="Database connection URL")
    api_key: Optional[str] = Field(default=None, description="API key for external services")
    max_connections: int = Field(default=10, description="Maximum number of connections")
    timeout: float = Field(default=30.0, description="Request timeout in seconds")
    custom: Dict[str, Any] = Field(default_factory=dict, description="Custom configuration")


def load_yaml_config(path: Optional[Union[str, Path]] = None) -> Dict[str, Any]:
    """Load configuration from a YAML file.
    
    Args:
        path: Path to the YAML configuration file.
        
    Returns:
        Dictionary containing the parsed YAML configuration.
    """
    if path is None:
        return {}
    
    try:
        with open(path, 'r') as f:
            return yaml.safe_load(f) or {}
    except FileNotFoundError:
        structlog.get_logger().warning("Config file not found", path=path)
        return {}
    except yaml.YAMLError as e:
        structlog.get_logger().error("Failed to parse YAML config", path=path, error=str(e))
        return {}


def load_env_config(prefix: str = "APP_") -> Dict[str, Any]:
    """Load configuration from environment variables.
    
    Converts environment variables to a dictionary by:
    - Taking variables that start with the specified prefix
    - Removing the prefix
    - Converting to lowercase and replacing underscores with dots
    
    Args:
        prefix: Environment variable prefix to filter by.
        
    Returns:
        Dictionary of environment configuration.
    """
    config = {}
    for key, value in os.environ.items():
        if key.startswith(prefix):
            # Remove prefix and convert to lowercase with dots
            config_key = key[len(prefix):].lower().replace('__', '.')
            # Try to convert values to appropriate types
            try:
                if value.lower() in ('true', 'false'):
                    value = value.lower() == 'true'
                elif value.isdigit():
                    value = int(value)
                else:
                    try:
                        value = float(value)
                    except ValueError:
                        pass
            except AttributeError:
                pass
            
            # Handle nested keys (e.g., database__url)
            nested_set(config, config_key.split('__'), value)
    
    return config


def nested_set(d: Dict[str, Any], keys: list, value: Any) -> None:
    """Set a value in a nested dictionary using a list of keys."""
    for key in keys[:-1]:
        d = d.setdefault(key, {})
    d[keys[-1]] = value


def load_cli_config(args: Optional[ argparse.Namespace ] = None) -> Dict[str, Any]:
    """Load configuration from command-line arguments.
    
    Args:
        args: Parsed arguments namespace. If None, parses sys.argv.
        
    Returns:
        Dictionary of command-line configuration.
    """
    parser = argparse.ArgumentParser(description="Application configuration")
    parser.add_argument("--app-name", type=str, help="Application name")
    parser.add_argument("--debug", action="store_true", help="Enable debug mode")
    parser.add_argument("--log-level", type=str, help="Logging level (DEBUG, INFO, WARNING, ERROR)")
    parser.add_argument("--database-url", type=str, help="Database connection URL")
    parser.add_argument("--api-key", type=str, help="API key for external services")
    parser.add_argument("--max-connections", type=int, help="Maximum number of connections")
    parser.add_argument("--timeout", type=float, help="Request timeout in seconds")
    parser.add_argument("--config", type=str, help="Path to YAML config file")
    
    parsed_args = parser.parse_args() if args is None else args
    
    # Convert parsed args to dict, excluding None values
    config = {
        k.replace("-", "_"): v 
        for k, v in vars(parsed_args).items() 
        if v is not None
    }
    
    # Handle --config flag separately to load YAML after CLI args
    if hasattr(parsed_args, 'config') and parsed_args.config:
        yaml_config = load_yaml_config(parsed_args.config)
        # CLI args should override YAML config, so we apply YAML first, then CLI
        return {**yaml_config, **config}
    
    return config


def merge_configs(*configs: Dict[str, Any]) -> Dict[str, Any]:
    """Deep merge multiple configuration dictionaries.
    
    Args:
        *configs: Configuration dictionaries to merge.
        
    Returns:
        Merged configuration dictionary with later configs taking precedence.
    """
    result = {}
    for config in configs:
        for key, value in config.items():
            if key in result and isinstance(result[key], dict) and isinstance(value, dict):
                result[key] = merge_configs(result[key], value)
            else:
                result[key] = value
    return result


def load_config(
    yaml_path: Optional[str] = None,
    cli_args: Optional[argparse.Namespace] = None,
    env_prefix: str = "APP_"
) -> Config:
    """Load and merge configuration from YAML file, environment variables, and CLI arguments.
    
    Priority (lowest to highest): YAML file < environment variables < CLI arguments
    
    Args:
        yaml_path: Path to YAML configuration file.
        cli_args: Pre-parsed CLI arguments.
        env_prefix: Environment variable prefix to filter by.
        
    Returns:
        Pydantic Config model with merged configuration.
    """
    # Load configurations from each source
    yaml_config = load_yaml_config(yaml_path)
    env_config = load_env_config(env_prefix)
    cli_config = load_cli_config(cli_args)
    
    # Merge configurations with proper priority
    merged_config = merge_configs(yaml_config, env_config, cli_config)
    
    # Create and return validated Config model
    return Config(**merged_config)

   - CLI arguments with `load_cli_config()`
3. **Proper Merging**: Deep merge with priority (YAML < env vars < CLI args)
4. **Debug Output**: Uses structlog for logging the final configuration
