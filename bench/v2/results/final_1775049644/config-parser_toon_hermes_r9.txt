```python
# config_loader.py
from __future__ import annotations

import os
import sys
from pathlib import Path
from typing import Any, Dict, Optional, Type, TypeVar

import yaml
import structlog
from pydantic import BaseModel, ValidationError

# Configure structlog for clean structured logging
structlog.configure(
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.format_exc_info,
        structlog.dev.ConsoleRenderer(colors=True),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.WARNING),  # default level
)
logger = structlog.get_logger(__name__)

T = TypeVar("T", bound=BaseModel)


def load_config(
    config_path: Optional[str] = None,
    model_class: Type[T] = None,
) -> T:
    """
    Load configuration from YAML file, environment variables, and CLI flags,
    merging with priority: CLI > env vars > YAML > defaults.

    Args:
        config_path: Optional path to YAML config file.
        model_class: Pydantic model class defining expected config schema.

    Returns:
        Validated and merged config instance.

    Raises:
        ValueError: If no model class is provided or config is invalid.
        FileNotFoundError: If YAML file is specified but not found.
    """
    if model_class is None:
        raise ValueError("model_class must be provided to define config schema.")

    try:
        # 1. Load defaults from model
        config_dict = model_class.model_dump()  # defaults only
        logger.info("loaded_default_config", defaults=config_dict)

        # 2. Load from YAML file if path provided
        if config_path:
            path = Path(config_path)
            if not path.is_absolute():
                path = Path.cwd() / path
            if not path.exists():
                raise FileNotFoundError(f"Config file not found: {path}")
            with path.open("r", encoding="utf-8") as f:
                yaml_data = yaml.safe_load(f) or {}
            config_dict = _deep_merge(config_dict, yaml_data)
            logger.info("loaded_yaml_config", file=str(path), config=config_dict)

        # 3. Override with environment variables (prefix: CONFIG_)
        env_dict = _load_env_vars(model_class, prefix="CONFIG_")
        config_dict = _deep_merge(config_dict, env_dict)
        logger.info("loaded_env_config", config=config_dict)

        # 4. Override with CLI flags (e.g., --config.db.host=localhost)
        cli_dict = _load_cli_flags(model_class)
        config_dict = _deep_merge(config_dict, cli_dict)
        logger.info("loaded_cli_config", config=config_dict)

        # 5. Validate and instantiate final config
        final_config = model_class.model_validate(config_dict)
        logger.info("final_config_validated", config=final_config.model_dump())
        return final_config

    except ValidationError as e:
        logger.error("config_validation_error", errors=e.errors(), exc_info=True)
        raise
    except Exception as e:
        logger.error("config_loading_failed", error=str(e), exc_info=True)
        raise


def _load_env_vars(model_class: Type[BaseModel], prefix: str = "CONFIG_") -> Dict[str, Any]:
    """Load config from environment variables matching model field names (with optional prefix)."""
    env_config = {}
    for field_name in model_class.model_fields:
        env_key = f"{prefix}{field_name.upper()}"
        if env_key in os.environ:
            # For nested models, handle via dot notation later if needed
            env_config[field_name] = os.environ[env_key]
        # For nested models, you could recursively inspect sub-models
        # This is a simplified version assuming flat fields or flat nesting
    return env_config


def _load_cli_flags(model_class: Type[BaseModel]) -> Dict[str, Any]:
    """Parse CLI flags like --config.field=value or --field=value."""
    import argparse

    parser = argparse.ArgumentParser(add_help=False)
    # Collect all fields with nested support (flat for now)
    for field_name in model_class.model_fields:
        parser.add_argument(f"--{field_name}", dest=field_name, default=None)

    args, _ = parser.parse_known_args()
    cli_dict = {k: v for k, v in vars(args).items() if v is not None}
    return cli_dict


def _deep_merge(base: Dict[str, Any], override: Dict[str, Any]) -> Dict[str, Any]:
    """Deep merge `override` into `base`, with `override` taking precedence."""
    result = base.copy()
    for key, value in override.items():
        if (
            key in result
            and isinstance(result[key], dict)
            and isinstance(value, dict)
        ):
            result[key] = _deep_merge(result[key], value)
        else:
            result[key] = value
    return result


# Example usage
if __name__ == "__main__":
    from pydantic import Field

    class DatabaseConfig(BaseModel):
        host: str = "localhost"
        port: int = 5432
        name: str = "app_db"

    class Config(BaseModel):
        env: str = "development"
        debug: bool = False
        db: DatabaseConfig = Field(default_factory=DatabaseConfig)

    # Example CLI: python config_loader.py --config.db.port=5433 --debug=true
    cfg = load_config("config.yaml", model_class=Config)
    print(cfg.model_dump())  # for demo only; in real app, avoid print()
```

### 🧪 Example Usage

**1. `config.yaml`:**
```yaml
env: production
debug: false
db:
  host: prod-db.example.com
```

**2. Run:**
```bash
CONFIG_DB_PORT=5433 python config_loader.py --db.port=5433
```

**3. Output (via structlog):**
