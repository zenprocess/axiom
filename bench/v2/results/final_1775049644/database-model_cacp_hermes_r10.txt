```python
"""
User management data models and database fetching utilities.

This module provides Pydantic models for User, Role, and Permission,
along with async functions to fetch users from a database URL.
"""

import re
from datetime import datetime
from typing import List, Optional
from urllib.parse import urlparse

import structlog
from pydantic import BaseModel, EmailStr, Field, field_validator

logger = structlog.get_logger(__name__)


class Permission(BaseModel):
    """
    Represents a permission in the system.

    Attributes:
        id (int): Unique identifier for the permission.
        name (str): Name of the permission (e.g., "read_users").
        description (Optional[str]): Description of what the permission allows.
    """
    id: int = Field(..., gt=0)
    name: str = Field(..., min_length=1, max_length=100)
    description: Optional[str] = Field(default=None, max_length=255)


class Role(BaseModel):
    """
    Represents a role in the system.

    Attributes:
        id (int): Unique identifier for the role.
        name (str): Name of the role (e.g., "admin").
        permissions (List[Permission]): List of permissions associated with the role.
    """
    id: int = Field(..., gt=0)
    name: str = Field(..., min_length=1, max_length=50)
    permissions: List[Permission] = Field(default_factory=list)

    @field_validator("name")
    @classmethod
    def validate_role_name(cls, v: str) -> str:
        """Validate role name format (alphanumeric with underscores/hyphens)."""
        if not re.match(r"^[a-zA-Z][a-zA-Z0-9_-]*$", v):
            raise ValueError(
                "Role name must start with a letter and contain only "
                "alphanumeric characters, underscores, or hyphens"
            )
        return v


class User(BaseModel):
    """
    Represents a user in the system.

    Attributes:
        id (int): Unique identifier for the user.
        email (EmailStr): User's email address.
        username (str): User's username.
        role_id (int): ID of the user's primary role.
        is_active (bool): Whether the user account is active.
        created_at (datetime): When the user account was created.
        updated_at (datetime): When the user account was last updated.
    """
    id: int = Field(..., gt=0)
    email: EmailStr
    username: str = Field(..., min_length=1, max_length=50)
    role_id: int = Field(..., gt=0)
    is_active: bool = True
    created_at: datetime
    updated_at: datetime

    @field_validator("username")
    @classmethod
    def validate_username(cls, v: str) -> str:
        """Validate username format (alphanumeric with underscores/hyphens, 3-50 chars)."""
        if len(v) < 3:
            raise ValueError("Username must be at least 3 characters long")
        if not re.match(r"^[a-zA-Z][a-zA-Z0-9_-]*$", v):
            raise ValueError(
                "Username must start with a letter and contain only "
                "alphanumeric characters, underscores, or hyphens"
            )
        return v


class UserListResponse(BaseModel):
    """
    Response model for a list of users.

    Attributes:
        users (List[User]): List of users.
        total (int): Total number of users.
    """
    users: List[User]
    total: int = Field(..., ge=0)


async def fetch_users(database_url: str) -> UserListResponse:
    """
    Fetch users from a database URL.

    Args:
        database_url (str): Database connection URL (e.g., "postgres://user:pass@host:5432/db").

    Returns:
        UserListResponse: Response containing the list of users and total count.

    Raises:
        ValueError: If the database URL is invalid or not supported.
        RuntimeError: If there's an error connecting to the database or fetching data.
    """
    # Validate database URL
    try:
        parsed = urlparse(database_url)
        if not parsed.scheme or not parsed.netloc:
            raise ValueError("Invalid database URL format")
    except Exception as exc:
        logger.error("database_url_validation_failed", error=str(exc))
        raise ValueError("Invalid database URL format") from exc

    # Mock implementation - in production, this would use async database driver
    try:
        # Simulate database operation
        users = [
            User(
                id=1,
                email="admin@example.com",
                username="admin",
                role_id=1,
                is_active=True,
                created_at=datetime(2023, 1, 1),
                updated_at=datetime(2023, 1, 1),
            ),
            User(
                id=2,
                email="user@example.com",
                username="john_doe",
                role_id=2,
                is_active=True,
                created_at=datetime(2023, 2, 15),
                updated_at=datetime(2023, 3, 1),
            ),
        ]
        total = len(users)

        logger.info("users_fetched", count=total, database_url=database_url)
        return UserListResponse(users=users, total=total)

    except Exception as exc:
        logger.error("database_fetch_failed", error=str(exc))
        raise RuntimeError("Failed to fetch users from database") from exc
```
