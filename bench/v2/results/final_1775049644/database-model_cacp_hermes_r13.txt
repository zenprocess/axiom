```python
"""User management system data models and database integration."""

from __future__ import annotations

from datetime import datetime
from typing import List, Optional
import asyncio
import aiohttp
import structlog
from pydantic import BaseModel, EmailStr, Field, validator

logger = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission in the system."""
    
    id: int
    name: str = Field(..., min_length=1, max_length=100)
    description: Optional[str] = Field(default=None, max_length=255)

    class Config:
        orm_mode = True


class Role(BaseModel):
    """Represents a role in the system."""
    
    id: int
    name: str = Field(..., min_length=1, max_length=50)
    description: Optional[str] = Field(default=None, max_length=255)
    permissions: List[Permission] = Field(default_factory=list)

    class Config:
        orm_mode = True


class User(BaseModel):
    """Represents a user in the system."""
    
    id: int
    username: str = Field(..., min_length=3, max_length=30)
    email: EmailStr
    first_name: Optional[str] = Field(default=None, max_length=50)
    last_name: Optional[str] = Field(default=None, max_length=50)
    is_active: bool = Field(default=True)
    role: Optional[Role] = None
    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: Optional[datetime] = None

    @validator("updated_at")
    def validate_updated_at(cls, v: Optional[datetime], values: dict) -> Optional[datetime]:
        """Ensure updated_at is not before created_at."""
        created_at = values.get("created_at")
        if v is not None and created_at is not None and v < created_at:
            raise ValueError("updated_at cannot be before created_at")
        return v

    def to_dict(self) -> dict:
        """Serialize user to dictionary."""
        return self.dict(exclude_unset=True)

    class Config:
        orm_mode = True


async def fetch_users(database_url: str) -> List[User]:
    """
    Fetch users from a database URL endpoint.

    Args:
        database_url: The URL to fetch users from.

    Returns:
        A list of User objects.

    Raises:
        RuntimeError: If the request fails or the response is invalid.
    """
    try:
        async with aiohttp.ClientSession() as session:
            async with session.get(database_url) as response:
                response.raise_for_status()
                data = await response.json()
                
                if not isinstance(data, list):
                    raise ValueError("Expected a list of users in response")
                
                users = [User.parse_obj(user_data) for user_data in data]
                logger.info("users_fetched", count=len(users))
                return users
                
    except aiohttp.ClientError as exc:
        logger.error("http_error", error=str(exc))
        raise RuntimeError(f"HTTP error occurred while fetching users: {exc}") from exc
    except ValueError as exc:
        logger.error("validation_error", error=str(exc))
        raise RuntimeError(f"Validation error while parsing users: {exc}") from exc
    except Exception as exc:
        logger.error("unexpected_error", error=str(exc))
        raise RuntimeError(f"Unexpected error while fetching users: {exc}") from exc
```
