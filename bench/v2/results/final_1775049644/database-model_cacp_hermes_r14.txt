"""User management system data models and database utilities."""

from datetime import datetime
from typing import List, Optional
import urllib.parse

import structlog
from pydantic import BaseModel, EmailStr, Field, validator, root_validator

logger = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission in the system."""
    name: str = Field(..., min_length=1, max_length=50)
    description: Optional[str] = Field(None, max_length=255)

    class Config:
        extra = "forbid"
        validate_assignment = True


class Role(BaseModel):
    """Represents a role in the system."""
    id: int = Field(..., gt=0)
    name: str = Field(..., min_length=1, max_length=50)
    description: Optional[str] = Field(None, max_length=255)
    permissions: List[Permission] = Field(default_factory=list)

    @validator("name")
    def name_must_be_alphanumeric(cls, v: str) -> str:
        """Ensure role name contains only alphanumeric characters and underscores."""
        if not v.replace("_", "").isalnum():
            raise ValueError("Role name must contain only alphanumeric characters and underscores")
        return v

    class Config:
        extra = "forbid"
        validate_assignment = True


class User(BaseModel):
    """Represents a user in the system."""
    id: int = Field(..., gt=0)
    email: EmailStr
    username: str = Field(..., min_length=3, max_length=30)
    full_name: Optional[str] = Field(None, max_length=100)
    role: Role
    created_at: Optional[datetime] = Field(default_factory=datetime.utcnow)
    is_active: bool = True
    permissions_override: Optional[List[Permission]] = Field(default=None)

    @validator("username")
    def username_must_be_alphanumeric(cls, v: str) -> str:
        """Ensure username contains only alphanumeric characters and underscores."""
        if not v.replace("_", "").isalnum():
            raise ValueError("Username must contain only alphanumeric characters and underscores")
        return v

    @root_validator
    def validate_user_permissions(cls, values: dict) -> dict:
        """Validate that overridden permissions don't conflict with role permissions."""
        role = values.get("role")
        override = values.get("permissions_override")
        
        if override and role:
            role_perms = {p.name for p in role.permissions}
            override_names = {p.name for p in override}
            
            if not override_names.issubset(role_perms):
                logger.warning(
                    "user_permissions_validation_failed",
                    user_id=values.get("id"),
                    invalid_permissions=list(override_names - role_perms)
                )
                raise ValueError("Override permissions must be subset of role permissions")
        
        return values

    class Config:
        extra = "forbid"
        validate_assignment = True
