```python
"""User management system data models and database interaction functions."""

from typing import List, Optional
from datetime import datetime
from pydantic import BaseModel, EmailStr, validator, Field
import structlog
import httpx

logger = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission in the system."""
    id: int
    name: str = Field(..., min_length=1, max_length=100)
    description: Optional[str] = Field(default=None, max_length=255)

    class Config:
        extra = "forbid"


class Role(BaseModel):
    """Represents a role in the system."""
    id: int
    name: str = Field(..., min_length=1, max_length=50)
    permissions: List[Permission] = Field(default_factory=list)
    description: Optional[str] = Field(default=None, max_length=255)

    @validator("permissions")
    def validate_permissions(cls, v):
        """Ensure all permissions have unique names."""
        permission_names = [p.name for p in v]
        if len(permission_names) != len(set(permission_names)):
            raise ValueError("Permission names must be unique within a role")
        return v

    class Config:
        extra = "forbid"


class User(BaseModel):
    """Represents a user in the system."""
    id: int
    username: str = Field(..., min_length=3, max_length=50)
    email: EmailStr
    role: Role
    is_active: bool = True
    created_at: datetime
    updated_at: Optional[datetime] = None

    @validator("updated_at")
    def validate_timestamps(cls, v, values):
        """Ensure updated_at is not earlier than created_at if both are present."""
        if v is not None and "created_at" in values and v < values["created_at"]:
            raise ValueError("updated_at cannot be earlier than created_at")
        return v

    def to_dict(self) -> dict:
        """Serialize the user model to a dictionary."""
        return self.dict()

    class Config:
        extra = "forbid"
        json_encoders = {
            datetime: lambda dt: dt.isoformat()
        }


async def fetch_users_from_db(database_url: str) -> List[User]:
    """
    Fetch users from a database API endpoint.

    Args:
        database_url: URL of the database API endpoint returning user data

    Returns:
        List of User models

    Raises:
        RuntimeError: If the request fails or the response is invalid
    """
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(database_url)
            response.raise_for_status()
            data = response.json()
            
            if not isinstance(data, list):
                raise ValueError("Expected a list of users in the response")
            
            return [User.parse_obj(user_data) for user_data in data]
    except httpx.HTTPError as exc:
        logger.error("http_error", url=database_url, error=str(exc))
        raise RuntimeError(f"Failed to fetch users: {exc}") from exc
    except ValueError as exc:
        logger.error("validation_error", error=str(exc))
        raise RuntimeError(f"Invalid user data received: {exc}") from exc
```
