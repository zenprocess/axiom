"""
User management data models and database fetching utilities.

This module provides Pydantic models for User, Role, and Permission entities,
along with validation, serialization, and async database fetching functionality.
"""

import asyncio
from typing import List, Optional
from datetime import datetime
from urllib.parse import urlparse
import structlog

from pydantic import BaseModel, Field, validator, root_validator, EmailStr, HttpUrl

logger = structlog.get_logger(__name__)


class Permission(BaseModel):
    """
    Represents a permission in the system.
    
    Attributes:
        id: Unique identifier for the permission
        name: Name of the permission (e.g., "read_users", "write_settings")
        description: Description of what the permission allows
    """
    id: int = Field(..., ge=1)
    name: str = Field(..., min_length=1, max_length=100)
    description: Optional[str] = Field(None, max_length=255)

    class Config:
        orm_mode = True


class Role(BaseModel):
    """
    Represents a role in the system.
    
    Attributes:
        id: Unique identifier for the role
        name: Name of the role (e.g., "admin", "editor")
        description: Description of the role's responsibilities
        permissions: List of permissions assigned to this role
    """
    id: int = Field(..., ge=1)
    name: str = Field(..., min_length=1, max_length=100)
    description: Optional[str] = Field(None, max_length=255)
    permissions: List[Permission] = Field(default_factory=list)

    class Config:
        orm_mode = True


class User(BaseModel):
    """
    Represents a user in the system.
    
    Attributes:
        id: Unique identifier for the user
        email: User's email address (validated)
        username: User's username
        full_name: User's full name
        role: Role assigned to the user
        is_active: Whether the user account is active
        created_at: When the user was created
        updated_at: When the user was last updated
    """
    id: int = Field(..., ge=1)
    email: EmailStr
    username: str = Field(..., min_length=3, max_length=50)
    full_name: Optional[str] = Field(None, max_length=100)
    role: Role
    is_active: bool = True
    created_at: datetime
    updated_at: datetime

    @validator("username")
    def username_must_be_alphanumeric(cls, v):
        """Ensure username contains only alphanumeric characters and underscores."""
        if not v.replace("_", "").isalnum():
            raise ValueError("Username must contain only alphanumeric characters and underscores")
        return v

    @root_validator(skip_on_failure=True)
    def check_dates(cls, values):
        """Ensure updated_at is not before created_at."""
        created = values.get("created_at")
        updated = values.get("updated_at")
        if created and updated and updated < created:
            raise ValueError("updated_at cannot be before created_at")
        return values

    class Config:
        orm_mode = True
