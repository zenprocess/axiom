"""User management system data models and database interaction utilities."""

from datetime import datetime
from typing import List, Optional
from urllib.parse import urlparse

import structlog
from pydantic import BaseModel, EmailStr, Field, field_validator, model_validator

logger = structlog.get_logger(__name__)


class Permission(BaseModel):
    """Represents a permission that can be granted to roles."""
    id: int
    name: str = Field(..., min_length=1, max_length=50)
    description: Optional[str] = Field(default=None, max_length=200)


class Role(BaseModel):
    """Represents a role that can be assigned to users."""
    id: int
    name: str = Field(..., min_length=1, max_length=50)
    description: Optional[str] = Field(default=None, max_length=200)
    permissions: List[Permission] = Field(default_factory=list)

    @field_validator("name")
    @classmethod
    def validate_name(cls, v: str) -> str:
        """Ensure role name is not a reserved name."""
        reserved_names = {"admin", "superuser", "system"}
        if v.lower() in reserved_names:
            raise ValueError(f"Role name '{v}' is reserved and cannot be used")
        return v


class User(BaseModel):
    """Represents a user in the system."""
    id: int
    username: str = Field(..., min_length=3, max_length=30)
    email: EmailStr
    role_id: int
    is_active: bool = True
    created_at: datetime
    updated_at: Optional[datetime] = None

    # Optional nested role for convenience
    role: Optional[Role] = None

    @field_validator("username")
    @classmethod
    def validate_username(cls, v: str) -> str:
        """Ensure username contains only alphanumeric characters and underscores."""
        if not v.replace("_", "").isalnum():
            raise ValueError("Username must contain only alphanumeric characters and underscores")
        return v

    @model_validator(mode="after")
    def validate_updated_at(self) -> "User":
        """Ensure updated_at is not before created_at."""
        if self.updated_at is not None and self.updated_at < self.created_at:
            raise ValueError("updated_at cannot be earlier than created_at")
        return self

    def to_dict(self) -> dict:
        """Serialize the user model to a dictionary."""
        return self.model_dump(mode="json")

    @classmethod
    def from_dict(cls, data: dict) -> "User":
        """Deserialize a dictionary to a User model."""
        return cls.model_validate(data)


async def fetch_users_from_db(database_url: str) -> List[User]:
    """
    Fetch users from a database using the provided database URL.

    Args:
        database_url: Database connection string (e.g., "postgresql://user:pass@host/db")

    Returns:
        List of User models fetched from the database

    Raises:
        ValueError: If database URL is invalid or connection fails
        RuntimeError: If database connection or query execution fails
    """
    # Validate database URL format
    try:
        parsed = urlparse(database_url)
        if not all([parsed.scheme, parsed.netloc]):
            raise ValueError("Invalid database URL: missing scheme or netloc")
    except Exception as exc:
        logger.error("url_validation_failed", error=str(exc))
        raise ValueError(f"Invalid database URL format: {database_url}") from exc

    # Simulate database connection and query execution
    try:
        # In a real implementation, this would use an async database driver
        # For now, we'll simulate successful connection and return sample data
        sample_users = [
            User(
                id=1,
                username="john_doe",
                email="john.doe@example.com",
                role_id=1,
                is_active=True,
                created_at=datetime(2023, 1, 15, 10, 30, 0),
                updated_at=datetime(2023, 6, 20, 14, 45, 0),
            ),
            User(
                id=2,
                username="jane_smith",
                email="jane.smith@example.org",
                role_id=2,
                is_active=True,
                created_at=datetime(2023, 3, 10, 9, 15, 0),
            )
        ]
        
        logger.info("users_fetched", count=len(sample_users), url=database_url)
        return sample_users
        
    except Exception as exc:
        logger.error("database_connection_failed", error=str(exc))
        raise RuntimeError(f"Failed to fetch users from database: {exc}") from exc
