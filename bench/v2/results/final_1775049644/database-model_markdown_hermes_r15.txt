"""User management system data models and database utilities."""

import os
from typing import Optional

import structlog
from pydantic import BaseModel, EmailStr, validator, root_validator

log = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission that can be assigned to roles."""
    id: int
    name: str
    description: Optional[str] = None

    class Config:
        orm_mode = True


class Role(BaseModel):
    """Represents a role in the system with associated permissions."""
    id: int
    name: str
    description: Optional[str] = None
    permissions: list[Permission] = []

    class Config:
        orm_mode = True

    @validator('name')
    def name_must_not_be_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError('Role name cannot be empty or whitespace only')
        return v.strip()


class User(BaseModel):
    """Represents a user in the system."""
    id: int
    email: EmailStr
    username: str
    full_name: Optional[str] = None
    role_id: Optional[int] = None
    is_active: bool = True
    is_admin: bool = False

    # Nested models
    role: Optional[Role] = None

    class Config:
        orm_mode = True

    @validator('username')
    def username_must_not_be_empty(cls, v: str) -> str:
        if not v.strip():
            raise ValueError('Username cannot be empty or whitespace only')
        return v.strip()

    @validator('full_name')
    def full_name_must_not_be_empty(cls, v: Optional[str]) -> Optional[str]:
        if v is not None and not v.strip():
            return None
        return v

    @root_validator
    def admin_requires_admin_role(cls, values):
        if values.get('is_admin') and not values.get('role_id'):
            raise ValueError('Admin users must have a role assigned')
        return values
