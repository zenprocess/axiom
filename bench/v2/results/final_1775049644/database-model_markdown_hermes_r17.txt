class RoleName(Enum):
    """Enumeration of system roles."""
    ADMIN = "admin"
    EDITOR = "editor"
    VIEWER = "viewer"


class PermissionName(Enum):
    """Enumeration of system permissions."""
    CREATE_USER = "create_user"
    EDIT_USER = "edit_user"
    DELETE_USER = "delete_user"
    VIEW_USERS = "view_users"
    MANAGE_ROLES = "manage_roles"
    MANAGE_PERMISSIONS = "manage_permissions"


class Permission(BaseModel):
    """Represents a system permission."""
    name: PermissionName = Field(..., description="The permission name")
    description: str = Field("", description="Human-readable description of the permission")

    class Config:
        use_enum_values = True

    def __str__(self) -> str:
        return f"Permission(name={self.name.value!r}, description={self.description!r})"


class Role(BaseModel):
    """Represents a system role with associated permissions."""
    id: int = Field(..., ge=1, description="Unique role identifier")
    name: RoleName = Field(..., description="Role name")
    description: str = Field("", description="Human-readable description of the role")
    permissions: List[Permission] = Field(default_factory=list, description="List of permissions granted to this role")

    class Config:
        use_enum_values = True
        validate_assignment = True

    @validator("permissions", pre=True)
    def ensure_permissions_list(cls, v) -> List[Permission]:
        """Ensure permissions is always a list of Permission objects."""
        if v is None:
            return []
        if isinstance(v, list):
            return [p if isinstance(p, Permission) else Permission(**p) if isinstance(p, dict) else p for p in v]
        raise ValueError("permissions must be a list")

    def __str__(self) -> str:
        return f"Role(id={self.id}, name={self.name.value!r}, description={self.description!r})"
