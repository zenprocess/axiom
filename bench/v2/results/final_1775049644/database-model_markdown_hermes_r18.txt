"""User management system data models and database operations."""

from __future__ import annotations

import os
from datetime import datetime
from typing import List, Optional

import structlog
from pydantic import BaseModel, EmailStr, Field, validator

log = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission in the system."""
    
    id: int = Field(..., description="Unique identifier for the permission")
    name: str = Field(..., min_length=1, max_length=100, description="Name of the permission")
    description: Optional[str] = Field(None, description="Description of what the permission allows")
    
    class Config:
        orm_mode = True


class Role(BaseModel):
    """Represents a role in the system."""
    
    id: int = Field(..., description="Unique identifier for the role")
    name: str = Field(..., min_length=1, max_length=50, description="Name of the role")
    description: Optional[str] = Field(None, description="Description of the role")
    permissions: List[Permission] = Field(default_factory=list, description="List of permissions assigned to this role")
    
    @validator('name')
    def name_must_not_be_reserved(cls, v: str) -> str:
        """Ensure role name is not a reserved name."""
        reserved_names = {'admin', 'superuser', 'root'}
        if v.lower() in reserved_names:
            raise ValueError(f'Role name "{v}" is reserved')
        return v
    
    class Config:
        orm_mode = True


class User(BaseModel):
    """Represents a user in the system."""
    
    id: int = Field(..., description="Unique identifier for the user")
    email: EmailStr = Field(..., description="User's email address")
    username: str = Field(..., min_length=3, max_length=30, description="User's username")
    full_name: Optional[str] = Field(None, description="User's full name")
    role: Role = Field(..., description="User's primary role")
    is_active: bool = Field(default=True, description="Whether the user account is active")
    created_at: datetime = Field(..., description="When the user account was created")
    updated_at: Optional[datetime] = Field(None, description="When the user account was last updated")
    
    @validator('username')
    def username_must_be_alphanumeric(cls, v: str) -> str:
        """Ensure username contains only alphanumeric characters and underscores."""
        if not v.replace('_', '').isalnum():
            raise ValueError('Username must contain only alphanumeric characters and underscores')
        return v
    
    def to_dict(self) -> dict:
        """Serialize user to dictionary."""
        return self.dict(exclude={'id'})
    
    class Config:
        orm_mode = True

async def fetch_users_from_api(api_url: str) -> List[User]:
    """
    Fetch users from an API endpoint.
    
    Args:
        api_url: URL of the users API endpoint
        
    Returns:
        List of User models fetched from the API
        
    Raises:
        RuntimeError: If API call fails or response parsing fails
    """
    try:
        import httpx
        
        async with httpx.AsyncClient() as client:
            response = await client.get(api_url, timeout=30.0)
            response.raise_for_status()
            
            users_data = response.json()
            
            # Parse the response into User models
            users = [User(**user_data) for user_data in users_data]
            
            log.info('users_fetched', count=len(users), api_url=api_url)
            return users
            
    except httpx.HTTPError as e:
        log.error('api_call_failed', error=str(e), api_url=api_url)
        raise RuntimeError(f"Failed to fetch users from API: {str(e)}") from e
    except Exception as e:
        log.error('response_parsing_failed', error=str(e), api_url=api_url)
        raise RuntimeError(f"Failed to parse user data from API response: {str(e)}") from e
