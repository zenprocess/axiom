"""User management data models with validation and serialization."""

from typing import List, Optional
from pydantic import BaseModel, Field, validator, conint
import structlog
import httpx

log = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission in the system."""
    
    id: int = Field(..., description="Permission ID")
    name: str = Field(..., min_length=1, max_length=100, description="Permission name")
    description: Optional[str] = Field(None, description="Permission description")
    
    class Config:
        extra = "forbid"
        schema_extra = {
            "example": {
                "id": 1,
                "name": "read_users",
                "description": "Ability to read user information"
            }
        }


class Role(BaseModel):
    """Represents a role with associated permissions."""
    
    id: int = Field(..., description="Role ID")
    name: str = Field(..., min_length=1, max_length=50, description="Role name")
    permissions: List[Permission] = Field(default_factory=list, description="Permissions assigned to this role")
    
    @validator("name")
    def name_must_not_be_empty(cls, v):
        if not v.strip():
            raise ValueError("Role name cannot be empty or whitespace")
        return v.strip()
    
    class Config:
        extra = "forbid"
        schema_extra = {
            "example": {
                "id": 1,
                "name": "admin",
                "permissions": [
                    {"id": 1, "name": "read_users", "description": "Ability to read user information"}
                ]
            }
        }


class User(BaseModel):
    """Represents a user in the system."""
    
    id: int = Field(..., description="User ID")
    username: str = Field(..., min_length=3, max_length=50, description="Unique username")
    email: str = Field(..., description="User email address")
    role_id: int = Field(..., description="ID of the user's role")
    is_active: bool = Field(True, description="Whether the user is active")
    created_at: Optional[str] = Field(None, description="ISO 8601 timestamp of user creation")
    roles: Optional[List[Role]] = Field(None, description="Roles assigned to the user (for nested responses)")
    
    @validator("email")
    def email_must_be_valid(cls, v):
        if "@" not in v or "." not in v.split("@")[-1]:
            raise ValueError("Invalid email format")
        return v
    
    @validator("username")
    def username_must_be_valid(cls, v):
        if not v.replace("_", "").isalnum():
            raise ValueError("Username must contain only alphanumeric characters and underscores")
        return v
    
    class Config:
        extra = "forbid"
        schema_extra = {
            "example": {
                "id": 1,
                "username": "alice",
                "email": "alice@example.com",
                "role_id": 1,
                "is_active": True,
                "created_at": "2023-01-15T10:30:00Z"
            }
        }


async def fetch_users(database_url: str) -> List[User]:
    """
    Fetch users from the specified database URL.
    
    Args:
        database_url: The URL endpoint to fetch users from
        
    Returns:
        List[User]: List of users fetched from the database
        
    Raises:
        httpx.HTTPError: If the HTTP request fails
        ValueError: If the response data is invalid
    """
    log.info("fetching_users", database_url=database_url)
    
    try:
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.get(database_url)
            response.raise_for_status()
            
            users_data = response.json()
            users = [User(**user_data) for user_data in users_data]
            
            log.info("users_fetched", count=len(users))
            return users
            
    except httpx.HTTPError as e:
        log.error("http_error", url=database_url, error=str(e))
        raise
    except ValueError as e:
        log.error("invalid_data", error=str(e))
        raise
    except Exception as e:
        log.error("unexpected_error", error=str(e))
        raise
