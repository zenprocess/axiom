import os
from typing import List, Optional
from datetime import datetime
from enum import Enum

import structlog
from pydantic import BaseModel, EmailStr, Field, validator
import httpx

log = structlog.get_logger()


class Permission(Enum):
    """User permission levels."""
    READ = "read"
    WRITE = "write"
    DELETE = "delete"
    ADMIN = "admin"


class Role(BaseModel):
    """User role with associated permissions."""
    name: str = Field(..., min_length=1, max_length=50)
    permissions: List[Permission]

    class Config:
        use_enum_values = True


class User(BaseModel):
    """User model with validation and serialization support."""
    id: int
    username: str = Field(..., min_length=3, max_length=30)
    email: EmailStr
    role_id: int = Field(..., ge=1)
    role: Optional[Role] = None
    created_at: Optional[datetime] = None
    is_active: bool = True

    @validator('username')
    def validate_username(cls, v: str) -> str:
        if not v.replace('_', '').isalnum():
            raise ValueError('Username must contain only alphanumeric characters and underscores')
        return v

    @validator('created_at', pre=True, always=True)
    def set_created_at(cls, v: Optional[datetime]) -> datetime:
        return v or datetime.utcnow()

    def to_dict(self) -> dict:
        """Serialize user model to dictionary."""
        return self.dict(exclude_none=True)

    @classmethod
    def from_dict(cls, data: dict) -> "User":
        """Deserialize dictionary to User model."""
        return cls(**data)


async def fetch_users(database_url: str) -> List[User]:
    """
    Fetch users from a database endpoint.

    Args:
        database_url: URL of the database endpoint returning user data

    Returns:
        List of User objects

    Raises:
        httpx.HTTPError: If the request fails
        ValueError: If the response data is invalid
    """
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(database_url, timeout=30.0)
            response.raise_for_status()
            users_data = response.json()

            # Validate and convert to User models
            users = []
            for user_data in users_data:
                try:
                    user = User(**user_data)
                    users.append(user)
                except Exception as e:
                    log.error('user_validation_failed', error=str(e), data=user_data)
                    raise ValueError(f"Invalid user data: {user_data}") from e

            return users

    except httpx.HTTPError as e:
        log.error('database_fetch_failed', url=database_url, error=str(e))
        raise


# Example usage (not executed when imported)
if __name__ == "__main__":
    # Example of how to use the module
    import asyncio

    async def main():
        try:
            users = await fetch_users("https://api.example.com/users")

                print(user.to_dict())
        except Exception as e:
            log.error("Failed to fetch users", error=str(e))
