```python
"""User management system data models and utilities."""

import httpx
from typing import List, Optional
from pydantic import BaseModel, Field, validator, EmailStr
import structlog

logger = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission in the system."""
    id: int = Field(..., ge=1, description="Unique identifier for the permission")
    name: str = Field(..., min_length=1, max_length=100, description="Name of the permission")
    description: Optional[str] = Field(None, max_length=255, description="Description of the permission")

    class Config:
        orm_mode = True


class Role(BaseModel):
    """Represents a role in the system."""
    id: int = Field(..., ge=1, description="Unique identifier for the role")
    name: str = Field(..., min_length=1, max_length=50, description="Name of the role")
    description: Optional[str] = Field(None, max_length=255, description="Description of the role")
    permissions: List[Permission] = Field(default_factory=list, description="List of permissions assigned to the role")

    class Config:
        orm_mode = True


class User(BaseModel):
    """Represents a user in the system."""
    id: int = Field(..., ge=1, description="Unique identifier for the user")
    username: str = Field(..., min_length=3, max_length=50, description="User's unique username")
    email: EmailStr = Field(..., description="User's email address")
    full_name: Optional[str] = Field(None, max_length=100, description="User's full name")
    role_id: int = Field(..., ge=1, description="ID of the user's role")
    is_active: bool = Field(default=True, description="Whether the user account is active")
    created_at: Optional[str] = Field(None, description="ISO 8601 timestamp of user creation")

    # Validate role_id is not zero or negative
    @validator('role_id')
    def validate_role_id(cls, v):
        if v <= 0:
            raise ValueError('role_id must be positive')
        return v

    class Config:
        orm_mode = True


async def fetch_users(database_url: str) -> List[User]:
    """
    Fetch users from a database API endpoint.

    Args:
        database_url: URL of the API endpoint returning user data

    Returns:
        List of User objects

    Raises:
        httpx.HTTPError: If the request fails
        ValueError: If the response data fails validation
    """
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(database_url)
            response.raise_for_status()
            data = response.json()
            
            # Handle both list and dict responses
            if isinstance(data, dict):
                users_data = data.get("users", [])
            else:
                users_data = data
            
            users = [User.parse_obj(user_data) for user_data in users_data]
            logger.info("users_fetched", count=len(users), url=database_url)
            return users
            
    except httpx.HTTPError as exc:
        logger.error("http_error", url=database_url, error=str(exc))
        raise
    except Exception as exc:
        logger.error("validation_error", url=database_url, error=str(exc))
        raise ValueError(f"Failed to parse user data: {exc}")
```
