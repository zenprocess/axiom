```python
"""
User management system data models and utilities.

This module defines Pydantic models for User, Role, and Permission,
along with a function to fetch users from a database URL.
"""

from typing import List, Optional
from datetime import datetime
import httpx
import structlog

from pydantic import BaseModel, EmailStr, validator, Field

logger = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission granting specific access rights."""
    id: int
    name: str = Field(..., min_length=1, max_length=50)
    description: Optional[str] = Field(None, max_length=200)

    class Config:
        orm_mode = True


class Role(BaseModel):
    """Represents a role that can be assigned to users."""
    id: int
    name: str = Field(..., min_length=1, max_length=50)
    description: Optional[str] = Field(None, max_length=200)
    permissions: List[Permission] = []

    @validator('name')
    def validate_name(cls, v):
        """Ensure role name is not a reserved word."""
        if v.lower() in ['admin', 'superuser']:
            raise ValueError('Role name cannot be reserved word')
        return v

    class Config:
        orm_mode = True


class User(BaseModel):
    """Represents a user in the system."""
    id: int
    username: str = Field(..., min_length=3, max_length=30)
    email: EmailStr
    role_id: Optional[int] = None
    role: Optional[Role] = None
    is_active: bool = True
    created_at: datetime
    updated_at: Optional[datetime] = None

    @validator('username')
    def validate_username(cls, v):
        """Ensure username contains only alphanumeric characters and underscores."""
        if not v.replace('_', '').isalnum():
            raise ValueError('Username must contain only alphanumeric characters and underscores')
        return v

    def to_dict(self) -> dict:
        """Serialize model to dictionary."""
        return self.dict()

    class Config:
        orm_mode = True


async def fetch_users(database_url: str) -> List[User]:
    """
    Fetch users from a database API endpoint.

    Args:
        database_url: URL to the users API endpoint.

    Returns:
        List of User objects.

    Raises:
        httpx.HTTPError: If the request fails.
        ValueError: If the response data is invalid.
    """
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(database_url)
            response.raise_for_status()
            data = response.json()

            # Validate response structure
            if not isinstance(data, list):
                raise ValueError("Expected list of users in response")

            # Parse and validate each user
            users = []
            for user_data in data:
                try:
                    user = User.parse_obj(user_data)
                    users.append(user)
                except Exception as e:
                    logger.warning("Failed to parse user", error=str(e), user_data=user_data)
                    continue

            logger.info("Successfully fetched users", count=len(users))
            return users

    except httpx.HTTPError as e:
        logger.error("HTTP error fetching users", url=database_url, error=str(e))
        raise
    except Exception as e:
        logger.error("Unexpected error fetching users", url=database_url, error=str(e))
        raise
```
