```python
# user_models.py

import httpx
from typing import List, Optional
from pydantic import BaseModel, Field, validator, EmailStr
import structlog

logger = structlog.get_logger()


class Permission(BaseModel):
    """Represents a permission granted to a role."""
    id: int = Field(..., ge=1, description="Unique identifier for the permission")
    name: str = Field(..., min_length=1, max_length=64, description="Name of the permission (e.g., 'read_users')")
    description: Optional[str] = Field(None, max_length=256, description="Description of the permission")

    class Config:
        validate_assignment = True


class Role(BaseModel):
    """Represents a role in the system with associated permissions."""
    id: int = Field(..., ge=1, description="Unique identifier for the role")
    name: str = Field(..., min_length=1, max_length=64, description="Name of the role (e.g., 'admin', 'editor')")
    description: Optional[str] = Field(None, max_length=256, description="Description of the role")
    permissions: List[Permission] = Field(default_factory=list, description="List of permissions assigned to this role")

    @validator('permissions', each_item=True)
    def validate_permissions(cls, perm: Permission) -> Permission:
        if not perm.name or not perm.name.strip():
            raise ValueError("Permission name cannot be empty")
        return perm

    class Config:
        validate_assignment = True


class User(BaseModel):
    """Represents a user in the system."""
    id: int = Field(..., ge=1, description="Unique identifier for the user")
    username: str = Field(..., min_length=3, max_length=32, pattern=r'^[a-zA-Z0-9_.-]+$', description="User's unique username")
    email: EmailStr = Field(..., description="User's email address")
    role_id: int = Field(..., ge=1, description="ID of the user's role")
    role_name: str = Field(..., min_length=1, max_length=64, description="Name of the user's role")
    is_active: bool = Field(default=True, description="Whether the user account is active")
    created_at: Optional[str] = Field(None, description="ISO 8601 formatted creation timestamp")

    @validator('username')
    def validate_username(cls, v: str) -> str:
        if v.lower() in ['admin', 'root', 'system']:
            raise ValueError("Username cannot be a reserved word")
        return v

    @validator('email')
    def validate_email(cls, v: str) -> str:
        if v.endswith('@example.com'):
            raise ValueError("Email cannot be an example email")
        return v

    class Config:
        validate_assignment = True


async def fetch_users(database_url: str) -> List[User]:
    """
    Fetch users from a database API endpoint.

    Args:
        database_url (str): The URL of the database API endpoint.

    Returns:
        List[User]: A list of User objects fetched from the database.

    Raises:
        httpx.HTTPStatusError: If the HTTP request returned an unsuccessful status code.
        httpx.RequestError: For network-related errors.
        ValueError: If the response data cannot be validated as User objects.
    """
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(database_url)
            response.raise_for_status()
            users_data = response.json()

            # Validate and convert raw data to User models
            users = [User.parse_obj(user_data) for user_data in users_data]
            logger.info("users_fetched", count=len(users), url=database_url)
            return users

    except httpx.HTTPStatusError as exc:
        logger.error("http_error", status_code=exc.response.status_code, url=database_url)
        raise
    except httpx.RequestError as exc:
        logger.error("request_error", url=database_url, error=str(exc))
        raise
    except Exception as exc:
        logger.error("validation_error", url=database_url, error=str(exc))
        raise ValueError(f"Failed to validate user data from {database_url}: {exc}")
```
