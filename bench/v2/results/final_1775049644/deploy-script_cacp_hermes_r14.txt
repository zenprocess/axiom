Here's a production-ready bash deployment script that complies with your strict coding standards. It uses best practices for deployment while ensuring no `print()`, no hardcoded secrets, proper error handling, and rollback capability.

```bash
#!/bin/bash
set -euo pipefail

# Deployment script for Docker-based application
# Requirements: docker, git, kubectl (if deploying to Kubernetes), or equivalent
# Assumes Docker registry authentication is handled via environment or config

# Environment variables (must be set externally or via .env file)
# DOCKER_REGISTRY, IMAGE_NAME, IMAGE_TAG, KUBECONFIG (if applicable)

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
LOG_FILE="${SCRIPT_DIR}/deploy.log"

# Import structlog-style logging (simulated via helper functions)
log_info() {
  echo "[$(date -Iseconds)] [INFO] $*" >> "$LOG_FILE" 2>/dev/null || true
}

log_error() {
  echo "[$(date -Iseconds)] [ERROR] $*" >&2
  echo "[$(date -Iseconds)] [ERROR] $*" >> "$LOG_FILE" 2>/dev/null || true
}

# Validate required environment variables
validate_env() {
  local required_vars=("DOCKER_REGISTRY" "IMAGE_NAME" "IMAGE_TAG")
  for var in "${required_vars[@]}"; do
    if [[ -z "${!var:-}" ]]; then
      log_error "Required environment variable '$var' is not set"
      exit 1
    fi
  done
}

# Build Docker image
build_image() {
  log_info "Building Docker image: ${DOCKER_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
  docker build -t "${DOCKER_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}" .
}

# Push image to registry
push_image() {
  log_info "Pushing image to registry: ${DOCKER_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
  docker push "${DOCKER_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
}

# Update git tag
update_git_tag() {
  local tag_name="v${IMAGE_TAG}"
  log_info "Updating git tag to ${tag_name}"
  git tag -d "$tag_name" 2>/dev/null || true
  git tag "$tag_name"
  git push origin "$tag_name"
}

# Deploy to production (generic placeholder)
# Replace with Kubernetes, ECS, or other deployment mechanism
deploy_production() {
  if [[ -n "${KUBECONFIG:-}" ]]; then
    log_info "Deploying to Kubernetes using KUBECONFIG"
    kubectl --kubeconfig="$KUBECONFIG" set image "deployment/${IMAGE_NAME}" \
      "${IMAGE_NAME}=${DOCKER_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}" \
      --record
  else
    log_error "No deployment method configured (KUBECONFIG not set)"
    exit 1
  fi
}

# Rollback production deployment
rollback_production() {
  local previous_tag="${1:-}"
  if [[ -n "${KUBECONFIG:-}" ]]; then
    log_info "Rolling back deployment to previous version"
    kubectl --kubeconfig="$KUBECONFIG" rollout undo "deployment/${IMAGE_NAME}"
  else
    log_error "Rollback failed: no deployment method configured"
    exit 1
  fi
}

# Main deployment flow
main() {
  validate_env
  local rollback_required=false

  # Ensure clean working directory
  if ! git diff-index --quiet HEAD --; then
    log_error "Working directory has uncommitted changes"
    exit 1
  fi

  # Step 1: Build image
  if ! build_image; then
    log_error "Image build failed"
    exit 1
  fi

  # Step 2: Push image
  if ! push_image; then
    log_error "Image push failed"
    exit 1
  fi

  # Step 3: Update git tag
  if ! update_git_tag; then
    log_error "Git tag update failed"
    rollback_required=true
    rollback_production "previous"
    exit 1
  fi

  # Step 4: Deploy to production
  if ! deploy_production; then
    log_error "Production deployment failed"
    rollback_required=true
  fi

  # Rollback if deployment failed
  if [[ "$rollback_required" == true ]]; then
    rollback_production
    exit 1
  fi

  log_info "Deployment completed successfully"
}

# Handle errors and cleanup
trap 'log_error "Deployment interrupted"; exit 1' INT TERM

main "$@"
```

### Key Compliance Notes:

- ✅ **No `print()`**: Uses `log_info`/`log_error` for logging to file only
- ✅ **No secrets**: All credentials must be provided via environment variables
- ✅ **Async I/O**: Not applicable for bash scripts (this is shell deployment logic)
- ✅ **structlog-only**: Simulated via custom logging functions
- ✅ **Error handling**: All steps wrapped in `if ! command; then ...; fi`
- ✅ **Rollback support**: Automatic rollback on deployment failure
- ✅ **Git safety**: No `--force` push, uses `git diff-index` for clean state check
- ✅ **Type hints**: Not applicable for bash (script uses parameter validation)
- ✅ **No star imports**: N/A (bash doesn't have imports)

