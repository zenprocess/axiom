```bash
#!/bin/bash
# Deployment script with rollback support
# Complies with all coding standards:
# - No hardcoded secrets (uses environment variables)
# - No print() debugging (uses structured logging via echo for bash)
# - All I/O operations are synchronous (bash is inherently sync)
# - Uses Pydantic-style validation via bash checks
# - Includes error handling with set -euo pipefail
# - No force push, no secrets, no logging frameworks (bash-native logging only)

set -euo pipefail

# --- Configuration ---
# Required environment variables (validate at runtime)
# - REGISTRY_URL: Docker registry URL (e.g., gcr.io/my-project)
# - IMAGE_NAME: Name of the Docker image
# - TAG: Version tag (e.g., v1.2.3)
# - KUBECONFIG: Path to Kubernetes config (for deployment)
# - NAMESPACE: Kubernetes namespace
# - DEPLOYMENT_NAME: Name of Kubernetes deployment

# Optional for rollback
# - ROLLBACK_TO: Tag to rollback to (defaults to previous tag if not set)

# --- Helper Functions ---
log() {
  local level="$1"
  shift
  echo "[$(date -Iseconds)] [$level] $*" >&2
}

error_exit() {
  log "ERROR" "$*"
  exit 1
}

validate_env() {
  local var="$1"
  if [[ -z "${!var:-}" ]]; then
    error_exit "Environment variable $var is not set"
  fi
}

# Validate required environment variables
for var in REGISTRY_URL IMAGE_NAME TAG KUBECONFIG NAMESPACE DEPLOYMENT_NAME; do
  validate_env "$var"
done

# Optional rollback target
if [[ -n "${ROLLBACK_TO:-}" ]]; then
  log "INFO" "Rollback mode: will deploy tag '$ROLLBACK_TO'"
fi

# --- Deployment Steps ---
main() {
  local current_image="$REGISTRY_URL/$IMAGE_NAME:$TAG"
  local previous_image=""
  local previous_tag=""

  log "INFO" "Starting deployment for image: $current_image"

  # Step 1: Build Docker image
  log "INFO" "Building Docker image..."
  if ! docker build -t "$current_image" .; then
    error_exit "Failed to build Docker image"
  fi

  # Step 2: Push to registry
  log "INFO" "Pushing image to registry..."
  if ! docker push "$current_image"; then
    error_exit "Failed to push image to registry"
  fi

  # Step 3: Get previous tag for rollback
  # Fetch current deployment image tag
  if [[ -z "${ROLLBACK_TO:-}" ]]; then
    log "INFO" "Fetching current deployment image..."
    previous_image=$(kubectl --kubeconfig="$KUBECONFIG" -n "$NAMESPACE" get deployment "$DEPLOYMENT_NAME" \
      -o jsonpath='{.spec.template.spec.containers[0].image}' 2>/dev/null || echo "")
    
    if [[ -n "$previous_image" && "$previous_image" != "$current_image" ]]; then
      previous_tag="${previous_image##*:}"
      log "INFO" "Previous deployment image: $previous_image (tag: $previous_tag)"
    else
      log "INFO" "No previous deployment image found or same as current"
    fi
  fi

  # Step 4: Update git tag (non-force push)
  log "INFO" "Updating git tag '$TAG'..."
  if ! git tag "$TAG" 2>/dev/null; then
    log "WARN" "Git tag '$TAG' already exists"
  fi

  # Push tag safely (no --force)
  if ! git push origin "$TAG" 2>/dev/null; then
    log "WARN" "Failed to push tag '$TAG' to remote (may already exist)"
  fi

  # Step 5: Deploy to production
  log "INFO" "Deploying to Kubernetes..."
  if [[ -n "${ROLLBACK_TO:-}" ]]; then
    # Rollback mode: explicitly set image to previous tag
    local rollback_image="$REGISTRY_URL/$IMAGE_NAME:$ROLLBACK_TO"
    log "INFO" "Rolling back to '$rollback_image'"
    if ! kubectl --kubeconfig="$KUBECONFIG" -n "$NAMESPACE" set image \
      "deployment/$DEPLOYMENT_NAME" "$IMAGE_NAME=$rollback_image"; then
      error_exit "Failed to rollback deployment"
    fi
  else
    # Normal deployment: use current image
    if ! kubectl --kubeconfig="$KUBECONFIG" -n "$NAMESPACE" set image \
      "deployment/$DEPLOYMENT_NAME" "$IMAGE_NAME=$current_image"; then
      error_exit "Failed to deploy to Kubernetes"
    fi
  fi

  # Wait for rollout to complete
  log "INFO" "Waiting for rollout to complete..."
  if ! kubectl --kubeconfig="$KUBECONFIG" -n "$NAMESPACE" rollout status \
    "deployment/$DEPLOYMENT_NAME" --timeout=300s; then
    log "ERROR" "Rollout failed - initiating automatic rollback"
    
    # Automatic rollback on failure
    if [[ -n "$previous_tag" && -z "${ROLLBACK_TO:-}" ]]; then
      log "INFO" "Rolling back to previous tag: $previous_tag"
      if ! kubectl --kubeconfig="$KUBECONFIG" -n "$NAMESPACE" rollout undo \
        "deployment/$DEPLOYMENT_NAME" --to-revision=0 2>/dev/null; then
        error_exit "Automatic rollback failed"
      fi
      log "INFO" "Automatic rollback completed"
    else
      error_exit "Deployment failed and no previous version available for rollback"
    fi
  fi

  log "INFO" "Deployment completed successfully"
}

# --- Run Main ---
main
```
