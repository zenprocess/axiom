Here's a robust bash deployment script that follows your development rules and includes proper error handling, logging, and rollback support.

```bash
#!/bin/bash

# deployment.sh
set -euo pipefail

# Use structlog-compatible logging (example: output JSON for CI/CD tooling)
log_info() {
    echo "{\"level\":\"info\",\"timestamp\":\"$(date -Iseconds)\",\"message\":\"$1\"}"
}

log_error() {
    echo "{\"level\":\"error\",\"timestamp\":\"$(date -Iseconds)\",\"message\":\"$1\"}" >&2
}

log_warning() {
    echo "{\"level\":\"warning\",\"timestamp\":\"$(date -Iseconds)\",\"message\":\"$1\"}" >&2
}

# Configuration
REGISTRY="${REGISTRY:-ghcr.io}"
IMAGE_NAME="${IMAGE_NAME:-myapp}"
IMAGE_TAG="${IMAGE_TAG:-latest}"
GIT_TAG_PREFIX="${GIT_TAG_PREFIX:-v}"
ROLLBACK_VERSION="${ROLLBACK_VERSION:-}"

# Validate required environment variables
if [[ -z "${REGISTRY}" || -z "${IMAGE_NAME}" ]]; then
    log_error "Required environment variables REGISTRY and IMAGE_NAME must be set"
    exit 1
fi

# Parse command-line arguments
ACTION="${1:-deploy}"
if [[ "$ACTION" != "deploy" && "$ACTION" != "rollback" ]]; then
    log_error "Usage: $0 [deploy|rollback]"
    exit 1
fi

# Get current directory and ensure we're in a git repo
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
cd "$SCRIPT_DIR"

if ! git rev-parse --git-dir > /dev/null 2>&1; then
    log_error "Not a git repository"
    exit 1
fi

# Get current commit SHA for tagging
CURRENT_COMMIT=$(git rev-parse HEAD)

# Deploy function
do_deploy() {
    log_info "Starting deployment"
    
    # Build Docker image
    log_info "Building Docker image"
    if ! docker build -t "${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}" .; then
        log_error "Docker build failed"
        exit 1
    fi
    
    # Push to registry
    log_info "Pushing image to registry"
    if ! docker push "${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"; then
        log_error "Failed to push image to registry"
        exit 1
    fi
    
    # Create and push git tag
    GIT_TAG="${GIT_TAG_PREFIX}${CURRENT_COMMIT:0:7}"
    log_info "Creating git tag: ${GIT_TAG}"
    
    if ! git tag -a "${GIT_TAG}" -m "Deploy ${CURRENT_COMMIT}"; then
        log_error "Failed to create git tag"
        exit 1
    fi
    
    if ! git push "${GIT_REMOTE:-origin}" "${GIT_TAG}"; then
        log_error "Failed to push git tag"
        git tag -d "${GIT_TAG}" 2>/dev/null || true
        exit 1
    fi
    
    log_info "Deployment successful: ${IMAGE_NAME}:${IMAGE_TAG} tagged as ${GIT_TAG}"
}

# Rollback function
do_rollback() {
    # If rollback version not specified, use the previous tag
    if [[ -z "${ROLLBACK_VERSION}" ]]; then
        # Get the previous tag (2nd most recent)
        PREV_TAG=$(git describe --tags --abbrev=0 "$(git rev-list --tags --skip=1 --max-count=1)" 2>/dev/null || \
                   git tag --list "${GIT_TAG_PREFIX}*" --sort=-version:refname | head -n 2 | tail -n 1)
        
        if [[ -z "${PREV_TAG}" ]]; then
            log_error "No previous tag found for rollback"
            exit 1
        fi
        
        ROLLBACK_VERSION="${PREV_TAG}"
    fi
    
    log_info "Rolling back to tag: ${ROLLBACK_VERSION}"
    
    # Extract commit SHA from tag
    ROLLBACK_COMMIT=$(git rev-list -n 1 "${ROLLBACK_VERSION}")
    
    if [[ -z "${ROLLBACK_COMMIT}" ]]; then
        log_error "Tag ${ROLLBACK_VERSION} not found"
        exit 1
    fi
    
    # Extract image tag from git tag (assuming format: v<hash>)
    IMAGE_TAG="${ROLLBACK_VERSION#${GIT_TAG_PREFIX}}"
    
    # Deploy the old version
    log_info "Deploying image from rollback tag"
    if ! docker pull "${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"; then
        log_error "Failed to pull image for rollback"
        exit 1
    fi
    
    # Update deployment (example: kubectl apply -f k8s/deployment.yaml)
    # This would be replaced with your actual deployment mechanism
    log_info "Updating deployment to ${IMAGE_NAME}:${IMAGE_TAG}"
    
    # If using Kubernetes:
    # kubectl set image deployment/${IMAGE_NAME} ${IMAGE_NAME}="${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
    
    log_info "Rollback successful: deployed ${IMAGE_NAME}:${IMAGE_TAG}"
}

# Main execution
case "${ACTION}" in
    deploy)
        do_deploy
        ;;
    rollback)
        do_rollback
        ;;
esac
```

## Key Features & Compliance

1. **No secrets** - Uses environment variables for configuration
2. **Error handling** - Uses `set -euo pipefail`, checks command exit codes, and provides detailed error messages
3. **Logging** - Uses `structlog`-style JSON output for structured logging (adaptable to your logging infrastructure)
4. **Rollback support** - Creates versioned git tags and supports rolling back to previous versions
5. **Type hints not applicable** - Bash doesn't support type hints, but uses proper variable quoting and validation
6. **Git safety** - Uses `git push` (not force push) for tags

## Usage

```bash
# Deploy the current version
./deployment.sh deploy

# Rollback to previous version
./deployment.sh rollback

# With environment variables
REGISTRY=registry.example.com IMAGE_NAME=webapp IMAGE_TAG=v1.2.3 ./deployment.sh deploy
```

## Notes

1. Replace the deployment step (currently commented) with your actual deployment mechanism (Kubernetes, ECS, etc.)
2. Ensure your CI/CD environment has:
   - Docker authentication configured (`docker login`)
   - Git push permissions
   - Required environment variables set
3. The script assumes tags follow the pattern `v<commit-short-hash>` but can be customized via `GIT_TAG_PREFIX`
