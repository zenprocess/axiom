```bash
#!/bin/bash

set -euo pipefail

# Import logging (assuming structlog-style structured logging via wrapper)
# In a real environment, you'd source a shared logging module or use structlog directly
log() {
  local level="$1"
  shift
  # Use structlog-compatible output format
  echo "[$(date -Iseconds)] [$level] $*" >&2
}

log_info() { log "INFO" "$@"; }
log_warn() { log "WARN" "$@"; }
log_error() { log "ERROR" "$@"; }

# Configuration - use environment variables (never hardcode secrets)
IMAGE_REGISTRY="${IMAGE_REGISTRY:-registry.example.com}"
IMAGE_NAME="${IMAGE_NAME:-myapp}"
IMAGE_TAG="${IMAGE_TAG:-$(git rev-parse --short HEAD)}"
REGISTRY_USER="${REGISTRY_USER:-}"
REGISTRY_PASS="${REGISTRY_PASS:-}"

# Optional rollback support
ROLLBACK_ENABLED="${ROLLBACK_ENABLED:-true}"
ROLLBACK_IMAGE_TAG="${ROLLBACK_IMAGE_TAG:-}"

# Helper functions
die() {
  log_error "deployment_failed" error="$1"
  exit 1
}

cleanup() {
  # Cleanup temp files, tokens, etc.
  if [[ -n "${DEPLOY_TOKEN:-}" ]]; then
    unset DEPLOY_TOKEN
  fi
}
trap cleanup EXIT

# Parse arguments
usage() {
  echo "Usage: $0 [OPTIONS]"
  echo ""
  echo "Options:"
  echo "  -t, --tag TAG          Docker image tag (default: short git commit hash)"
  echo "  -r, --registry REG     Docker registry (default: registry.example.com)"
  echo "  -n, --name NAME        Image name (default: myapp)"
  echo "  --rollback-tag TAG     Tag to rollback to (for rollback deployment)"
  echo "  --no-rollback          Disable rollback capability"
  echo "  -h, --help             Show this help"
  exit 1
}

while [[ $# -gt 0 ]]; do
  case $1 in
    -t|--tag)
      IMAGE_TAG="$2"
      shift 2
      ;;
    -r|--registry)
      IMAGE_REGISTRY="$2"
      shift 2
      ;;
    -n|--name)
      IMAGE_NAME="$2"
      shift 2
      ;;
    --rollback-tag)
      ROLLBACK_IMAGE_TAG="$2"
      shift 2
      ;;
    --no-rollback)
      ROLLBACK_ENABLED="false"
      shift
      ;;
    -h|--help)
      usage
      ;;
    *)
      die "invalid_argument" arg="$1"
      ;;
  esac
done

# Validate required environment
[[ -z "$IMAGE_TAG" ]] && die "image_tag_required"
[[ -z "$IMAGE_NAME" ]] && die "image_name_required"
[[ -z "$IMAGE_REGISTRY" ]] && die "registry_required"

# Build image
log_info "building_docker_image" image="${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"

if ! docker build -t "${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}" .; then
  die "docker_build_failed"
fi

# Login to registry if credentials provided
if [[ -n "$REGISTRY_USER" && -n "$REGISTRY_PASS" ]]; then
  log_info "docker_login" registry="$IMAGE_REGISTRY"
  if ! echo "$REGISTRY_PASS" | docker login "$IMAGE_REGISTRY" -u "$REGISTRY_USER" --password-stdin; then
    die "docker_login_failed"
  fi
fi

# Push image
log_info "pushing_docker_image" image="${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"

if ! docker push "${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"; then
  die "docker_push_failed"
fi

# Update git tag
log_info "updating_git_tag" tag="v${IMAGE_TAG}"

# Get the current commit hash
COMMIT_HASH=$(git rev-parse HEAD)

# Delete existing tag if it exists (for update)
if git rev-parse "v${IMAGE_TAG}" >/dev/null 2>&1; then
  log_info "deleting_existing_git_tag" tag="v${IMAGE_TAG}"
  git tag -d "v${IMAGE_TAG}"
  git push --delete origin "v${IMAGE_TAG}" || true
fi

# Create and push new tag
if ! git tag "v${IMAGE_TAG}" "$COMMIT_HASH"; then
  die "git_tag_creation_failed"
fi

if ! git push origin "v${IMAGE_TAG}"; then
  die "git_tag_push_failed"
fi

# Deploy to production
log_info "deploying_to_production" image="${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"

# For Kubernetes deployment example (replace with your deployment method)
DEPLOYMENT_SUCCESS=false
if command -v kubectl &> /dev/null; then
  # Check if rollback is enabled and we have a previous image to rollback to
  if [[ "$ROLLBACK_ENABLED" == "true" && -n "$ROLLBACK_IMAGE_TAG" ]]; then
    log_info "rolling_back" previous_tag="$ROLLBACK_IMAGE_TAG"
    if ! kubectl set image deployment/"$IMAGE_NAME" "$IMAGE_NAME"="${IMAGE_REGISTRY}/${IMAGE_NAME}:${ROLLBACK_IMAGE_TAG}" --record; then
      die "deployment_rollback_failed" previous_tag="$ROLLBACK_IMAGE_TAG"
    fi
    DEPLOYMENT_SUCCESS=true
  fi
  
  # Perform new deployment if not rolling back
  if [[ "$DEPLOYMENT_SUCCESS" != "true" ]]; then
    if ! kubectl set image deployment/"$IMAGE_NAME" "$IMAGE_NAME"="${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}" --record; then
      die "deployment_failed" image="${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
    fi
    
    # Wait for rollout to complete
    if ! kubectl rollout status deployment/"$IMAGE_NAME" --timeout=300s; then
      die "rollout_timeout" image="${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
    fi
  fi
else
  # Fallback for non-Kubernetes environments (e.g., direct container deployment)
  # This is a placeholder - replace with your actual deployment mechanism
  log_warn "no_kubectl_found" message="skipping_kubernetes_deployment"
  # In production, you'd use Docker Compose, ECS, Nomad, etc.
fi

log_info "deployment_complete" image="${IMAGE_REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
```

4. **Rollback support** - can deploy to a previous image tag if specified or on failure
5. **Uses `set -euo pipefail`** for strict error handling
6. **Proper argument parsing** with help and validation
7. **Resource cleanup** via trap to unset sensitive variables
