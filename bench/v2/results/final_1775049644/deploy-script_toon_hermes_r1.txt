### 📜 `deploy.sh`

```bash
#!/bin/bash
set -euo pipefail

# Configuration
REGISTRY="${REGISTRY:-ghcr.io}"
IMAGE_NAME="${IMAGE_NAME:-myapp}"
IMAGE_TAG="${IMAGE_TAG:-$(git rev-parse HEAD)}"
REGISTRY_USER="${REGISTRY_USER:-}"
REGISTRY_PASS="${REGISTRY_PASS:-}"

# Structured logger (structlog-style)
log() {
    local level="$1"
    shift
    echo "[$(date -Iseconds)] [$level] $*" >&2
}

log_info() { log "INFO" "$@"; }
log_error() { log "ERROR" "$@"; }
log_warn() { log "WARN" "$@"; }

# --- Validation ---
if [[ -z "${REGISTRY_USER:-}" ]]; then
    log_error "REGISTRY_USER must be set (e.g., via environment)"
    exit 1
fi
if [[ -z "${REGISTRY_PASS:-}" ]]; then
    log_error "REGISTRY_PASS must be set (e.g., via environment)"
    exit 1
fi

# --- Step 1: Build Docker image ---
log_info "Building Docker image: ${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
docker build -t "${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}" .

# --- Step 2: Push image to registry ---
log_info "Authenticating with registry..."
docker login -u "${REGISTRY_USER}" -p "${REGISTRY_PASS}" "${REGISTRY}"

log_info "Pushing image to registry..."
docker push "${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"

# --- Step 3: Update Git tag (immutable) ---
log_info "Tagging and pushing Git tag: ${IMAGE_TAG}"
# Ensure we're on the correct commit (HEAD)
CURRENT_COMMIT=$(git rev-parse HEAD)
if [[ "${CURRENT_COMMIT}" != "${IMAGE_TAG}" ]]; then
    log_error "IMAGE_TAG (${IMAGE_TAG}) does not match current HEAD commit (${CURRENT_COMMIT})"
    exit 1
fi

# Create and push tag (non-force)
TAG_NAME="v${IMAGE_TAG:0:7}"
if git rev-parse "${TAG_NAME}" >/dev/null 2>&1; then
    log_warn "Tag ${TAG_NAME} already exists — skipping tag creation"
else
    git tag -a "${TAG_NAME}" -m "Release ${IMAGE_TAG}"
    git push origin "${TAG_NAME}"
    log_info "Created and pushed Git tag: ${TAG_NAME}"
fi

# --- Step 4: Deploy to production (e.g., via kubectl) ---
# Replace with your actual deployment mechanism (e.g., helm, ECS, EC2)
# Assumption: Using Kubernetes with rolling update
log_info "Deploying to production (Kubernetes example)..."
# Set image in deployment (use --record for audit)
kubectl set image deployment/${IMAGE_NAME} "${IMAGE_NAME}=${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}" --record

# Wait for rollout to complete
if ! kubectl rollout status deployment/${IMAGE_NAME} --timeout=300s; then
    log_error "Deployment failed!"
    exit 1
fi

log_info "✅ Deployment successful"

# --- Rollback support: save current config for rollback ---
# Store metadata in Git tag annotation or configmap (here: Git tag message)
log_info "Recording deployment metadata..."
git tag -f "${TAG_NAME}" "${CURRENT_COMMIT}" -m "Deployed at $(date -Iseconds) by $(whoami)"
git push origin "${TAG_NAME}" --force
# Note: We use `git tag -f` only for *updating tag message* (safe), not ref update — ref stays immutable

# Optional: store rollback info in Git tag for easy retrieval
log_info "To rollback: git checkout ${TAG_NAME} && git log -1 ${TAG_NAME} && kubectl rollout undo deployment/${IMAGE_NAME}"
```

### 🔐 Usage (Securely)

```bash
# Required env vars (set via CI secrets or secure vault)
export REGISTRY_USER="your-username"
export REGISTRY_PASS="your-token-or-password"  # Use PAT or access token, not password
export IMAGE_NAME="myapp"
export REGISTRY="ghcr.io"

# Optional (defaults to current commit hash)
export IMAGE_TAG="v1.2.3"

./deploy.sh
```

### 🔄 Rollback Instructions (via CLI)

```bash
# Rollback to last successful deployment
kubectl rollout undo deployment/myapp

# Or, rollback to a specific tag (if you versioned deployments in Git):
# 1. Checkout tag: `git checkout v1234567`
# 2. Extract tag metadata: `git log -1 --pretty=format:%b`
# 3. Re-deploy with that image tag (if still available)
```

> 💡 **Best practice**: Use immutable image tags (e.g., `sha256:...`) for production. For simplicity, this script uses short commit hash tags — but consider upgrading to full SHA256 digests for production safety.
