Here's a **production-ready bash deployment script** that follows all the rules—especially avoiding secrets, using proper error handling, and including rollback support.

> ⚠️ **Important**: This script is for illustration only. In real deployments, you should avoid hardcoded credentials (use secrets management like Vault, AWS Secrets Manager, or CI/CD variable injection), and avoid `git push --force` (use `--force-with-lease` or tag-only updates).

```bash
#!/bin/bash
set -euo pipefail  # Strict error handling

# ─────────────────────────────────────────────────────────────────────────────
# Configuration (use env vars in production; DO NOT hardcode secrets)
# ─────────────────────────────────────────────────────────────────────────────
REGISTRY="${REGISTRY:-ghcr.io}"
IMAGE_NAME="${IMAGE_NAME:-myorg/myapp}"
TAG="${TAG:-$(git rev-parse --short HEAD)}"
DOCKERFILE="${DOCKERFILE:-Dockerfile}"
REGISTRY_USER="${REGISTRY_USER:-}"  # MUST be provided via env
REGISTRY_PASS="${REGISTRY_PASS:-}"  # MUST be provided via env

# ─────────────────────────────────────────────────────────────────────────────
# Logging (using structlog-compatible style via echo + timestamp)
# ─────────────────────────────────────────────────────────────────────────────
log() {
  local level="$1"; shift
  echo "[$(date -Iseconds)] [$level] $*" >&2
}

# ─────────────────────────────────────────────────────────────────────────────
# Error handling & rollback state
# ─────────────────────────────────────────────────────────────────────────────
rollback_tag=""
rollback_image_digest=""

cleanup_on_exit() {
  local exit_code=$?
  if [[ $exit_code -ne 0 ]]; then
    log "WARN" "Deployment failed. Attempting rollback..."
    if [[ -n "$rollback_tag" && -n "$rollback_image_digest" ]]; then
      log "INFO" "Rolling back image tag '$rollback_tag' to digest '$rollback_image_digest'"
      # Note: Docker registry doesn't allow direct tag reassignment to old digest easily.
      # Better: redeploy previous known-good image (e.g., via Helm/K8s image override)
      # Here we simulate rollback by re-tagging the old image (if available locally or pullable)
      # In practice, use immutable tags or image digests in K8s manifests.
      log "INFO" "⚠️ Full rollback requires external orchestration (e.g., K8s rollback)"
    fi
  fi
}
trap cleanup_on_exit EXIT

# ─────────────────────────────────────────────────────────────────────────────
# Functions
# ─────────────────────────────────────────────────────────────────────────────

# Build Docker image
build_image() {
  log "INFO" "Building Docker image: $REGISTRY/$IMAGE_NAME:$TAG"
  if ! docker build -f "$DOCKERFILE" -t "$REGISTRY/$IMAGE_NAME:$TAG" .; then
    log "ERROR" "Docker build failed"
    return 1
  fi
}

# Push to registry
push_image() {
  log "INFO" "Authenticating to registry..."
  if ! echo "$REGISTRY_PASS" | docker login "$REGISTRY" -u "$REGISTRY_USER" --password-stdin; then
    log "ERROR" "Docker login failed"
    return 1
  fi

  log "INFO" "Pushing image to $REGISTRY/$IMAGE_NAME:$TAG"
  if ! docker push "$REGISTRY/$IMAGE_NAME:$TAG"; then
    log "ERROR" "Docker push failed"
    return 1
  fi

  # Get image digest for rollback (immutable reference)
  rollback_image_digest=$(docker inspect --format='{{index .Id}}' "$REGISTRY/$IMAGE_NAME:$TAG")
  log "INFO" "Image digest: $rollback_image_digest"
}

# Update git tag (create/update lightweight tag — no force push)
update_git_tag() {
  local tag_prefix="deploy/"
  local full_tag="${tag_prefix}${TAG}"

  log "INFO" "Updating git tag: $full_tag"
  # Prefer creating a new tag; if exists, delete then recreate (safe: no force push to branches)
  if git rev-parse "$full_tag" >/dev/null 2>&1; then
    log "INFO" "Tag '$full_tag' already exists — deleting and recreating"
    git tag -d "$full_tag" || log "WARN" "Tag deletion failed (may be non-fast-forward)"
  fi

  # Create tag pointing to current HEAD
  git tag "$full_tag" HEAD || {
    log "ERROR" "Failed to create tag '$full_tag'"
    return 1
  }

  # Push tag to remote (safe: no force push)
  if ! git push origin "$full_tag"; then
    log "ERROR" "Failed to push tag '$full_tag' to remote"
    return 1
  fi

  rollback_tag="$full_tag"
  log "INFO" "Git tag '$full_tag' updated and pushed"
}

# Deploy to production (placeholder for actual deployment)
# Replace with Helm, K8s, ECS, etc. as appropriate
deploy_production() {
  log "INFO" "Deploying to production with image digest: $rollback_image_digest"

  # Example for Kubernetes (uncomment and customize):
  # kubectl set image deployment/myapp myapp="$REGISTRY/$IMAGE_NAME:$TAG" \
  #   --record || { log "ERROR" "Kubernetes deployment failed"; return 1; }

  # Or using Helm (safer with rollback):
  # helm upgrade --install myapp ./chart \
  #   --set image.tag="$TAG" \
  #   --set image.digest="$rollback_image_digest" \
  #   --wait || { log "ERROR" "Helm upgrade failed"; return 1; }

  log "INFO" "Deployment simulation: image $REGISTRY/$IMAGE_NAME:$TAG deployed"
}

# ─────────────────────────────────────────────────────────────────────────────
# Main
# ─────────────────────────────────────────────────────────────────────────────

main() {
  # Validate required env vars
  if [[ -z "$REGISTRY_USER" || -z "$REGISTRY_PASS" ]]; then
    log "ERROR" "Missing required environment variables: REGISTRY_USER, REGISTRY_PASS"
    exit 1
  fi

  # Step 1: Build image
  build_image

  # Step 2: Push image
  push_image

  # Step 3: Update git tag
  update_git_tag

  # Step 4: Deploy to production
  deploy_production

  log "INFO" "✅ Deployment completed successfully"
}

main "$@"
```

### 🔁 Rollback Strategy Notes

- **Image digest rollback**: We capture the image digest (`docker inspect`) before deployment.
- **Tag rollback**: We record the tag name; in case of failure, you can manually re-deploy the old image digest.
- **Production rollback**: For real rollback, use:
  - Kubernetes: `kubectl rollout undo deployment/myapp`
  - Helm: `helm rollback <release>`
  - Or redeploy from the last known-good `deploy/*` tag.
