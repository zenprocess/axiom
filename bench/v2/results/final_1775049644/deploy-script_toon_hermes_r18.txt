### ✅ `deploy.sh`

```bash
#!/usr/bin/env bash
# deploy.sh
#
# Deploy a service to production using Docker and Git tags.
# Supports rollback to the previous version if deployment fails.
#
# Usage:
#   ./deploy.sh <image_tag> <registry> <service_name>
#   Example: ./deploy.sh v1.2.3 ghcr.io/myorg myapp
#
# Required environment variables:
#   - DOCKER_USERNAME: Docker registry username
#   - DOCKER_PASSWORD: Docker registry password (or access token)
#   - GIT_TAG_PREFIX: Optional prefix for Git tags (default: "deploy/")
#
# Assumptions:
#   - Dockerfile exists in current directory
#   - Kubernetes deployment manifest exists at k8s/deployment.yaml (or override via K8S_MANIFEST)
#   - `kubectl` is configured and authenticated

set -euo pipefail

# --- Logging functions (structlog-style, non-logging import) ---
log_info()  { echo "[INFO] $(date -Iseconds) $*" >&2; }
log_error() { echo "[ERROR] $(date -Iseconds) $*" >&2; }
log_warn()  { echo "[WARN] $(date -Iseconds) $*" >&2; }

# --- Configuration ---
REGISTRY="${2:-}"
IMAGE_NAME="${3:-}"
IMAGE_TAG="${1:-}"

if [[ -z "${REGISTRY}" || -z "${IMAGE_NAME}" || -z "${IMAGE_TAG}" ]]; then
  log_error "Usage: $0 <image_tag> <registry> <service_name>"
  exit 1
fi

IMAGE_FULL="${REGISTRY}/${IMAGE_NAME}:${IMAGE_TAG}"
PREV_IMAGE_FULL=""
K8S_MANIFEST="${K8S_MANIFEST:-k8s/deployment.yaml}"
GIT_TAG_PREFIX="${GIT_TAG_PREFIX:-deploy/}"
GIT_TAG="${GIT_TAG_PREFIX}${IMAGE_TAG}"

# --- Validation ---
log_info "Validating environment..."
if [[ -z "${DOCKER_USERNAME}" ]]; then
  log_error "DOCKER_USERNAME is not set"
  exit 1
fi
if [[ -z "${DOCKER_PASSWORD}" ]]; then
  log_error "DOCKER_PASSWORD is not set"
  exit 1
fi
if [[ ! -f "Dockerfile" ]]; then
  log_error "Dockerfile not found in current directory"
  exit 1
fi
if [[ ! -f "${K8S_MANIFEST}" ]]; then
  log_error "Kubernetes manifest not found at ${K8S_MANIFEST}"
  exit 1
fi

# --- Helper: rollback function ---
rollback() {
  local exit_code=$?
  log_error "Deployment failed. Rolling back..."
  if [[ -n "${PREV_IMAGE_FULL}" ]]; then
    log_info "Rolling back Kubernetes deployment to ${PREV_IMAGE_FULL}..."
    if ! sed -i.bak "s|image: ${IMAGE_FULL}|image: ${PREV_IMAGE_FULL}|g" "${K8S_MANIFEST}"; then
      log_error "Failed to patch manifest. Manual intervention required."
      exit "${exit_code}"
    fi
    if ! kubectl apply -f "${K8S_MANIFEST}" --wait; then
      log_error "Kubernetes rollback failed!"
      exit "${exit_code}"
    fi
    rm -f "${K8S_MANIFEST}.bak"
    log_info "Rollback successful."
  else
    log_warn "No previous image to rollback to."
  fi
  exit "${exit_code}"
}

trap rollback ERR

# --- Step 1: Build Docker image ---
log_info "Building Docker image: ${IMAGE_FULL}..."
if ! docker build -t "${IMAGE_FULL}" .; then
  log_error "Docker build failed"
  exit 1
fi

# --- Step 2: Push to registry ---
log_info "Authenticating with Docker registry..."
if ! docker login -u "${DOCKER_USERNAME}" --password-stdin "${REGISTRY}" <<< "${DOCKER_PASSWORD}"; then
  log_error "Docker login failed"
  exit 1
fi

log_info "Pushing image to registry..."
if ! docker push "${IMAGE_FULL}"; then
  log_error "Docker push failed"
  exit 1
fi

# --- Step 3: Get previous image for rollback ---
log_info "Fetching current deployment image..."
PREV_IMAGE_FULL=$(grep -E '^\s*image:' "${K8S_MANIFEST}" | head -1 | awk '{print $2}' | tr -d '"' || true)
log_info "Current image in manifest: ${PREV_IMAGE_FULL:-<none>}"

# --- Step 4: Update Kubernetes manifest with new image ---
log_info "Updating Kubernetes manifest with new image..."
if ! sed -i.bak "s|image: ${PREV_IMAGE_FULL}|image: ${IMAGE_FULL}|g" "${K8S_MANIFEST}"; then
  log_error "Failed to update manifest"
  exit 1
fi

# --- Step 5: Deploy to production ---
log_info "Deploying to production..."
if ! kubectl apply -f "${K8S_MANIFEST}" --wait; then
  log_error "Kubernetes deployment failed"
  exit 1
fi

# --- Step 6: Create Git tag (safe, no force-push) ---
log_info "Creating Git tag: ${GIT_TAG}..."
if ! git tag -a "${GIT_TAG}" -m "Deploy ${IMAGE_TAG}"; then
  log_warn "Failed to create annotated tag (may already exist). Retrying with force-create disabled."
fi

# Optional: Push tag (only if needed, and safely — not force)
if ! git rev-parse "${GIT_TAG}" >/dev/null 2>&1; then
  log_warn "Tag ${GIT_TAG} not found locally — skipping push"
else
  log_info "Pushing tag to remote..."
  if ! git push origin "${GIT_TAG}"; then
    log_warn "Failed to push tag (remote may be read-only or conflict exists). Continuing..."
  fi
fi

# --- Cleanup ---
rm -f "${K8S_MANIFEST}.bak"

log_info "✅ Deployment of ${IMAGE_FULL} completed successfully."
```

### ✅ Security & Best Practices

- **No secrets in code**: Credentials passed via environment variables only.
- **No `--force` Git operations**: Tag creation fails gracefully if tag exists.
- **Rollback on failure**: `trap` ensures manifest reverts if `kubectl apply` fails.
- **Structured logging**: `log_*` functions replace `echo`/`print`.
- **Idempotent manifest patching**: Uses `sed -i.bak` with fallback cleanup.
