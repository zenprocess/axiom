### 📜 `deploy.sh`

```bash
#!/bin/bash
set -euo pipefail

# Deploy script for containerized application
# Usage: ./deploy.sh <version-tag> [registry-url]
# Example: ./deploy.sh v1.2.3 docker.io/myorg

# Required environment variables (pass via shell or CI secret injection)
# - REGISTRY_USER: Docker registry username
# - REGISTRY_PASS: Docker registry password or access token

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
APP_NAME="${APP_NAME:-app}"
REGISTRY_URL="${2:-docker.io}"
IMAGE_NAME="${REGISTRY_URL}/${APP_NAME}"

# --- Functions ---

usage() {
    echo "Usage: $0 <version-tag> [registry-url]"
    echo "  e.g. $0 v1.2.3 docker.io/myorg"
    exit 1
}

log() {
    # Info-level logging (no debug prints)
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*"
}

error() {
    echo "[ERROR] $*" >&2
}

# Check preconditions
check_requirements() {
    for cmd in docker git; do
        if ! command -v "$cmd" >/dev/null 2>&1; then
            error "Required command '$cmd' not found. Install it first."
            exit 1
        fi
    done

    if [[ -z "${REGISTRY_USER:-}" ]]; then
        error "Environment variable REGISTRY_USER is not set."
        exit 1
    fi

    if [[ -z "${REGISTRY_PASS:-}" ]]; then
        error "Environment variable REGISTRY_PASS is not set."
        exit 1
    fi
}

# 1) Build the Docker image
build_image() {
    local version_tag="$1"
    log "Building Docker image: ${IMAGE_NAME}:${version_tag}..."
    docker build \
        --no-cache \
        --build-arg BUILD_DATE="$(date -u +'%Y-%m-%dT%H:%M:%SZ')" \
        -t "${IMAGE_NAME}:${version_tag}" \
        -t "${IMAGE_NAME}:latest" \
        "${SCRIPT_DIR}"
    log "Build completed."
}

# 2) Push image to registry
push_image() {
    local version_tag="$1"
    log "Logging into registry ${REGISTRY_URL}..."
    echo "${REGISTRY_PASS}" | docker login "${REGISTRY_URL}" --username "${REGISTRY_USER}" --password-stdin

    log "Pushing image ${IMAGE_NAME}:${version_tag}..."
    docker push "${IMAGE_NAME}:${version_tag}"
    docker push "${IMAGE_NAME}:latest"

    log "Image pushed successfully."
}

# 3) Update Git tag (safe, non-force push)
tag_git() {
    local version_tag="$1"
    log "Creating and pushing Git tag ${version_tag}..."

    # Check if tag already exists locally or remotely
    if git rev-parse "${version_tag}" >/dev/null 2>&1; then
        log "Tag ${version_tag} already exists locally."
    else
        git tag "${version_tag}"
    fi

    # Push tag (will fail if already exists on remote, but that's okay)
    if ! git push origin "${version_tag}" 2>/dev/null; then
        log "Tag ${version_tag} already exists on remote (safe to ignore)."
    fi
}

# 4) Deploy to production (Kubernetes example)
deploy_production() {
    local version_tag="$1"
    log "Deploying to production with image tag: ${version_tag}..."

    # Example for Kubernetes: update deployment image
    kubectl set image "deployment/${APP_NAME}" "${APP_NAME}=${IMAGE_NAME}:${version_tag}" \
        || { error "Deployment failed. Initiating rollback..."; rollback_production; exit 1; }

    log "Deployment successful."
}

# Rollback support: revert to previous stable version
rollback_production() {
    log "Rolling back deployment..."
    # For Kubernetes: rollback to previous revision
    if kubectl rollout undo "deployment/${APP_NAME}" 2>/dev/null; then
        log "Rollback completed."
    else
        error "Rollback failed. Manual intervention required."
    fi
}

# --- Main Flow ---

main() {
    if [[ $# -lt 1 ]]; then
        usage
    fi

    local version_tag="$1"
    check_requirements

    # Ensure git is clean and on main (optional safety)
    if [[ "$(git rev-parse --abbrev-ref HEAD)" != "main" ]]; then
        error "You must be on the 'main' branch to deploy."
        exit 1
    fi

    build_image "${version_tag}"
    push_image "${version_tag}"
    tag_git "${version_tag}"
    deploy_production "${version_tag}"

    log "✅ Deployment completed: ${IMAGE_NAME}:${version_tag}"
}

main "$@"
```

---

### 🔐 **Security & Best Practices**

- **No secrets in scripts**: Credentials passed via environment variables.
- **Non-force git tag**: Avoids rewriting history (`git push origin tag` is safe).
- **Atomic deployment + rollback**: Kubernetes example shows rollback on failure.
- **Strict error handling**: `set -euo pipefail` ensures failures fail fast.
- **Idempotency**: Checks for existing tags to avoid errors on re-runs.
