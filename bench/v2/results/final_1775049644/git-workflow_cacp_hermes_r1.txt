```python
import asyncio
import os
import subprocess
from typing import Optional
from pathlib import Path

import structlog
from pydantic import BaseModel

# Configure structlog
structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging_level="INFO"),
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.dev.ConsoleRenderer(),
    ],
)
logger = structlog.get_logger()


class GitConfig(BaseModel):
    repo_path: Path
    remote_name: str = "origin"
    base_branch: str = "main"
    feature_branch_prefix: str = "feature/"


class MRConfig(BaseModel):
    title: str
    description: str
    target_branch: str
    source_branch: str
    gitlab_token: Optional[str] = None
    github_token: Optional[str] = None
    project_id: Optional[str] = None  # Required for GitLab API
    repo_owner: Optional[str] = None  # Required for GitHub API
    repo_name: Optional[str] = None   # Required for GitHub API


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""


class GitWorkflow:
    def __init__(self, config: GitConfig):
        self.config = config

    async def _run_cmd(self, cmd: list[str], cwd: Optional[Path] = None) -> str:
        """Run a shell command asynchronously and return stdout."""
        try:
            proc = await asyncio.create_subprocess_exec(
                *cmd,
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
                cwd=str(cwd or self.config.repo_path),
            )
            stdout, stderr = await proc.communicate()
            if proc.returncode != 0:
                raise GitWorkflowError(
                    f"Command failed: {' '.join(cmd)}\nstderr: {stderr.decode().strip()}"
                )
            return stdout.decode().strip()
        except Exception as e:
            logger.error("cmd_failed", cmd=cmd, error=str(e))
            raise

    async def create_feature_branch(self, feature_name: str) -> str:
        """Create a new feature branch from base branch."""
        feature_branch = f"{self.config.feature_branch_prefix}{feature_name}"
        logger.info("creating_feature_branch", feature=feature_branch)

        # Checkout base branch and pull latest
        await self._run_cmd(["git", "checkout", self.config.base_branch])
        await self._run_cmd(["git", "pull", self.config.remote_name, self.config.base_branch])

        # Create and checkout new branch
        await self._run_cmd(["git", "checkout", "-b", feature_branch])
        logger.info("branch_created", branch=feature_branch)
        return feature_branch

    async def commit_changes(self, message: str, files: Optional[list[str]] = None) -> None:
        """Stage and commit changes."""
        logger.info("committing_changes", message=message)
        try:
            if files:
                await self._run_cmd(["git", "add", *files])
            else:
                await self._run_cmd(["git", "add", "."])

            await self._run_cmd(["git", "commit", "-m", message])
        except GitWorkflowError as e:
            logger.warning("commit_skipped", reason=str(e))

    async def push_branch(self, branch: str, force: bool = False) -> None:
        """Push branch to remote. Force is disallowed per policy."""
        if force:
            raise ValueError("Force push is prohibited per coding standards.")
        logger.info("pushing_branch", branch=branch)
        await self._run_cmd(["git", "push", self.config.remote_name, branch])

    async def create_merge_request(self, mr_config: MRConfig) -> str:
        """Create a merge request (GitHub or GitLab)."""
        logger.info("creating_merge_request", source=mr_config.source_branch, target=mr_config.target_branch)

        # GitHub API example
        if mr_config.github_token and mr_config.repo_owner and mr_config.repo_name:
            return await self._create_github_mr(mr_config)

        # GitLab API example
        if mr_config.gitlab_token and mr_config.project_id:
            return await self._create_gitlab_mr(mr_config)

        raise ValueError("No valid MR configuration provided (GitHub or GitLab).")

    async def _create_github_mr(self, mr: MRConfig) -> str:
        """Create a GitHub pull request via REST API."""
        import aiohttp

        url = f"https://api.github.com/repos/{mr.repo_owner}/{mr.repo_name}/pulls"
        headers = {
            "Authorization": f"token {mr.github_token}",
            "Accept": "application/vnd.github.v3+json",
        }
        data = {
            "title": mr.title,
            "head": mr.source_branch,
            "base": mr.target_branch,
            "body": mr.description,
        }

        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(url, headers=headers, json=data) as resp:
                    if not resp.ok:
                        text = await resp.text()
                        raise GitWorkflowError(f"GitHub PR creation failed: {resp.status} - {text}")
                    result = await resp.json()
                    return result.get("html_url", "")
        except Exception as e:
            logger.error("github_mr_failed", error=str(e))
            raise

    async def _create_gitlab_mr(self, mr: MRConfig) -> str:
        """Create a GitLab merge request via REST API."""
        import aiohttp

        url = f"https://gitlab.com/api/v4/projects/{mr.project_id}/merge_requests"
        headers = {"PRIVATE-TOKEN": mr.gitlab_token}

        data = {
            "title": mr.title,
            "source_branch": mr.source_branch,
            "target_branch": mr.target_branch,
            "description": mr.description,
        }

        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(url, headers=headers, json=data) as resp:
                    if not resp.ok:
                        text = await resp.text()
                        raise GitWorkflowError(f"GitLab MR creation failed: {resp.status} - {text}")
                    result = await resp.json()
                    return result.get("web_url", "")
        except Exception as e:
            logger.error("gitlab_mr_failed", error=str(e))
            raise


async def main():
    # Example usage
    config = GitConfig(repo_path=Path.cwd())
    workflow = GitWorkflow(config)

    # Create feature branch
    feature_branch = await workflow.create_feature_branch("add-auth")

    # Simulate changes and commit
    await workflow.commit_changes("Add authentication module", ["src/auth.py"])

    # Push branch (safe, no force)
    await workflow.push_branch(feature_branch)

    # Create MR (example for GitHub)
    mr_config = MRConfig(
        title="feat: add authentication module",
        description="Implements JWT-based auth",
        target_branch=config.base_branch,
        source_branch=feature_branch,
        github_token=os.getenv("GITHUB_TOKEN"),
        repo_owner="your-org",
        repo_name="your-repo",
    )
    pr_url = await workflow.create_merge_request(mr_config)
    logger.info("merge_request_created", url=pr_url)


if __name__ == "__main__":
    asyncio.run(main())
```
