```python
import asyncio
import os
import subprocess
from pathlib import Path
from typing import List, Optional

import structlog
from pydantic import BaseModel, Field

structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.stdlib.add_logger_name,
        structlog.stdlib.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.stdlib.LoggerFactory(),
    cache_logger_on_first_use=True,
)

logger = structlog.get_logger(__name__)


class GitConfig(BaseModel):
    """Configuration for Git repository operations."""

    repo_path: Path = Field(default_factory=Path.cwd)
    remote_name: str = "origin"
    base_branch: str = "main"
    branch_prefix: str = "feature/"


class MRConfig(BaseModel):
    """Configuration for merge request creation."""

    title: str
    description: str = ""
    target_branch: str = "main"
    source_branch: str = ""
    draft: bool = False


class GitWorkflowError(Exception):
    """Custom exception for Git workflow failures."""


async def _run_cmd(*args: str, cwd: Optional[Path] = None) -> str:
    """
    Run a shell command asynchronously and return stdout.

    Args:
        *args: Command and its arguments.
        cwd: Working directory (optional).

    Returns:
        Command output as string.

    Raises:
        GitWorkflowError: If command fails.
    """
    try:
        proc = await asyncio.create_subprocess_exec(
            *args,
            cwd=str(cwd) if cwd else None,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        stdout, stderr = await proc.communicate()
        if proc.returncode != 0:
            raise GitWorkflowError(f"Command failed: {' '.join(args)}\n{stderr.decode().strip()}")
        return stdout.decode().strip()
    except Exception as exc:
        logger.error("command_failed", cmd=args, error=str(exc))
        raise GitWorkflowError(f"Failed to execute command: {' '.join(args)}") from exc


async def ensure_clean_working_tree(config: GitConfig) -> None:
    """
    Ensure the working tree is clean before proceeding.

    Args:
        config: Git configuration.

    Raises:
        GitWorkflowError: If working tree is not clean.
    """
    status = await _run_cmd("git", "status", "--porcelain", cwd=config.repo_path)
    if status:
        raise GitWorkflowError(f"Working tree is not clean:\n{status}")


async def create_feature_branch(config: GitConfig, feature_name: str) -> str:
    """
    Create and checkout a new feature branch.

    Args:
        config: Git configuration.
        feature_name: Name of the feature (e.g., 'auth-fix').

    Returns:
        Full branch name (e.g., 'feature/auth-fix').
    """
    branch_name = f"{config.branch_prefix}{feature_name}"
    await _run_cmd("git", "checkout", "-b", branch_name, cwd=config.repo_path)
    logger.info("branch_created", branch=branch_name)
    return branch_name


async def commit_changes(config: GitConfig, message: str) -> None:
    """
    Stage and commit changes with the given message.

    Args:
        config: Git configuration.
        message: Commit message.
    """
    await _run_cmd("git", "add", ".", cwd=config.repo_path)
    await _run_cmd("git", "commit", "-m", message, cwd=config.repo_path)
    logger.info("changes_committed", message=message)


async def push_branch(config: GitConfig, branch_name: str) -> None:
    """
    Push branch to remote without force.

    Args:
        config: Git configuration.
        branch_name: Name of the branch to push.
    """
    try:
        await _run_cmd(
            "git",
            "push",
            "--no-verify",
            config.remote_name,
            branch_name,
            cwd=config.repo_path,
        )
        logger.info("branch_pushed", remote=config.remote_name, branch=branch_name)
    except GitWorkflowError as exc:
        if "non-fast-forward" in str(exc):
            # Recommend safe alternative: merge/rebase locally first
            logger.warning(
                "branch_push_rejected",
                branch=branch_name,
                advice="Consider rebasing or merging first, then push.",
            )
        raise


async def create_merge_request(config: GitConfig, mr_config: MRConfig) -> str:
    """
    Create a merge request via GitLab/GitHub CLI (placeholder).

    This function assumes `gh` (GitHub CLI) or `glab` (GitLab CLI) is installed and configured.

    Args:
        config: Git configuration.
        mr_config: Merge request configuration.

    Returns:
        URL of the created merge request (or placeholder if CLI not found).
    """
    # Try GitHub CLI first
    try:
        await _run_cmd("gh", "--version")
        cmd = [
            "gh",
            "pr",
            "create",
            "--title",
            mr_config.title,
            "--body",
            mr_config.description,
            "--base",
            mr_config.target_branch,
            "--head",
            mr_config.source_branch,
        ]
        if mr_config.draft:
            cmd.append("--draft")
        output = await _run_cmd(*cmd, cwd=config.repo_path)
        logger.info("merge_request_created", output=output)
        return output
    except GitWorkflowError:
        pass

    # Try GitLab CLI
    try:
        await _run_cmd("glab", "--version")
        cmd = [
            "glab",
            "mr",
            "create",
            "--title",
            mr_config.title,
            "--description",
            mr_config.description,
            "--target-branch",
            mr_config.target_branch,
            "--source-branch",
            mr_config.source_branch,
        ]
        if mr_config.draft:
            cmd.append("--draft")
        output = await _run_cmd(*cmd, cwd=config.repo_path)
        logger.info("merge_request_created", output=output)
        return output
    except GitWorkflowError:
        pass

    # Fallback: log but do not fail
    logger.warning(
        "merge_request_cli_missing",
        mr_title=mr_config.title,
        advice="Install `gh` or `glab` CLI and configure auth.",
    )
    return "https://example.com/merge-request-placeholder"


async def run_workflow(
    feature_name: str,
    commit_message: str,
    mr_title: str,
    mr_description: str = "",
    repo_path: Optional[Path] = None,
) -> str:
    """
    Run the full Git workflow: branch, commit, push, and MR.

    Args:
        feature_name: Name for the feature branch.
        commit_message: Commit message.
        mr_title: Title for the merge request.
        mr_description: Description for the merge request.
        repo_path: Path to the Git repository (default: current directory).

    Returns:
        URL or identifier of the created merge request.
    """
    config = GitConfig(repo_path=repo_path or Path.cwd())
    mr_config = MRConfig(
        title=mr_title,
        description=mr_description,
        source_branch=f"{config.branch_prefix}{feature_name}",
    )

    try:
        await ensure_clean_working_tree(config)
        branch = await create_feature_branch(config, feature_name)
        await commit_changes(config, commit_message)
        await push_branch(config, branch)
        return await create_merge_request(config, mr_config)
    except GitWorkflowError as exc:
        logger.error("workflow_failed", error=str(exc))
        raise


# Example usage (not executed if imported)
if __name__ == "__main__":
    asyncio.run(
        run_workflow(
            feature_name="user-auth",
            commit_message="Implement user authentication",
            mr_title="feat: add user authentication",
            mr_description="Adds JWT-based auth for users.",
        )
    )
```
