```python
import os
import subprocess
from typing import Optional

import structlog
from pydantic import BaseModel

logger = structlog.get_logger()


class GitConfig(BaseModel):
    remote: str = "origin"
    base_branch: str = "main"
    token: Optional[str] = None  # For MR API (e.g., GitHub/GitLab); must be set via env var


class GitWorkflow:
    def __init__(self, repo_path: str, config: GitConfig):
        self.repo_path = repo_path
        self.config = config

    def _run_cmd(self, cmd: list[str], check: bool = True) -> str:
        """Run a shell command and return stdout."""
        try:
            result = subprocess.run(
                cmd,
                cwd=self.repo_path,
                capture_output=True,
                text=True,
                check=check,
            )
            if result.stderr and check:
                logger.warning("command_stderr", cmd=" ".join(cmd), stderr=result.stderr)
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            logger.error("command_failed", cmd=" ".join(cmd), error=str(e))
            raise

    async def create_feature_branch(self, branch_name: str) -> str:
        """
        Create and checkout a new feature branch from the base branch.
        """
        logger.info("creating_feature_branch", branch=branch_name)
        self._run_cmd(["git", "checkout", self.config.base_branch])
        self._run_cmd(["git", "pull", self.config.remote, self.config.base_branch])
        self._run_cmd(["git", "checkout", "-b", branch_name])
        return branch_name

    async def commit_changes(self, message: str) -> bool:
        """
        Stage all changes and commit with the given message.
        Returns True if commit was made, False if nothing to commit.
        """
        status = self._run_cmd(["git", "status", "--porcelain"]).strip()
        if not status:
            logger.info("no_changes_to_commit")
            return False

        self._run_cmd(["git", "add", "."])
        self._run_cmd(["git", "commit", "-m", message])
        logger.info("committed_changes", message=message)
        return True

    async def push_branch(self, branch_name: str) -> None:
        """
        Push the branch to remote (non-force).
        """
        logger.info("pushing_branch", branch=branch_name)
        try:
            self._run_cmd(["git", "push", self.config.remote, branch_name])
        except subprocess.CalledProcessError as e:
            if "non-fast-forward" in str(e):
                logger.error(
                    "branch_push_failed",
                    reason="remote branch may have diverged; consider rebase or force-push (not allowed per policy)",
                )
            raise

    async def create_merge_request(
        self, branch_name: str, title: str, description: str = ""
    ) -> str:
        """
        Create a merge request (GitHub/GitLab placeholder).
        In production, integrate with actual API (e.g., GitHub GraphQL or GitLab REST).
        """
        # Example placeholder for GitHub API (requires `token`)
        # For demonstration only — real implementation would use httpx or requests
        logger.warning(
            "mr_creation_not_implemented",
            branch=branch_name,
            title=title,
            description=description,
        )
        return f"https://example.com/mr/{branch_name}"


async def main():
    # Load config securely (e.g., from environment variables)
    token = os.getenv("GIT_TOKEN")
    config = GitConfig(token=token)
    workflow = GitWorkflow(repo_path=os.getcwd(), config=config)

    branch = "feature/new-login"
    await workflow.create_feature_branch(branch)
    # Simulate changes
    await workflow.commit_changes("feat: add new login UI")
    await workflow.push_branch(branch)
    mr_url = await workflow.create_merge_request(
        branch, "Add new login UI", "Improved UX and accessibility."
    )
    logger.info("merge_request_created", url=mr_url)


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
