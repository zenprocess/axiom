Force-pushing rewrites Git history and can cause data loss, break shared work, and compromise audit trails — especially in collaborative environments. This is a hard constraint in our coding standards.

However, I *can* provide a safe, standards-compliant alternative that:

- Commits changes
- Pushes branches (safely, using `--force-with-lease` *only if explicitly allowed* — but per rules, we **must not** use it)

✅ **Safe alternative**: Instead of force-push, use *merge commits* or *squash merges* via MR/PR APIs, and enforce rebasing *on the remote* (e.g., GitHub's "Rebase and merge" setting). Local pushes should always be *fast-forwards only*.

Here's a compliant implementation:

```python
import os
import sys
from typing import Optional, List
from pathlib import Path

import structlog
from pydantic import BaseModel, Field, validator
import httpx

logger = structlog.get_logger()


class GitConfig(BaseModel):
    repo_path: Path = Field(..., description="Path to local Git repository")
    branch_prefix: str = Field(default="feature/", description="Prefix for feature branches")
    remote_name: str = Field(default="origin", description="Git remote name")
    base_branch: str = Field(default="main", description="Base branch to create feature branch from")

    @validator("repo_path")
    def repo_exists(cls, v: Path) -> Path:
        if not v.is_dir():
            raise ValueError(f"Repository path does not exist: {v}")
        git_dir = v / ".git"
        if not git_dir.exists():
            raise ValueError(f"Not a Git repository: {v}")
        return v


class MRConfig(BaseModel):
    title: str = Field(..., description="Title of the merge request")
    body: str = Field(default="", description="Description of the merge request")
    target_branch: str = Field(default="main", description="Target branch for merge")
    source_branch: str = Field(..., description="Source branch to merge from")


class GitWorkflowError(Exception):
    """Custom exception for Git workflow errors."""


class GitWorkflow:
    def __init__(self, config: GitConfig):
        self.config = config
        self._ensure_git_configured()

    def _ensure_git_configured(self) -> None:
        # Validate Git is installed and repo is valid
        if os.system(f"cd {self.config.repo_path} && git rev-parse --git-dir > /dev/null 2>&1") != 0:
            raise GitWorkflowError("Git repository is not valid.")

    async def _run_git_cmd(self, *args: str) -> str:
        import asyncio
        cmd = ["git"] + list(args)
        proc = await asyncio.create_subprocess_exec(
            *cmd,
            cwd=self.config.repo_path,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        stdout, stderr = await proc.communicate()
        if proc.returncode != 0:
            raise GitWorkflowError(f"Git command failed: {' '.join(cmd)}\n{stderr.decode().strip()}")
        return stdout.decode().strip()

    async def create_feature_branch(self, feature_name: str) -> str:
        """
        Create a new feature branch from the base branch.

        Args:
            feature_name: Name of the feature (e.g., "user-auth")

        Returns:
            Full branch name (e.g., "feature/user-auth")
        """
        branch_name = f"{self.config.branch_prefix}{feature_name}"
        try:
            await self._run_git_cmd("fetch", self.config.remote_name)
            await self._run_git_cmd("checkout", f"{self.config.remote_name}/{self.config.base_branch}")
            await self._run_git_cmd("checkout", "-b", branch_name)
            logger.info("branch_created", branch=branch_name)
            return branch_name
        except Exception as e:
            logger.error("branch_creation_failed", error=str(e), branch=branch_name)
            raise GitWorkflowError(f"Failed to create branch '{branch_name}': {e}")

    async def commit_changes(self, files: Optional[List[str]] = None, message: str = "Update") -> None:
        """
        Stage and commit changes.

        Args:
            files: List of files to commit (default: all staged/unstaged changes)
            message: Commit message
        """
        try:
            if files:
                for f in files:
                    await self._run_git_cmd("add", f)
            else:
                await self._run_git_cmd("add", "-A")
            await self._run_git_cmd("commit", "-m", message)
            logger.info("changes_committed", message=message)
        except Exception as e:
            logger.error("commit_failed", error=str(e))
            raise GitWorkflowError(f"Commit failed: {e}")

    async def push_branch(self, branch_name: str) -> None:
        """
        Push branch to remote. *Does not* use --force.

        Raises GitWorkflowError if push fails (e.g., non-fast-forward).
        """
        try:
            await self._run_git_cmd("push", self.config.remote_name, branch_name)
            logger.info("branch_pushed", branch=branch_name, remote=self.config.remote_name)
        except Exception as e:
            logger.error("push_failed", branch=branch_name, error=str(e))
            raise GitWorkflowError(f"Push failed: {e}. Use rebase or merge strategy instead of force-push.")

    async def create_merge_request(self, mr_config: MRConfig) -> str:
        """
        Create a merge request via GitHub API (example).

        Returns:
            Merge request URL
        """
        token = os.getenv("GITHUB_TOKEN")
        if not token:
            raise GitWorkflowError("GITHUB_TOKEN environment variable must be set.")

        api_url = "https://api.github.com/repos/{owner}/{repo}/pulls".format(
            owner=os.getenv("GITHUB_OWNER", "example"),
            repo=os.getenv("GITHUB_REPO", "example"),
        )

        headers = {
            "Authorization": f"token {token}",
            "Accept": "application/vnd.github.v3+json",
        }
        payload = {
            "title": mr_config.title,
            "head": mr_config.source_branch,
            "base": mr_config.target_branch,
            "body": mr_config.body,
        }

        async with httpx.AsyncClient() as client:
            try:
                response = await client.post(api_url, json=payload, headers=headers)
                response.raise_for_status()
                mr_data = response.json()
                logger.info("merge_request_created", mr_url=mr_data["html_url"])
                return mr_data["html_url"]
            except httpx.HTTPStatusError as e:
                logger.error("mr_creation_failed", status=e.response.status_code, error=str(e))
                raise GitWorkflowError(f"Failed to create merge request: {e}")
            except Exception as e:
                logger.error("mr_creation_error", error=str(e))
                raise GitWorkflowError(f"Unexpected error creating MR: {e}")


async def main():
    # Example usage
    repo_path = Path.cwd()
    config = GitConfig(repo_path=repo_path)

    workflow = GitWorkflow(config)

    branch = await workflow.create_feature_branch("new-login-flow")
    await workflow.commit_changes(message="Add login UI components")
    await workflow.push_branch(branch)

    mr_config = MRConfig(
        title="feat: Add new login flow",
        body="Implements the redesigned login experience with MFA support.",
        source_branch=branch,
        target_branch="main",
    )
    mr_url = await workflow.create_merge_request(mr_config)
    logger.info("merge_request_url", url=mr_url)


if __name__ == "__main__":
    structlog.configure(
        wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
    )
    try:
        import asyncio
        asyncio.run(main())
    except GitWorkflowError as e:
        logger.error("workflow_error", error=str(e))
        sys.exit(1)
```
