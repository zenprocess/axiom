#!/usr/bin/env python3
"""
Git workflow automation module for managing feature branches and merge requests.

This module provides utilities for:
- Creating feature branches
- Committing and pushing changes
- Creating merge requests
"""

import os
import subprocess
import sys
from typing import Optional

import structlog
from pydantic import BaseModel

logger = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = "origin"
    base_branch: str = "main"
    feature_prefix: str = "feature/"


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""
    pass


class GitWorkflow:
    """Automated git workflow management."""

    def __init__(self, config: Optional[GitConfig] = None):
        """
        Initialize the Git workflow manager.

        Args:
            config: Git configuration settings (optional)
        """
        self.config = config or GitConfig()
        self._ensure_git_repo()

    def _run_command(self, command: list[str], check: bool = True) -> subprocess.CompletedProcess:
        """
        Run a shell command and return the result.

        Args:
            command: List of command arguments
            check: Whether to raise exception on non-zero exit code

        Returns:
            CompletedProcess instance with command results

        Raises:
            GitWorkflowError: If command fails and check=True
        """
        try:
            result = subprocess.run(
                command,
                capture_output=True,
                text=True,
                check=check,
            )
            return result
        except subprocess.CalledProcessError as e:
            logger.error("command_failed", command=command, error=e.stderr.strip())
            raise GitWorkflowError(f"Command failed: {' '.join(command)}") from e

    def _ensure_git_repo(self) -> None:
        """Verify current directory is a git repository."""
        result = self._run_command(["git", "rev-parse", "--is-inside-work-tree"], check=False)
        if result.returncode != 0:
            raise GitWorkflowError("Current directory is not a git repository")

    def create_feature_branch(self, feature_name: str) -> str:
        """
        Create a new feature branch from the base branch.

        Args:
            feature_name: Name for the feature branch (without prefix)

        Returns:
            The full branch name created

        Raises:
            GitWorkflowError: If branch creation fails
        """
        branch_name = f"{self.config.feature_prefix}{feature_name}"
        
        # Ensure we're on the base branch and up-to-date
        self._run_command(["git", "checkout", self.config.base_branch])
        self._run_command(["git", "pull", self.config.remote, self.config.base_branch])
        
        # Create and checkout the feature branch
        self._run_command(["git", "checkout", "-b", branch_name])
        logger.info("branch_created", branch=branch_name)
        return branch_name

    def add_files(self, files: list[str]) -> None:
        """
        Add files to the staging area.

        Args:
            files: List of file paths to add
        """
        if not files:
            logger.warning("no_files_specified", action="add")
            return
        
        self._run_command(["git", "add"] + files)
        logger.info("files_added", files=files)

    def commit_changes(self, message: str) -> None:
        """
        Commit staged changes with the provided message.

        Args:
            message: Commit message
        """
        self._run_command(["git", "commit", "-m", message])
        logger.info("changes_committed", message=message)

    def push_branch(self, branch_name: str, force: bool = False) -> None:
        """
        Push a branch to the remote repository.

        Args:
            branch_name: Name of the branch to push
            force: Whether to use force push (use with caution)

        Raises:
            GitWorkflowError: If push fails
        """
        command = ["git", "push"]
        if force:
            command.append("--force-with-lease")
        command.extend([self.config.remote, branch_name])
        
        self._run_command(command)
        logger.info("branch_pushed", branch=branch_name, force=force)

    def create_merge_request(
        self,
        branch_name: str,
        title: str,
        description: str = "",
        assignees: Optional[list[str]] = None,
        labels: Optional[list[str]] = None,
    ) -> str:
        """
        Create a merge request for the feature branch.

        This is a simplified implementation using git commands.
        In production, you would integrate with platforms like GitHub/GitLab APIs.

        Args:
            branch_name: Name of the feature branch
            title: Title for the merge request
            description: Description for the merge request
            assignees: List of usernames to assign (optional)
            labels: List of labels to apply (optional)

        Returns:
            URL or identifier for the created merge request
        """
        # This is a placeholder implementation
        # In real scenarios, you would use platform-specific APIs
        # For example, GitHub API: https://docs.github.com/en/rest/pulls/pulls#create-a-pull-request
        # For GitLab API: https://docs.gitlab.com/ee/api/merge_requests.html#create-merge-request
        
        logger.info(
            "merge_request_created",
            branch=branch_name,
            title=title,
            description=description[:100] + "..." if len(description) > 100 else description,
        )
        
        # Return a placeholder MR reference
        mr_url = f"{self.config.remote}/merge_requests/new?merge_request%5Bsource_branch%5D={branch_name}&merge_request%5Btarget_branch%5D={self.config.base_branch}"
        return mr_url


async def main() -> None:
    """
    Main entry point for the git workflow automation.

    This function demonstrates the complete workflow:
    1. Create feature branch
    2. Add and commit changes
    3. Push branch
    4. Create merge request

    Note: In production, this would be replaced with actual CLI parsing
    and integration with CI/CD systems.
    """
    # Example usage
    try:
        # Initialize workflow manager
        workflow = GitWorkflow()
        
        # Create feature branch
        branch = workflow.create_feature_branch("user-authentication")
        
        # Add and commit changes
        workflow.add_files(["src/auth.py", "tests/test_auth.py"])
        workflow.commit_changes("Add user authentication module")
        
        # Push branch (using force-with-lease for rebased branches)
        workflow.push_branch(branch, force=False)
        
        # Create merge request
        mr_url = workflow.create_merge_request(
            branch_name=branch,
            title="Add user authentication module",
            description="Implements user authentication with JWT tokens",
            assignees=["dev1", "dev2"],
            labels=["feature", "security"],
        )
        
        logger.info("workflow_completed", mr_url=mr_url)
        
    except GitWorkflowError as e:
        logger.error("workflow_failed", error=str(e))
        sys.exit(1)
    except Exception as e:
        logger.error("unexpected_error", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
