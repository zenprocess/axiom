```python
#!/usr/bin/env python3
"""
Git workflow automation script.

Creates feature branches, commits changes, pushes to remote (safely),
and creates merge requests via GitLab/GitHub APIs.
"""

import os
import sys
import subprocess
from pathlib import Path
from typing import Optional, List, Dict, Any

import structlog
from pydantic import BaseModel, Field, validator

structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
)
logger = structlog.get_logger()


class MRConfig(BaseModel):
    """Configuration for a merge request."""
    title: str
    description: str = ""
    source_branch: str
    target_branch: str = "main"
    labels: List[str] = Field(default_factory=list)


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""
    pass


class GitWorkflow:
    """Automates git workflows safely."""

    def __init__(self, repo_path: Path):
        self.repo_path = repo_path
        self.current_branch: Optional[str] = None

    async def run_git(self, *args: str, check: bool = True) -> str:
        """Run a git command asynchronously."""
        try:
            result = subprocess.run(
                ["git", *args],
                cwd=self.repo_path,
                capture_output=True,
                text=True,
                check=check,
            )
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            logger.error(
                "git_command_failed",
                args=args,
                stderr=e.stderr,
                stdout=e.stdout,
                returncode=e.returncode,
            )
            raise GitWorkflowError(f"Git command failed: {e.stderr or e.stdout}") from e

    async def get_current_branch(self) -> str:
        """Get current branch name."""
        return await self.run_git("rev-parse", "--abbrev-ref", "HEAD")

    async def create_feature_branch(self, branch_name: str) -> None:
        """Create and checkout a new feature branch."""
        logger.info("creating_feature_branch", branch=branch_name)
        await self.run_git("checkout", "-b", branch_name)

    async def commit_changes(self, message: str) -> None:
        """Stage all changes and commit."""
        logger.info("committing_changes", message=message)
        await self.run_git("add", ".")
        await self.run_git("commit", "-m", message)

    async def push_to_remote(
        self, branch: str, remote: str = "origin", force_with_lease: bool = True
    ) -> None:
        """Push branch to remote repository."""
        logger.info(
            "pushing_to_remote", branch=branch, remote=remote, force_with_lease=force_with_lease
        )
        if force_with_lease:
            await self.run_git("push", "--force-with-lease", remote, branch)
        else:
            await self.run_git("push", remote, branch)

    async def create_merge_request(
        self,
        config: MRConfig,
        api_url: str,
        token: str,
    ) -> Dict[str, Any]:
        """
        Create a merge request using GitLab or GitHub API.

        Uses GitLab API by default; switch to GitHub by modifying the request format.
        """
        headers = {
            "PRIVATE-TOKEN": token,
            "Content-Type": "application/json",
        }

        payload = {
            "title": config.title,
            "description": config.description,
            "source_branch": config.source_branch,
            "target_branch": config.target_branch,
            "labels": ",".join(config.labels) if config.labels else "",
        }

        try:
            import aiohttp
            async with aiohttp.ClientSession() as session:
                async with session.post(
                    f"{api_url}/merge_requests",
                    headers=headers,
                    json=payload,
                ) as response:
                    if response.status >= 400:
                        error_text = await response.text()
                        logger.error("mr_creation_failed", status=response.status, error=error_text)
                        raise GitWorkflowError(
                            f"Failed to create MR: {response.status} - {error_text}"
                        )
                    return await response.json()
        except ImportError:
            logger.error("aiohttp_not_installed")
            raise GitWorkflowError("aiohttp is required for MR creation") from None


async def main():
    """Main entry point."""
    repo_path = Path(os.getcwd())
    if not (repo_path / ".git").exists():
        logger.error("not_a_git_repository", path=str(repo_path))
        sys.exit(1)

    workflow = GitWorkflow(repo_path)

    # Get current branch to avoid conflicts
    current_branch = await workflow.get_current_branch()
    if current_branch != "main":
        logger.warning("not_on_main_branch", current_branch=current_branch)

    # Create feature branch
    feature_branch = "feature/new-api-integration"
    await workflow.create_feature_branch(feature_branch)

    # Commit changes (simulate)
    # In real usage, you'd have actual file changes
    await workflow.commit_changes("feat: add new API integration")

    # Push safely with --force-with-lease
    await workflow.push_to_remote(feature_branch, force_with_lease=True)

    # Create merge request (requires valid API URL and token)
    mr_config = MRConfig(
        title="Feature: New API Integration",
        description="Implements support for external payment gateway.",
        source_branch=feature_branch,
        target_branch="main",
        labels=["feature", "backend"],
    )

    # For demonstration: this would normally use real values
    api_url = os.getenv("GITLAB_API_URL", "https://gitlab.com/api/v4/projects/123")
    token = os.getenv("GITLAB_TOKEN")

    if not token:
        logger.warning("no_gitlab_token_provided")
    else:
        try:
            mr = await workflow.create_merge_request(mr_config, api_url, token)
            logger.info("merge_request_created", mr_id=mr.get("iid"), url=mr.get("web_url"))
        except GitWorkflowError as e:
            logger.error("mr_creation_error", message=str(e))


if __name__ == "__main__":
    import asyncio
    import logging

    # Set up logging (structlog will be used, but we need to silence asyncio warnings)
    logging.basicConfig(level=logging.WARNING)
    asyncio.run(main())
```
