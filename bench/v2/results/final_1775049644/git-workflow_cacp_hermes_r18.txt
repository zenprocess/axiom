```python
#!/usr/bin/env python3
"""
Git Workflow Automation Script

Creates feature branches, commits changes, pushes to remote (non-force),
and creates merge requests via GitHub/GitLab APIs.
"""

import os
import sys
import subprocess
from pathlib import Path
from typing import Optional, Literal

import structlog
from pydantic import BaseModel, Field, HttpUrl, SecretStr
from git import Repo, GitCommandError


structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.dev.ConsoleRenderer(colors=True),
    ],
)
logger = structlog.get_logger()


class MRConfig(BaseModel):
    """Configuration for creating a merge request."""
    title: str = Field(..., description="Title of the merge request")
    description: str = Field(default="", description="Description of the merge request")
    target_branch: str = Field(default="main", description="Target branch for merge")
    source_branch: str = Field(..., description="Source feature branch")
    project_id: str = Field(..., description="GitLab project ID or GitHub repo slug")


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""
    pass


async def run_cmd(cmd: list[str], cwd: Optional[Path] = None) -> str:
    """Run a shell command and return stdout."""
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd,
            capture_output=True,
            text=True,
            check=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        logger.error("Command failed", cmd=cmd, stderr=e.stderr, stdout=e.stdout)
        raise GitWorkflowError(f"Command failed: {' '.join(cmd)}") from e


async def create_feature_branch(repo: Repo, feature_name: str) -> str:
    """
    Create and checkout a new feature branch.

    Args:
        repo: GitPython Repo instance
        feature_name: Name of the feature branch (e.g., "add-login")

    Returns:
        Full branch name (e.g., "feature/add-login")
    """
    branch_name = f"feature/{feature_name}"
    logger.info("Creating feature branch", branch=branch_name)
    repo.git.checkout("HEAD", b=branch_name)
    return branch_name


async def commit_changes(repo: Repo, message: str) -> None:
    """
    Stage and commit all changes in the repo.

    Args:
        repo: GitPython Repo instance
        message: Commit message
    """
    logger.info("Committing changes", message=message)
    repo.git.add(A=True)
    repo.index.commit(message)


async def push_branch(repo: Repo, branch: str, remote: str = "origin") -> None:
    """
    Push branch to remote (safe push, never force).

    Args:
        repo: GitPython Repo instance
        branch: Branch name to push
        remote: Remote name (default: 'origin')
    """
    logger.info("Pushing branch", branch=branch, remote=remote)
    try:
        origin = repo.remotes[remote]
        origin.push(refspec=f"{branch}:{branch}", force=False)
        logger.info("Branch pushed successfully", branch=branch)
    except GitCommandError as e:
        raise GitWorkflowError(f"Failed to push branch '{branch}': {e}") from e


async def create_mr_github(
    repo_path: Path,
    config: MRConfig,
    token: SecretStr,
) -> HttpUrl:
    """
    Create a GitHub pull request via API.

    Args:
        repo_path: Path to local git repo
        config: MR configuration
        token: GitHub personal access token

    Returns:
        URL of the created pull request
    """
    import aiohttp

    repo = Repo(repo_path)
    remote_url = repo.remotes.origin.url
    if not remote_url.endswith(".git"):
        raise GitWorkflowError("Remote URL must end with .git")
    owner_repo = remote_url[:-4].split("/")[-2:]
    owner, repo_name = owner_repo[0], owner_repo[1].replace(".git", "")

    url = f"https://api.github.com/repos/{owner}/{repo_name}/pulls"

    headers = {
        "Authorization": f"token {token.get_secret_value()}",
        "Accept": "application/vnd.github.v3+json",
    }

    data = {
        "title": config.title,
        "head": config.source_branch,
        "base": config.target_branch,
        "body": config.description,
    }

    async with aiohttp.ClientSession() as session:
        try:
            async with session.post(url, headers=headers, json=data) as resp:
                if resp.status == 201:
                    result = await resp.json()
                    pr_url = HttpUrl(result["html_url"])
                    logger.info("Pull request created", url=str(pr_url))
                    return pr_url
                else:
                    text = await resp.text()
                    raise GitWorkflowError(
                        f"GitHub API error: {resp.status} - {text}"
                    )
        except aiohttp.ClientError as e:
            raise GitWorkflowError("Network error while creating PR") from e


async def create_mr_gitlab(
    repo_path: Path,
    config: MRConfig,
    token: SecretStr,
    gitlab_url: str = "https://gitlab.com",
) -> HttpUrl:
    """
    Create a GitLab merge request via API.

    Args:
        repo_path: Path to local git repo
        config: MR configuration
        token: GitLab access token
        gitlab_url: GitLab instance URL (default: gitlab.com)

    Returns:
        URL of the created merge request
    """
    import aiohttp

    url = f"{gitlab_url.rstrip('/')}/api/v4/projects/{config.project_id}/merge_requests"

    headers = {
        "PRIVATE-TOKEN": token.get_secret_value(),
    }

    data = {
        "title": config.title,
        "source_branch": config.source_branch,
        "target_branch": config.target_branch,
        "description": config.description,
    }

    async with aiohttp.ClientSession() as session:
        try:
            async with session.post(url, headers=headers, json=data) as resp:
                if resp.status == 201:
                    result = await resp.json()
                    mr_url = HttpUrl(result["web_url"])
                    logger.info("Merge request created", url=str(mr_url))
                    return mr_url
                else:
                    text = await resp.text()
                    raise GitWorkflowError(
                        f"GitLab API error: {resp.status} - {text}"
                    )
        except aiohttp.ClientError as e:
            raise GitWorkflowError("Network error while creating MR") from e


async def main(
    feature_name: str,
    commit_msg: str,
    mr_type: Literal["github", "gitlab"] = "github",
    mr_config: Optional[MRConfig] = None,
    token: Optional[SecretStr] = None,
) -> Optional[HttpUrl]:
    """
    Automate git workflow: feature branch → commit → push → MR.

    Args:
        feature_name: Name for the feature branch
        commit_msg: Commit message
        mr_type: Target platform ('github' or 'gitlab')
        mr_config: MR configuration (required if creating MR)
        token: API token (required if creating MR)

    Returns:
        URL of created MR or None if not created
    """
    if mr_config is None or token is None:
        logger.warning("MR creation skipped: missing config or token")
        mr_type = None

    repo_path = Path.cwd()
    if not (repo_path / ".git").exists():
        raise GitWorkflowError("Current directory is not a git repository")

    repo = Repo(str(repo_path))

    # Ensure clean working directory
    if repo.is_dirty():
        raise GitWorkflowError("Repository has uncommitted changes")

    # Create feature branch
    branch = await create_feature_branch(repo, feature_name)

    # Commit changes (simulated here as no real changes exist)
    # In real usage, user would modify files before calling this
    await commit_changes(repo, commit_msg or f"feat: {feature_name}")

    # Push branch safely (no force)
    await push_branch(repo, branch)

    # Create MR if configured
    if mr_type and mr_config and token:
        if mr_type == "github":
            return await create_mr_github(repo_path, mr_config, token)
        elif mr_type == "gitlab":
            return await create_mr_gitlab(repo_path, mr_config, token)
        else:
            raise GitWorkflowError(f"Unsupported MR type: {mr_type}")

    logger.info("Workflow complete (MR not created)")
    return None


if __name__ == "__main__":
    import asyncio
    import argparse
    import logging

    logging.basicConfig(level=logging.INFO)

    parser = argparse.ArgumentParser(
        description="Automate git workflow with safe branching and MR creation"
    )
    parser.add_argument("feature", help="Feature branch name (e.g., 'user-auth')")
    parser.add_argument(
        "--commit-msg", default=None, help="Commit message (default: auto-generated)"
    )
    parser.add_argument(
        "--mr-type",
        choices=["github", "gitlab"],
        help="Create MR on GitHub or GitLab",
    )
    parser.add_argument("--mr-title", help="Merge request title")
    parser.add_argument("--mr-desc", default="", help="Merge request description")
    parser.add_argument(
        "--target-branch", default="main", help="Target branch for MR"
    )
    parser.add_argument(
        "--source-branch", help="Source branch (defaults to feature/{feature})"
    )
    parser.add_argument(
        "--project-id", help="GitLab project ID (required for GitLab MRs)"
    )
    parser.add_argument(
        "--token",
        default=os.getenv("GITHUB_TOKEN") or os.getenv("GITLAB_TOKEN"),
        help="API token (from env var or argument)",
    )

    args = parser.parse_args()

    if args.mr_type and not args.token:
        logger.error("API token required for MR creation")
        sys.exit(1)

    mr_config = None
    if args.mr_type and args.token:
        mr_config = MRConfig(
            title=args.mr_title or f"feat: {args.feature}",
            description=args.mr_desc,
            target_branch=args.target_branch,
            source_branch=args.source_branch or f"feature/{args.feature}",
            project_id=args.project_id or "",
        )

    async def run():
        try:
            result = await main(
                feature_name=args.feature,
                commit_msg=args.commit_msg,
                mr_type=args.mr_type,
                mr_config=mr_config,
                token=SecretStr(args.token),
            )
            if result:
                print(f"✅ MR created: {result}")
        except GitWorkflowError as e:
            logger.error("Workflow failed", error=str(e))
            sys.exit(1)

    asyncio.run(run())
```
