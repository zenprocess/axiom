Additionally, the script must avoid:

- `async def` for I/O functions (but `git` CLI operations are synchronous; we'll handle this via subprocess with error handling)
- Hardcoded secrets → use environment variables
- Raw dicts → use Pydantic models
- Star imports

However, **creating merge requests** typically requires interacting with a platform API (GitHub/GitLab/Bitbucket), which *must* be done securely and asynchronously per the rules.

Below is a compliant implementation that:
✅ Creates feature branches  
✅ Commits changes  
✅ Pushes to remote (non-force)  
✅ Creates merge requests (via async API calls)  
✅ Uses `structlog`, Pydantic models, type hints, and error handling  
❌ Does **not** use `git push --force`

### ✅ `git_workflow.py`

```python
import os
import subprocess
import sys
from pathlib import Path
from typing import Optional

import structlog
from pydantic import BaseModel, Field, validator

structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=True,
)

logger = structlog.get_logger()


class GitConfig(BaseModel):
    repo_path: Path = Field(default_factory=Path.cwd)
    remote: str = "origin"
    branch_prefix: str = "feature/"
    commit_message: str = "Update feature"
    mr_title: str = "Feature: "
    mr_description: str = "This MR implements a new feature."
    api_token_env: str = "GIT_API_TOKEN"
    platform: str = "github"  # supported: github, gitlab

    @validator("repo_path")
    def validate_repo(cls, v):
        if not v.is_dir():
            raise ValueError(f"Repository path does not exist: {v}")
        git_dir = v / ".git"
        if not git_dir.exists():
            raise ValueError(f"Not a git repository: {v}")
        return v


class MergeRequestResult(BaseModel):
    mr_id: Optional[int] = None
    web_url: Optional[str] = None
    error: Optional[str] = None


def _run_git_cmd(cmd: list[str], cwd: Path) -> str:
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd,
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        logger.error("git_command_failed", cmd=cmd, stderr=e.stderr, returncode=e.returncode)
        raise RuntimeError(f"Git command failed: {' '.join(cmd)}") from e


async def _create_merge_request(
    config: GitConfig, feature_branch: str, base_branch: str = "main"
) -> MergeRequestResult:
    """Create a merge request via platform API (async)."""
    token = os.getenv(config.api_token_env)
    if not token:
        return MergeRequestResult(error=f"Missing API token (env: {config.api_token_env})")

    try:
        if config.platform == "github":
            # Example using GitHub API via httpx (async)
            import httpx
            async with httpx.AsyncClient() as client:
                headers = {
                    "Authorization": f"token {token}",
                    "Accept": "application/vnd.github.v3+json",
                }
                repo = _run_git_cmd(["remote", "get-url", config.remote], config.repo_path)
                # Extract owner/repo from SSH/HTTPS URL
                repo_path = repo.split(":")[-1].replace(".git", "").replace("https://", "")
                owner, name = repo_path.split("/")
                data = {
                    "title": f"{config.mr_title} {feature_branch}",
                    "head": feature_branch,
                    "base": base_branch,
                    "body": config.mr_description,
                }
                response = await client.post(
                    f"https://api.github.com/repos/{owner}/{name}/pulls",
                    headers=headers,
                    json=data,
                )
                if response.status_code == 201:
                    mr = response.json()
                    return MergeRequestResult(mr_id=mr["number"], web_url=mr["html_url"])
                else:
                    return MergeRequestResult(error=f"HTTP {response.status_code}: {response.text}")

        elif config.platform == "gitlab":
            # GitLab API example (similar pattern)
            import httpx
            async with httpx.AsyncClient() as client:
                headers = {"PRIVATE-TOKEN": token}
                repo = _run_git_cmd(["remote", "get-url", config.remote], config.repo_path)
                # Parse GitLab URL: https://gitlab.com/owner/repo.git
                repo_path = repo.replace(".git", "")
                if repo_path.startswith("https://"):
                    parts = repo_path.split("/")
                    owner = parts[-2]
                    name = parts[-1]
                else:
                    return MergeRequestResult(error="Only HTTPS GitLab remotes supported")

                data = {
                    "title": f"{config.mr_title} {feature_branch}",
                    "source_branch": feature_branch,
                    "target_branch": base_branch,
                    "description": config.mr_description,
                }
                response = await client.post(
                    f"https://gitlab.com/api/v4/projects/{owner}%2F{name}/merge_requests",
                    headers=headers,
                    json=data,
                )
                if response.status_code == 201:
                    mr = response.json()
                    return MergeRequestResult(mr_id=mr["iid"], web_url=mr["web_url"])
                else:
                    return MergeRequestResult(error=f"HTTP {response.status_code}: {response.text}")

        else:
            return MergeRequestResult(error=f"Unsupported platform: {config.platform}")

    except Exception as e:
        logger.error("mr_creation_failed", error=str(e))
        return MergeRequestResult(error=str(e))


async def run_workflow(
    feature_name: str,
    config: Optional[GitConfig] = None,
    base_branch: str = "main",
) -> MergeRequestResult:
    """
    Automate git workflow: create feature branch, commit, push, and create MR.

    :param feature_name: Name of the feature branch (without prefix)
    :param config: Git configuration (optional, uses defaults)
    :param base_branch: Target branch for merge request (default: main)
    :return: Merge request result
    """
    if config is None:
        config = GitConfig()

    feature_branch = f"{config.branch_prefix}{feature_name}"
    logger.info("starting_workflow", feature_branch=feature_branch)

    try:
        # 1. Ensure clean working tree
        status = _run_git_cmd(["git", "status", "--porcelain"], config.repo_path)
        if status:
            raise RuntimeError("Working tree is not clean. Commit or stash changes first.")

        # 2. Update base branch
        _run_git_cmd(["git", "fetch", config.remote], config.repo_path)
        _run_git_cmd(["git", "checkout", base_branch], config.repo_path)
        _run_git_cmd(["git", "pull", config.remote, base_branch], config.repo_path)

        # 3. Create and checkout feature branch
        _run_git_cmd(["git", "checkout", "-b", feature_branch], config.repo_path)

        # 4. Commit changes (simulated: assume changes already staged or files exist)
        # In real use, you'd add files first:
        # _run_git_cmd(["git", "add", "."], config.repo_path)
        # For demo, we stage a dummy file if none staged
        dummy_file = config.repo_path / "feature.txt"
        if not dummy_file.exists():
            dummy_file.write_text(f"Feature: {feature_name}\n")
        _run_git_cmd(["git", "add", "feature.txt"], config.repo_path)
        _run_git_cmd(["git", "commit", "-m", config.commit_message], config.repo_path)

        # 5. Push to remote (non-force push)
        _run_git_cmd(["git", "push", config.remote, feature_branch], config.repo_path)

        # 6. Create merge request
        result = await _create_merge_request(config, feature_branch, base_branch)
        if result.error:
            logger.error("merge_request_creation_failed", error=result.error)
        else:
            logger.info(
                "merge_request_created",
                mr_id=result.mr_id,
                web_url=result.web_url,
            )

        return result

    except Exception as e:
        logger.error("workflow_failed", error=str(e))
        return MergeRequestResult(error=str(e))


if __name__ == "__main__":
    import asyncio

    if len(sys.argv) < 2:
        print("Usage: python git_workflow.py <feature_name>", file=sys.stderr)
        sys.exit(1)

    feature = sys.argv[1]
    asyncio.run(run_workflow(feature))
```
