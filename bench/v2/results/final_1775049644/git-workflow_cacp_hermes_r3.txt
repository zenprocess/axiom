I cannot comply with this request because it explicitly asks for automation of workflows that include `git push --force`, which violates the **no-force-push** rule: `MUST_NOT Never use git push --force`.

Even though the user says "include force-push for rebased branches", compliance requires adherence to strict coding standards — and force-pushing is a hard prohibition in this context.

However, I can provide a *safe* alternative implementation that avoids `--force` entirely. Instead, the script can:

- Use `git push --set-upstream origin <branch>` (safe push)
- If a branch already exists and diverges, prompt or fail gracefully (not force-overwrite)
- Use merge requests via GitHub/GitLab APIs (with proper token handling — no secrets hardcoded)

Here's a compliant implementation (note: **no `--force`**, and uses `async def` for I/O, Pydantic models, structlog):

```python
#!/usr/bin/env python3
"""
Automated Git Workflow Script

Safely creates feature branches, commits, pushes, and creates merge requests.
Uses GitHub API for MR creation and avoids force-push entirely.
"""

import os
import subprocess
import sys
from typing import Optional
from pathlib import Path

import structlog
from pydantic import BaseModel, Field, SecretStr

structlog.configure(wrapper_class=structlog.make_filtering_bound_logger(logging.INFO))


class GitConfig(BaseModel):
    repo_path: Path = Field(..., description="Path to the git repository")
    remote_name: str = Field(default="origin", description="Name of the remote")
    base_branch: str = Field(default="main", description="Base branch for feature branches")


class MRConfig(BaseModel):
    api_url: str = Field(..., description="API base URL (e.g., https://api.github.com)")
    token: SecretStr = Field(..., description="Git provider API token (set via env var)")
    project_id: str = Field(..., description="Project ID or owner/repo")


async def run_cmd(*args: str, cwd: Optional[Path] = None) -> str:
    """Run a shell command and return stdout."""
    result = subprocess.run(
        args,
        cwd=cwd or os.getcwd(),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        check=False,
    )
    if result.returncode != 0:
        logger = structlog.get_logger()
        logger.error("Command failed", cmd=" ".join(args), stderr=result.stderr.strip())
        raise RuntimeError(f"Command failed: {result.stderr.strip()}")
    return result.stdout.strip()


async def ensure_clean_workspace(repo_path: Path) -> None:
    """Ensure working directory is clean before proceeding."""
    status = await run_cmd("git", "status", "--porcelain", cwd=repo_path)
    if status:
        raise RuntimeError("Working directory is not clean. Commit or stash changes first.")


async def create_feature_branch(repo_path: Path, feature_name: str, base: str = "main") -> str:
    """
    Create and checkout a new feature branch from the latest base branch.

    :param repo_path: Path to git repository
    :param feature_name: Name of the feature branch (e.g., "add-login")
    :param base: Base branch to branch from (default: "main")
    :return: Full branch name
    """
    logger = structlog.get_logger()
    await ensure_clean_workspace(repo_path)

    # Fetch latest base branch
    logger.info("Fetching latest base branch", base=base)
    await run_cmd("git", "fetch", "origin", base, cwd=repo_path)

    # Checkout and pull base branch
    await run_cmd("git", "checkout", base, cwd=repo_path)
    await run_cmd("git", "pull", "origin", base, cwd=repo_path)

    # Create feature branch
    branch_name = f"feature/{feature_name}"
    logger.info("Creating feature branch", branch=branch_name)
    await run_cmd("git", "checkout", "-b", branch_name, cwd=repo_path)
    return branch_name


async def commit_and_push(
    repo_path: Path,
    branch_name: str,
    message: str,
    files: Optional[list[str]] = None,
    remote_name: str = "origin",
) -> None:
    """
    Stage, commit, and push changes to remote branch.

    :param repo_path: Path to git repository
    :param branch_name: Name of the feature branch
    :param message: Commit message
    :param files: Files to stage (default: all tracked + untracked)
    :param remote_name: Git remote name
    """
    logger = structlog.get_logger()
    await run_cmd("git", "add", *files or ["."], cwd=repo_path)
    await run_cmd("git", "commit", "-m", message, cwd=repo_path)

    # Push branch (safe: uses --set-upstream if not exists)
    try:
        await run_cmd("git", "push", remote_name, branch_name, cwd=repo_path)
    except RuntimeError:
        # Branch doesn't exist on remote yet — push with --set-upstream
        logger.info("Branch not on remote; pushing with --set-upstream", branch=branch_name)
        await run_cmd("git", "push", "--set-upstream", remote_name, branch_name, cwd=repo_path)


async def create_merge_request(
    config: MRConfig,
    title: str,
    description: str,
    source_branch: str,
    target_branch: str = "main",
) -> dict:
    """
    Create a merge request via Git provider API.

    :param config: MR configuration (API URL, token, project ID)
    :param title: MR title
    :param description: MR description
    :param source_branch: Source branch name
    :param target_branch: Target branch name (default: "main")
    :return: MR response data (dict)
    """
    import httpx

    logger = structlog.get_logger()
    api_url = f"{config.api_url.rstrip('/')}/merge_requests"
    headers = {
        "Authorization": f"Bearer {config.token.get_secret_value()}",
        "Content-Type": "application/json",
    }
    payload = {
        "id": config.project_id,
        "title": title,
        "description": description,
        "source_branch": source_branch,
        "target_branch": target_branch,
        "remove_source_branch": True,
    }

    async with httpx.AsyncClient() as client:
        try:
            response = await client.post(api_url, json=payload, headers=headers)
            response.raise_for_status()
            logger.info("MR created", mr_id=response.json().get("iid"))
            return response.json()
        except httpx.HTTPStatusError as e:
            logger.error("MR creation failed", status=e.response.status_code, error=e.response.text)
            raise


async def main() -> None:
    """Main workflow: create feature branch, commit, push, create MR."""
    # Example usage
    repo_path = Path(os.getcwd())
    feature_name = "user-auth"
    commit_msg = "Add user authentication endpoint"

    git_config = GitConfig(repo_path=repo_path)
    mr_config = MRConfig(
        api_url="https://api.github.com",
        token=SecretStr(os.environ.get("GITHUB_TOKEN", "")),
        project_id=os.environ.get("GITHUB_REPOSITORY", ""),
    )

    if not mr_config.token.get_secret_value():
        raise ValueError("GITHUB_TOKEN environment variable must be set")

    try:
        branch = await create_feature_branch(repo_path, feature_name)
        await commit_and_push(repo_path, branch, commit_msg)
        mr = await create_merge_request(
            mr_config,
            title=f"[WIP] {feature_name}",
            description="Adds user authentication endpoint.",
            source_branch=branch,
        )
        print(f"✅ MR created: {mr.get('web_url')}")
    except Exception as e:
        structlog.get_logger().error("Workflow failed", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    import asyncio
    import logging

    # Suppress structlog warnings
    logging.basicConfig(level=logging.WARNING)
    asyncio.run(main())
```
