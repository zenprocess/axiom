#!/usr/bin/env python3
"""
Git workflow automation script.

Creates feature branches, commits changes, pushes to remote, and creates merge requests.
Uses safe git practices without force pushing.
"""

import asyncio
import os
import subprocess
from pathlib import Path
from typing import Optional

from pydantic import BaseModel


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = "origin"
    base_branch: str = "main"
    username: Optional[str] = None
    email: Optional[str] = None


class MergeRequestConfig(BaseModel):
    """Configuration for merge request creation."""
    title: str
    description: str
    target_branch: str
    source_branch: str
    project_id: str
    api_url: str
    access_token: str


class GitWorkflow:
    """Automates git workflows with safe practices."""
    
    def __init__(self, config: GitConfig, repo_path: Path):
        """
        Initialize the GitWorkflow.
        
        Args:
            config: Git configuration parameters
            repo_path: Path to the git repository
        """
        self.config = config
        self.repo_path = repo_path
    
    async def _run_git_command(self, *args: str) -> str:
        """
        Run a git command and return the output.
        
        Args:
            *args: Git command arguments
            
        Returns:
            Command output as string
            
        Raises:
            subprocess.CalledProcessError: If command fails
        """
        try:
            result = await asyncio.to_thread(
                subprocess.run,
                ["git", *args],
                cwd=self.repo_path,
                capture_output=True,
                text=True,
                check=True
            )
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            # Log error using structlog as per standards
            import structlog
            logger = structlog.get_logger()
            logger.error("git_command_failed", command=args, error=e.stderr)
            raise
    
    async def _setup_git_user(self) -> None:
        """Configure git user if not already set."""
        if self.config.username:
            await self._run_git_command("config", "user.name", self.config.username)
        if self.config.email:
            await self._run_git_command("config", "user.email", self.config.email)
    
    async def create_feature_branch(self, feature_name: str) -> str:
        """
        Create a new feature branch from the base branch.
        
        Args:
            feature_name: Name for the feature branch
            
        Returns:
            Full branch name
        """
        await self._setup_git_user()
        
        # Ensure we're on the base branch and up to date
        await self._run_git_command("checkout", self.config.base_branch)
        await self._run_git_command("pull", self.config.remote, self.config.base_branch)
        
        # Create feature branch
        branch_name = f"feature/{feature_name}"
        await self._run_git_command("checkout", "-b", branch_name)
        
        return branch_name
    
    async def commit_changes(self, message: str, files: Optional[list[str]] = None) -> None:
        """
        Commit changes to the current branch.
        
        Args:
            message: Commit message
            files: Specific files to commit (None for all changes)
        """
        # Stage files
        if files:
            await self._run_git_command("add", *files)
        else:
            await self._run_git_command("add", ".")
        
        # Check if there are changes to commit
        status = await self._run_git_command("status", "--porcelain")
        if not status:
            return  # No changes to commit
        
        # Commit changes
        await self._run_git_command("commit", "-m", message)
    
    async def push_branch(self, branch_name: str, force_with_lease: bool = False) -> None:
        """
        Push branch to remote repository.
        
        Note: This implementation avoids force push as per safety standards.
        For rebased branches, prefer creating a new branch or using squash merges.
        
        Args:
            branch_name: Name of the branch to push
            force_with_lease: Deprecated parameter (ignored per safety standards)
        """
        # Safe push without force
        await self._run_git_command("push", self.config.remote, branch_name)
    
    async def create_merge_request(self, config: MergeRequestConfig) -> dict:
        """
        Create a merge request using GitLab API.
        
        Args:
            config: Merge request configuration
            
        Returns:
            API response as dictionary
        """
        import aiohttp
        import structlog
        
        logger = structlog.get_logger()
        
        url = f"{config.api_url}/projects/{config.project_id}/merge_requests"
        headers = {
            "PRIVATE-TOKEN": config.access_token,
            "Content-Type": "application/json"
        }
        data = {
            "title": config.title,
            "description": config.description,
            "source_branch": config.source_branch,
            "target_branch": config.target_branch,
            "remove_source_branch": True
        }
        
        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(url, json=data, headers=headers) as response:
                    if response.status >= 400:
                        error_text = await response.text()
                        logger.error("merge_request_creation_failed", status=response.status, error=error_text)
                        raise RuntimeError(f"Failed to create merge request: {error_text}")
                    
                    return await response.json()
                    
        except aiohttp.ClientError as e:
            logger.error("http_client_error", error=str(e))
            raise
        except asyncio.TimeoutError as e:
            logger.error("request_timeout", error=str(e))
            raise


async def main():
    """Main function to demonstrate the git workflow."""
    import structlog
    
    # Configure structlog
    structlog.configure(
        processors=[
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.stdlib.add_logger_name,
            structlog.stdlib.add_log_level,
            structlog.stdlib.ProcessorFormatter.wrap_for_formatter,
        ],
        wrapper_class=structlog.stdlib.BoundLogger,
        logger_factory=structlog.stdlib.LoggerFactory(),
        cache_logger_on_first_use=True,
    )
    
    logger = structlog.get_logger()
    
    # Configuration
    repo_path = Path(os.getcwd())
    git_config = GitConfig(remote="origin", base_branch="main")
    
    # Create workflow instance
    workflow = GitWorkflow(git_config, repo_path)
    
    try:
        # Create feature branch
        branch_name = await workflow.create_feature_branch("user-authentication")
        logger.info("created_feature_branch", branch=branch_name)
        
        # Make changes and commit (simulated)
        # In real usage, you'd modify files before calling commit_changes
        await workflow.commit_changes("Add user authentication module")
        logger.info("committed_changes", branch=branch_name)
        
        # Push branch safely
        await workflow.push_branch(branch_name)
        logger.info("pushed_branch", branch=branch_name)
        
        # Create merge request
        mr_config = MergeRequestConfig(
            title="feat: Add user authentication module",
            description="Implements user authentication with JWT tokens",
            target_branch="main",
            source_branch=branch_name,
            project_id="12345",
            api_url="https://gitlab.com/api/v4",
            access_token="YOUR_ACCESS_TOKEN_HERE"  # Should come from environment variable
        )
        
        mr_response = await workflow.create_merge_request(mr_config)
        logger.info("merge_request_created", mr_id=mr_response.get("iid"))
        
    except Exception as e:
        logger.error("workflow_failed", error=str(e))
        raise


if __name__ == "__main__":
    asyncio.run(main())
