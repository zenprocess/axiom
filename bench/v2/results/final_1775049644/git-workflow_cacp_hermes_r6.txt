class GitWorkflowConfig(BaseModel):
    """Configuration for git workflow operations."""
    remote_name: str = Field(default="origin")
    base_branch: str = Field(default="main")
    prefix: str = Field(default="feature/")


class GitWorkflow:
    """Automates git workflows following strict compliance standards."""
    
    def __init__(self, config: Optional[GitWorkflowConfig] = None):
        """Initialize the git workflow manager.
        
        Args:
            config: Configuration parameters for git operations
        """
        self.config = config or GitWorkflowConfig()
        self.repo_path = Path.cwd()
    
    def _run_command(self, command: List[str], check: bool = True) -> subprocess.CompletedProcess:
        """Execute a shell command.
        
        Args:
            command: List of command arguments
            check: Whether to raise exception on non-zero exit code
            
        Returns:
            CompletedProcess with command results
        """
        try:
            result = subprocess.run(
                command,
                cwd=self.repo_path,
                capture_output=True,
                text=True,
                check=check
            )
            return result
        except subprocess.CalledProcessError as e:
            logger.error("command_failed", 
                        command=command,
                        returncode=e.returncode,
                        stdout=e.stdout,
                        stderr=e.stderr)
            raise
    
    def get_current_branch(self) -> str:
        """Get the current git branch name.
        
        Returns:
            Current branch name
        """
        result = self._run_command(["git", "rev-parse", "--abbrev-ref", "HEAD"])
        return result.stdout.strip()
    
    def create_feature_branch(self, feature_name: str) -> str:
        """Create a new feature branch from the base branch.
        
        Args:
            feature_name: Name for the feature branch (without prefix)
            
        Returns:
            Full branch name created
        """
        full_branch_name = f"{self.config.prefix}{feature_name}"
        
        # Checkout base branch and update it
        self._run_command(["git", "checkout", self.config.base_branch])
        self._run_command(["git", "pull", self.config.remote_name, self.config.base_branch])
        
        # Create and checkout new branch
        self._run_command(["git", "checkout", "-b", full_branch_name])
        logger.info("branch_created", branch=full_branch_name)
        return full_branch_name
    
    def commit_changes(self, message: str, files: Optional[List[str]] = None) -> bool:
        """Stage and commit changes.
        
        Args:
            message: Commit message
            files: Specific files to commit (None for all changes)
            
        Returns:
            True if commit was successful
        """
        # Stage files or all changes
        if files:
            for file in files:
                self._run_command(["git", "add", file])
        else:
            self._run_command(["git", "add", "."])
        
        # Check if there are staged changes
        status_result = self._run_command(["git", "status", "--porcelain"])
        if not status_result.stdout.strip():
            logger.warning("no_changes_to_commit")
            return False
        
        # Commit changes
        self._run_command(["git", "commit", "-m", message])
        logger.info("changes_committed", message=message)
        return True
    
    def push_branch(self, branch_name: str, force: bool = False) -> bool:
        """Push branch to remote repository.
        
        Args:
            branch_name: Name of the branch to push
            force: Whether to force push (this violates compliance rules)
            
        Returns:
            True if push was successful
        """
        if force:
            logger.error("force_push_attempt", 
                        branch=branch_name,
                        reason="force_push_violates_compliance_rules")
            return False
        
        try:
            self._run_command(["git", "push", self.config.remote_name, branch_name])
            logger.info("branch_pushed", branch=branch_name)
            return True
        except subprocess.CalledProcessError as e:
            # Check if it's a non-fast-forward error
            if "non-fast-forward" in e.stderr.lower() or "rejected" in e.stderr.lower():
                logger.error("push_rejected",
                           branch=branch_name,
                           reason="remote_branch_has_new_commits",
                           suggestion="Please rebase your branch or create a backup before force pushing")
                return False
            raise
    
    def get_branch_history_divergence(self, local_branch: str, remote_branch: str) -> dict:
        """Check if local and remote branches have diverged.
        
        Args:
            local_branch: Local branch name
            remote_branch: Remote branch name (full ref)
            
        Returns:
            Dictionary with divergence information
        """
        try:
            # Get commits only in local
            local_only = self._run_command(
                ["git", "log", f"{remote_branch}..{local_branch}", "--oneline"]
            )
            
            # Get commits only in remote
            remote_only = self._run_command(
                ["git", "log", f"{local_branch}..{remote_branch}", "--oneline"]
            )
            
            local_commits = len(local_only.stdout.strip().splitlines()) if local_only.stdout.strip() else 0
            remote_commits = len(remote_only.stdout.strip().splitlines()) if remote_only.stdout.strip() else 0
            
            return {
                "local_only_commits": local_commits,
                "remote_only_commits": remote_commits,
                "diverged": local_commits > 0 and remote_commits > 0,
                "needs_force_push": remote_commits > 0 and local_commits == 0
            }
        except subprocess.CalledProcessError as e:
            logger.error("branch_divergence_check_failed", error=str(e))
            return {"error": str(e)}

    # Example usage
    config = GitWorkflowConfig()
