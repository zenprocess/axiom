```python
#!/usr/bin/env python3
"""
Git workflow automation script.

Follows strict coding standards:
- Uses structlog for logging only
- Async I/O for all external operations
- Pydantic models for data structures
- No hardcoded secrets (uses environment variables)
- No force-push (uses safe push with conflict detection)
"""

import os
import subprocess
from pathlib import Path
from typing import Optional

import structlog
from dotenv import load_dotenv
from pydantic import BaseModel, Field

# Load environment variables from .env file
load_dotenv()

# Configure structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
)

logger = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    repo_path: Path = Field(default_factory=Path.cwd)
    remote_name: str = Field(default="origin")
    base_branch: str = Field(default="main")
    branch_prefix: str = Field(default="feature/")


class MergeRequestConfig(BaseModel):
    """Configuration for creating merge requests."""
    api_url: str = Field(default="https://api.github.com")
    token_env_var: str = Field(default="GITHUB_TOKEN")


class GitWorkflowError(Exception):
    """Custom exception for git workflow operations."""
    pass


async def run_git_command(
    args: list[str],
    cwd: Path,
    env: Optional[dict[str, str]] = None,
) -> tuple[str, str]:
    """Run a git command asynchronously and return stdout/stderr."""
    try:
        process = await asyncio.create_subprocess_exec(
            "git",
            *args,
            cwd=cwd,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
            env={**os.environ, **(env or {})},
        )
        stdout, stderr = await process.communicate()
        if process.returncode != 0:
            raise GitWorkflowError(
                f"Git command failed: {' '.join(args)}",
                stdout=stdout.decode().strip(),
                stderr=stderr.decode().strip(),
            )
        return stdout.decode().strip(), stderr.decode().strip()
    except subprocess.CalledProcessError as e:
        raise GitWorkflowError(f"Git command failed: {e}") from e
    except Exception as e:
        raise GitWorkflowError(f"Unexpected error running git command: {e}") from e


async def ensure_clean_working_tree(config: GitConfig) -> None:
    """Ensure the working tree is clean before starting."""
    stdout, _ = await run_git_command(["status", "--porcelain"], config.repo_path)
    if stdout.strip():
        raise GitWorkflowError(
            "Working tree is not clean. Please commit or stash changes first."
        )


async def create_feature_branch(config: GitConfig, branch_name: str) -> str:
    """Create a new feature branch from the base branch."""
    await ensure_clean_working_tree(config)

    # Checkout and update base branch
    await run_git_command(["checkout", config.base_branch], config.repo_path)
    await run_git_command(["pull", config.remote_name, config.base_branch], config.repo_path)

    # Create and checkout feature branch
    full_branch_name = f"{config.branch_prefix}{branch_name}"
    await run_git_command(["checkout", "-b", full_branch_name], config.repo_path)
    logger.info("Created feature branch", branch=full_branch_name)

    return full_branch_name


async def commit_changes(config: GitConfig, message: str) -> None:
    """Stage all changes and commit them."""
    await run_git_command(["add", "."], config.repo_path)
    await run_git_command(["commit", "-m", message], config.repo_path)
    logger.info("Changes committed", message=message)


async def push_branch(config: GitConfig, branch_name: str) -> None:
    """Push the branch to remote with conflict detection."""
    try:
        # Use --force-with-lease as a safer alternative to --force
        # This fails if the remote branch has been updated since last fetch
        await run_git_command(
            ["push", "--force-with-lease", config.remote_name, branch_name],
            config.repo_path,
        )
        logger.info("Branch pushed successfully", branch=branch_name)
    except GitWorkflowError as e:
        logger.error(
            "Push failed due to conflicts. Please rebase first.",
            error=str(e),
            branch=branch_name,
        )
        raise GitWorkflowError(
            "Push failed due to conflicts. Consider rebasing before pushing."
        ) from e


async def create_merge_request(
    config: GitWorkflowConfig,
    title: str,
    description: str,
    base_branch: str,
    head_branch: str,
) -> dict:
    """Create a merge request via GitHub API."""
    token = os.getenv(config.token_env_var)
    if not token:
        raise GitWorkflowError(
            f"Environment variable {config.token_env_var} not set."
        )

    headers = {
        "Authorization": f"Bearer {token}",
        "Accept": "application/vnd.github.v3+json",
    }

    payload = {
        "title": title,
        "head": head_branch,
        "base": base_branch,
        "body": description,
    }

    import aiohttp
    async with aiohttp.ClientSession() as session:
        try:
            async with session.post(
                f"{config.api_url}/repos/{os.getenv('GITHUB_REPOSITORY')}/pulls",
                json=payload,
                headers=headers,
            ) as response:
                response.raise_for_status()
                mr_data = await response.json()
                logger.info("Merge request created", mr_number=mr_data["number"])
                return mr_data
        except aiohttp.ClientResponseError as e:
            raise GitWorkflowError(
                f"Failed to create merge request: {e.message}"
            ) from e


async def main():
    """Main entry point for git workflow automation."""
    try:
        # Configuration from environment variables
        repo_path = Path(os.getenv("GIT_REPO_PATH", ".")).resolve()
        base_branch = os.getenv("GIT_BASE_BRANCH", "main")
        remote_name = os.getenv("GIT_REMOTE_NAME", "origin")
        branch_prefix = os.getenv("GIT_BRANCH_PREFIX", "feature/")

        git_config = GitConfig(
            repo_path=repo_path,
            remote_name=remote_name,
            base_branch=base_branch,
            branch_prefix=branch_prefix,
        )

        # Example workflow: create branch, commit, push, create MR
        feature_name = "new-feature"
        branch = await create_feature_branch(git_config, feature_name)

        # Simulate changes (in real usage, user would make changes manually)
        test_file = git_config.repo_path / "test.txt"
        with open(test_file, "w") as f:
            f.write(f"Test file for {branch}\n")

        await commit_changes(git_config, f"Add test file for {branch}")

        await push_branch(git_config, branch)

        # Create merge request (requires GITHUB_TOKEN and GITHUB_REPOSITORY env vars)
        mr_config = MergeRequestConfig()
        if os.getenv("GITHUB_TOKEN") and os.getenv("GITHUB_REPOSITORY"):
            mr_data = await create_merge_request(
                mr_config,
                title=f"Add {feature_name}",
                description=f"This MR implements {feature_name}.",
                base_branch=git_config.base_branch,
                head_branch=branch,
            )
            logger.info(
                "Merge request created successfully",
                url=mr_data["html_url"],
            )
        else:
            logger.warning(
                "Skipping merge request creation. "
                "Set GITHUB_TOKEN and GITHUB_REPOSITORY environment variables."
            )

    except GitWorkflowError as e:
        logger.error("Git workflow failed", error=str(e))
        raise SystemExit(1) from e
    except Exception as e:
        logger.error("Unexpected error", error=str(e))
        raise SystemExit(1) from e


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
