```python
#!/usr/bin/env python3
"""
Automated Git Workflow Script

This script automates common Git workflows:
- Create feature branches
- Stage and commit changes
- Push branches to remote (non-forcefully)
- Create merge requests (e.g., GitHub pull requests)

⚠️ Per project rules: no-force-push is enforced — this script never uses `--force`.
"""

import asyncio
import os
import sys
from pathlib import Path
from typing import List, Optional, Dict

import httpx
from pydantic import BaseModel, Field
import structlog

# Configure structlog only (no logging/print)
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.stdlib.add_logger_name,
        structlog.stdlib.add_log_level,
        structlog.stdlib.PositionalArgumentsFormatter(),
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.processors.UnicodeDecoder(),
        structlog.stdlib.ProcessorFormatter.wrap_for_formatter,
    ],
    context_class=dict,
    logger_factory=structlog.stdlib.LoggerFactory(),
    cache_logger_on_first_use=True,
)
logger = structlog.get_logger(__name__)


class GitConfig(BaseModel):
    """Configuration for Git repository."""
    repo_path: Path = Field(..., description="Path to the Git repository")
    remote: str = Field(default="origin", description="Remote name")
    base_branch: str = Field(default="main", description="Base branch to branch from")


class MRConfig(BaseModel):
    """Configuration for Merge Request (e.g., GitHub PR)."""
    api_url: str = Field(..., description="API base URL (e.g., https://api.github.com)")
    repo_owner: str = Field(..., description="Repository owner")
    repo_name: str = Field(..., description="Repository name")
    token: Optional[str] = Field(default=None, description="GitHub token — MUST be set via env var")


class FeatureBranchConfig(BaseModel):
    """Feature branch creation parameters."""
    branch_name: str = Field(..., description="Name of the feature branch")
    commit_message: str = Field(default="Update feature", description="Commit message")
    files_to_add: List[str] = Field(default_factory=list, description="Relative paths to add")


class GitClient:
    """Handles Git operations safely (no force-push)."""

    def __init__(self, config: GitConfig):
        self.config = config

    async def _run_git(self, *args: str) -> str:
        """Run a Git command and return stdout."""
        import subprocess
        cmd = ["git", "-C", str(self.config.repo_path)] + list(args)
        try:
            result = await asyncio.to_thread(
                subprocess.run,
                cmd,
                capture_output=True,
                text=True,
                check=True
            )
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            logger.error("git_command_failed", cmd=" ".join(cmd), stderr=e.stderr, returncode=e.returncode)
            raise

    async def create_branch(self, branch_name: str) -> str:
        """Create and checkout a new feature branch from HEAD."""
        try:
            logger.info("creating_feature_branch", branch=branch_name)
            await self._run_git("checkout", "-b", branch_name)
            return branch_name
        except Exception as e:
            logger.error("branch_creation_failed", branch=branch_name, error=str(e))
            raise

    async def add_files(self, files: List[str]) -> None:
        """Stage specified files."""
        if not files:
            return
        try:
            logger.info("staging_files", count=len(files))
            await self._run_git("add", *files)
        except Exception as e:
            logger.error("file_staging_failed", files=files, error=str(e))
            raise

    async def commit(self, message: str) -> None:
        """Commit staged changes."""
        try:
            logger.info("committing_changes", message=message)
            await self._run_git("commit", "-m", message)
        except Exception as e:
            logger.error("commit_failed", message=message, error=str(e))
            raise

    async def push_branch(self, branch_name: str) -> None:
        """Push branch to remote — never force."""
        try:
            logger.info("pushing_branch", branch=branch_name, remote=self.config.remote)
            await self._run_git("push", self.config.remote, branch_name)
        except Exception as e:
            logger.error("push_failed", branch=branch_name, remote=self.config.remote, error=str(e))
            raise


class MRClient:
    """Handles Merge Request creation (GitHub API example)."""

    def __init__(self, config: MRConfig):
        self.config = config
        self._token = os.environ.get("GITHUB_TOKEN")
        if not self._token:
            logger.warning("GITHUB_TOKEN_env_var_missing")

    async def create_mr(
        self,
        title: str,
        body: str,
        head_branch: str,
        base_branch: str = "main"
    ) -> Dict[str, str]:
        """Create a pull request via GitHub API."""
        async with httpx.AsyncClient(
            timeout=30.0,
            headers={
                "Authorization": f"token {self._token}" if self._token else "",
                "Accept": "application/vnd.github.v3+json"
            }
        ) as client:
            payload = {
                "title": title,
                "body": body,
                "head": head_branch,
                "base": base_branch
            }
            url = f"{self.config.api_url}/repos/{self.config.repo_owner}/{self.config.repo_name}/pulls"
            try:
                response = await client.post(url, json=payload)
                response.raise_for_status()
                pr_data = response.json()
                logger.info("merge_request_created", pr_url=pr_data["html_url"])
                return pr_data
            except httpx.HTTPStatusError as e:
                logger.error("mr_creation_failed", status=e.response.status_code, error=e.response.text)
                raise
            except Exception as e:
                logger.error("unexpected_mr_error", error=str(e))
                raise


async def run_workflow(
    feature_config: FeatureBranchConfig,
    git_config: GitConfig,
    mr_config: MRConfig
) -> Optional[Dict]:
    """
    Run the full Git workflow: branch → add/commit → push → create MR.

    Args:
        feature_config: Parameters for the feature branch
        git_config: Git repository configuration
        mr_config: Merge request configuration

    Returns:
        MR data on success, None on failure
    """
    try:
        client = GitClient(git_config)
        mr_client = MRClient(mr_config)

        # 1. Create branch
        await client.create_branch(feature_config.branch_name)

        # 2. Add files
        if feature_config.files_to_add:
            await client.add_files(feature_config.files_to_add)

        # 3. Commit
        await client.commit(feature_config.commit_message)

        # 4. Push branch (non-force)
        await client.push_branch(feature_config.branch_name)

        # 5. Create MR
        title = f"feat: {feature_config.branch_name}"
        body = f"Automatic PR for feature branch `{feature_config.branch_name}`"
        return await mr_client.create_mr(
            title=title,
            body=body,
            head_branch=feature_config.branch_name,
            base_branch=git_config.base_branch
        )

    except Exception as e:
        logger.error("workflow_failed", error=str(e))
        return None


# Example usage (for CLI invocation)
if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="Automate Git workflows")
    parser.add_argument("--branch", required=True, help="Feature branch name")
    parser.add_argument("--files", nargs="+", default=[], help="Files to stage")
    parser.add_argument("--message", default="Update feature", help="Commit message")
    parser.add_argument("--repo-path", type=Path, default=Path.cwd(), help="Repo path")
    parser.add_argument("--remote", default="origin", help="Git remote")
    parser.add_argument("--base-branch", default="main", help="Base branch")
    parser.add_argument("--api-url", default="https://api.github.com", help="MR API URL")
    parser.add_argument("--repo-owner", required=True, help="Repo owner (GitHub)")
    parser.add_argument("--repo-name", required=True, help="Repo name (GitHub)")
    args = parser.parse_args()

    git_config = GitConfig(
        repo_path=args.repo_path,
        remote=args.remote,
        base_branch=args.base_branch
    )
    mr_config = MRConfig(
        api_url=args.api_url,
        repo_owner=args.repo_owner,
        repo_name=args.repo_name,
        token=os.environ.get("GITHUB_TOKEN")
    )
    feature_config = FeatureBranchConfig(
        branch_name=args.branch,
        commit_message=args.message,
        files_to_add=args.files
    )

    async def main():
        result = await run_workflow(feature_config, git_config, mr_config)
        if result:
            print(f"✅ MR created: {result['html_url']}")
        else:
            sys.exit(1)

    asyncio.run(main())
```
