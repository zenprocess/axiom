```python
#!/usr/bin/env python3
"""
Git workflow automation script.

This script automates common git workflows including:
- Creating feature branches
- Committing changes
- Pushing to remote
- Creating merge requests
"""

import os
import subprocess
import sys
from dataclasses import dataclass
from typing import Optional

import structlog
from pydantic import BaseModel, field_validator

log = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = "origin"
    base_branch: str = "main"
    commit_message: str = "Automated commit"
    
    @field_validator("remote", "base_branch")
    @classmethod
    def validate_non_empty(cls, v: str) -> str:
        """Ensure non-empty strings."""
        if not v:
            raise ValueError("Cannot be empty")
        return v


@dataclass
class BranchInfo:
    """Information about a git branch."""
    name: str
    is_remote: bool = False
    is_current: bool = False


class GitWorkflow:
    """Automates common git workflows."""
    
    def __init__(self, config: Optional[GitConfig] = None):
        """Initialize with optional configuration."""
        self.config = config or GitConfig()
    
    def _run_command(self, cmd: list[str], cwd: Optional[str] = None) -> tuple[int, str, str]:
        """
        Run a git command and return the result.
        
        Args:
            cmd: Command to run as a list of arguments
            cwd: Working directory (defaults to current directory)
            
        Returns:
            Tuple of (return_code, stdout, stderr)
        """
        try:
            result = subprocess.run(
                cmd,
                capture_output=True,
                text=True,
                cwd=cwd or os.getcwd(),
                check=False
            )
            return result.returncode, result.stdout.strip(), result.stderr.strip()
        except FileNotFoundError:
            log.error("git_not_found", command=cmd[0])
            raise
        except Exception as e:
            log.error("command_failed", command=cmd, error=str(e))
            raise
    
    def _get_current_branch(self) -> str:
        """Get the name of the current branch."""
        return_code, stdout, stderr = self._run_command(["git", "rev-parse", "--abbrev-ref", "HEAD"])
        if return_code != 0:
            log.error("failed_to_get_current_branch", error=stderr)
            raise RuntimeError(f"Failed to get current branch: {stderr}")
        
        return stdout
    
    def create_feature_branch(self, feature_name: str) -> BranchInfo:
        """
        Create a new feature branch from the base branch.
        
        Args:
            feature_name: Name for the new feature branch
            
        Returns:
            BranchInfo about the created branch
        """
        log.info("creating_feature_branch", feature_name=feature_name)
        
        # Get current branch to return to later
        current_branch = self._get_current_branch()
        
        # Switch to base branch and pull latest changes
        return_code, _, stderr = self._run_command(["git", "checkout", self.config.base_branch])
        if return_code != 0:
            log.error("failed_to_checkout_base_branch", branch=self.config.base_branch, error=stderr)
            raise RuntimeError(f"Failed to checkout base branch '{self.config.base_branch}': {stderr}")
        
        # Pull latest changes
        return_code, stdout, stderr = self._run_command(["git", "pull", self.config.remote, self.config.base_branch])
        if return_code != 0:
            log.warning("failed_to_pull", error=stderr)
            # Continue anyway if pull fails (might be first time)
        
        # Create and checkout new feature branch
        branch_name = f"feature/{feature_name}"
        return_code, _, stderr = self._run_command(["git", "checkout", "-b", branch_name])
        if return_code != 0:
            log.error("failed_to_create_branch", branch=branch_name, error=stderr)
            raise RuntimeError(f"Failed to create branch '{branch_name}': {stderr}")
        
        log.info("feature_branch_created", branch=branch_name)
        return BranchInfo(name=branch_name, is_current=True)
    
    def add_files(self, files: list[str] | None = None) -> None:
        """
        Add files to the staging area.
        
        Args:
            files: List of files to add (None adds all changes)
        """
        if files is None:
            cmd = ["git", "add", "."]
        else:
            cmd = ["git", "add"] + files
        
        return_code, _, stderr = self._run_command(cmd)
        if return_code != 0:
            log.error("failed_to_add_files", files=files, error=stderr)
            raise RuntimeError(f"Failed to add files: {stderr}")
        
        log.info("files_added", files=files or "all")
    
    def commit_changes(self, message: Optional[str] = None) -> None:
        """
        Commit staged changes.
        
        Args:
            message: Commit message (uses config default if not provided)
        """
        commit_msg = message or self.config.commit_message
        
        return_code, _, stderr = self._run_command(["git", "commit", "-m", commit_msg])
        if return_code != 0:
            log.error("failed_to_commit", error=stderr)
            raise RuntimeError(f"Failed to commit changes: {stderr}")
        
        log.info("changes_committed", message=commit_msg)
    
    def push_branch(self, branch: str, force: bool = False) -> None:
        """
        Push branch to remote repository.
        
        Args:
            branch: Branch to push
            force: Whether to force push (use with caution)
        """
        log.info("pushing_branch", branch=branch, force=force)
        
        if force:
            cmd = ["git", "push", "--force-with-lease", self.config.remote, branch]
        else:
            cmd = ["git", "push", self.config.remote, branch]
        
        return_code, stdout, stderr = self._run_command(cmd)
        if return_code != 0:
            log.error("failed_to_push", branch=branch, force=force, error=stderr)
            raise RuntimeError(f"Failed to push branch '{branch}': {stderr}")
        
        log.info("branch_pushed", branch=branch, force=force)
    
    def get_merge_request_url(self, branch: str, title: Optional[str] = None) -> str:
        """
        Generate a merge request URL for the current repository.
        
        This is a simplified version - in practice you'd use the GitLab/GitHub API.
        
        Args:
            branch: Source branch for the merge request
            title: Title for the merge request
            
        Returns:
            URL where the merge request can be created
        """
        # Get remote URL to construct merge request URL
        return_code, stdout, _ = self._run_command(["git", "remote", "get-url", self.config.remote])
        if return_code != 0:
            log.error("failed_to_get_remote_url", error=stdout)
            raise RuntimeError("Failed to get remote URL")
        
        remote_url = stdout.strip()
        
        # Extract repository info from URL (simplified for GitHub/GitLab)
        if "gitlab.com" in remote_url:
            repo_path = remote_url.replace("git@gitlab.com:", "").replace(".git", "")
            base_url = f"https://gitlab.com/{repo_path}/-/merge_requests/new"
        elif "github.com" in remote_url:
            repo_path = remote_url.replace("git@github.com:", "").replace(".git", "")
            base_url = f"https://github.com/{repo_path}/compare/{self.config.base_branch}...{branch}?expand=1"
        else:
            # Generic fallback
            base_url = f"https://git.example.com/{remote_url.replace('.git', '')}/merge_requests/new"
        
        # Construct URL with title parameter if provided
        if title:
            import urllib.parse
            return f"{base_url}&title={urllib.parse.quote(title)}"
        return base_url
    
    def create_merge_request(self, branch: str, title: str, description: str = "") -> str:
        """
        Create a merge request via the appropriate API.
        
        This is a placeholder that returns the URL where the MR can be created.
        For production use, implement actual API calls to GitHub/GitLab.
        
        Args:
            branch: Source branch
            title: MR title
            description: MR description
            
        Returns:
            URL where the merge request can be created
        """
        log.info("creating_merge_request", branch=branch, title=title)
        
        # For production, you would use the GitHub/GitLab API here
        # Example for GitHub:
        # import requests
        # token = os.environ['GITHUB_TOKEN']
        # headers = {'Authorization': f'token {token}'}
        # payload = {
        #     'title': title,
        #     'head': branch,
        #     'base': self.config.base_branch,
        #     'body': description
        # }
        # response = requests.post(f'https://api.github.com/repos/{repo}/pulls', json=payload, headers=headers)
        
        mr_url = self.get_merge_request_url(branch, title)
        log.info("merge_request_url_generated", url=mr_url)
        return mr_url


def main():
    """Main entry point for the git workflow script."""
    import argparse
    
    parser = argparse.ArgumentParser(description="Automate git workflows")
    parser.add_argument("feature", help="Feature name for the branch")
    parser.add_argument("--message", "-m", default="Automated commit", help="Commit message")
    parser.add_argument("--base-branch", default="main", help="Base branch to branch from")
    parser.add_argument("--remote", default="origin", help="Remote repository name")
    parser.add_argument("--force", action="store_true", help="Force push (use with caution)")
    parser.add_argument("--files", nargs="+", help="Specific files to commit")
    parser.add_argument("--mr-title", help="Title for merge request")
    parser.add_argument("--mr-description", default="", help="Description for merge request")
    
    args = parser.parse_args()
    
    # Validate that we're in a git repository
    if subprocess.run(["git", "rev-parse", "--git-dir"], capture_output=True).returncode != 0:
        log.error("not_a_git_repository")
        sys.exit(1)
    
    # Configure and run workflow
    config = GitConfig(
        remote=args.remote,
        base_branch=args.base_branch,
        commit_message=args.message
    )
    
    workflow = GitWorkflow(config)
    
    try:
        # Step 1: Create feature branch
        branch_info = workflow.create_feature_branch(args.feature)
        
        # Step 2: Add and commit changes
        workflow.add_files(args.files)
        workflow.commit_changes()
        
        # Step 3: Push branch
        workflow.push_branch(branch_info.name, force=args.force)
        
        # Step 4: Create merge request
        title = args.mr_title or f"Feature: {args.feature}"
        mr_url = workflow.create_merge_request(
            branch_info.name, 
            title=title, 
            description=args.mr_description
        )
        
        log.info("workflow_completed", merge_request_url=mr_url)
        print(f"\nMerge request can be created at: {mr_url}")
        
    except Exception as e:
        log.error("workflow_failed", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()
```
