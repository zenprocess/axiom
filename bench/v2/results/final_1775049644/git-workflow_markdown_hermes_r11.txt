I cannot write a script that automates git workflows including force-push, because that would violate the **no-force-push** rule (severity: critical). Force pushing rewrites remote history and can destroy other developers' work.

Instead, I will write a safe, production-grade script that follows best practices:

- Creates feature branches from the correct base
- Commits changes with clear messages
- Pushes branches to remote (without force)
- Creates merge requests via the GitHub API (or GitLab API)
- Includes robust error handling and logging using `structlog`

Here is the script:

```python
#!/usr/bin/env python3
"""
Automated Git Workflow Script

Creates feature branches, commits changes, pushes to remote, and creates merge requests.
Does NOT use force push — safer alternative: rebase locally and push with --force-with-lease if needed.
"""

import os
import subprocess
import sys
from pathlib import Path
from typing import Optional

import requests
import structlog
from pydantic import BaseModel, Field

# Configure structlog
structlog.configure(
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.dev.ConsoleRenderer(),
    ],
)
log = structlog.get_logger()


class GitConfig(BaseModel):
    repo_path: str = Field(default=".")
    base_branch: str = Field(default="main")
    remote_name: str = Field(default="origin")


class GitHubConfig(BaseModel):
    owner: str
    repo: str
    token: str = Field(default_factory=lambda: os.environ.get("GITHUB_TOKEN", ""))
    api_url: str = Field(default="https://api.github.com")


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""


def run_git_command(*args: str, cwd: Optional[str] = None) -> str:
    """Run a git command and return stdout."""
    try:
        result = subprocess.run(
            ["git", *args],
            cwd=cwd,
            capture_output=True,
            text=True,
            check=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        log.error("git_command_failed", cmd=args, stderr=e.stderr, stdout=e.stdout)
        raise GitWorkflowError(f"Git command failed: {' '.join(args)}") from e


def ensure_clean_working_tree(cwd: str) -> None:
    """Ensure working tree is clean before proceeding."""
    status = run_git_command("status", "--porcelain", cwd=cwd)
    if status:
        log.error("working_tree_dirty", status=status)
        raise GitWorkflowError("Working tree is not clean. Please commit or stash changes first.")


def create_feature_branch(feature_name: str, config: GitConfig) -> str:
    """Create and checkout a new feature branch."""
    cwd = config.repo_path
    ensure_clean_working_tree(cwd)

    # Update base branch first
    log.info("updating_base_branch", base=config.base_branch)
    run_git_command("fetch", config.remote_name, config.base_branch, cwd=cwd)
    run_git_command("checkout", f"{config.remote_name}/{config.base_branch}", cwd=cwd)
    run_git_command("pull", config.remote_name, config.base_branch, cwd=cwd)

    branch_name = f"feature/{feature_name.lower().replace(' ', '-')}"
    log.info("creating_feature_branch", branch=branch_name)
    run_git_command("checkout", "-b", branch_name, cwd=cwd)
    return branch_name


def commit_all_changes(message: str, cwd: str) -> None:
    """Stage and commit all changes."""
    run_git_command("add", ".", cwd=cwd)
    run_git_command("commit", "-m", message, cwd=cwd)
    log.info("changes_committed", message=message)


def push_branch(branch_name: str, config: GitConfig) -> None:
    """Push branch to remote (safely)."""
    log.info("pushing_branch", branch=branch_name, remote=config.remote_name)
    run_git_command("push", config.remote_name, branch_name, cwd=config.repo_path)


def create_merge_request_github(
    branch_name: str,
    title: str,
    body: str,
    config: GitHubConfig,
    base: str = "main",
) -> dict:
    """Create a pull request on GitHub via API."""
    if not config.token:
        raise ValueError("GITHUB_TOKEN environment variable is required")

    url = f"{config.api_url}/repos/{config.owner}/{config.repo}/pulls"
    payload = {
        "title": title,
        "head": branch_name,
        "base": base,
        "body": body,
    }

    headers = {
        "Authorization": f"token {config.token}",
        "Accept": "application/vnd.github.v3+json",
    }

    try:
        response = requests.post(url, json=payload, headers=headers, timeout=30)
        response.raise_for_status()
        pr = response.json()
        log.info("merge_request_created", pr_url=pr["html_url"], pr_number=pr["number"])
        return pr
    except requests.RequestException as e:
        log.error(
            "merge_request_failed",
            status=getattr(e.response, "status_code", None),
            error=str(e),
        )
        raise GitWorkflowError(f"Failed to create merge request: {e}") from e


def main():
    """Main entry point for the git workflow script."""
    import argparse

    parser = argparse.ArgumentParser(description="Automated Git Workflow")
    parser.add_argument("feature_name", help="Name of the feature branch")
    parser.add_argument("--commit-message", default="feat: add new feature", help="Commit message")
    parser.add_argument("--title", help="Merge request title (defaults to feature name)")
    parser.add_argument("--body", default="", help="Merge request description")
    parser.add_argument("--owner", default=os.environ.get("GITHUB_OWNER", ""))
    parser.add_argument("--repo", default=os.environ.get("GITHUB_REPO", ""))
    parser.add_argument("--base-branch", default="main", help="Base branch to merge into")
    parser.add_argument("--repo-path", default=".", help="Path to git repository")

    args = parser.parse_args()

    if not args.owner or not args.repo:
        log.error("missing_github_config", owner=args.owner, repo=args.repo)
        sys.exit("Error: GitHub owner and repo must be provided (via args or env vars)")

    # Validate repository path
    repo_path = Path(args.repo_path).resolve()
    if not (repo_path / ".git").exists():
        log.error("invalid_repo", path=str(repo_path))
        sys.exit(f"Error: Not a git repository: {repo_path}")

    git_config = GitConfig(repo_path=str(repo_path), base_branch=args.base_branch)
    gh_config = GitHubConfig(owner=args.owner, repo=args.repo)

    try:
        # Step 1: Create feature branch
        branch = create_feature_branch(args.feature_name, git_config)

        # Step 2: Commit changes (user must have made changes first)
        commit_all_changes(args.commit_message, str(repo_path))

        # Step 3: Push branch
        push_branch(branch, git_config)

        # Step 4: Create merge request
        title = args.title or f"feat: {args.feature_name}"
        pr = create_merge_request_github(
            branch_name=branch,
            title=title,
            body=args.body,
            config=gh_config,
            base=args.base_branch,
        )

        log.info("workflow_complete", pr_url=pr["html_url"])

    except GitWorkflowError as e:
        log.error("workflow_failed", error=str(e))
        sys.exit(1)
    except Exception as e:
        log.error("unexpected_error", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()
```

### Key Safety Features:

1. **No force push** — Only uses safe `git push` (no `--force`)
2. **Uses `structlog`** for structured logging as required
3. **Uses `pydantic`** for configuration validation
4. **Includes error handling** for all external operations (git commands, HTTP requests)
5. **Requires environment variables** for secrets (GitHub token)
6. **Validates repository state** (ensures clean working tree)
7. **Type hints** on all functions

### How to Use Safely:

- Always commit your changes before running the script
- If you need to rebase, do it locally first, then push with `git push --force-with-lease` (which is safer than `--force` but still carries risk)
- For production use, consider using a CI/CD pipeline instead of local automation

This script follows all the rules while providing a robust, production-ready workflow automation tool.
