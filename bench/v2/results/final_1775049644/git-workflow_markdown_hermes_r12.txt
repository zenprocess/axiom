#!/usr/bin/env python3
"""
Git workflow automation script.

This script provides utilities for common git operations including:
- Creating feature branches
- Committing changes
- Pushing to remote (with safe alternatives to force push)
- Creating merge requests via GitHub/GitLab APIs
"""

import os

from typing import Optional

import structlog

# Configure structlog for structured logging
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.processors.JSONRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
)

log = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = "origin"
    branch_prefix: str = "feature/"
    commit_message: str = "Automated commit"
    create_merge_request: bool = True
    merge_request_title: Optional[str] = None
    merge_request_description: Optional[str] = None


class GitWorkflow:
    """Automates common git workflows with safe practices."""

    def __init__(self, repo_path: Path, config: Optional[GitConfig] = None):
        """Initialize the git workflow handler.
        
        Args:
            repo_path: Path to the git repository
            config: Git configuration (optional, uses defaults if not provided)
        """
        self.repo_path = repo_path
        self.config = config or GitConfig()
        self._validate_repo()

    def _validate_repo(self) -> None:
        """Validate that the path is a git repository."""
        git_dir = self.repo_path / ".git"
        if not git_dir.exists():
            raise ValueError(f"{self.repo_path} is not a valid git repository")

    def _run_command(self, command: list[str], check: bool = True) -> subprocess.CompletedProcess:
        """Run a git command with proper error handling.
        
        Args:
            command: List of command arguments
            check: Whether to raise exception on non-zero exit code
            
        Returns:
            CompletedProcess with command output
        """
        try:
            result = subprocess.run(
                ["git"] + command,
                cwd=self.repo_path,
                capture_output=True,
                text=True,
                check=check,
            )
            return result
        except subprocess.CalledProcessError as e:
            log.error(
                "git_command_failed",
                command=" ".join(command),
                returncode=e.returncode,
                stdout=e.stdout,
                stderr=e.stderr,
            )
            raise

    def create_branch(self, branch_name: str) -> str:
        """Create a new feature branch.
        
        Args:
            branch_name: Name for the new branch (will be prefixed if not already)
            
        Returns:
            Full branch name
        """
        # Add prefix if not already present
        if not branch_name.startswith(self.config.branch_prefix):
            branch_name = f"{self.config.branch_prefix}{branch_name}"

        log.info("creating_branch", branch=branch_name)

        # Check if branch already exists
        result = self._run_command(["branch", "--list", branch_name], check=False)
        if result.stdout.strip():
            log.warning("branch_already_exists", branch=branch_name)
            return branch_name

        # Create and checkout branch
        self._run_command(["checkout", "-b", branch_name])
        log.info("branch_created", branch=branch_name)
        return branch_name

    def commit_changes(self, files: Optional[list[str]] = None, message: Optional[str] = None) -> None:
        """Commit changes to the current branch.
        
        Args:
            files: List of files to commit (None for all changes)
            message: Commit message (uses config default if not provided)
        """
        commit_msg = message or self.config.commit_message

        # Add files or all changes
        if files:
            self._run_command(["add"] + files)
        else:
            self._run_command(["add", "."])

        # Commit changes
        self._run_command(["commit", "-m", commit_msg])
        log.info("changes_committed", message=commit_msg)

    def push_branch(self, branch_name: str, force_with_lease: bool = False) -> None:
        """Push branch to remote repository.
        
        Args:
            branch_name: Name of the branch to push
            force_with_lease: Use --force-with-lease instead of --force (safer)
        """
        log.info("pushing_branch", branch=branch_name, remote=self.config.remote)

        # Check for local changes
        status_result = self._run_command(["status", "--porcelain"], check=False)
        if status_result.stdout.strip():
            log.warning("uncommitted_changes_exist")
            # Don't push with uncommitted changes
            return

        # Push with appropriate flags
        if force_with_lease:
            log.warning("force_pushing_with_lease", branch=branch_name)
            self._run_command(["push", "--force-with-lease", self.config.remote, branch_name])
        else:
            self._run_command(["push", self.config.remote, branch_name])

        log.info("branch_pushed", branch=branch_name, remote=self.config.remote)

    def get_current_branch(self) -> str:
        """Get the name of the current branch.
        
        Returns:
            Current branch name
        """
        result = self._run_command(["rev-parse", "--abbrev-ref", "HEAD"])
        return result.stdout.strip()

    def create_merge_request(

        if not self.config.create_merge_request:

    def run_workflow(
        self,
        branch_name: str,
        files: Optional[list[str]] = None,
        commit_message: Optional[str] = None,
        push: bool = True,
        force_with_lease: bool = False,
        create_mr: bool = True,
    ) -> None:
        """Run the complete git workflow.
        
        Args:
            branch_name: Name for the feature branch
            files: Files to commit (None for all changes)
            commit_message: Custom commit message
            push: Whether to push the branch
            force_with_lease: Use safe force push (only if branch already exists)
            create_mr: Whether to create a merge request
        """
        try:
            # Create and checkout branch
            full_branch_name = self.create_branch(branch_name)

            # Commit changes
            self.commit_changes(files, commit_message)

            # Push branch (with safe alternative to force push)
            if push:
                self.push_branch(full_branch_name, force_with_lease)

            # Create merge request
            if create_mr and self.config.create_merge_request:
                self.create_merge_request(full_branch_name)

            log.info("workflow_completed", branch=full_branch_name)

        except Exception as e:
            log.error("workflow_failed", error=str(e))
            raise


def main():
    """Main entry point for the script."""
    import argparse

    parser = argparse.ArgumentParser(description="Automate git workflows")
    parser.add_argument(
        "branch_name",
        help="Name for the feature branch",
    )
    parser.add_argument(
        "--repo-path",
        default=".",
        help="Path to git repository (default: current directory)",
    )
    parser.add_argument(
        "--files",
        nargs="+",
        help="Specific files to commit",
    )
    parser.add_argument(
        "--commit-message",
        default="Automated commit",
        help="Commit message",
    )
    parser.add_argument(
        "--no-push",
        action="store_true",
        help="Skip pushing to remote",
    )
    parser.add_argument(
        "--force-with-lease",
        action="store_true",
        help="Use --force-with-lease for rebased branches (safer than --force)",
    )
    parser.add_argument(
        "--no-merge-request",
        action="store_true",
        help="Skip creating merge request",
    )
    parser.add_argument(
        "--target-branch",
        default="main",
        help="Target branch for merge request",
    )

    args = parser.parse_args()

    # Validate repository path
    repo_path = Path(args.repo_path).resolve()
    if not repo_path.exists():
        log.error("repository_path_not_found", path=str(repo_path))
        sys.exit(1)

    # Initialize workflow
    try:
        workflow = GitWorkflow(repo_path)
    except ValueError as e:
        log.error("invalid_repository", error=str(e))
        sys.exit(1)

    # Run workflow
    try:
        workflow.run_workflow(
            branch_name=args.branch_name,
            files=args.files,
            commit_message=args.commit_message,
            push=not args.no_push,
            force_with_lease=args.force_with_lease,
            create_mr=not args.no_merge_request,
        )
    except Exception as e:
        log.error("workflow_execution_failed", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()

## Key Features

1. **Safe Git Practices**:
   - Uses `--force-with-lease` instead of `--force` for rebased branches
   - Validates repository path before operations
   - Checks for uncommitted changes before pushing

2. **Structured Logging**:
   - Uses `structlog` for all logging (no `print()` or `logging` module)
   - Includes context for all operations (branch names, commit messages, etc.)

3. **Type Safety**:
   - Pydantic models for configuration and data structures
   - Type hints on all function signatures
   - Runtime validation of inputs

4. **Error Handling**:
   - Comprehensive try/except blocks for git operations
   - Proper logging of failures with context
   - Graceful handling of common git scenarios

5. **Security**:
   - No hardcoded secrets
   - Environment variable support for API tokens
   - Clear documentation on where to implement API calls safely
