#!/usr/bin/env python3
"""
Git workflow automation script for creating feature branches, committing changes,
pushing to remote, and creating merge requests.

Uses structlog for structured logging and async I/O for git operations.
"""

import asyncio
import os
import subprocess
import sys
from typing import List, Optional

import structlog
from pydantic import BaseModel, Field

# Configure structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.processors.UnicodeDecoder(),
        structlog.processors.JSONRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=True,
)

log = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = "origin"
    branch_prefix: str = "feature/"
    commit_message: str = "Automated commit"
    mr_title: str = "Automated merge request"
    mr_description: str = "This merge request was automatically created"


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""
    pass


class GitWorkflow:
    """Automates git workflows including branch creation, commits, pushes, and MRs."""

    def __init__(self, config: Optional[GitConfig] = None):
        """Initialize the Git workflow with configuration."""
        self.config = config or GitConfig()
        self.base_dir = os.getcwd()

    async def _run_git_command(self, *args: str, check: bool = True) -> str:
        """Run a git command asynchronously."""
        try:
            result = await asyncio.to_thread(
                subprocess.run,
                ["git", *args],
                capture_output=True,
                text=True,
                check=check,
                cwd=self.base_dir,
            )
            if result.returncode != 0:
                log.error(
                    "git_command_failed",
                    command=" ".join(args),
                    stdout=result.stdout,
                    stderr=result.stderr,
                    returncode=result.returncode,
                )
                raise GitWorkflowError(
                    f"Git command failed: {' '.join(args)}"
                )
            log.info("git_command_succeeded", command=" ".join(args))
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            log.error("git_command_error", error=str(e))
            raise GitWorkflowError(f"Git command failed: {e}") from e

    async def get_current_branch(self) -> str:
        """Get the current git branch."""
        return await self._run_git_command("rev-parse", "--abbrev-ref", "HEAD")

    async def get_remote_branches(self) -> List[str]:
        """Get list of remote branches."""
        output = await self._run_git_command("branch", "-r")
        return [line.strip() for line in output.split("\n") if line.strip()]

    async def create_feature_branch(self, name: str) -> str:
        """Create a new feature branch."""
        branch_name = f"{self.config.branch_prefix}{name}"
        log.info("creating_feature_branch", branch=branch_name)
        
        # Check if branch already exists
        branches = await self.get_remote_branches()
        if f"origin/{branch_name}" in branches:
            log.warning("branch_already_exists", branch=branch_name)
            return branch_name
        
        # Create and checkout branch
        await self._run_git_command("checkout", "-b", branch_name)
        log.info("feature_branch_created", branch=branch_name)
        return branch_name

    async def add_files(self, patterns: List[str]) -> None:
        """Add files matching patterns to staging area."""
        if not patterns:
            patterns = ["."]
        
        log.info("adding_files", patterns=patterns)
        for pattern in patterns:
            await self._run_git_command("add", pattern)
        log.info("files_added", count=len(patterns))

    async def commit_changes(self, message: Optional[str] = None) -> None:
        """Commit staged changes."""
        commit_msg = message or self.config.commit_message
        log.info("committing_changes", message=commit_msg)
        
        # Check if there are changes to commit
        status = await self._run_git_command("status", "--porcelain")
        if not status:
            log.warning("no_changes_to_commit")
            return
        
        await self._run_git_command("commit", "-m", commit_msg)
        log.info("changes_committed", message=commit_msg)

    async def push_branch(
        self, 
        branch: str, 
        force: bool = False, 
        force_with_lease: bool = False
    ) -> None:
        """Push branch to remote repository."""
        log.info(
            "pushing_branch", 
            branch=branch, 
            force=force, 
            force_with_lease=force_with_lease
        )
        
        if force and force_with_lease:
            raise ValueError("Cannot use both force and force_with_lease")
        
        if force:
            log.warning("using_force_push", branch=branch)
            await self._run_git_command("push", "-f", self.config.remote, branch)
        elif force_with_lease:
            log.info("using_force_with_lease", branch=branch)
            await self._run_git_command("push", "--force-with-lease", self.config.remote, branch)
        else:
            await self._run_git_command("push", self.config.remote, branch)
        
        log.info("branch_pushed", branch=branch)

    async def create_merge_request(
        self,
        branch: str,
        title: Optional[str] = None,
        description: Optional[str] = None,
        target_branch: str = "main"
    ) -> str:
        """Create a merge request (GitHub/GitLab specific - placeholder)."""
        log.info(
            "creating_merge_request",
            branch=branch,
            target_branch=target_branch,
            title=title or self.config.mr_title,
        )
        
        # This is a placeholder - in real implementation you'd use:
        # - GitHub API: github3.py or PyGithub
        # - GitLab API: python-gitlab
        # - Bitbucket API: bitbucket
        # - Or use git commands if self-hosted solution provides CLI
        
        mr_url = f"https://github.com/{os.environ.get('GITHUB_REPOSITORY', 'user/repo')}/compare/{target_branch}...{branch}"
        log.info("merge_request_created", url=mr_url)
        return mr_url

    async def run_workflow(
        self,
        feature_name: str,
        files_to_add: Optional[List[str]] = None,
        commit_message: Optional[str] = None,
        create_mr: bool = True,
        force_push: bool = False,
        force_with_lease: bool = False,
    ) -> dict:
        """Run the complete git workflow."""
        log.info(
            "starting_git_workflow",
            feature_name=feature_name,
            force_push=force_push,
            force_with_lease=force_with_lease,
        )
        
        try:
            # Check git is available
            await self._run_git_command("--version")
            
            # Get current branch to return to later
            original_branch = await self.get_current_branch()
            
            # Create feature branch
            feature_branch = await self.create_feature_branch(feature_name)
            
            # Add files
            await self.add_files(files_to_add or [])
            
            # Commit changes
            await self.commit_changes(commit_message)
            
            # Push branch
            await self.push_branch(
                feature_branch, 
                force=force_push, 
                force_with_lease=force_with_lease
            )
            
            # Create merge request
            mr_url = None
            if create_mr:
                mr_url = await self.create_merge_request(feature_branch)
            
            # Return to original branch
            if original_branch != feature_branch:
                await self._run_git_command("checkout", original_branch)
            
            log.info(
                "git_workflow_completed",
                feature_name=feature_name,
                branch=feature_branch,
                mr_url=mr_url,
            )
            
            return {
                "success": True,
                "feature_branch": feature_branch,
                "merge_request_url": mr_url,
                "original_branch": original_branch,
            }
            
        except Exception as e:
            log.error("git_workflow_failed", error=str(e))
            raise


async def main() -> None:
    """Main entry point for the script."""
    import argparse
    
    parser = argparse.ArgumentParser(
        description="Automate git workflows for feature branches and merge requests"
    )
    parser.add_argument(
        "feature_name",
        help="Name for the feature branch (will be prefixed with 'feature/')"
    )
    parser.add_argument(
        "--files", 
        nargs="+", 
        default=[],
        help="Files to add (default: all changes)"
    )
    parser.add_argument(
        "--commit-message",
        default=None,
        help="Commit message (default: 'Automated commit')"
    )
    parser.add_argument(
        "--no-mr",
        action="store_true",
        help="Skip creating merge request"
    )
    parser.add_argument(
        "--force",
        action="store_true",
        help="Force push (dangerous - use only for rebased branches)"
    )
    parser.add_argument(
        "--force-with-lease",
        action="store_true",
        help="Force push with lease (safer than force)"
    )
    parser.add_argument(
        "--target-branch",
        default="main",
        help="Target branch for merge request (default: main)"
    )
    
    args = parser.parse_args()
    
    # Validate force options
    if args.force and args.force_with_lease:
        log.error("invalid_force_options")
        print("Error: Cannot use both --force and --force-with-lease", file=sys.stderr)
        sys.exit(1)
    
    # Check for secrets in environment
    if any(key in os.environ for key in ['API_KEY', 'SECRET_TOKEN', 'GITHUB_TOKEN']):
        log.warning("potential_secrets_in_environment")
    
    config = GitConfig(
        commit_message=args.commit_message or "Automated commit"
    )
    
    workflow = GitWorkflow(config)
    
    try:
        result = await workflow.run_workflow(
            feature_name=args.feature_name,
            files_to_add=args.files,
            create_mr=not args.no_mr,
            force_push=args.force,
            force_with_lease=args.force_with_lease,
        )
        
        if result["success"]:
            print(f"✓ Feature branch '{result['feature_branch']}' created and pushed")
            if result["merge_request_url"]:
                print(f"✓ Merge request created: {result['merge_request_url']}")
        else:
            print("✗ Workflow failed", file=sys.stderr)
            sys.exit(1)
            
    except GitWorkflowError as e:
        log.error("workflow_error", error=str(e))
        print(f"Error: {e}", file=sys.stderr)
        sys.exit(1)
    except Exception as e:
        log.error("unexpected_error", error=str(e))
        print(f"Unexpected error: {e}", file=sys.stderr)
        sys.exit(1)


if __name__ == "__main__":
    asyncio.run(main())
