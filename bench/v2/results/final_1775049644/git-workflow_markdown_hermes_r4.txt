#!/usr/bin/env python3
"""
Git Workflow Automation Script

This script automates common Git workflows:
- Create feature branches
- Stage and commit changes
- Push to remote (safely)
- Create merge requests via GitHub/GitLab APIs

It follows best practices:
- No force push by default; --force-with-lease only when explicitly requested
- Structured logging with structlog
- Pydantic models for configuration and results
- Proper error handling
"""

import os
import subprocess
import sys
from typing import Optional, List, Dict, Any

import structlog
from pydantic import BaseModel, Field, field_validator
from dotenv import load_dotenv

# Load environment variables from .env if present
load_dotenv()

# Initialize structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.WARNING),
)
log = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for Git repository operations."""

    repo_path: str = Field(default=".")
    base_branch: str = Field(default="main")
    remote_name: str = Field(default="origin")
    commit_message: str = Field(default="Update feature branch")
    force_with_lease: bool = Field(default=False, description="Use --force-with-lease for push")


class MergeRequestConfig(BaseModel):
    """Configuration for creating a merge request."""

    title: str = Field(..., min_length=1)
    description: str = Field(default="")
    target_branch: str = Field(..., min_length=1)
    source_branch: str = Field(..., min_length=1)
    git_provider: str = Field(default="github")  # "github" or "gitlab"


class GitWorkflowError(Exception):
    """Custom exception for Git workflow errors."""
    pass


def run_git_command(
    args: List[str],
    cwd: str = ".",
    check: bool = True,
    capture_output: bool = True,
) -> subprocess.CompletedProcess:
    """Run a Git command safely with structured logging."""
    try:
        result = subprocess.run(
            ["git"] + args,
            cwd=cwd,
            check=check,
            capture_output=capture_output,
            text=True,
        )
        log.info("git_command_executed", args=args, returncode=result.returncode)
        return result
    except subprocess.CalledProcessError as e:
        log.error("git_command_failed", args=args, stderr=e.stderr, stdout=e.stdout)
        raise GitWorkflowError(f"Git command failed: {' '.join(args)}") from e


def create_feature_branch(config: GitConfig, feature_name: str) -> str:
    """Create a new feature branch from base branch."""
    full_branch_name = f"{feature_name}"
    log.info("creating_feature_branch", branch=full_branch_name, base=config.base_branch)

    # Ensure we're on base branch and it's up to date
    run_git_command(["checkout", config.base_branch], cwd=config.repo_path)
    run_git_command(["pull", config.remote_name, config.base_branch], cwd=config.repo_path)

    # Create and checkout feature branch
    run_git_command(["checkout", "-b", full_branch_name], cwd=config.repo_path)
    log.info("feature_branch_created", branch=full_branch_name)
    return full_branch_name


def commit_changes(config: GitConfig, files: Optional[List[str]] = None) -> None:
    """Stage and commit changes."""
    log.info("committing_changes", commit_message=config.commit_message)

    # Stage all changes or specific files
    if files:
        for file in files:
            run_git_command(["add", file], cwd=config.repo_path)
    else:
        run_git_command(["add", "."], cwd=config.repo_path)

    # Commit with message
    run_git_command(["commit", "-m", config.commit_message], cwd=config.repo_path)
    log.info("changes_committed", message=config.commit_message)


def push_branch(config: GitConfig, branch_name: str) -> None:
    """Push branch to remote repository safely."""
    log.info("pushing_branch", branch=branch_name, remote=config.remote_name)

    # Use --force-with-lease only if explicitly requested
    push_args = ["push", config.remote_name, branch_name]
    if config.force_with_lease:
        log.warning("using_force_with_lease", branch=branch_name)
        push_args.insert(2, "--force-with-lease")
    else:
        # Default safe push (does not rewrite history)
        push_args.append("--set-upstream")

    run_git_command(push_args, cwd=config.repo_path)
    log.info("branch_pushed", branch=branch_name, remote=config.remote_name)

def main():
    """Main entry point for the Git workflow automation."""
    import argparse
    import logging  # Only for log level configuration

    parser = argparse.ArgumentParser(
        description="Automate Git workflows: create branches, commit, push, and create merge requests."
    )
    parser.add_argument("feature_name", help="Name for the feature branch")
    parser.add_argument(
        "--commit-message",
        "-m",
        default="Update feature branch",
        help="Commit message for changes",
    )
    parser.add_argument(
        "--base-branch",
        "-b",
        default="main",
        help="Base branch to branch from (default: main)",
    )
    parser.add_argument(
        "--remote",
        "-r",
        default="origin",
        help="Remote repository name (default: origin)",
    )
    parser.add_argument(
        "--force-with-lease",
        action="store_true",
        help="Use --force-with-lease for push (use with caution)",
    )
    parser.add_argument(
        "--mr-title",
        required=True,
        help="Title for the merge request",
    )
    parser.add_argument(
        "--mr-target",
        required=True,
        help="Target branch for the merge request",
    )
    parser.add_argument(
        "--mr-provider",
        choices=["github", "gitlab"],
        default="github",
        help="Git provider (default: github)",
    )
    parser.add_argument(
        "--log-level",
        choices=["DEBUG", "INFO", "WARNING", "ERROR"],
        default="INFO",
        help="Logging level (default: INFO)",
    )

    args = parser.parse_args()

    # Configure logging level
    structlog.configure(
        wrapper_class=structlog.make_filtering_bound_logger(getattr(logging, args.log_level))
    )

    # Validate required environment variables for MR creation
    if args.mr_provider == "github" and not os.environ.get("GITHUB_TOKEN"):
        log.error("missing_env_var", var="GITHUB_TOKEN")
        sys.exit(1)
    if args.mr_provider == "gitlab" and not os.environ.get("GITLAB_TOKEN"):
        log.error("missing_env_var", var="GITLAB_TOKEN")
        sys.exit(1)

    try:
        # Create configuration objects
        git_config = GitConfig(
            repo_path=os.getcwd(),
            base_branch=args.base_branch,
            remote_name=args.remote,
            commit_message=args.commit_message,
            force_with_lease=args.force_with_lease,
        )

        # Execute workflow
        feature_branch = create_feature_branch(git_config, args.feature_name)
        commit_changes(git_config)
        push_branch(git_config, feature_branch)

        mr_config = MergeRequestConfig(
            title=args.mr_title,
            target_branch=args.mr_target,
            source_branch=feature_branch,
            git_provider=args.mr_provider,
        )
        mr_result = create_merge_request(mr_config)

        log.info(
            "workflow_completed",
            branch=feature_branch,
            merge_request=mr_result,
        )

    except GitWorkflowError as e:
        log.error("workflow_failed", error=str(e))
        sys.exit(1)
    except Exception as e:
        log.error("unexpected_error", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()
