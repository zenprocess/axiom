```python
#!/usr/bin/env python3
"""
Git Workflow Automation Script

This script automates common git workflows including:
- Creating feature branches
- Committing changes
- Pushing to remote (with safe force-with-lease for rebased branches)
- Creating merge requests via GitHub/GitLab API
"""

import os
import subprocess
import sys
from typing import Optional

import structlog
from pydantic import BaseModel, Field, field_validator
import httpx

# Initialize structured logging
log = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = "origin"
    base_branch: str = "main"
    branch_prefix: str = "feature/"
    commit_message: str = "Automated commit"
    mr_title: str = "Merge changes"
    mr_description: str = ""


class MergeRequestConfig(BaseModel):
    """Configuration for merge request creation."""
    platform: str = Field(default="github", pattern="^(github|gitlab)$")
    api_url: Optional[str] = None
    token: str = Field(default_factory=lambda: os.environ.get("GIT_API_TOKEN", ""))
    project_id: Optional[str] = None
    target_branch: str = "main"
    source_branch: str = "feature/my-feature"


class GitWorkflowError(Exception):
    """Custom exception for git workflow operations."""
    pass


class GitWorkflow:
    """Automates git workflows for feature branches and merge requests."""

    def __init__(
        self,
        git_config: GitConfig,
        mr_config: Optional[MergeRequestConfig] = None,
    ):
        """Initialize the GitWorkflow with configuration."""
        self.git_config = git_config
        self.mr_config = mr_config
        self.repo_path = os.getcwd()

    async def run_command(self, command: list[str]) -> str:
        """
        Run a git command and return the output.

        Args:
            command: List of command arguments

        Returns:
            Command output as string

        Raises:
            GitWorkflowError: If command fails
        """
        try:
            result = subprocess.run(
                command,
                capture_output=True,
                text=True,
                check=True,
                cwd=self.repo_path,
            )
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            log.error(
                "git_command_failed",
                command=" ".join(command),
                stderr=e.stderr,
                returncode=e.returncode,
            )
            raise GitWorkflowError(f"Git command failed: {e.stderr}") from e

    def ensure_clean_working_directory(self) -> None:
        """
        Check if the working directory is clean.

        Raises:
            GitWorkflowError: If there are uncommitted changes
        """
        status = subprocess.run(
            ["git", "status", "--porcelain"],
            capture_output=True,
            text=True,
            cwd=self.repo_path,
        )
        if status.stdout.strip():
            log.error("uncommitted_changes_detected", output=status.stdout)
            raise GitWorkflowError(
                "Working directory is not clean. Commit or stash changes first."
            )

    async def create_feature_branch(self, branch_name: str) -> str:
        """
        Create a new feature branch from the base branch.

        Args:
            branch_name: Name of the feature branch (without prefix)

        Returns:
            Full branch name
        """
        full_branch_name = f"{self.git_config.branch_prefix}{branch_name}"
        log.info("creating_feature_branch", branch=full_branch_name)

        # Ensure clean working directory
        self.ensure_clean_working_directory()

        # Checkout base branch and pull latest
        await self.run_command(["git", "checkout", self.git_config.base_branch])
        await self.run_command(["git", "pull", self.git_config.remote, self.git_config.base_branch])

        # Create and checkout new feature branch
        await self.run_command(["git", "checkout", "-b", full_branch_name])
        log.info("feature_branch_created", branch=full_branch_name)

        return full_branch_name

    async def commit_changes(self, message: Optional[str] = None) -> None:
        """
        Commit all staged changes.

        Args:
            message: Commit message (uses config default if not provided)
        """
        commit_msg = message or self.git_config.commit_message
        log.info("committing_changes", message=commit_msg)

        # Check if there are changes to commit
        status = subprocess.run(
            ["git", "status", "--porcelain"],
            capture_output=True,
            text=True,
            cwd=self.repo_path,
        )
        if not status.stdout.strip():
            log.warning("no_changes_to_commit")
            return

        await self.run_command(["git", "add", "."])
        await self.run_command(["git", "commit", "-m", commit_msg])
        log.info("changes_committed", message=commit_msg)

    async def push_branch(
        self, branch_name: str, force_with_lease: bool = False
    ) -> None:
        """
        Push branch to remote repository.

        Args:
            branch_name: Name of the branch to push
            force_with_lease: Use safe force push for rebased branches
        """
        log.info(
            "pushing_branch",
            branch=branch_name,
            force_with_lease=force_with_lease,
        )

        if force_with_lease:
            await self.run_command(
                [
                    "git",
                    "push",
                    "--force-with-lease",
                    self.git_config.remote,
                    branch_name,
                ]
            )
            log.info("branch_pushed_force_with_lease", branch=branch_name)
        else:
            await self.run_command(
                ["git", "push", self.git_config.remote, branch_name]
            )
            log.info("branch_pushed", branch=branch_name)

    async def create_merge_request(self) -> dict:
        """
        Create a merge request via GitHub or GitLab API.

        Returns:
            Merge request response data
        """
        if not self.mr_config:
            raise GitWorkflowError("Merge request configuration not provided")

        log.info(
            "creating_merge_request",
            platform=self.mr_config.platform,
            source=self.mr_config.source_branch,
            target=self.mr_config.target_branch,
        )

        # Validate required configuration
        if not self.mr_config.token:
            raise GitWorkflowError("API token not provided")

        if not self.mr_config.project_id:
            raise GitWorkflowError("Project ID not provided")

        # Set up API endpoint
        api_url = self.mr_config.api_url or (
            "https://api.github.com"
            if self.mr_config.platform == "github"
            else "https://gitlab.com/api/v4"
        )

        headers = {
            "Authorization": f"Bearer {self.mr_config.token}",
            "Accept": "application/json",
        }

        if self.mr_config.platform == "github":
            data = {
                "title": self.mr_config.mr_title,
                "head": self.mr_config.source_branch,
                "base": self.mr_config.target_branch,
                "body": self.mr_config.mr_description,
            }
            endpoint = f"{api_url}/repos/{self.mr_config.project_id}/pulls"
        else:  # gitlab
            data = {
                "title": self.mr_config.mr_title,
                "source_branch": self.mr_config.source_branch,
                "target_branch": self.mr_config.target_branch,
                "description": self.mr_config.mr_description,
            }
            endpoint = f"{api_url}/projects/{self.mr_config.project_id}/merge_requests"

        try:
            async with httpx.AsyncClient() as client:
                response = await client.post(endpoint, json=data, headers=headers)
                response.raise_for_status()
                mr_data = response.json()

                log.info(
                    "merge_request_created",
                    mr_id=mr_data.get("id"),
                    mr_url=mr_data.get("web_url"),
                    platform=self.mr_config.platform,
                )
                return mr_data

        except httpx.HTTPError as e:
            log.error(
                "merge_request_creation_failed",
                error=str(e),
                platform=self.mr_config.platform,
                project_id=self.mr_config.project_id,
            )
            raise GitWorkflowError(
                f"Failed to create merge request: {str(e)}"
            ) from e


async def main():
    """Main entry point for the git workflow automation."""
    # Configure logging
    structlog.configure(
        processors=[
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.add_log_level,
            structlog.processors.JSONRenderer(),
        ],
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=True,
    )

    try:
        # Create configuration
        git_config = GitConfig(
            remote="origin",
            base_branch="main",
            branch_prefix="feature/",
            commit_message="Automated commit with feature changes",
        )

        mr_config = MergeRequestConfig(
            platform="github",
            project_id=os.environ.get("GITHUB_REPO", "owner/repo"),
            source_branch="feature/my-new-feature",
            target_branch="main",
            mr_title="Add new feature",
            mr_description="This MR implements the new feature",
        )

        # Initialize workflow
        workflow = GitWorkflow(git_config, mr_config)

        # Execute workflow
        branch_name = await workflow.create_feature_branch("my-new-feature")
        await workflow.commit_changes()
        await workflow.push_branch(branch_name)

        # For rebased branches, use safe force push
        # await workflow.push_branch(branch_name, force_with_lease=True)

        # Create merge request
        mr_data = await workflow.create_merge_request()
        log.info("merge_request_created_successfully", mr_url=mr_data.get("web_url"))

    except GitWorkflowError as e:
        log.error("git_workflow_failed", error=str(e))
        sys.exit(1)
    except Exception as e:
        log.error("unexpected_error", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
