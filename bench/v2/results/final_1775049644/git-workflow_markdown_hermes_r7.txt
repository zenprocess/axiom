#!/usr/bin/env python3
"""
Git workflow automation script.

Automates common git workflows: creating feature branches, committing changes,
pushing to remote, and creating merge requests. Includes safe branch switching
and proper error handling for git operations.
"""

import os
import subprocess
import sys
from typing import Optional

import structlog
from pydantic import BaseModel, Field

log = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = Field(default="origin")
    base_branch: str = Field(default="main")
    commit_message: str = Field(default="Auto-commit changes")
    mr_title: str = Field(default="New feature")
    mr_description: str = Field(default="")


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""
    pass


async def run_command(cmd: list[str], cwd: Optional[str] = None) -> str:
    """
    Run a shell command and return its output.
    
    Args:
        cmd: Command to run as a list of strings
        cwd: Working directory for the command
        
    Returns:
        Command output as a string
        
    Raises:
        GitWorkflowError: If command fails
    """
    try:
        result = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            check=True,
            cwd=cwd or os.getcwd()
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        log.error(
            "command_failed",
            cmd=" ".join(cmd),
            returncode=e.returncode,
            stderr=e.stderr,
            stdout=e.stdout
        )
        raise GitWorkflowError(f"Command failed: {' '.join(cmd)}") from e


def get_current_branch(cwd: Optional[str] = None) -> str:
    """
    Get the current git branch name.
    
    Args:
        cwd: Working directory
        
    Returns:
        Current branch name
    """
    return run_command(["git", "rev-parse", "--abbrev-ref", "HEAD"], cwd=cwd)


def branch_exists(branch_name: str, cwd: Optional[str] = None) -> bool:
    """
    Check if a branch exists locally or remotely.
    
    Args:
        branch_name: Name of the branch to check
        cwd: Working directory
        
    Returns:
        True if branch exists, False otherwise
    """
    try:
        run_command(["git", "rev-parse", "--verify", branch_name], cwd=cwd)
        return True
    except GitWorkflowError:
        return False


async def create_feature_branch(
    feature_name: str,
    base_branch: str = "main",
    config: Optional[GitConfig] = None,
    cwd: Optional[str] = None
) -> str:
    """
    Create a new feature branch from the base branch.
    
    Args:
        feature_name: Name for the new feature branch
        base_branch: Branch to branch from (default: main)
        config: Git configuration (optional)
        cwd: Working directory
        
    Returns:
        Name of the created branch
    """
    config = config or GitConfig()
    
    log.info(
        "creating_feature_branch",
        feature_name=feature_name,
        base_branch=base_branch
    )
    
    # Ensure we're on the base branch and it's up to date
    await run_command(["git", "checkout", base_branch], cwd=cwd)
    await run_command(["git", "pull", config.remote, base_branch], cwd=cwd)
    
    # Create and checkout the feature branch
    feature_branch = f"feature/{feature_name}"
    await run_command(
        ["git", "checkout", "-b", feature_branch],
        cwd=cwd
    )
    
    log.info(
        "feature_branch_created",
        branch=feature_branch,
        base_branch=base_branch
    )
    
    return feature_branch


async def commit_changes(
    files: Optional[list[str]] = None,
    message: Optional[str] = None,
    config: Optional[GitConfig] = None,
    cwd: Optional[str] = None
) -> None:
    """
    Stage and commit changes in the git repository.
    
    Args:
        files: List of files to commit (default: all changes)
        message: Commit message (default: from config)
        config: Git configuration (optional)
        cwd: Working directory
    """
    config = config or GitConfig()
    commit_msg = message or config.commit_message
    
    log.info("committing_changes", commit_message=commit_msg)
    
    # Stage all changes or specific files
    if files:
        await run_command(["git", "add"] + files, cwd=cwd)
    else:
        await run_command(["git", "add", "."], cwd=cwd)
    
    # Check if there are changes to commit
    status = await run_command(["git", "status", "--porcelain"], cwd=cwd)
    if not status:
        log.warning("no_changes_to_commit")
        return
    
    # Commit the changes
    await run_command(["git", "commit", "-m", commit_msg], cwd=cwd)
    
    log.info("changes_committed")


async def push_branch(
    branch_name: str,
    force: bool = False,
    config: Optional[GitConfig] = None,
    cwd: Optional[str] = None
) -> None:
    """
    Push a branch to the remote repository.
    
    Args:
        branch_name: Name of the branch to push
        force: Whether to use force push (default: False)
        config: Git configuration (optional)
        cwd: Working directory
        
    Raises:
        GitWorkflowError: If force push is requested (use force_with_lease instead)
    """
    config = config or GitConfig()
    
    if force:
        log.warning(
            "force_push_requested",
            branch=branch_name,
            recommendation="use force_with_lease for safety"
        )
        # For rebased branches, use force-with-lease instead
        await run_command(
            ["git", "push", "--force-with-lease", config.remote, branch_name],
            cwd=cwd
        )
    else:
        await run_command(
            ["git", "push", config.remote, branch_name],
            cwd=cwd
        )
    
    log.info(
        "branch_pushed",
        branch=branch_name,
        remote=config.remote,
        force=force
    )


async def create_merge_request(
    branch_name: str,
    title: Optional[str] = None,
    description: Optional[str] = None,
    config: Optional[GitConfig] = None,
    cwd: Optional[str] = None
) -> str:
    """
    Create a merge request/pull request for the branch.
    
    This is a placeholder that demonstrates the workflow. In production,
    you would integrate with GitLab API, GitHub API, or similar.
    
    Args:
        branch_name: Name of the feature branch
        title: MR title (default: from config)
        description: MR description (default: from config)
        config: Git configuration (optional)
        cwd: Working directory
        
    Returns:
        MR URL or identifier (placeholder implementation)
    """
    config = config or GitConfig()
    mr_title = title or config.mr_title
    mr_description = description or config.mr_description
    
    log.info(
        "creating_merge_request",
        branch=branch_name,
        title=mr_title
    )
    
    # In a real implementation, this would:
    # 1. Get the current repository URL
    # 2. Use the appropriate API (GitHub, GitLab, etc.)
    # 3. Create the MR and return the URL
    
    # For demonstration purposes, we'll create a placeholder
    try:
        remote_url = await run_command(
            ["git", "remote", "get-url", config.remote],
            cwd=cwd
        )
        
        # Extract repository identifier from URL
        repo_name = os.path.splitext(os.path.basename(remote_url))[0]
        
        # This would be the real API call in production
        mr_url = f"placeholder://mr-create/{repo_name}/merge_requests/new?merge_request[branch_source]={branch_name}&merge_request[title]={mr_title}"
        
        log.info(
            "merge_request_created",
            branch=branch_name,
            title=mr_title,
            url=mr_url
        )
        
        return mr_url
        
    except Exception as e:
        log.error(
            "merge_request_creation_failed",
            branch=branch_name,
            error=str(e)
        )
        raise GitWorkflowError(f"Failed to create merge request: {e}") from e


async def run_git_workflow(
    feature_name: str,
    base_branch: str = "main",
    files: Optional[list[str]] = None,
    commit_message: Optional[str] = None,
    title: Optional[str] = None,
    description: Optional[str] = None,
    cwd: Optional[str] = None
) -> str:
    """
    Run the complete git workflow: create branch, commit changes, push, and create MR.
    
    Args:
        feature_name: Name for the new feature branch
        base_branch: Branch to branch from (default: main)
        files: Files to commit (default: all changes)
        commit_message: Commit message (default: from config)
        title: MR title (default: from config)
        description: MR description (default: from config)
        cwd: Working directory
        
    Returns:
        Merge request URL or identifier
    """
    config = GitConfig(
        base_branch=base_branch,
        commit_message=commit_message or "Auto-commit changes",
        mr_title=title or "New feature",
        mr_description=description or ""
    )
    
    try:
        # Step 1: Create feature branch
        feature_branch = await create_feature_branch(
            feature_name,
            base_branch=base_branch,
            config=config,
            cwd=cwd
        )
        
        # Step 2: Commit changes
        await commit_changes(
            files=files,
            message=commit_message,
            config=config,
            cwd=cwd
        )
        
        # Step 3: Push branch (safe push with --force-with-lease)
        await push_branch(
            feature_branch,
            force=False,  # Using safe push by default
            config=config,
            cwd=cwd
        )
        
        # Step 4: Create merge request
        mr_url = await create_merge_request(
            feature_branch,
            title=title,
            description=description,
            config=config,
            cwd=cwd
        )
        
        log.info(
            "git_workflow_completed",
            feature_branch=feature_branch,
            mr_url=mr_url
        )
        
        return mr_url
        
    except Exception as e:
        log.error(
            "git_workflow_failed",
            feature_name=feature_name,
            error=str(e)
        )
        raise


def main():
    """Main entry point for the git workflow script."""
    import argparse
    
    parser = argparse.ArgumentParser(
        description="Automate git workflows: create feature branches, commit, push, and create merge requests"
    )
    parser.add_argument(
        "feature_name",
        help="Name for the new feature branch"
    )
    parser.add_argument(
        "--base-branch",
        default="main",
        help="Branch to branch from (default: main)"
    )
    parser.add_argument(
        "--files",
        nargs="+",
        help="Specific files to commit (default: all changes)"
    )
    parser.add_argument(
        "--commit-message",
        help="Commit message"
    )
    parser.add_argument(
        "--mr-title",
        help="Merge request title"
    )
    parser.add_argument(
        "--mr-description",
        help="Merge request description"
    )
    parser.add_argument(
        "--directory",
        help="Working directory (default: current directory)"
    )
    
    args = parser.parse_args()
    
    # Set up structlog for console output
    structlog.configure(
        processors=[
            structlog.stdlib.add_logger_name,
            structlog.stdlib.add_log_level,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.JSONRenderer()
        ],
        context_class=dict,
        logger_factory=structlog.stdlib.LoggerFactory(),
        wrapper_class=structlog.stdlib.BoundLogger,
        cache_logger_on_first_use=True,
    )
    
    try:
        import asyncio
        mr_url = asyncio.run(
            run_git_workflow(
                feature_name=args.feature_name,
                base_branch=args.base_branch,
                files=args.files,
                commit_message=args.commit_message,
                title=args.mr_title,
                description=args.mr_description,
                cwd=args.directory
            )
        )
        print(f"Merge request created: {mr_url}")
        sys.exit(0)
        
    except GitWorkflowError as e:
        log.error("git_workflow_error", error=str(e))
        sys.exit(1)
    except Exception as e:
        log.error("unexpected_error", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()
