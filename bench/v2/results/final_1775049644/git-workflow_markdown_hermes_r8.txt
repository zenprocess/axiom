```python
#!/usr/bin/env python3
"""
Git workflow automation script.

This module provides functionality for creating feature branches,
committing changes, pushing to remote, and creating merge requests.
"""

import os
import subprocess
import sys
from typing import Optional

import structlog
from pydantic import BaseModel, Field, field_validator

log = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations."""
    remote: str = Field(default="origin")
    branch_prefix: str = Field(default="feature/")
    commit_message: str = Field(default="Auto-generated commit")
    mr_title: str = Field(default="Auto-generated merge request")
    mr_description: str = Field(default="")


class GitWorkflowError(Exception):
    """Custom exception for git workflow operations."""
    pass


async def run_command(cmd: list[str], cwd: Optional[str] = None) -> str:
    """Run a shell command and return its output."""
    try:
        result = subprocess.run(
            cmd,
            capture_output=True,
            text=True,
            check=True,
            cwd=cwd,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        log.error(
            "command_failed",
            cmd=cmd,
            stdout=e.stdout,
            stderr=e.stderr,
            returncode=e.returncode,
        )
        raise GitWorkflowError(f"Command failed: {' '.join(cmd)}") from e


async def get_current_branch(cwd: Optional[str] = None) -> str:
    """Get the current git branch."""
    return await run_command(["git", "rev-parse", "--abbrev-ref", "HEAD"], cwd=cwd)


async def is_git_repo(cwd: Optional[str] = None) -> bool:
    """Check if the current directory is a git repository."""
    try:
        await run_command(["git", "rev-parse", "--git-dir"], cwd=cwd)
        return True
    except GitWorkflowError:
        return False


async def create_feature_branch(
    branch_name: str,
    config: GitConfig,
    cwd: Optional[str] = None,
) -> str:
    """Create a new feature branch from the current branch."""
    full_branch_name = f"{config.branch_prefix}{branch_name}"
    current_branch = await get_current_branch(cwd=cwd)
    
    log.info(
        "creating_feature_branch",
        branch=full_branch_name,
        source_branch=current_branch,
    )
    
    # Create and checkout the feature branch
    await run_command(["git", "checkout", "-b", full_branch_name], cwd=cwd)
    
    return full_branch_name


async def commit_changes(
    files: Optional[list[str]] = None,
    config: GitConfig = None,
    cwd: Optional[str] = None,
) -> None:
    """Commit changes to the current branch."""
    if config is None:
        config = GitConfig()
    
    # Add files or all changes
    if files:
        for file in files:
            await run_command(["git", "add", file], cwd=cwd)
    else:
        await run_command(["git", "add", "."], cwd=cwd)
    
    # Check if there are changes to commit
    status = await run_command(["git", "status", "--porcelain"], cwd=cwd)
    if not status:
        log.warning("no_changes_to_commit", cwd=cwd)
        return
    
    # Commit the changes
    await run_command(
        ["git", "commit", "-m", config.commit_message],
        cwd=cwd,
    )
    
    log.info("changes_committed", message=config.commit_message)


async def push_branch(
    branch_name: str,
    config: GitConfig = None,
    force: bool = False,
    cwd: Optional[str] = None,
) -> None:
    """Push a branch to the remote repository."""
    if config is None:
        config = GitConfig()
    
    log.info(
        "pushing_branch",
        branch=branch_name,
        remote=config.remote,
        force=force,
    )
    
    cmd = ["git", "push"]
    if force:
        # Use --force-with-lease for safety
        cmd.append("--force-with-lease")
    
    cmd.extend([config.remote, branch_name])
    
    await run_command(cmd, cwd=cwd)


async def get_remote_url(cwd: Optional[str] = None) -> str:
    """Get the URL of the remote repository."""
    return await run_command(
        ["git", "remote", "get-url", "origin"], cwd=cwd
    )


async def create_merge_request(
    branch_name: str,
    config: GitConfig = None,
    cwd: Optional[str] = None,
) -> str:
    """Create a merge request (GitHub/GitLab)."""
    if config is None:
        config = GitConfig()
    
    log.info("creating_merge_request", branch=branch_name)
    
    # Get repository information
    remote_url = await get_remote_url(cwd=cwd)
    
    # Determine the platform (GitHub or GitLab)
    if "github.com" in remote_url:
        return await create_github_pr(branch_name, config, cwd=cwd)
    elif "gitlab.com" in remote_url:
        return await create_gitlab_mr(branch_name, config, cwd=cwd)
    else:
        raise GitWorkflowError(
            "Unsupported git hosting platform. Only GitHub and GitLab are supported."
        )


async def create_github_pr(
    branch_name: str,
    config: GitConfig = None,
    cwd: Optional[str] = None,
) -> str:
    """Create a GitHub Pull Request."""
    if config is None:
        config = GitConfig()
    
    # Get the base branch (typically 'main' or 'master')
    base_branch = "main"
    try:
        await run_command(["git", "rev-parse", "--verify", "main"], cwd=cwd)
    except GitWorkflowError:
        base_branch = "master"
    
    # Create PR using GitHub CLI if available
    try:
        # Check if gh CLI is installed
        await run_command(["gh", "--version"])
        
        pr_url = await run_command(
            [
                "gh",
                "pr",
                "create",
                "--base",
                base_branch,
                "--head",
                branch_name,
                "--title",
                config.mr_title,
                "--body",
                config.mr_description or config.commit_message,
            ],
            cwd=cwd,
        )
        log.info("created_github_pr", url=pr_url)
        return pr_url
    except GitWorkflowError:
        # Fallback: generate the URL manually
        remote_url = await get_remote_url(cwd=cwd)
        repo_path = remote_url.replace("git@github.com:", "").replace(".git", "")
        return f"https://github.com/{repo_path}/pull/new/{branch_name}"


async def create_gitlab_mr(
    branch_name: str,
    config: GitConfig = None,
    cwd: Optional[str] = None,
) -> str:
    """Create a GitLab Merge Request."""
    if config is None:
        config = GitConfig()
    
    # Get the base branch (typically 'main' or 'master')
    base_branch = "main"
    try:
        await run_command(["git", "rev-parse", "--verify", "main"], cwd=cwd)
    except GitWorkflowError:
        base_branch = "master"
    
    # Create MR using GitLab CLI if available
    try:
        # Check if glab CLI is installed
        await run_command(["glab", "--version"])
        
        mr_url = await run_command(
            [
                "glab",
                "mr",
                "create",
                "--base",
                base_branch,
                "--head",
                branch_name,
                "--title",
                config.mr_title,
                "--description",
                config.mr_description or config.commit_message,
            ],
            cwd=cwd,
        )
        log.info("created_gitlab_mr", url=mr_url)
        return mr_url
    except GitWorkflowError:
        # Fallback: generate the URL manually
        remote_url = await get_remote_url(cwd=cwd)
        repo_path = remote_url.replace("git@gitlab.com:", "").replace(".git", "")
        return f"https://gitlab.com/{repo_path}/-/merge_requests/new?merge_request[source_branch]={branch_name}&merge_request[target_branch]={base_branch}"


async def run_git_workflow(
    branch_name: str,
    files: Optional[list[str]] = None,
    config: GitConfig = None,
    force_push: bool = False,
    cwd: Optional[str] = None,
) -> dict:
    """
    Run the complete git workflow: create branch, commit changes, push, and create MR.
    
    Args:
        branch_name: Name for the feature branch (will be prefixed)
        files: List of files to commit (None for all changes)
        config: Git configuration
        force_push: Whether to use force push (for rebased branches)
        cwd: Working directory (None for current directory)
    
    Returns:
        Dictionary with branch name and MR URL
    """
    if config is None:
        config = GitConfig()
    
    # Verify we're in a git repository
    if not await is_git_repo(cwd=cwd):
        raise GitWorkflowError("Current directory is not a git repository")
    
    # Create feature branch
    feature_branch = await create_feature_branch(branch_name, config, cwd=cwd)
    
    # Commit changes
    await commit_changes(files=files, config=config, cwd=cwd)
    
    # Push branch (with force if requested)
    await push_branch(feature_branch, config=config, force=force_push, cwd=cwd)
    
    # Create merge request
    mr_url = await create_merge_request(feature_branch, config=config, cwd=cwd)
    
    log.info(
        "git_workflow_completed",
        branch=feature_branch,
        mr_url=mr_url,
        force_push=force_push,
    )
    
    return {
        "branch": feature_branch,
        "mr_url": mr_url,
    }


async def main() -> None:
    """Main entry point for the script."""
    import argparse
    
    parser = argparse.ArgumentParser(
        description="Automate git workflows: create branches, commit, push, and create merge requests."
    )
    parser.add_argument(
        "branch_name",
        help="Name for the feature branch (will be prefixed with 'feature/')",
    )
    parser.add_argument(
        "--files",
        nargs="+",
        help="Specific files to commit (default: all changes)",
    )
    parser.add_argument(
        "--force-push",
        action="store_true",
        help="Force push the branch (use with caution, typically for rebased branches)",
    )
    parser.add_argument(
        "--commit-message",
        default="Auto-generated commit",
        help="Commit message to use",
    )
    parser.add_argument(
        "--mr-title",
        default="Auto-generated merge request",
        help="Title for the merge request",
    )
    parser.add_argument(
        "--mr-description",
        default="",
        help="Description for the merge request",
    )
    parser.add_argument(
        "--remote",
        default="origin",
        help="Git remote to push to",
    )
    parser.add_argument(
        "--branch-prefix",
        default="feature/",
        help="Prefix for feature branches",
    )
    parser.add_argument(
        "--directory",
        "-C",
        default=None,
        help="Run as if git was started in this directory",
    )
    
    args = parser.parse_args()
    
    # Configure logging
    structlog.configure(
        processors=[
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.add_log_level,
            structlog.processors.JSONRenderer(),
        ],
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=True,
    )
    
    # Create config
    config = GitConfig(
        remote=args.remote,
        commit_message=args.commit_message,
        mr_title=args.mr_title,
        mr_description=args.mr_description,
        branch_prefix=args.branch_prefix,
    )
    
    try:
        result = await run_git_workflow(
            branch_name=args.branch_name,
            files=args.files,
            config=config,
            force_push=args.force_push,
            cwd=args.directory,
        )
        print(f"✅ Workflow completed successfully!")
        print(f"Branch: {result['branch']}")
        print(f"Merge Request: {result['mr_url']}")
    except GitWorkflowError as e:
        log.error("git_workflow_failed", error=str(e))
        print(f"❌ Git workflow failed: {e}", file=sys.stderr)
        sys.exit(1)
    except Exception as e:
        log.error("unexpected_error", error=str(e))
        print(f"❌ Unexpected error: {e}", file=sys.stderr)
        sys.exit(1)


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
