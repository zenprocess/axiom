```python
#!/usr/bin/env python3
"""
Automated Git Workflow Script

Creates feature branches, commits staged changes, pushes branches safely,
and creates merge requests via GitHub or GitLab APIs.

⚠️ NO FORCED PUSH — to prevent history rewriting.
⚠️ Credentials must be provided via environment variables:
    - GITHUB_TOKEN or GITLAB_TOKEN
    - GITHUB_API_URL or GITLAB_API_URL (optional, defaults to public APIs)
"""

import os
import subprocess
import sys
from pathlib import Path
from typing import Optional, Literal

import httpx
from pydantic import BaseModel, Field
from structlog import get_logger

# Configure structlog
import structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.INFO),
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
)

logger = get_logger("git_workflow")


class MRDetails(BaseModel):
    title: str = Field(..., description="Title of the merge request")
    description: str = Field("", description="Description of the MR")
    target_branch: str = Field("main", description="Target branch (e.g., 'main' or 'develop')")
    source_branch: str = Field(..., description="Source branch to merge from")


def run_git(*args: str, check: bool = True) -> subprocess.CompletedProcess[str]:
    """Run a git command and return the result."""
    try:
        result = subprocess.run(
            ["git"] + list(args),
            capture_output=True,
            text=True,
            check=check,
        )
        logger.info("git_command", cmd=args, stdout=result.stdout, stderr=result.stderr)
        return result
    except subprocess.CalledProcessError as e:
        logger.error("git_command_failed", cmd=args, error=e.stderr)
        raise


def get_current_branch() -> str:
    """Get the current branch name."""
    result = run_git("rev-parse", "--abbrev-ref", "HEAD")
    return result.stdout.strip()


def create_feature_branch(branch_name: str) -> None:
    """Create and checkout a new feature branch."""
    current = get_current_branch()
    if current != "main":  # or "master" — adjust as needed
        logger.warning("not_on_main", current_branch=current)
    run_git("checkout", "-b", branch_name)
    logger.info("feature_branch_created", branch=branch_name)


def commit_changes(message: str) -> None:
    """Stage all changes and commit with message."""
    run_git("add", ".")
    run_git("commit", "-m", message)
    logger.info("changes_committed", message=message)


def push_branch(branch_name: str, remote: str = "origin") -> None:
    """
    Push branch to remote **without force**.

    Raises if remote branch exists and is not fast-forward (to avoid overwriting).
    """
    # Check if remote branch exists
    result = run_git("ls-remote", "--heads", remote, branch_name, check=False)
    if result.stdout.strip():
        logger.warning("remote_branch_exists", branch=branch_name)

    # Push normally (no --force)
    try:
        run_git("push", remote, branch_name)
        logger.info("branch_pushed", branch=branch_name, remote=remote)
    except subprocess.CalledProcessError as e:
        if "non-fast-forward" in e.stderr.lower():
            logger.error(
                "push_failed_non_fast_forward",
                branch=branch_name,
                hint="Branch was rebased or rewritten. Consider deleting remote branch first (if intentional), or avoid force-push.",
            )
        raise


def create_merge_request(
    details: MRDetails,
    provider: Literal["github", "gitlab"] = "github",
) -> dict:
    """
    Create a merge request via GitHub or GitLab REST API.

    Requires environment variables:
    - GITHUB_TOKEN (for GitHub)
    - GITLAB_TOKEN (for GitLab)
    """
    token_env = f"{provider.upper()}_TOKEN"
    token = os.getenv(token_env)
    if not token:
        raise ValueError(f"Missing required env var: {token_env}")

    api_url_env = f"{provider.upper()}_API_URL"
    base_url = os.getenv(api_url_env, f"https://{provider}.com/api/v4" if provider == "gitlab" else "https://api.github.com")

    # Get repo info
    repo_path = run_git("config", "--get", "remote.origin.url").stdout.strip()
    # Normalize: strip .git, convert ssh to https
    repo_path = repo_path.replace(".git", "").replace("git@", "https://").replace(":", "/")

    if provider == "github":
        # GitHub uses owner/repo format
        if repo_path.startswith("https://"):
            repo_path = repo_path.replace("https://github.com/", "")
        headers = {"Authorization": f"token {token}", "Accept": "application/vnd.github.v3+json"}
        url = f"https://api.github.com/repos/{repo_path}/pulls"
        payload = {
            "title": details.title,
            "head": details.source_branch,
            "base": details.target_branch,
            "body": details.description,
        }
    else:  # gitlab
        # GitLab uses project ID or namespace/project
        if repo_path.startswith("https://"):
            repo_path = repo_path.replace("https://gitlab.com/", "")
        headers = {"PRIVATE-TOKEN": token}
        url = f"{base_url}/projects/{repo_path.replace('/', '%2F')}/merge_requests"
        payload = {
            "title": details.title,
            "source_branch": details.source_branch,
            "target_branch": details.target_branch,
            "description": details.description,
        }

    with httpx.Client(headers=headers, timeout=30.0) as client:
        try:
            response = client.post(url, json=payload)
            response.raise_for_status()
            mr = response.json()
            logger.info("merge_request_created", mr_id=mr.get("iid") or mr.get("id"), url=mr.get("web_url"))
            return mr
        except httpx.HTTPStatusError as e:
            logger.error("mr_creation_failed", status=e.response.status_code, body=e.response.text)
            raise


def main() -> None:
    """Main workflow."""
    branch_name = "feature/automated-fix"
    commit_msg = "chore: automated commit for testing"

    # Ensure we're in a git repo
    if not Path(".git").exists():
        logger.error("not_a_git_repo")
        sys.exit(1)

    try:
        create_feature_branch(branch_name)
        commit_changes(commit_msg)
        push_branch(branch_name)

        # Create MR
        mr_details = MRDetails(
            title="Fix: Automated feature branch",
            description="This MR was auto-created by the git workflow script.",
            source_branch=branch_name,
        )
        mr = create_merge_request(mr_details, provider="github")  # or "gitlab"
        logger.info("success", mr_url=mr.get("web_url"))

    except Exception as e:
        logger.error("workflow_failed", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    main()
```
