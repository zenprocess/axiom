```python
#!/usr/bin/env python3
"""
Git Workflow Automation Script

This script automates common git workflows including:
- Creating feature branches
- Committing changes
- Pushing branches to remote
- Creating merge requests via GitHub/GitLab API

Note: Force-push is intentionally NOT implemented per security policy.
"""

import asyncio
import os
import subprocess
import sys
from pathlib import Path
from typing import Optional, List

import httpx
from pydantic import BaseModel, Field
import structlog

# Configure structlog for logging
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
)

logger = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations"""
    remote: str = Field(default="origin")
    base_branch: str = Field(default="main")
    username: Optional[str] = Field(default=None)
    token: Optional[str] = Field(default=None)


class MergeRequestConfig(BaseModel):
    """Configuration for merge request creation"""
    title: str
    description: str = Field(default="")
    target_branch: str = Field(default="main")
    source_branch: str
    platform: str = Field(default="github")  # "github" or "gitlab"


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors"""
    pass


class GitWorkflow:
    """Handles automated git workflows"""

    def __init__(self, repo_path: str, config: Optional[GitConfig] = None):
        """
        Initialize the GitWorkflow manager.

        Args:
            repo_path: Path to the git repository
            config: Git configuration (optional)
        """
        self.repo_path = Path(repo_path)
        self.config = config or GitConfig()
        self._ensure_git_repo()

    def _ensure_git_repo(self) -> None:
        """Verify that the path is a valid git repository."""
        git_dir = self.repo_path / ".git"
        if not git_dir.exists():
            raise GitWorkflowError(f"{self.repo_path} is not a valid git repository")

    def _run_git_command(self, args: List[str], cwd: Optional[Path] = None) -> str:
        """
        Run a git command and return its output.

        Args:
            args: List of arguments for the git command
            cwd: Working directory (defaults to repo_path)

        Returns:
            Command output as a string

        Raises:
            GitWorkflowError: If the command fails
        """
        try:
            result = subprocess.run(
                ["git"] + args,
                cwd=cwd or self.repo_path,
                capture_output=True,
                text=True,
                check=True,
            )
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            logger.error("git_command_failed", error=e.stderr, args=args)
            raise GitWorkflowError(f"Git command failed: {e.stderr}") from e

    async def create_feature_branch(self, feature_name: str) -> str:
        """
        Create a new feature branch from the base branch.

        Args:
            feature_name: Name for the new feature branch

        Returns:
            Name of the created branch
        """
        logger.info("creating_feature_branch", feature_name=feature_name)

        # Ensure we're on the base branch and up-to-date
        self._run_git_command(["checkout", self.config.base_branch])
        self._run_git_command(["pull", self.config.remote, self.config.base_branch])

        # Create and checkout the feature branch
        branch_name = f"feature/{feature_name}"
        self._run_git_command(["checkout", "-b", branch_name])
        logger.info("feature_branch_created", branch_name=branch_name)

        return branch_name

    async def commit_changes(self, files: Optional[List[str]] = None, message: str = "Automated commit") -> None:
        """
        Commit changes to the current branch.

        Args:
            files: Specific files to commit (default: all staged changes)
            message: Commit message
        """
        logger.info("committing_changes", message=message)

        # Stage files or all changes
        if files:
            self._run_git_command(["add"] + files)
        else:
            self._run_git_command(["add", "."])

        # Commit the changes
        self._run_git_command(["commit", "-m", message])
        logger.info("changes_committed")

    async def push_branch(self, branch_name: str) -> None:
        """
        Push a branch to the remote repository.

        Args:
            branch_name: Name of the branch to push
        """
        logger.info("pushing_branch", branch_name=branch_name)

        try:
            self._run_git_command(["push", self.config.remote, branch_name])
            logger.info("branch_pushed", branch_name=branch_name)
        except GitWorkflowError as e:
            logger.error("push_failed", branch_name=branch_name, error=str(e))
            raise

    async def create_merge_request(self, config: MergeRequestConfig) -> dict:
        """
        Create a merge request via GitHub or GitLab API.

        Args:
            config: Merge request configuration

        Returns:
            Response data from the API
        """
        logger.info("creating_merge_request", config=config)

        if config.platform == "github":
            return await self._create_github_pr(config)
        elif config.platform == "gitlab":
            return await self._create_gitlab_mr(config)
        else:
            raise ValueError(f"Unsupported platform: {config.platform}")

    async def _create_github_pr(self, config: MergeRequestConfig) -> dict:
        """
        Create a GitHub pull request.

        Args:
            config: Merge request configuration

        Returns:
            Response data from GitHub API
        """
        # Get repository info
        repo_url = self._run_git_command(["remote", "get-url", self.config.remote])
        # Extract owner/repo from URL (e.g., "https://github.com/owner/repo.git")
        repo_path = repo_url.replace(".git", "").split("github.com/")[-1]
        owner, repo = repo_path.split("/")

        # Create the PR using GitHub API
        url = f"https://api.github.com/repos/{owner}/{repo}/pulls"
        headers = {
            "Authorization": f"token {self.config.token}",
            "Accept": "application/vnd.github.v3+json",
        }
        data = {
            "title": config.title,
            "head": config.source_branch,
            "base": config.target_branch,
            "body": config.description,
        }

        try:
            async with httpx.AsyncClient() as client:
                response = await client.post(url, json=data, headers=headers)
                response.raise_for_status()
                logger.info("pull_request_created", pr_number=response.json()["number"])
                return response.json()
        except httpx.HTTPError as e:
            logger.error("github_pr_creation_failed", error=str(e))
            raise GitWorkflowError(f"Failed to create GitHub PR: {e}") from e

    async def _create_gitlab_mr(self, config: MergeRequestConfig) -> dict:
        """
        Create a GitLab merge request.

        Args:
            config: Merge request configuration

        Returns:
            Response data from GitLab API
        """
        # Get repository info
        repo_url = self._run_git_command(["remote", "get-url", self.config.remote])
        # Extract project path from URL (e.g., "https://gitlab.com/owner/repo.git")
        repo_path = repo_url.replace(".git", "").split("gitlab.com/")[-1]
        project_id = repo_path.replace("/", "%2F")

        # Create the MR using GitLab API
        url = f"https://gitlab.com/api/v4/projects/{project_id}/merge_requests"
        headers = {
            "PRIVATE-TOKEN": self.config.token,
        }
        data = {
            "title": config.title,
            "source_branch": config.source_branch,
            "target_branch": config.target_branch,
            "description": config.description,
        }

        try:
            async with httpx.AsyncClient() as client:
                response = await client.post(url, json=data, headers=headers)
                response.raise_for_status()
                logger.info("merge_request_created", mr_iid=response.json()["iid"])
                return response.json()
        except httpx.HTTPError as e:
            logger.error("gitlab_mr_creation_failed", error=str(e))
            raise GitWorkflowError(f"Failed to create GitLab MR: {e}") from e


async def main():
    """Main entry point for the script."""
    # Example usage
    repo_path = os.getcwd()
    
    # Create workflow instance
    workflow = GitWorkflow(repo_path)
    
    # Create a feature branch
    feature_name = "add-new-feature"
    branch_name = await workflow.create_feature_branch(feature_name)
    
    # Commit changes (example)
    # await workflow.commit_changes(message=f"Add {feature_name}")
    
    # Push the branch
    await workflow.push_branch(branch_name)
    
    # Create merge request
    mr_config = MergeRequestConfig(
        title=f"Add {feature_name}",
        description=f"Implementation of {feature_name}",
        source_branch=branch_name,
        target_branch="main",
    )
    
    # Note: For this to work, you need to set GITHUB_TOKEN or GITLAB_TOKEN environment variable
    token = os.getenv("GITHUB_TOKEN") or os.getenv("GITLAB_TOKEN")
    if token:
        workflow.config.token = token
    
    try:
        mr = await workflow.create_merge_request(mr_config)
        logger.info("merge_request_created_successfully", mr_url=mr.get("html_url", mr.get("web_url")))
    except GitWorkflowError as e:
        logger.error("merge_request_creation_failed", error=str(e))


if __name__ == "__main__":
    asyncio.run(main())
```
