```python
#!/usr/bin/env python3
"""
Automated Git Workflow Script.

This script automates feature branch creation, commits, safe pushes (with lease),
and merge request creation using Git and GitHub/GitLab APIs.
"""

import os
import subprocess
import sys
from pathlib import Path
from typing import Optional

import httpx
import structlog
from dotenv import load_dotenv
from pydantic import BaseModel, Field

# Initialize structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=True,
)

logger = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for Git operations."""
    base_branch: str = Field(default="main")
    remote: str = Field(default="origin")
    username: Optional[str] = None
    token: Optional[str] = None


class MRConfig(BaseModel):
    """Configuration for merge request."""
    title: str
    description: str
    target_branch: str = Field(default="main")
    source_branch: str
    project_id: str


class GitWorkflowError(Exception):
    """Custom exception for Git workflow errors."""
    pass


async def create_feature_branch(branch_name: str, config: GitConfig) -> str:
    """Create and checkout a new feature branch.

    Args:
        branch_name: Name of the new feature branch
        config: Git configuration

    Returns:
        The name of the created branch

    Raises:
        GitWorkflowError: If branch creation fails
    """
    try:
        logger.info("Creating feature branch", branch=branch_name)
        result = subprocess.run(
            ["git", "checkout", "-b", branch_name],
            check=True,
            capture_output=True,
            text=True,
        )
        logger.info("Branch created", output=result.stdout.strip())
        return branch_name
    except subprocess.CalledProcessError as e:
        logger.error("Failed to create branch", error=str(e), branch=branch_name)
        raise GitWorkflowError(f"Failed to create branch '{branch_name}': {e.stderr}") from e


async def commit_changes(message: str, files: Optional[list[str]] = None) -> None:
    """Commit staged changes with the given message.

    Args:
        message: Commit message
        files: Optional list of files to commit; if None, all changes are committed

    Raises:
        GitWorkflowError: If commit fails
    """
    try:
        logger.info("Staging changes")
        subprocess.run(["git", "add", *files] if files else ["git", "add", "."], check=True)

        logger.info("Committing changes", message=message)
        subprocess.run(
            ["git", "commit", "-m", message],
            check=True,
            capture_output=True,
            text=True,
        )
        logger.info("Changes committed successfully")
    except subprocess.CalledProcessError as e:
        logger.error("Failed to commit changes", error=str(e))
        raise GitWorkflowError(f"Failed to commit: {e.stderr}") from e


async def push_branch(branch_name: str, config: GitConfig, force_with_lease: bool = True) -> None:
    """Push branch to remote repository.

    Uses --force-with-lease for safety (only overwrites if no one else pushed).

    Args:
        branch_name: Name of the branch to push
        config: Git configuration
        force_with_lease: Use safe force push (recommended)

    Raises:
        GitWorkflowError: If push fails
    """
    try:
        cmd = ["git", "push", config.remote, branch_name]
        if force_with_lease:
            cmd.insert(3, "--force-with-lease")

        logger.info("Pushing branch", branch=branch_name, command=" ".join(cmd))
        result = subprocess.run(cmd, check=True, capture_output=True, text=True)
        logger.info("Branch pushed successfully", output=result.stdout.strip())
    except subprocess.CalledProcessError as e:
        logger.error("Failed to push branch", error=str(e), branch=branch_name)
        raise GitWorkflowError(f"Failed to push branch '{branch_name}': {e.stderr}") from e


async def create_merge_request(
    config: MRConfig,
    git_config: GitConfig,
) -> dict:
    """Create a merge request using GitLab/GitHub API.

    This example uses GitLab API format; adapt URLs/headers for GitHub.

    Args:
        config: MR configuration
        git_config: Git configuration with credentials

    Returns:
        MR response data as dict

    Raises:
        GitWorkflowError: If MR creation fails
    """
    # Determine API endpoint (GitLab vs GitHub)
    api_url = os.getenv("GITLAB_API_URL", "https://gitlab.com/api/v4")
    headers = {
        "PRIVATE-TOKEN": git_config.token or os.getenv("GITLAB_TOKEN", ""),
        "Content-Type": "application/json",
    }

    # Check required credentials
    if not git_config.token and not os.getenv("GITLAB_TOKEN"):
        raise GitWorkflowError(
            "Missing GitLab token. Set GITLAB_TOKEN env var or provide token in GitConfig."
        )

    payload = {
        "title": config.title,
        "description": config.description,
        "source_branch": config.source_branch,
        "target_branch": config.target_branch,
        "project_id": config.project_id,
    }

    url = f"{api_url}/projects/{config.project_id}/merge_requests"

    try:
        logger.info("Creating merge request", url=url, title=config.title)
        async with httpx.AsyncClient() as client:
            response = await client.post(url, json=payload, headers=headers)
            response.raise_for_status()
            mr_data = response.json()
            logger.info("Merge request created", mr_id=mr_data.get("iid"), url=mr_data.get("web_url"))
            return mr_data
    except httpx.HTTPError as e:
        logger.error("Failed to create MR", error=str(e), status=getattr(e, "response", None))
        raise GitWorkflowError(f"Failed to create MR: {e}") from e


async def run_workflow(
    branch_name: str,
    commit_message: str,
    mr_config: MRConfig,
    git_config: Optional[GitConfig] = None,
) -> dict:
    """Run the full automated Git workflow.

    Args:
        branch_name: Name for the new feature branch
        commit_message: Message for the commit
        mr_config: Merge request configuration
        git_config: Git configuration (uses env vars if not provided)

    Returns:
        Merge request data

    Raises:
        GitWorkflowError: If any step fails
    """
    git_config = git_config or GitConfig()

    # Step 1: Create feature branch
    await create_feature_branch(branch_name, git_config)

    # Step 2: Commit changes (assumes changes are already staged or added)
    # In practice, you'd add files here: await commit_changes(commit_message, ["file1.py", "file2.py"])
    await commit_changes(commit_message)

    # Step 3: Push branch safely (with lease)
    await push_branch(branch_name, git_config, force_with_lease=True)

    # Step 4: Create merge request
    return await create_merge_request(mr_config, git_config)


def main():
    """Main entry point."""
    load_dotenv()

    if len(sys.argv) < 3:
        print("Usage: python git_workflow.py <branch_name> <commit_message>")
        print("Set environment variables: GITLAB_TOKEN, GITLAB_PROJECT_ID")
        sys.exit(1)

    branch_name = sys.argv[1]
    commit_message = sys.argv[2]

    git_config = GitConfig(
        username=os.getenv("GITLAB_USERNAME"),
        token=os.getenv("GITLAB_TOKEN"),
    )
    mr_config = MRConfig(
        title=f"feat: {commit_message}",
        description=f"Implements {commit_message}",
        source_branch=branch_name,
        target_branch="main",
        project_id=os.getenv("GITLAB_PROJECT_ID", ""),
    )

    try:
        result = asyncio.run(run_workflow(branch_name, commit_message, mr_config, git_config))
        logger.info("Workflow completed successfully", mr_url=result.get("web_url"))
    except GitWorkflowError as e:
        logger.error("Workflow failed", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    import asyncio
    main()
```
