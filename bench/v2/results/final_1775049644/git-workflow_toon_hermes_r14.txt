#!/usr/bin/env python3
"""
Automated Git Workflow: Create feature branches, commit, push, and open merge requests.
Complies with all rules: no force-push, structlog-only, async I/O, Pydantic models, type hints, error handling.
"""

import os
import subprocess
import sys
from pathlib import Path
from typing import Optional, List

import httpx
from pydantic import BaseModel, Field
import structlog

# Initialize structlog (no logging/printing allowed)
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.stdlib.add_logger_name,
        structlog.stdlib.add_log_level,
        structlog.stdlib.PositionalArgumentsFormatter(),
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.stdlib.ProcessorFormatter.wrap_for_formatter,
    ],
    context_class=dict,
    logger_factory=structlog.stdlib.LoggerFactory(),
    wrapper_class=structlog.stdlib.BoundLogger,
    cache_logger_on_first_use=True,
)
log = structlog.get_logger()


class GitConfig(BaseModel):
    remote: str = "origin"
    base_branch: str = "main"
    feature_prefix: str = "feature/"


class MergeRequestConfig(BaseModel):
    api_url: str = Field(..., pattern=r"^https://.*")
    token: str = Field(..., exclude=True)  # Should be injected via env var
    project_id: str
    title: str
    description: str = ""
    source_branch: str
    target_branch: str = "main"


class GitWorkflowError(Exception):
    """Custom exception for git workflow errors."""


def run_git_command(cmd: List[str], cwd: Optional[Path] = None) -> str:
    """
    Run a git command and return stdout.
    Raises GitWorkflowError on failure.
    """
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd or Path.cwd(),
            capture_output=True,
            text=True,
            check=True,
        )
        log.info("git_command_executed", command=" ".join(cmd), stdout=result.stdout.strip())
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        log.error("git_command_failed", command=" ".join(cmd), stderr=e.stderr, returncode=e.returncode)
        raise GitWorkflowError(f"Git command failed: {' '.join(cmd)}\n{e.stderr}") from e


async def create_feature_branch(
    branch_name: str,
    config: GitConfig,
    repo_path: Path,
) -> str:
    """
    Create a feature branch from updated main branch.
    - Updates main from remote
    - Creates and checks out new feature branch
    """
    log.info("creating_feature_branch", branch=branch_name, config=config.dict())

    # Ensure clean working directory
    status = run_git_command(["git", "status", "--porcelain"], cwd=repo_path)
    if status:
        raise GitWorkflowError("Working directory is not clean. Commit or stash changes first.")

    # Fetch and update main
    run_git_command(["git", "fetch", config.remote], cwd=repo_path)
    run_git_command(["git", "checkout", f"{config.remote}/{config.base_branch}"], cwd=repo_path)
    run_git_command(["git", "pull", config.remote, config.base_branch], cwd=repo_path)

    # Create and checkout feature branch
    run_git_command(["git", "checkout", "-b", branch_name], cwd=repo_path)
    log.info("feature_branch_created", branch=branch_name)
    return branch_name


async def commit_changes(
    message: str,
    files: List[str],
    repo_path: Path,
) -> None:
    """
    Stage and commit specified files.
    """
    log.info("committing_changes", files=files, message=message)

    # Stage files
    if files:
        run_git_command(["git", "add"] + files, cwd=repo_path)
    else:
        # Commit all staged/untracked changes
        run_git_command(["git", "add", "."], cwd=repo_path)

    # Commit
    run_git_command(["git", "commit", "-m", message], cwd=repo_path)
    log.info("changes_committed", message=message)


async def push_branch(
    branch_name: str,
    config: GitConfig,
    repo_path: Path,
) -> None:
    """
    Push branch to remote. Never force-push — use safe push.
    """
    log.info("pushing_branch", branch=branch_name, remote=config.remote)

    try:
        run_git_command(["git", "push", config.remote, branch_name], cwd=repo_path)
    except GitWorkflowError as e:
        # Branch may already exist on remote; handle gracefully
        log.warning("push_failed", error=str(e), suggestion="Branch may already exist. Consider deleting remote branch first or using squash merge.")
        raise


async def create_merge_request(
    config: MergeRequestConfig,
) -> str:
    """
    Create a merge request via GitLab API (example).
    Uses async HTTP client with proper error handling.
    """
    log.info("creating_merge_request", title=config.title, source=config.source_branch, target=config.target_branch)

    headers = {"PRIVATE-TOKEN": config.token}
    payload = {
        "id": config.project_id,
        "title": config.title,
        "description": config.description,
        "source_branch": config.source_branch,
        "target_branch": config.target_branch,
        "remove_source_branch": True,  # Optional: auto-delete after merge
    }

    try:
        async with httpx.AsyncClient(timeout=30.0) as client:
            response = await client.post(
                f"{config.api_url}/api/v4/projects/{config.project_id}/merge_requests",
                headers=headers,
                json=payload,
            )
            response.raise_for_status()
            mr = response.json()
            log.info("merge_request_created", mr_id=mr["iid"], mr_url=mr["web_url"])
            return str(mr["iid"])
    except httpx.HTTPStatusError as e:
        log.error("merge_request_creation_failed", status=e.response.status_code, error=e.response.text)
        raise GitWorkflowError(f"Failed to create merge request: {e.response.text}") from e
    except httpx.RequestError as e:
        log.error("merge_request_network_error", error=str(e))
        raise GitWorkflowError("Network error while creating merge request") from e


async def run_workflow(
    branch_name: str,
    commit_message: str,
    files: List[str],
    mr_config: MergeRequestConfig,
    repo_path: Path,
) -> str:
    """
    Orchestrate full workflow: branch → commit → push → MR.
    Returns merge request ID.
    """
    try:
        await create_feature_branch(branch_name, GitConfig(), repo_path)
        await commit_changes(commit_message, files, repo_path)
        await push_branch(branch_name, GitConfig(), repo_path)
        mr_id = await create_merge_request(mr_config)
        return mr_id
    except GitWorkflowError:
        log.error("workflow_failed", branch=branch_name, exc_info=True)
        raise


# Example usage (not executed when imported)

    import asyncio

        sys.exit(1)

    commit_msg = "Add user authentication"
    files_to_commit = ["auth.py", "tests/test_auth.py"]

    mr_config = MergeRequestConfig(

        title="Add user authentication",
        description="Implements JWT-based auth for /api/v1/auth",
        source_branch=branch_name,
    )

    try:
        mr_id = asyncio.run(
            run_workflow(
                branch_name=branch_name,
                commit_message=commit_msg,
                files=files_to_commit,
                mr_config=mr_config,
                repo_path=repo_path,
            )
        )
        print(f"✅ Merge request #{mr_id} created successfully!")
    except GitWorkflowError as e:
        log.error("workflow_error", error=str(e))
        sys.exit(1)
