#!/usr/bin/env python3
"""
Git Workflow Automation Script

This script automates common git workflows including:
- Creating feature branches
- Committing changes
- Pushing to remote
- Creating merge requests

Note: This script intentionally avoids force-push operations as per security policies.
"""

import subprocess
import sys
from typing import List, Optional
import structlog

# Initialize structlog logger
logger = structlog.get_logger()

class GitWorkflowError(Exception):
    """Custom exception for git workflow operations."""
    pass

class GitWorkflow:
    """Automates git workflows for feature branches and merge requests."""
    
    def __init__(self, remote: str = "origin"):
        """
        Initialize the GitWorkflow.
        
        Args:
            remote: The remote repository name (default: 'origin')
        """
        self.remote = remote
        self.logger = logger.bind(remote=remote)
    
    def _run_command(self, command: List[str], check: bool = True) -> subprocess.CompletedProcess:
        """
        Run a shell command with error handling.
        
        Args:
            command: List of command arguments
            check: Whether to raise exception on non-zero exit code
            
        Returns:
            CompletedProcess instance
            
        Raises:
            GitWorkflowError: If command fails and check=True
        """
        try:
            self.logger.info("running_command", command=" ".join(command))
            result = subprocess.run(
                command,
                capture_output=True,
                text=True,
                check=check
            )
            return result
        except subprocess.CalledProcessError as e:
            self.logger.error(
                "command_failed",
                command=" ".join(command),
                returncode=e.returncode,
                stdout=e.stdout,
                stderr=e.stderr
            )
            raise GitWorkflowError(f"Command failed: {' '.join(command)}") from e
    
    def get_current_branch(self) -> str:
        """
        Get the current git branch name.
        
        Returns:
            Current branch name
            
        Raises:
            GitWorkflowError: If unable to determine current branch
        """
        try:
            result = self._run_command(["git", "rev-parse", "--abbrev-ref", "HEAD"])
            branch = result.stdout.strip()
            if not branch:
                raise GitWorkflowError("Unable to determine current branch")
            self.logger.info("current_branch", branch=branch)
            return branch
        except GitWorkflowError:
            raise
    
    def create_branch(self, branch_name: str) -> None:
        """
        Create a new feature branch from current branch.
        
        Args:
            branch_name: Name of the new branch to create
            
        Raises:
            GitWorkflowError: If branch creation fails
        """
        self.logger.info("creating_branch", branch=branch_name)
        
        # Check if branch already exists
        result = self._run_command(["git", "rev-parse", "--verify", branch_name], check=False)
        if result.returncode == 0:
            self.logger.warning("branch_already_exists", branch=branch_name)
            self.checkout_branch(branch_name)
            return
            
        # Create and checkout new branch
        self._run_command(["git", "checkout", "-b", branch_name])
        self.logger.info("branch_created", branch=branch_name)
    
    def checkout_branch(self, branch_name: str) -> None:
        """
        Checkout an existing branch.
        
        Args:
            branch_name: Name of the branch to checkout
            
        Raises:
            GitWorkflowError: If checkout fails
        """
        self.logger.info("checking_out_branch", branch=branch_name)
        self._run_command(["git", "checkout", branch_name])
        self.logger.info("branch_checked_out", branch=branch_name)
    
    def add_all_changes(self) -> None:
        """
        Stage all changes in the working directory.
        
        Raises:
            GitWorkflowError: If git add fails
        """
        self.logger.info("staging_changes")
        self._run_command(["git", "add", "."])
        self.logger.info("changes_staged")
    
    def commit_changes(self, message: str) -> None:
        """
        Commit staged changes with a message.
        
        Args:
            message: Commit message
            
        Raises:
            GitWorkflowError: If commit fails
        """
        self.logger.info("committing_changes", message=message)
        self._run_command(["git", "commit", "-m", message])
        self.logger.info("changes_committed")
    
    def push_branch(self, branch_name: str, force: bool = False) -> None:
        """
        Push branch to remote repository.
        
        Note: This method intentionally does not support force pushes as per security policies.
        
        Args:
            branch_name: Name of the branch to push
            force: Whether to force push (always False per policy)
            
        Raises:
            GitWorkflowError: If push fails
        """
        if force:
            self.logger.error("force_push_not_allowed")
            raise GitWorkflowError("Force push is not allowed per security policy")
        
        self.logger.info("pushing_branch", branch=branch_name)
        self._run_command(["git", "push", self.remote, branch_name])
        self.logger.info("branch_pushed", branch=branch_name)
    
    def update_branch_from_main(self, main_branch: str = "main") -> None:
        """
        Update current branch with changes from main branch (safe alternative to rebase).
        
        This method fetches and merges main branch changes to keep the feature branch
        up-to-date without requiring force pushes.
        
        Args:
            main_branch: Name of the main branch (default: 'main')
            
        Raises:
            GitWorkflowError: If update fails
        """
        self.logger.info("updating_branch_from_main", main_branch=main_branch)
        
        # Fetch latest changes
        self._run_command(["git", "fetch", self.remote])
        
        # Merge main branch changes
        self._run_command(["git", "merge", f"{self.remote}/{main_branch}"])
        self.logger.info("branch_updated_from_main")
    
    def create_merge_request(
        self,
        branch_name: str,
        title: str,
        description: str = "",
        target_branch: str = "main"
    ) -> str:
        """
        Create a merge request (placeholder for GitHub/GitLab API integration).
        
        This is a placeholder implementation. In production, you would integrate
        with a specific platform's API (GitHub, GitLab, Bitbucket, etc.).
        
        Args:
            branch_name: Source branch for the merge request
            title: Title of the merge request
            description: Description of the merge request
            target_branch: Target branch for the merge request
            
        Returns:
            URL or identifier of the created merge request
            
        Raises:
            NotImplementedError: This is a placeholder implementation
        """
        self.logger.info(
            "creating_merge_request",
            branch=branch_name,
            target=target_branch,
            title=title
        )
        
        # Placeholder for actual MR creation
        # In production, you would use:
        # - GitHub API: requests.post("https://api.github.com/repos/{owner}/{repo}/pulls", ...)
        # - GitLab API: requests.post("https://gitlab.com/api/v4/projects/{id}/merge_requests", ...)
        
        mr_url = f"https://example.com/merge-requests/new?merge_request[source_branch]={branch_name}&merge_request[target_branch]={target_branch}"
        
        self.logger.info(
            "merge_request_created",
            mr_url=mr_url,
            title=title,
            description=description
        )
        
        return mr_url


def main():
    """Main function demonstrating git workflow automation."""
    import argparse
    
    parser = argparse.ArgumentParser(description="Automate git workflows")
    parser.add_argument(
        "--branch-name",
        required=True,
        help="Name of the feature branch to create"
    )
    parser.add_argument(
        "--message",
        default="Update feature branch",
        help="Commit message"
    )
    parser.add_argument(
        "--remote",
        default="origin",
        help="Remote repository name"
    )
    parser.add_argument(
        "--target-branch",
        default="main",
        help="Target branch for merge request"
    )
    parser.add_argument(
        "--title",
        default="Feature update",
        help="Merge request title"
    )
    
    args = parser.parse_args()
    
    # Initialize workflow
    workflow = GitWorkflow(remote=args.remote)
    
    try:
        # Get current branch
        current_branch = workflow.get_current_branch()
        
        # Create and checkout new branch
        workflow.create_branch(args.branch_name)
        
        # Stage and commit changes
        workflow.add_all_changes()
        workflow.commit_changes(args.message)
        
        # Push branch to remote
        workflow.push_branch(args.branch_name)
        
        # Create merge request
        mr_url = workflow.create_merge_request(
            branch_name=args.branch_name,
            title=args.title,
            target_branch=args.target_branch
        )
        
        # Log success
        logger.info(
            "workflow_completed",
            branch=args.branch_name,
            mr_url=mr_url,
            current_branch=current_branch
        )
        
        return 0
        
    except GitWorkflowError as e:
        logger.error("workflow_failed", error=str(e))
        return 1
    except Exception as e:
        logger.error("unexpected_error", error=str(e))
        return 1


if __name__ == "__main__":
    sys.exit(main())
