```python
import os
import subprocess
import sys
from pathlib import Path
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel, Field

# Configure structlog (rule: structlog-only)
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.dev.ConsoleRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=False,
)
logger = structlog.get_logger()

# Rule: pydantic-models + type-hints
class GitConfig(BaseModel):
    repo_path: Path = Field(..., description="Path to local git repository")
    branch_prefix: str = Field("feature/", description="Prefix for feature branches")
    remote: str = Field("origin", description="Git remote name")
    base_branch: str = Field("main", description="Base branch to branch from")

class MergeRequestConfig(BaseModel):
    title: str = Field(..., description="Title of the merge request")
    description: str = Field("", description="Description of the merge request")
    labels: list[str] = Field(default_factory=list, description="Labels to apply")
    assignees: list[str] = Field(default_factory=list, description="Usernames to assign")
    draft: bool = Field(False, description="Create as draft MR")

class GitHubClient(BaseModel):
    token: str = Field(..., description="GitHub personal access token (from env)")
    owner: str = Field(..., description="Repository owner")
    repo: str = Field(..., description="Repository name")

    class Config:
        arbitrary_types_allowed = True

    async def create_merge_request(
        self,
        head_branch: str,
        config: MergeRequestConfig,
        base_branch: str = "main",
    ) -> Optional[dict]:
        """
        Create a pull/merge request on GitHub.

        Args:
            head_branch: Branch containing changes (e.g., 'feature/my-feature')
            config: Merge request details
            base_branch: Target branch (default: 'main')

        Returns:
            MR data dict on success, None on failure (rule: error-handling)
        """
        url = f"https://api.github.com/repos/{self.owner}/{self.repo}/pulls"
        headers = {
            "Authorization": f"token {self.token}",
            "Accept": "application/vnd.github.v3+json",
        }
        payload = {
            "title": config.title,
            "head": head_branch,
            "base": base_branch,
            "body": config.description,
            "draft": config.draft,
        }
        if config.labels:
            payload["labels"] = config.labels
        if config.assignees:
            payload["assignees"] = config.assignees

        try:
            async with httpx.AsyncClient() as client:
                response = await client.post(url, headers=headers, json=payload)
                response.raise_for_status()
                mr_data = response.json()
                logger.info("Merge request created", mr_url=mr_data["html_url"])
                return mr_data
        except httpx.HTTPStatusError as e:
            logger.error("Failed to create MR", status=e.response.status_code, error=e.response.text)
            return None
        except Exception as e:
            logger.exception("Unexpected error creating MR", exc_info=e)
            return None


def run_git_command(cmd: list[str], cwd: Path) -> str:
    """
    Run a git command and return stdout.

    Args:
        cmd: Command arguments (e.g., ['status'])
        cwd: Working directory

    Returns:
        Command output as string

    Raises:
        RuntimeError: On non-zero exit code
    """
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd,
            capture_output=True,
            text=True,
            check=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        logger.error("Git command failed", cmd=cmd, stderr=e.stderr, stdout=e.stdout)
        raise RuntimeError(f"Git command failed: {e.stderr or e.stdout}")


def create_feature_branch(
    repo_path: Path,
    feature_name: str,
    config: GitConfig,
) -> str:
    """
    Create and checkout a new feature branch from the base branch.

    Args:
        repo_path: Path to git repository
        feature_name: Name of the feature (e.g., 'auth-fix')
        config: Git configuration

    Returns:
        Full branch name (e.g., 'feature/auth-fix')
    """
    branch_name = f"{config.branch_prefix}{feature_name}"
    logger.info("Creating feature branch", branch=branch_name)

    # Ensure we're on base branch
    run_git_command(["git", "checkout", config.base_branch], cwd=repo_path)
    run_git_command(["git", "pull", config.remote, config.base_branch], cwd=repo_path)

    # Create and checkout branch
    run_git_command(["git", "checkout", "-b", branch_name], cwd=repo_path)
    logger.info("Branch created and checked out", branch=branch_name)
    return branch_name


def commit_changes(
    repo_path: Path,
    message: str,
) -> None:
    """
    Stage and commit all changes.

    Args:
        repo_path: Path to git repository
        message: Commit message
    """
    run_git_command(["git", "add", "."], cwd=repo_path)
    run_git_command(["git", "commit", "-m", message], cwd=repo_path)
    logger.info("Changes committed", message=message)


def push_branch(
    repo_path: Path,
    branch_name: str,
    config: GitConfig,
) -> None:
    """
    Push branch to remote (safe: no force-push).

    Args:
        repo_path: Path to git repository
        branch_name: Name of branch to push
        config: Git configuration
    """
    logger.info("Pushing branch", branch=branch_name)
    try:
        run_git_command(
            ["git", "push", config.remote, branch_name, "--set-upstream"],
            cwd=repo_path,
        )
    except RuntimeError as e:
        # Branch already exists on remote? Update it with regular push
        logger.warning("Initial push failed, trying update", error=str(e))
        run_git_command(["git", "push", config.remote, branch_name], cwd=repo_path)


def main():
    """Main workflow: create feature branch, commit, push, and create MR."""
    # Get config from environment (rule: no-secrets)
    token = os.getenv("GITHUB_TOKEN")
    owner = os.getenv("GITHUB_OWNER")
    repo = os.getenv("GITHUB_REPO")
    repo_path = Path(os.getenv("GIT_REPO_PATH", "."))

    if not all([token, owner, repo]):
        logger.error("Missing required environment variables: GITHUB_TOKEN, GITHUB_OWNER, GITHUB_REPO")
        sys.exit(1)

    git_config = GitConfig(repo_path=repo_path)
    mr_config = MergeRequestConfig(
        title="feat: Add new user authentication flow",
        description="Implements OAuth2 login and session management.",
        labels=["enhancement", "security"],
        assignees=["you"],
    )

    try:
        # Step 1: Create feature branch
        branch = create_feature_branch(repo_path, "auth-flow", git_config)

        # Step 2: Simulate changes (for demo)
        readme_path = repo_path / "README.md"
        with open(readme_path, "a") as f:
            f.write("\n\n## Authentication\n- OAuth2 login\n- Session tokens\n")

        # Step 3: Commit changes
        commit_changes(repo_path, "feat: Add OAuth2 authentication")

        # Step 4: Push branch (safely)
        push_branch(repo_path, branch, git_config)

        # Step 5: Create MR
        gh_client = GitHubClient(token=token, owner=owner, repo=repo)
        mr = asyncio.run(
            gh_client.create_merge_request(
                head_branch=branch,
                config=mr_config,
                base_branch=git_config.base_branch,
            )
        )
        if not mr:
            logger.error("Failed to create merge request")
            sys.exit(1)

    except Exception as e:
        logger.exception("Workflow failed", exc_info=e)
        sys.exit(1)


# Rule: async-io (main calls async function)
import asyncio
if __name__ == "__main__":
    main()
```
