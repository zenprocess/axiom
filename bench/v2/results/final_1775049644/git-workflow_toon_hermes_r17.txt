Git Workflow Automation Script

This script automates common git workflows including:
- Creating feature branches
- Committing changes
- Pushing branches to remote
- Creating merge requests via GitHub API

Note: This implementation follows the no-force-push rule and does not use
git push --force. For rebased branches, use git push --force-with-lease
instead, which is safer as it only succeeds if the remote branch hasn't
been updated by others.
"""

# Configure structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),

        structlog.processors.StackInfoRenderer(),

logger = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations"""
    remote_name: str = "origin"
    base_branch: str = "main"
    branch_prefix: str = "feature/"
    commit_message: str = "Auto-generated commit"
    push_strategy: str = "force-with-lease"  # Safer alternative to force


class MergeRequestConfig(BaseModel):
    """Configuration for merge request creation"""
    title: str = Field(..., min_length=1)
    description: str = ""
    base_branch: str = "main"
    labels: List[str] = Field(default_factory=list)
    assignees: List[str] = Field(default_factory=list)


class GitWorkflowError(Exception):
    """Custom exception for git workflow operations"""
    pass


async def run_git_command(
    *args: str,
    cwd: Optional[Path] = None,
    check: bool = True,
) -> subprocess.CompletedProcess:
    """Run a git command with proper error handling.
    
    Args:
        *args: Arguments to pass to git
        cwd: Working directory
        check: Whether to raise exception on non-zero exit code
        
    Returns:
        CompletedProcess with command results
        
    Raises:
        GitWorkflowError: If command fails and check=True
    """
    try:
        result = subprocess.run(
            ["git", *args],
            cwd=cwd or Path.cwd(),
            capture_output=True,
            text=True,
            check=check,
        )
        logger.info("git_command_executed", args=args, returncode=result.returncode)
        return result
    except subprocess.CalledProcessError as e:
        logger.error(
            "git_command_failed",
            args=args,
            stdout=e.stdout,
            stderr=e.stderr,
            returncode=e.returncode,
        )
        raise GitWorkflowError(f"Git command failed: {e.stderr}") from e


async def create_feature_branch(
    feature_name: str,
    config: GitConfig,
) -> str:
    """Create a new feature branch from the base branch.
    
    Args:
        feature_name: Name of the feature branch (without prefix)
        config: Git configuration
        
    Returns:
        Full branch name (with prefix)
    """
    full_branch_name = f"{config.branch_prefix}{feature_name}"
    
    # Check current branch
    current_branch_result = await run_git_command("branch", "--show-current")
    current_branch = current_branch_result.stdout.strip()
    
    # Update base branch
    logger.info("updating_base_branch", branch=config.base_branch)
    await run_git_command("fetch", config.remote_name, config.base_branch)
    await run_git_command("checkout", f"{config.remote_name}/{config.base_branch}")
    await run_git_command("checkout", "-b", full_branch_name)
    
    # Return to original branch if not already on it
    if current_branch != full_branch_name and current_branch != "":
        await run_git_command("checkout", current_branch)
    
    logger.info("feature_branch_created", branch=full_branch_name)
    return full_branch_name


async def commit_changes(
    message: Optional[str] = None,
    files: Optional[List[str]] = None,
) -> bool:
    """Commit changes to the current branch.
    
    Args:
        message: Commit message (uses default if not provided)
        files: Specific files to commit (all staged files if None)
        
    Returns:
        True if commit was successful
    """
    commit_msg = message or "Auto-generated commit"
    
    # Check if there are any changes to commit
    status_result = await run_git_command("status", "--porcelain")
    if not status_result.stdout.strip():
        logger.warning("no_changes_to_commit")
        return False
    
    # Add files if specified
    if files:
        for file_path in files:
            await run_git_command("add", file_path)
    else:
        # Add all changes
        await run_git_command("add", "-A")
    
    # Commit changes
    await run_git_command("commit", "-m", commit_msg)
    logger.info("changes_committed", message=commit_msg)
    return True


async def push_branch(
    branch_name: str,
    config: GitConfig,
    force_with_lease: bool = False,
) -> bool:
    """Push branch to remote repository.
    
    Args:
        branch_name: Name of the branch to push
        config: Git configuration
        force_with_lease: Use --force-with-lease instead of regular push
        
    Returns:
        True if push was successful
    """
    try:
        if force_with_lease:
            # Safer alternative to force push that respects remote changes
            await run_git_command("push", config.remote_name, branch_name, "--force-with-lease")
            logger.info("branch_pushed_with_lease", branch=branch_name)
        else:
            await run_git_command("push", config.remote_name, branch_name)
            logger.info("branch_pushed", branch=branch_name)
        return True
    except GitWorkflowError as e:
        logger.error("branch_push_failed", branch=branch_name, error=str(e))
        return False


async def create_merge_request(
    mr_config: MergeRequestConfig,
    branch_name: str,
    github_token: str,
    repo_owner: str,
    repo_name: str,
) -> Optional[int]:
    """Create a merge request via GitHub API.
    
    Args:
        mr_config: Merge request configuration
        branch_name: Source branch for the merge request
        github_token: GitHub personal access token
        repo_owner: Repository owner
        repo_name: Repository name
        
    Returns:
        Merge request number if successful, None otherwise
    """
    url = f"https://api.github.com/repos/{repo_owner}/{repo_name}/pulls"
    
    headers = {
        "Authorization": f"token {github_token}",
        "Accept": "application/vnd.github.v3+json",
    }
    
    data = {
        "title": mr_config.title,
        "head": branch_name,
        "base": mr_config.base_branch,
        "body": mr_config.description,
        "labels": mr_config.labels,
        "assignees": mr_config.assignees,
    }
    
    try:
        async with httpx.AsyncClient() as client:
            response = await client.post(url, json=data, headers=headers)
            response.raise_for_status()
            result = response.json()
            logger.info("merge_request_created", mr_number=result["number"])
            return result["number"]
    except httpx.HTTPStatusError as e:
        logger.error(
            "merge_request_creation_failed",
            status_code=e.response.status_code,
            response_text=e.response.text,
        )
        return None
    except Exception as e:
        logger.error("merge_request_creation_error", error=str(e))
        return None


async def run_git_workflow(
    feature_name: str,
    commit_message: Optional[str] = None,
    files: Optional[List[str]] = None,
    mr_config: Optional[MergeRequestConfig] = None,
    github_token: Optional[str] = None,
    repo_owner: Optional[str] = None,
    repo_name: Optional[str] = None,
) -> bool:
    """Run the complete git workflow.
    
    Args:
        feature_name: Name of the feature branch
        commit_message: Optional custom commit message
        files: Optional list of files to commit
        mr_config: Optional merge request configuration
        github_token: GitHub personal access token (required for MR creation)
        repo_owner: Repository owner (required for MR creation)
        repo_name: Repository name (required for MR creation)
        
    Returns:
        True if workflow completed successfully
    """
    try:
        # Validate required parameters for MR creation
        if mr_config and not all([github_token, repo_owner, repo_name]):
            logger.error("missing_parameters_for_merge_request")
            return False
        
        config = GitConfig(commit_message=commit_message or "Auto-generated commit")
        
        # Create feature branch
        branch_name = await create_feature_branch(feature_name, config)
        
        # Commit changes
        if not await commit_changes(commit_message, files):
            logger.warning("no_changes_to_commit_workflow_stopped")
            return False
        
        # Push branch
        if not await push_branch(branch_name, config, force_with_lease=True):
            logger.error("push_failed_workflow_stopped")
            return False
        
        # Create merge request if configured
        if mr_config and github_token and repo_owner and repo_name:
            mr_number = await create_merge_request(
                mr_config, branch_name, github_token, repo_owner, repo_name
            )
            if mr_number is None:
                logger.error("merge_request_creation_failed_workflow_stopped")
                return False
            logger.info("merge_request_created", mr_number=mr_number)
        
        logger.info("git_workflow_completed", feature_name=feature_name)
        return True
        
    except GitWorkflowError as e:
        logger.error("git_workflow_failed", error=str(e))
        return False
    except Exception as e:
        logger.error("unexpected_error", error=str(e))
        return False
