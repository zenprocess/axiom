#!/usr/bin/env python3
"""
Automated Git Workflow Script

This script automates common Git workflows:
- Create feature branches
- Commit staged changes
- Push to remote
- Create GitHub merge requests (pull requests)

Uses GitHub REST API v3 for merge request creation.
"""

import os
import subprocess
import sys
import httpx
from pathlib import Path
from typing import Optional, List, Dict, Any
from structlog import get_logger

log = get_logger(__name__)

# Pydantic models for structured data
from pydantic import BaseModel

class GitConfig(BaseModel):
    repo_path: Path
    remote: str = "origin"
    base_branch: str = "main"
    branch_prefix: str = "feature/"
    commit_message: str = "Automated commit"

class GitHubConfig(BaseModel):
    owner: str
    repo: str
    token: str  # Should be provided via environment variable or config file
    base_url: str = "https://api.github.com"

class MergeRequest(BaseModel):
    title: str
    head: str
    base: str
    body: str = ""

def run_git_command(cmd: List[str], cwd: Path) -> str:
    """Run a Git command and return stdout."""
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd,
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        log.error("git_command_failed", cmd=cmd, stderr=e.stderr, stdout=e.stdout, exc_info=True)
        raise

async def create_feature_branch(config: GitConfig) -> str:
    """Create and checkout a new feature branch."""
    branch_name = f"{config.branch_prefix}my-feature-{os.urandom(4).hex()}"
    run_git_command(["git", "checkout", "-b", branch_name], config.repo_path)
    log.info("feature_branch_created", branch=branch_name)
    return branch_name

async def commit_changes(config: GitConfig) -> None:
    """Stage and commit all changes."""
    run_git_command(["git", "add", "."], config.repo_path)
    run_git_command(["git", "commit", "-m", config.commit_message], config.repo_path)
    log.info("changes_committed", message=config.commit_message)

async def push_branch(config: GitConfig, branch_name: str) -> None:
    """Push branch to remote (without --force)."""
    try:
        run_git_command(["git", "push", config.remote, branch_name], config.repo_path)
        log.info("branch_pushed", branch=branch_name, remote=config.remote)
    except subprocess.CalledProcessError as e:
        if "non-fast-forward" in e.stderr:
            log.warning(
                "branch_push_failed_non_fast_forward",
                branch=branch_name,
                error=e.stderr,
                suggestion="Consider pulling remote changes first or using rebase.",
            )
        raise

async def create_merge_request(
    gh_config: GitHubConfig,
    mr: MergeRequest,
) -> Dict[str, Any]:
    """Create a GitHub pull request via API."""
    url = f"{gh_config.base_url}/repos/{gh_config.owner}/{gh_config.repo}/pulls"
    headers = {
        "Authorization": f"token {gh_config.token}",
        "Accept": "application/vnd.github.v3+json",
    }
    payload = {
        "title": mr.title,
        "head": mr.head,
        "base": mr.base,
        "body": mr.body,
    }

    async with httpx.AsyncClient() as client:
        try:
            response = await client.post(url, json=payload, headers=headers)
            response.raise_for_status()
            pr_data = response.json()
            log.info(
                "merge_request_created",
                pr_number=pr_data.get("number"),
                pr_url=pr_data.get("html_url"),
            )
            return pr_data
        except httpx.HTTPStatusError as e:
            log.error(
                "merge_request_creation_failed",
                status_code=e.response.status_code,
                error=e.response.text,
                exc_info=True,
            )
            raise

async def main():
    # Load config (example values — in practice, use env vars or config file)
    repo_path = Path("/path/to/your/repo")
    git_config = GitConfig(repo_path=repo_path)
    gh_config = GitHubConfig(
        owner="your-username",
        repo="your-repo",
        token=os.environ.get("GITHUB_TOKEN", ""),
    )

    if not gh_config.token:
        log.error("github_token_missing", message="GITHUB_TOKEN environment variable is required.")
        sys.exit(1)

    if not repo_path.is_dir():
        log.error("invalid_repo_path", path=str(repo_path))
        sys.exit(1)

    try:
        branch = await create_feature_branch(git_config)
        await commit_changes(git_config)
        await push_branch(git_config, branch)

        mr = MergeRequest(
            title=f"feat: add {branch.split('/')[-1]}",
            head=branch,
            base=git_config.base_branch,
            body="Auto-generated merge request",
        )
        await create_merge_request(gh_config, mr)

    except Exception as e:
        log.error("workflow_failed", error=str(e), exc_info=True)
        sys.exit(1)

if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
