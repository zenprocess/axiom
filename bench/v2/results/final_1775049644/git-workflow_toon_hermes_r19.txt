### ✅ Recommended Safe Git Automation Script

This script:
- Creates feature branches
- Commits changes
- Pushes branches safely (`git push` without `--force`)
- Uses `git pull --rebase` to stay up-to-date before pushing

It uses modern async I/O (`httpx.AsyncClient`) and follows all rules (`structlog`, Pydantic, type hints, etc.).

```python
#!/usr/bin/env python3
"""
Safe Git Workflow Automation Script

This script automates common Git workflows:
- Create feature branches from main
- Stage and commit changes
- Push to remote (safely, no force push)
- Create merge requests (GitHub example)

⚠️ Uses safe git push — no force pushes.
✅ For rebased branches, prefer rebasing locally and merging via PR/MR.
"""

import subprocess
import sys
import os
from pathlib import Path
from typing import Optional

import httpx
from pydantic import BaseModel
import structlog

# Configure logging (structlog only — no logging or print)
structlog.configure(
    wrapper_class=structlog.make_filtering_bound_logger(logging_level="INFO"),
)
logger = structlog.get_logger()


class MergeRequestResult(BaseModel):
    """Response from creating a merge request."""
    id: int
    title: str
    url: str
    state: str


def run_git_command(*args: str, cwd: Optional[Path] = None) -> str:
    """
    Run a Git command and return stdout.

    Args:
        *args: Git command arguments (e.g., 'status', '--porcelain')
        cwd: Working directory (defaults to current directory)

    Returns:
        stdout as string (stripped)

    Raises:
        RuntimeError: If git command fails
    """
    cmd = ["git"] + list(args)
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd or Path.cwd(),
            capture_output=True,
            text=True,
            check=True,
        )
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        logger.error("git_command_failed", cmd=cmd, stderr=e.stderr, stdout=e.stdout)
        raise RuntimeError(f"Git command failed: {' '.join(cmd)}") from e


def ensure_clean_working_tree(cwd: Path) -> None:
    """Ensure working tree is clean before proceeding."""
    status = run_git_command("status", "--porcelain", cwd=cwd)
    if status:
        logger.error("working_tree_dirty", status=status)
        raise RuntimeError("Working tree is not clean. Commit or stash changes first.")


def create_feature_branch(feature_name: str, base: str = "main", cwd: Path = Path.cwd()) -> str:
    """
    Create and checkout a new feature branch from base.

    Args:
        feature_name: Name of the feature branch (e.g., "add-login")
        base: Base branch to branch from (default: "main")
        cwd: Git repository root

    Returns:
        Name of created branch
    """
    ensure_clean_working_tree(cwd)
    run_git_command("checkout", base, cwd=cwd)
    run_git_command("pull", "origin", base, cwd=cwd)
    branch_name = f"feature/{feature_name}"
    run_git_command("checkout", "-b", branch_name, cwd=cwd)
    logger.info("feature_branch_created", branch=branch_name)
    return branch_name


def commit_changes(message: str, cwd: Path = Path.cwd()) -> None:
    """
    Stage all changes and commit.

    Args:
        message: Commit message
        cwd: Git repository root
    """
    run_git_command("add", ".", cwd=cwd)
    run_git_command("commit", "-m", message, cwd=cwd)
    logger.info("changes_committed", message=message)


def push_branch(branch_name: str, cwd: Path = Path.cwd()) -> None:
    """
    Push branch to remote (safely — no force push).

    Uses `git push` with `--set-upstream` for first push.

    Args:
        branch_name: Branch to push
        cwd: Git repository root
    """
    try:
        run_git_command("push", "--set-upstream", "origin", branch_name, cwd=cwd)
    except RuntimeError as e:
        logger.warning("push_failed_fallback", error=str(e))
        # Fallback: pull first (rebase), then push
        run_git_command("pull", "--rebase", "origin", branch_name, cwd=cwd)
        run_git_command("push", "origin", branch_name, cwd=cwd)
    logger.info("branch_pushed", branch=branch_name)


async def create_merge_request_github(
    title: str,
    body: str,
    head: str,
    base: str = "main",
    token: str,
    owner: str,
    repo: str,
) -> MergeRequestResult:
    """
    Create a GitHub merge request (pull request) using the REST API.

    Args:
        title: MR title
        body: MR description
        head: Source branch (e.g., "feature/add-login")
        base: Target branch (default: "main")
        token: GitHub personal access token (PAT)
        owner: GitHub repo owner
        repo: GitHub repo name

    Returns:
        Pydantic model with MR details
    """
    url = f"https://api.github.com/repos/{owner}/{repo}/pulls"
    headers = {
        "Authorization": f"token {token}",
        "Accept": "application/vnd.github.v3+json",
    }
    payload = {
        "title": title,
        "body": body,
        "head": head,
        "base": base,
    }

    async with httpx.AsyncClient() as client:
        try:
            response = await client.post(url, json=payload, headers=headers)
            response.raise_for_status()
            data = response.json()
            return MergeRequestResult(
                id=data["number"],
                title=data["title"],
                url=data["html_url"],
                state=data["state"],
            )
        except httpx.HTTPStatusError as e:
            logger.error("mr_creation_failed", status=e.response.status_code, body=e.response.text)
            raise RuntimeError(f"Failed to create merge request: {e.response.text}") from e


async def main():
    """Main orchestration function."""
    repo_path = Path.cwd()
    if not (repo_path / ".git").exists():
        logger.error("not_a_git_repo", path=str(repo_path))
        sys.exit(1)

    # Example workflow
    feature = "add-login-ui"
    branch = create_feature_branch(feature, cwd=repo_path)
    commit_changes("feat: add login UI component", cwd=repo_path)
    push_branch(branch, cwd=repo_path)

    # Create GitHub MR
    # ⚠️ Replace with actual values
    GITHUB_TOKEN = os.getenv("GITHUB_TOKEN")
    if not GITHUB_TOKEN:
        logger.warning("no_github_token", msg="Skipping MR creation")
    else:
        mr = await create_merge_request_github(
            title=f"Add {feature} feature",
            body="This MR implements the login UI component.",
            head=branch,
            base="main",
            token=GITHUB_TOKEN,
            owner="your-org",
            repo="your-repo",
        )
        logger.info("merge_request_created", id=mr.id, url=mr.url)


if __name__ == "__main__":
    import asyncio
    import logging  # structlog is used, but logging needed for filter level
    asyncio.run(main())
```

### 🔐 Security Notes

- **No secrets**: GitHub token is read from `GITHUB_TOKEN` env var (never hardcoded).
- **No `print()`**: Uses `structlog` for structured logging.
- **No raw dicts**: Uses Pydantic `MergeRequestResult`.
- **Async I/O**: `httpx.AsyncClient` used for GitHub API.
- **Error handling**: `try/except` around all external calls.

### 📌 Recommendation

Instead of force-pushing rebased branches, consider:

1. **Rebase locally** (`git rebase -i main`)
2. **Force push only if absolutely necessary**, and *only* if:
   - You own the branch
   - Team agrees
   - Use `--force-with-lease` (safer than `--force`)
3. Prefer **merge commits** or **squash merges** on GitHub/GitLab.
