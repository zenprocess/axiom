```python
#!/usr/bin/env python3
"""
Git Workflow Automation Script

This script automates common git workflows:
- Create feature branches
- Commit changes
- Push to remote
- Create merge requests (via GitHub API)

All operations follow safe git practices - no force pushes are used.
"""

import os
import sys
import subprocess
from typing import Optional, List
from pathlib import Path

import httpx
import structlog
from pydantic import BaseModel, Field, field_validator


# Initialize structlog for structured logging
logger = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for git operations"""
    remote: str = "origin"
    base_branch: str = "main"
    username: Optional[str] = None
    token: Optional[str] = Field(default=None, exclude=True)
    repo_path: Path = Field(default_factory=Path.cwd)


class MRConfig(BaseModel):
    """Configuration for merge request creation"""
    platform: str = "github"
    api_url: str = "https://api.github.com"
    owner: str
    repo: str
    title: str
    body: str = ""
    base: str = "main"
    head: str


class GitWorkflow:
    """Class to handle git workflow automation"""

    def __init__(self, config: GitConfig):
        self.config = config
        self.repo_path = config.repo_path

    def run_git_command(self, args: List[str], cwd: Optional[Path] = None) -> str:
        """Run a git command and return output"""
        try:
            result = subprocess.run(
                ["git"] + args,
                cwd=cwd or self.repo_path,
                capture_output=True,
                text=True,
                check=True
            )
            logger.info("git_command_success", command=" ".join(["git"] + args), 
                       output=result.stdout.strip())
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            logger.error("git_command_failed", 
                        command=" ".join(["git"] + args),
                        error=e.stderr.strip(),
                        exit_code=e.returncode)
            raise

    def create_feature_branch(self, branch_name: str) -> str:
        """Create a new feature branch from current branch"""
        # Ensure we're on base branch first
        self.run_git_command(["checkout", self.config.base_branch])
        self.run_git_command(["pull", self.config.remote, self.config.base_branch])
        
        # Create and checkout new feature branch
        branch = f"feature/{branch_name}"
        self.run_git_command(["checkout", "-b", branch])
        logger.info("feature_branch_created", branch=branch)
        return branch

    def add_files(self, files: List[str] = None) -> None:
        """Add files to staging area"""
        if files:
            self.run_git_command(["add"] + files)
        else:
            self.run_git_command(["add", "."])
        logger.info("files_added")

    def commit_changes(self, message: str) -> None:
        """Commit staged changes"""
        self.run_git_command(["commit", "-m", message])
        logger.info("changes_committed", message=message)

    def push_branch(self, branch: str, force: bool = False) -> None:
        """Push branch to remote repository
        
        IMPORTANT: Force pushing is disabled per security rules.
        Use safe push with --no-force option.
        """
        if force:
            logger.warning("force_push_attempted", 
                          message="Force push is disabled for safety. Using safe push instead.")
        
        self.run_git_command(["push", self.config.remote, branch])
        logger.info("branch_pushed", branch=branch)

    def get_current_branch(self) -> str:
        """Get current branch name"""
        return self.run_git_command(["rev-parse", "--abbrev-ref", "HEAD"])


class MRService:
    """Service to create merge requests via GitHub API"""

    def __init__(self, config: MRConfig):
        self.config = config
        self.client = httpx.AsyncClient(
            timeout=30.0,
            headers={
                "Accept": "application/vnd.github.v3+json",
                "User-Agent": "Git-Automation-Script"
            }
        )

    async def create_merge_request(self) -> dict:
        """Create a merge request on GitHub"""
        url = f"{self.config.api_url}/repos/{self.config.owner}/{self.config.repo}/pulls"
        
        payload = {
            "title": self.config.title,
            "head": self.config.head,
            "base": self.config.base,
            "body": self.config.body
        }
        
        try:
            response = await self.client.post(
                url,
                json=payload,
                headers={"Authorization": f"token {self.config.token}"} if self.config.token else {}
            )
            response.raise_for_status()
            
            mr_data = response.json()
            logger.info("merge_request_created", 
                       mr_number=mr_data.get("number"),
                       mr_url=mr_data.get("html_url"))
            return mr_data
            
        except httpx.HTTPStatusError as e:
            logger.error("merge_request_failed", 
                        status_code=e.response.status_code,
                        error=e.response.text)
            raise
        finally:
            await self.client.aclose()


async def main():
    """Main function to demonstrate git workflow automation"""
    try:
        # Get configuration from environment or use defaults
        repo_path = Path(os.environ.get("GIT_REPO_PATH", "."))
        remote = os.environ.get("GIT_REMOTE", "origin")
        base_branch = os.environ.get("GIT_BASE_BRANCH", "main")
        
        git_config = GitConfig(
            repo_path=repo_path,
            remote=remote,
            base_branch=base_branch
        )
        
        # Initialize workflow
        workflow = GitWorkflow(git_config)
        
        # Get current branch to ensure we're starting from a known state
        current_branch = workflow.get_current_branch()
        logger.info("started_workflow", current_branch=current_branch)
        
        # Create feature branch
        feature_branch = workflow.create_feature_branch("my-feature")
        
        # Make some changes (example)
        # In real usage, you would add files and commit changes here
        # For this example, we'll just simulate the workflow
        
        # Add all changes
        workflow.add_files()
        
        # Commit changes
        workflow.commit_changes("Add new feature implementation")
        
        # Push branch
        workflow.push_branch(feature_branch)
        
        # Create merge request (if configured)
        if os.environ.get("GITHUB_TOKEN") and os.environ.get("GITHUB_OWNER") and os.environ.get("GITHUB_REPO"):
            mr_config = MRConfig(
                owner=os.environ.get("GITHUB_OWNER", ""),
                repo=os.environ.get("GITHUB_REPO", ""),
                title=f"feat: Add {feature_branch} changes",
                body="This MR includes the new feature implementation.",
                base=base_branch,
                head=feature_branch
            )
            
            # Add token from environment
            mr_config.token = os.environ.get("GITHUB_TOKEN")
            
            mr_service = MRService(mr_config)
            mr = await mr_service.create_merge_request()
            
            logger.info("workflow_complete", 
                       branch=feature_branch,
                       mr_url=mr.get("html_url"))
        else:
            logger.info("workflow_complete_no_mr", 
                       branch=feature_branch,
                       message="Set GITHUB_TOKEN, GITHUB_OWNER, and GITHUB_REPO to create MRs")
        
    except Exception as e:
        logger.error("workflow_failed", error=str(e))
        sys.exit(1)


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
