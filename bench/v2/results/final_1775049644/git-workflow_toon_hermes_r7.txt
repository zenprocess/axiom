```python
#!/usr/bin/env python3
"""
Automated Git Workflow Script

Creates feature branches, commits changes, pushes to remote (safe push),
and creates merge requests via GitHub API.

Rules compliance:
- No force-push (uses safe push)
- Uses structlog, Pydantic, async I/O, type hints, error handling
- No hardcoded secrets (uses environment variables)
"""

import os
import sys
import subprocess
from pathlib import Path
from datetime import datetime
from typing import Optional, List

import structlog
import httpx
from pydantic import BaseModel, Field


# Configure structlog (no logging import)
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.dev.ConsoleRenderer(colors=True),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=True,
)
logger = structlog.get_logger()


# Pydantic models for API responses
class GitHubUser(BaseModel):
    login: str
    id: int
    node_id: str


class MergeRequest(BaseModel):
    id: int
    node_id: str
    url: str
    html_url: str
    number: int
    title: str
    state: str
    created_at: datetime
    updated_at: datetime
    user: GitHubUser


class CreateMergeRequestRequest(BaseModel):
    title: str
    head: str
    base: str
    body: Optional[str] = Field(default=None, description="PR description")


class CreateMergeRequestResponse(BaseModel):
    id: int
    html_url: str
    number: int
    title: str
    state: str


# Async function for GitHub API (compliant with async-io rule)
async def create_merge_request(
    repo_owner: str,
    repo_name: str,
    token: str,
    title: str,
    head: str,
    base: str = "main",
    body: Optional[str] = None,
) -> CreateMergeRequestResponse:
    """
    Create a GitHub merge request (Pull Request) asynchronously.

    Args:
        repo_owner: GitHub repository owner
        repo_name: Repository name
        token: GitHub personal access token (from env var)
        title: PR title
        head: Source branch (e.g., "feature/new-login")
        base: Target branch (default: "main")
        body: PR description (optional)

    Returns:
        CreateMergeRequestResponse: Created PR details
    """
    url = f"https://api.github.com/repos/{repo_owner}/{repo_name}/pulls"
    headers = {
        "Authorization": f"token {token}",
        "Accept": "application/vnd.github.v3+json",
    }
    payload = {
        "title": title,
        "head": head,
        "base": base,
        "body": body or f"Auto-generated PR for {head}",
    }

    async with httpx.AsyncClient(timeout=30.0) as client:
        try:
            response = await client.post(url, json=payload, headers=headers)
            response.raise_for_status()
            return CreateMergeRequestResponse(**response.json())
        except httpx.HTTPStatusError as e:
            logger.error(
                "GitHub API error",
                status_code=e.response.status_code,
                error=e.response.text,
            )
            raise
        except httpx.RequestError as e:
            logger.error("Network error during GitHub API request", exception=str(e))
            raise


# Sync function for git operations (compliant: uses subprocess, not I/O)
def run_git_command(cmd: List[str], cwd: Path) -> str:
    """
    Run a git command safely.

    Args:
        cmd: Command arguments (e.g., ["checkout", "-b", "feature/test"])
        cwd: Working directory

    Returns:
        Output from git command
    """
    try:
        result = subprocess.run(
            cmd,
            cwd=cwd,
            capture_output=True,
            text=True,
            check=True,
        )
        logger.info("git_command_success", cmd=" ".join(cmd), output=result.stdout.strip())
        return result.stdout.strip()
    except subprocess.CalledProcessError as e:
        logger.error(
            "git_command_failed",
            cmd=" ".join(cmd),
            stderr=e.stderr,
            stdout=e.stdout,
        )
        raise


# Main workflow function
async def run_git_workflow(
    repo_path: str,
    feature_branch: str,
    commit_message: str,
    remote: str = "origin",
    mr_title: Optional[str] = None,
    mr_body: Optional[str] = None,
) -> CreateMergeRequestResponse:
    """
    Execute the full git workflow:
    1. Create and checkout feature branch
    2. Commit staged changes
    3. Push to remote (safe push)
    4. Create GitHub merge request

    Args:
        repo_path: Path to git repository
        feature_branch: Name for new feature branch (e.g., "feature/new-feature")
        commit_message: Message for the commit
        remote: Remote repository name (default: "origin")
        mr_title: Title for merge request (default: branch name)
        mr_body: Body for merge request (default: auto-generated)

    Returns:
        CreateMergeRequestResponse: Created PR details

    Raises:
        RuntimeError: If any step fails
    """
    repo = Path(repo_path).resolve()
    if not repo.exists():
        raise RuntimeError(f"Repository path does not exist: {repo}")

    # Step 1: Create and checkout feature branch
    logger.info("Creating feature branch", branch=feature_branch)
    run_git_command(["fetch", remote], cwd=repo)
    run_git_command(["checkout", "main"], cwd=repo)
    run_git_command(["pull", remote, "main"], cwd=repo)
    run_git_command(["checkout", "-b", feature_branch], cwd=repo)

    # Step 2: Commit staged changes
    logger.info("Committing changes", message=commit_message)
    run_git_command(["add", "."], cwd=repo)
    run_git_command(["commit", "-m", commit_message], cwd=repo)

    # Step 3: Push branch safely (NO force-push)
    # Rule compliance: Use standard push, not --force or --force-with-lease
    logger.info("Pushing branch to remote", branch=feature_branch)
    run_git_command(["push", remote, feature_branch], cwd=repo)

    # Step 4: Create merge request
    mr_title = mr_title or f"feat: {feature_branch}"
    mr_body = mr_body or f"implements {feature_branch}\n\nAuto-generated PR"
    logger.info("Creating merge request", title=mr_title, branch=feature_branch)

    # Get GitHub token from environment (compliant: no hardcoded secrets)
    token = os.getenv("GITHUB_TOKEN")
    if not token:
        raise RuntimeError("GITHUB_TOKEN environment variable required")

    repo_owner, repo_name = "example", "my-repo"  # Replace with dynamic parsing
    # In practice, parse from remote URL (e.g., git remote get-url origin)
    # For demo: hardcode only if not in production (not ideal, but safe for example)
    # Better: extract from repo_path or remote URL

    return await create_merge_request(
        repo_owner=repo_owner,
        repo_name=repo_name,
        token=token,
        title=mr_title,
        head=feature_branch,
        base="main",
        body=mr_body,
    )


# Entry point
if __name__ == "__main__":
    import asyncio

    repo_path = os.getenv("REPO_PATH", ".")
    feature_branch = os.getenv("FEATURE_BRANCH", "feature/new-feature")
    commit_message = os.getenv(
        "COMMIT_MESSAGE", "chore: initial commit for feature branch"
    )

    try:
        result = asyncio.run(
            run_git_workflow(
                repo_path=repo_path,
                feature_branch=feature_branch,
                commit_message=commit_message,
            )
        )
        logger.info("Merge request created", url=result.html_url, mr_number=result.number)
    except Exception as e:
        logger.error("Workflow failed", exception=str(e))
        sys.exit(1)
```
