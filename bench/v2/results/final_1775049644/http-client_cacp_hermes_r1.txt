```python
"""Module for fetching user data from a REST API."""

import httpx
import structlog
from pydantic import BaseModel
from typing import Optional

# Initialize structlog
logger = structlog.get_logger()


class User(BaseModel):
    """User data model representing a user from the API."""
    id: int
    name: str
    email: str
    username: str
    phone: Optional[str] = None
    website: Optional[str] = None


class ApiResponse(BaseModel):
    """Response wrapper model for API responses."""
    user: Optional[User] = None
    error: Optional[str] = None


async def fetch_user(user_id: int) -> ApiResponse:
    """
    Fetch user data from the API.

    Args:
        user_id: The ID of the user to fetch.

    Returns:
        ApiResponse containing the user data or an error message.
    """
    url = f"https://api.example.com/users/{user_id}"
    
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(url)
            
            if response.status_code == 200:
                user_data = response.json()
                user = User(**user_data)
                logger.info("User fetched successfully", user_id=user_id)
                return ApiResponse(user=user)
            else:
                error_msg = f"API returned status {response.status_code}"
                logger.error("API request failed", 
                           user_id=user_id, 
                           status_code=response.status_code,
                           error=error_msg)
                return ApiResponse(error=error_msg)
                
    except httpx.RequestError as exc:
        error_msg = f"Request error: {str(exc)}"
        logger.error("Request failed", user_id=user_id, error=error_msg)
        return ApiResponse(error=error_msg)
    except Exception as exc:
        error_msg = f"Unexpected error: {str(exc)}"
        logger.error("Unexpected error occurred", user_id=user_id, error=error_msg)
        return ApiResponse(error=error_msg)


async def main() -> ApiResponse:
    """
    Main function that fetches user #42.

    Returns:
        ApiResponse containing user #42 data or an error message.
    """
    return await fetch_user(42)


if __name__ == "__main__":
    import asyncio
    
    async def run():
        result = await main()
        if result.user:
            print(f"User {result.user.id}: {result.user.name} ({result.user.email})")
        else:
            print(f"Error: {result.error}")
    
    asyncio.run(run())
```

This module implements:

1. **Pydantic models** for user data and API responses
2. **Async I/O** with `async def` for the fetch function
3. **Error handling** with try/except blocks for external API calls
4. **structlog** for logging (no print() or logging imports)
5. **Type hints** on all function signatures
6. **No force push, no secrets, no star imports**
7. **Docstrings** for public functions
8. **Proper error handling** for network issues and API responses

Note: The `print()` statements in the `run()` function at the bottom are for demonstration purposes only. In production code, you would replace them with proper logging using `structlog`.
