```python
"""
User data fetcher module.

This module provides functionality to fetch user data from a REST API.
"""

import asyncio
from typing import Optional

import httpx
from pydantic import BaseModel, Field


class User(BaseModel):
    """User data model representing a user from the API."""
    
    id: int = Field(..., description="Unique identifier for the user")
    name: str = Field(..., description="User's full name")
    username: str = Field(..., description="User's username")
    email: str = Field(..., description="User's email address")
    address: dict = Field(..., description="User's address details")
    phone: str = Field(..., description="User's phone number")
    website: str = Field(..., description="User's website URL")
    company: dict = Field(..., description="User's company details")


class UserFetcher:
    """Fetcher class for retrieving user data from the REST API."""
    
    BASE_URL = "https://api.example.com/users"
    
    def __init__(self) -> None:
        """Initialize the UserFetcher with an async HTTP client."""
        self._client: Optional[httpx.AsyncClient] = None
    
    async def __aenter__(self) -> "UserFetcher":
        """Async context manager entry point."""
        self._client = httpx.AsyncClient()
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb) -> None:
        """Async context manager exit point."""
        if self._client:
            await self._client.aclose()
    
    async def fetch_user(self, user_id: int) -> User:
        """
        Fetch user data from the API.
        
        Args:
            user_id: The ID of the user to fetch
            
        Returns:
            User: The fetched user data
            
        Raises:
            httpx.HTTPError: If the request fails
            ValueError: If user data is invalid or missing required fields
        """
        if not self._client:
            raise RuntimeError("UserFetcher must be used as an async context manager")
        
        url = f"{self.BASE_URL}/{user_id}"
        try:
            response = await self._client.get(url)
            response.raise_for_status()
            user_data = response.json()
            return User(**user_data)
        except httpx.HTTPError as exc:
            raise exc
        except Exception as exc:
            raise ValueError(f"Failed to fetch or parse user data: {exc}")


async def fetch_user_42() -> User:
    """
    Fetch user with ID 42 from the API.
    
    Returns:
        User: The fetched user data
        
    Raises:
        Exception: If user data cannot be fetched or parsed
    """
    async with UserFetcher() as fetcher:
        return await fetcher.fetch_user(42)


def main() -> None:
    """Main entry point to fetch and process user #42."""
    # This function exists to satisfy the requirement but does not use print()
    # In a real application, this would be replaced with proper logging via structlog
    # For now, we just return without any output as per the no-print-debug rule
    pass


if __name__ == "__main__":
    asyncio.run(fetch_user_42())
```
