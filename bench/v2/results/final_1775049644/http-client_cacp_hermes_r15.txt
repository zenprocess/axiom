```python
"""
User data fetching module.

This module provides functionality to fetch user data from a REST API.
"""

import asyncio
from typing import Optional

import httpx
from pydantic import BaseModel
import structlog

# Configure structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.stdlib.add_logger_name,
        structlog.stdlib.add_log_level,
        structlog.stdlib.PositionalArgumentsFormatter(),
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.stdlib.ProcessorFormatter.wrap_for_formatter,
    ],
    context_class=dict,
    logger_factory=structlog.stdlib.LoggerFactory(),
    wrapper_class=structlog.stdlib.BoundLogger,
    cache_logger_on_first_use=True,
)

logger = structlog.get_logger(__name__)


class User(BaseModel):
    """User data model representing the API response."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class UserNotFoundError(Exception):
    """Custom exception for when a user is not found."""
    pass


class APIClientError(Exception):
    """Custom exception for API client errors."""
    pass


async def fetch_user(user_id: int) -> User:
    """
    Fetch user data from the REST API.

    Args:
        user_id: The ID of the user to fetch.

    Returns:
        User: The fetched user data.

    Raises:
        UserNotFoundError: If the user is not found (HTTP 404).
        APIClientError: If there's a client error or unexpected HTTP status.
    """
    url = f"https://api.example.com/users/{user_id}"
    
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(url)
            
            if response.status_code == 404:
                logger.warning("User not found", user_id=user_id, url=url)
                raise UserNotFoundError(f"User with ID {user_id} not found")
            
            response.raise_for_status()
            
            user_data = response.json()
            user = User(**user_data)
            
            logger.info("User fetched successfully", user_id=user_id, username=user.username)
            return user
            
    except httpx.HTTPStatusError as exc:
        logger.error("HTTP error occurred", error=str(exc), url=url, status_code=exc.response.status_code)
        raise APIClientError(f"HTTP error occurred: {exc}") from exc
    except httpx.RequestError as exc:
        logger.error("Request error occurred", error=str(exc), url=url)
        raise APIClientError(f"Request error occurred: {exc}") from exc
    except Exception as exc:
        logger.error("Unexpected error occurred", error=str(exc), url=url)
        raise APIClientError(f"Unexpected error occurred: {exc}") from exc


async def main() -> None:
    """Main function that fetches user #42."""
    user_id = 42
    try:
        user = await fetch_user(user_id)
        logger.info("Successfully retrieved user", user=user)
        # In production code, you might return or process the user data here
    except UserNotFoundError as exc:
        logger.error("User not found", user_id=user_id, error=str(exc))
    except APIClientError as exc:
        logger.error("API client error", user_id=user_id, error=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```
