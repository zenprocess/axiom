```python
"""
User data fetcher module for retrieving user information from a REST API.
"""

import os
from typing import Optional
from dataclasses import dataclass

import structlog
from pydantic import BaseModel, ValidationError
import httpx

log = structlog.get_logger()


class User(BaseModel):
    """User data model representing the API response."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None


class UserService:
    """Service class for fetching user data from the REST API."""

    def __init__(self, base_url: str = "https://api.example.com"):
        """
        Initialize the UserService with the base API URL.
        
        Args:
            base_url: The base URL for the API endpoints.
        """
        self.base_url = base_url

    async def fetch_user(self, user_id: int) -> User:
        """
        Fetch a user by ID from the API.
        
        Args:
            user_id: The unique identifier of the user to fetch.
            
        Returns:
            User: A User model instance with the fetched data.
            
        Raises:
            UserNotFoundError: If the user with the given ID doesn't exist.
            APIError: If there's an error communicating with the API.
        """
        url = f"{self.base_url}/users/{user_id}"
        
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=30.0)
                response.raise_for_status()
                
                try:
                    user_data = response.json()
                    return User(**user_data)
                except ValidationError as e:
                    log.error("invalid_user_data", user_id=user_id, error=str(e))
                    raise APIError(f"Invalid user data received for ID {user_id}") from e
                    
        except httpx.TimeoutException as e:
            log.error("api_timeout", url=url, user_id=user_id, error=str(e))
            raise APIError(f"Request to {url} timed out") from e
        except httpx.NetworkError as e:
            log.error("network_error", url=url, user_id=user_id, error=str(e))
            raise APIError(f"Network error while accessing {url}") from e
        except httpx.HTTPStatusError as e:
            if e.response.status_code == 404:
                log.info("user_not_found", user_id=user_id)
                raise UserNotFoundError(f"User with ID {user_id} not found") from e
            log.error("http_error", url=url, status_code=e.response.status_code, error=str(e))
            raise APIError(f"HTTP {e.response.status_code} error from API") from e


class UserNotFoundError(Exception):
    """Exception raised when a user is not found in the API."""
    pass


class APIError(Exception):
    """Exception raised for API communication errors."""
    pass


async def main() -> User:
    """
    Main function that fetches user #42 from the API.
    
    Returns:
        User: The fetched user data.
        
    Raises:
        UserNotFoundError: If user #42 doesn't exist.
        APIError: If there's an error fetching the user data.
    """
    # Get API base URL from environment variable or use default
    base_url = os.environ.get("API_BASE_URL", "https://api.example.com")
    
    service = UserService(base_url=base_url)
    return await service.fetch_user(42)


if __name__ == "__main__":
    import asyncio
    
    async def run():
        try:
            user = await main()
            log.info("user_fetched_successfully", user_id=user.id, name=user.name)
            print(user)
        except UserNotFoundError as e:
            log.error("user_not_found", error=str(e))
        except APIError as e:
            log.error("api_error", error=str(e))
    
    asyncio.run(run())
```
