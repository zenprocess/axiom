```python
"""Module for fetching user data from a REST API."""

import os
from typing import Optional
import httpx
import structlog
from pydantic import BaseModel

# Configure structlog
log = structlog.get_logger()


class User(BaseModel):
    """User data model representing the API response."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class UserService:
    """Service for fetching user data from the REST API."""

    def __init__(self, base_url: Optional[str] = None):
        """
        Initialize the UserService.

        Args:
            base_url: Base URL of the API. Defaults to https://api.example.com
        """
        self.base_url = base_url or "https://api.example.com"
        self._client: Optional[httpx.AsyncClient] = None

    async def __aenter__(self):
        """Async context manager entry."""
        self._client = httpx.AsyncClient()
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Async context manager exit."""
        if self._client:
            await self._client.aclose()

    async def get_user(self, user_id: int) -> User:
        """
        Fetch a user by ID from the API.

        Args:
            user_id: The ID of the user to fetch.

        Returns:
            User: The fetched user data.

        Raises:
            UserNotFoundError: If the user is not found (404).
            APIError: For other API errors.
        """
        url = f"{self.base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=30.0)
                response.raise_for_status()
                user_data = response.json()
                user = User(**user_data)
                log.info("user_fetched", user_id=user.id, name=user.name)
                return user

        except httpx.HTTPStatusError as e:
            if e.response.status_code == 404:
                log.error("user_not_found", user_id=user_id, status_code=e.response.status_code)
                raise UserNotFoundError(f"User with ID {user_id} not found") from e
            else:
                log.error("api_status_error", url=url, status_code=e.response.status_code, error=str(e))
                raise APIError(f"API returned status {e.response.status_code}") from e
        except httpx.RequestError as e:
            log.error("api_request_error", url=url, error=str(e))
            raise APIError(f"Request failed: {str(e)}") from e


class UserNotFoundError(Exception):
    """Exception raised when a user is not found."""
    pass


class APIError(Exception):
    """Exception raised for API errors."""
    pass


async def main():
    """Main function to fetch user #42."""
    try:
        async with UserService() as service:
            user = await service.get_user(42)
            log.info("user_fetched_successfully", user_id=user.id, name=user.name)
            return user
    except (UserNotFoundError, APIError) as e:
        log.error("failed_to_fetch_user", user_id=42, error=str(e))
        raise


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
