```python
"""
User data fetching module.

This module provides functionality to fetch user data from a REST API.
"""

import os
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class User(BaseModel):
    """User data model."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class APIClientError(Exception):
    """Custom exception for API client errors."""
    pass


class APIClient:
    """API client for fetching user data."""

    def __init__(self, base_url: str = "https://api.example.com"):
        """Initialize the API client.
        
        Args:
            base_url: Base URL for the API.
        """
        self.base_url = base_url.rstrip("/")

    async def fetch_user(self, user_id: int) -> User:
        """Fetch user data by ID.
        
        Args:
            user_id: The ID of the user to fetch.
            
        Returns:
            User object with the fetched data.
            
        Raises:
            APIClientError: If the API call fails or returns unexpected data.
        """
        url = f"{self.base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)
        
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=30.0)
                
                # Raise an exception for 4xx/5xx responses
                response.raise_for_status()
                
                # Parse and validate the response
                user_data = response.json()
                user = User(**user_data)
                
                log.info("user_fetched", user_id=user.id, name=user.name)
                return user
                
        except httpx.TimeoutException as e:
            log.error("api_timeout", url=url, error=str(e))
            raise APIClientError(f"Request timed out for user {user_id}") from e
        except httpx.RequestError as e:
            log.error("api_request_error", url=url, error=str(e))
            raise APIClientError(f"Request failed for user {user_id}") from e
        except httpx.HTTPStatusError as e:
            log.error("api_http_error", url=url, status_code=e.response.status_code, error=str(e))
            raise APIClientError(f"HTTP error {e.response.status_code} for user {user_id}") from e
        except Exception as e:
            log.error("unexpected_error", url=url, error=str(e))
            raise APIClientError(f"Unexpected error fetching user {user_id}") from e


async def main():
    """Main function to fetch user #42."""
    # Use environment variable for API base URL if available, otherwise use default
    base_url = os.environ.get("API_BASE_URL", "https://api.example.com")
    
    client = APIClient(base_url=base_url)
    
    try:
        user = await client.fetch_user(42)
        print(f"User #42: {user.name} ({user.email})")
        return user
    except APIClientError as e:
        log.error("failed_to_fetch_user", user_id=42, error=str(e))
        raise


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
