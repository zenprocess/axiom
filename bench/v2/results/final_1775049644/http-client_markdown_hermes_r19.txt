```python
"""Module for fetching user data from a REST API."""

import os
from typing import Optional
import structlog
from pydantic import BaseModel
import httpx

log = structlog.get_logger()


class User(BaseModel):
    """User data model representing the API response."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class UserFetcher:
    """Class for fetching user data from the REST API."""

    def __init__(self, base_url: str = "https://api.example.com"):
        """Initialize the user fetcher with the base API URL."""
        self.base_url = base_url.rstrip('/')

    async def fetch_user(self, user_id: int) -> User:
        """
        Fetch a user by their ID from the API.

        Args:
            user_id: The ID of the user to fetch.

        Returns:
            User: The fetched user data.

        Raises:
            UserFetcherError: If the API call fails or returns unexpected data.
        """
        url = f"{self.base_url}/users/{user_id}"
        
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=30.0)
                response.raise_for_status()
                user_data = response.json()
                
                # Validate and parse the response into our User model
                return User(**user_data)
                
        except httpx.HTTPStatusError as e:
            log.error(
                "http_status_error",
                url=url,
                status_code=e.response.status_code,
                error=str(e)
            )
            raise UserFetcherError(
                f"API returned status {e.response.status_code} for user {user_id}"
            ) from e
        except httpx.RequestError as e:
            log.error(
                "http_request_error",
                url=url,
                error=str(e)
            )
            raise UserFetcherError(
                f"Request failed for user {user_id}: {str(e)}"
            ) from e
        except Exception as e:
            log.error(
                "unexpected_error",
                url=url,
                error=str(e)
            )
            raise UserFetcherError(
                f"Unexpected error fetching user {user_id}: {str(e)}"
            ) from e


class UserFetcherError(Exception):
    """Custom exception for user fetcher errors."""
    pass


async def main():
    """Main function that fetches user #42."""
    # Get API base URL from environment or use default
    base_url = os.environ.get("API_BASE_URL", "https://api.example.com")
    
    fetcher = UserFetcher(base_url=base_url)
    
    try:
        user = await fetcher.fetch_user(42)
        log.info("user_fetched", user_id=user.id, name=user.name, email=user.email)
        return user
    except UserFetcherError as e:
        log.error("failed_to_fetch_user", user_id=42, error=str(e))
        raise


if __name__ == "__main__":
    import asyncio
    user = asyncio.run(main())
    print(f"User 42: {user.name} ({user.email})")
```
