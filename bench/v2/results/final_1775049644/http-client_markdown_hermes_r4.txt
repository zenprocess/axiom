```python
"""Module for fetching user data from the example REST API."""

import os
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class User(BaseModel):
    """User data model representing the API response."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class APIClientError(Exception):
    """Custom exception for API client errors."""
    pass


class APIClient:
    """Client for interacting with the example REST API."""

    def __init__(self, base_url: str = "https://api.example.com"):
        self.base_url = base_url.rstrip("/")
        self.timeout = 30.0

    async def fetch_user(self, user_id: int) -> User:
        """
        Fetch user data from the API by ID.

        Args:
            user_id: The ID of the user to fetch.

        Returns:
            User: The user data model.

        Raises:
            APIClientError: If the API call fails or returns an unexpected response.
        """
        url = f"{self.base_url}/users/{user_id}"
        
        try:
            async with httpx.AsyncClient(timeout=self.timeout) as client:
                response = await client.get(url)
                response.raise_for_status()
                
                data = response.json()
                return User(**data)
                
        except httpx.HTTPStatusError as e:
            log.error(
                "http_status_error",
                url=url,
                status_code=e.response.status_code,
                error=str(e)
            )
            raise APIClientError(f"API returned status {e.response.status_code}") from e
        except httpx.TimeoutException as e:
            log.error("timeout_error", url=url, error=str(e))
            raise APIClientError(f"Request timed out after {self.timeout} seconds") from e
        except httpx.RequestError as e:
            log.error("request_error", url=url, error=str(e))
            raise APIClientError(f"Network error occurred: {str(e)}") from e
        except Exception as e:
            log.error("unexpected_error", url=url, error=str(e))
            raise APIClientError(f"Unexpected error occurred: {str(e)}") from e


async def main() -> User:
    """
    Main function that fetches user #42 from the API.

    Returns:
        User: The fetched user data.

    Raises:
        APIClientError: If fetching the user fails.
    """
    client = APIClient()
    return await client.fetch_user(42)


if __name__ == "__main__":
    import asyncio
    
    try:
        user = asyncio.run(main())
        print(f"Fetched user: {user}")
    except APIClientError as e:
        log.error("main_fetch_failed", error=str(e))
        raise
```

4. Implements comprehensive error handling with try/except blocks
5. Includes type hints for all functions
6. Uses custom exception `APIClientError` for API-related errors
7. Includes docstrings for all public functions
8. No hardcoded secrets (the base URL is configurable)
9. Follows all the specified development rules

The module is designed to be used as both a standalone script (fetching user #42 when run directly) and as an importable module for integration into larger applications.
