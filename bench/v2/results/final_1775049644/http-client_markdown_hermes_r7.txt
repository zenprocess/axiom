```python
"""Module for fetching user data from a REST API."""

import os
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class User(BaseModel):
    """User data model representing the API response."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class APIClientError(Exception):
    """Custom exception for API client errors."""
    pass


class UserAPIClient:
    """Client for fetching user data from the REST API."""
    
    def __init__(self, base_url: str = "https://api.example.com"):
        """Initialize the API client.
        
        Args:
            base_url: The base URL for the API.
        """
        self.base_url = base_url.rstrip('/')
    
    async def fetch_user(self, user_id: int) -> User:
        """Fetch a user by ID from the API.
        
        Args:
            user_id: The ID of the user to fetch.
            
        Returns:
            User: The fetched user data.
            
        Raises:
            APIClientError: If the API request fails or returns an unexpected response.
        """
        url = f"{self.base_url}/users/{user_id}"
        
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=30.0)
                response.raise_for_status()
                
                user_data = response.json()
                return User(**user_data)
                
        except httpx.TimeoutException as e:
            log.error("api_request_timeout", url=url, error=str(e))
            raise APIClientError(f"Request to {url} timed out") from e
        except httpx.HTTPStatusError as e:
            log.error("api_request_failed", url=url, status_code=e.response.status_code, error=str(e))
            raise APIClientError(f"API returned status code {e.response.status_code}: {e.response.text}") from e
        except httpx.RequestError as e:
            log.error("api_request_error", url=url, error=str(e))
            raise APIClientError(f"Request error for {url}: {str(e)}") from e
        except Exception as e:
            log.error("unexpected_error", url=url, error=str(e))
            raise APIClientError(f"Unexpected error while fetching user {user_id}: {str(e)}") from e


async def main():
    """Main function to fetch user #42."""
    client = UserAPIClient()
    
    try:
        user = await client.fetch_user(42)
        log.info("user_fetched", user_id=user.id, name=user.name, email=user.email)
        return user
    except APIClientError as e:
        log.error("failed_to_fetch_user", user_id=42, error=str(e))
        raise


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
