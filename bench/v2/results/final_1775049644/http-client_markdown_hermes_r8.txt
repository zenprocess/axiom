```python
"""User data fetcher module for retrieving user information from a REST API."""

import os
from typing import Optional
from dataclasses import dataclass

import structlog
from pydantic import BaseModel, Field
import httpx

log = structlog.get_logger()


class User(BaseModel):
    """User data model representing a user from the API."""
    id: int = Field(..., description="Unique identifier for the user")
    name: str = Field(..., description="Full name of the user")
    email: str = Field(..., description="Email address of the user")
    username: Optional[str] = Field(None, description="Username of the user")
    phone: Optional[str] = Field(None, description="Phone number of the user")
    website: Optional[str] = Field(None, description="Website of the user")


class UserFetcher:
    """Class responsible for fetching user data from the REST API."""

    def __init__(self, base_url: str = "https://api.example.com"):
        """
        Initialize the UserFetcher with the base API URL.

        Args:
            base_url: The base URL of the API (default: "https://api.example.com")
        """
        self.base_url = base_url
        self._client: Optional[httpx.AsyncClient] = None

    async def __aenter__(self):
        """Async context manager entry."""
        self._client = httpx.AsyncClient()
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Async context manager exit."""
        if self._client:
            await self._client.aclose()
        self._client = None

    async def fetch_user(self, user_id: int) -> User:
        """
        Fetch a user by ID from the API.

        Args:
            user_id: The ID of the user to fetch

        Returns:
            User: The user data model

        Raises:
            UserNotFoundError: If the user is not found (404)
            APIError: If there's a server error (5xx)
            ConnectionError: If there's a network/connection error
        """
        url = f"{self.base_url}/users/{user_id}"
        
        try:
            log.info("fetching_user", user_id=user_id, url=url)
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=30.0)
                
            response.raise_for_status()
            
            user_data = response.json()
            user = User(**user_data)
            
            log.info("user_fetched", user_id=user_id, user_name=user.name)
            return user
            
        except httpx.HTTPStatusError as exc:
            if exc.response.status_code == 404:
                log.error("user_not_found", user_id=user_id, status_code=exc.response.status_code)
                raise UserNotFoundError(f"User with ID {user_id} not found") from exc
            elif 500 <= exc.response.status_code < 600:
                log.error("server_error", user_id=user_id, status_code=exc.response.status_code)
                raise APIError(f"Server error: {exc.response.status_code}") from exc
            else:
                log.error("http_error", user_id=user_id, status_code=exc.response.status_code, error=str(exc))
                raise APIError(f"HTTP error: {exc.response.status_code}") from exc
        except httpx.RequestError as exc:
            log.error("connection_error", user_id=user_id, error=str(exc))
            raise ConnectionError(f"Connection error: {str(exc)}") from exc
        except Exception as exc:
            log.error("unexpected_error", user_id=user_id, error=str(exc))
            raise APIError(f"Unexpected error: {str(exc)}") from exc


class UserNotFoundError(Exception):
    """Exception raised when a user is not found."""
    pass


class APIError(Exception):
    """Exception raised for API-related errors."""
    pass


async def fetch_user_42():
    """
    Fetch user #42 from the API using the UserFetcher class.
    
    Returns:
        User: The user data model for user #42
        
    Raises:
        UserNotFoundError: If user #42 is not found
        APIError: If there's an API error
        ConnectionError: If there's a network error
    """
    async with UserFetcher() as fetcher:
        return await fetcher.fetch_user(42)


async def main():
    """Main function that fetches user #42 and prints the result."""
    try:
        user = await fetch_user_42()
        log.info("user_retrieved_successfully", user_id=user.id, name=user.name)
        return user
    except UserNotFoundError as e:
        log.error("user_not_found_error", error=str(e))
        raise
    except APIError as e:
        log.error("api_error", error=str(e))
        raise
    except ConnectionError as e:
        log.error("connection_error", error=str(e))
        raise


if __name__ == "__main__":
    import asyncio
    user = asyncio.run(main())
    print(user)
```
