```python
"""
User data fetching module.

This module provides functionality to fetch user data from a REST API.
"""

from typing import Optional
from pydantic import BaseModel, Field
import httpx


class User(BaseModel):
    """
    Pydantic model representing a user.
    
    Attributes:
        id: User ID
        name: User's full name
        username: User's username/handle
        email: User's email address
        address: User's address information
        phone: User's phone number
        website: User's website URL
        company: User's company information
    """
    id: int
    name: str
    username: str
    email: str
    address: dict = Field(default_factory=dict)
    phone: str = ""
    website: str = ""
    company: dict = Field(default_factory=dict)


class UserAPIError(Exception):
    """Custom exception for user API errors."""
    pass


class UserClient:
    """
    Client for fetching user data from the REST API.
    
    Attributes:
        base_url: The base URL for the API
        client: HTTPX async client instance
    """
    
    def __init__(self, base_url: str = "https://api.example.com"):
        """
        Initialize the UserClient.
        
        Args:
            base_url: The base URL for the API
        """
        self.base_url = base_url.rstrip("/")
        self.client = None
    
    async def __aenter__(self):
        """Async context manager entry."""
        self.client = httpx.AsyncClient()
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        """Async context manager exit."""
        if self.client:
            await self.client.aclose()
    
    async def fetch_user(self, user_id: int) -> User:
        """
        Fetch user data from the API.
        
        Args:
            user_id: The ID of the user to fetch
            
        Returns:
            User: The user data
            
        Raises:
            UserAPIError: If the API request fails or returns an error
        """
        if not self.client:
            self.client = httpx.AsyncClient()
            close_client = True
        else:
            close_client = False
        
        try:
            url = f"{self.base_url}/users/{user_id}"
            response = await self.client.get(url, timeout=10.0)
            
            if response.status_code == 404:
                raise UserAPIError(f"User with ID {user_id} not found")
            elif response.status_code != 200:
                raise UserAPIError(f"API returned status code {response.status_code}")
            
            user_data = response.json()
            return User(**user_data)
            
        except httpx.RequestError as exc:
            raise UserAPIError(f"Request error: {exc}") from exc
        except httpx.HTTPStatusError as exc:
            raise UserAPIError(f"HTTP error: {exc}") from exc
        except Exception as exc:
            raise UserAPIError(f"Unexpected error: {exc}") from exc
        finally:
            if close_client and self.client:
                await self.client.aclose()


async def main():
    """
    Main function to demonstrate fetching user #42.
    
    Returns:
        User: The fetched user data or None if an error occurred
    """
    client = UserClient()
    
    try:
        async with client:
            user = await client.fetch_user(42)
            return user
    except UserAPIError as exc:
        # In a real application, you'd use structlog for logging here
        print(f"Error fetching user: {exc}")
        return None


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
