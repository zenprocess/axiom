```python
"""User data fetching module using Pydantic models and async HTTP client."""

from typing import Optional
from pydantic import BaseModel
import httpx
import structlog

# Configure structlog
logger = structlog.get_logger()


class User(BaseModel):
    """User data model representing the API response structure."""
    id: int
    name: str
    username: str
    email: str
    address: dict
    phone: str
    website: str
    company: dict


class UserService:
    """Service for fetching user data from the REST API."""
    
    def __init__(self, base_url: str = "https://api.example.com"):
        """Initialize the service with the base API URL.
        
        Args:
            base_url: The base URL for the API endpoints.
        """
        self.base_url = base_url
    
    async def fetch_user(self, user_id: int) -> User:
        """Fetch a user by ID from the API.
        
        Args:
            user_id: The ID of the user to fetch.
            
        Returns:
            User: The fetched user data.
            
        Raises:
            httpx.HTTPError: If the HTTP request fails.
            ValueError: If the user is not found or the response is invalid.
        """
        url = f"{self.base_url}/users/{user_id}"
        
        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url)
                response.raise_for_status()
                
                user_data = response.json()
                user = User(**user_data)
                
                logger.info("user_fetched", user_id=user_id, user_name=user.name)
                return user
                
        except httpx.HTTPStatusError as exc:
            if exc.response.status_code == 404:
                logger.error("user_not_found", user_id=user_id, status_code=404)
                raise ValueError(f"User with ID {user_id} not found") from exc
            else:
                logger.error("http_error", user_id=user_id, status_code=exc.response.status_code)
                raise
        except httpx.RequestError as exc:
            logger.error("request_error", user_id=user_id, error=str(exc))
            raise
        except Exception as exc:
            logger.error("unexpected_error", user_id=user_id, error=str(exc))
            raise


async def main():
    """Main function to demonstrate fetching user #42."""
    service = UserService()
    
    try:
        user = await service.fetch_user(42)
        logger.info("user_data", user=user.dict())
        return user
    except (ValueError, httpx.HTTPError, httpx.RequestError) as exc:
        logger.error("failed_to_fetch_user", user_id=42, error=str(exc))
        raise


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
