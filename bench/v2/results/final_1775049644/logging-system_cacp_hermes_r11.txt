```python
"""Structured logging system for web applications using structlog."""

import json
import os
import sys
import time
import uuid
from typing import Any, Dict, Optional

import structlog
from structlog.processors import (
    JSONRenderer,
    add_log_level,
    format_exc_info,
    TimeStamper,
)
from structlog.stdlib import BoundLogger, add_logger_context

# Configuration for log rotation
LOG_FILE_PATH = os.getenv("LOG_FILE_PATH", "app.log")
MAX_LOG_SIZE = int(os.getenv("MAX_LOG_SIZE", "10485760"))  # 10 MB
BACKUP_COUNT = int(os.getenv("BACKUP_COUNT", "5"))


def _request_id_processor(logger: BoundLogger, method_name: str, event_dict: Dict[str, Any]) -> Dict[str, Any]:
    """Add request correlation ID to log event."""
    request_id = event_dict.get("request_id")
    if not request_id:
        # Generate new correlation ID if not present
        request_id = str(uuid.uuid4())
    event_dict["request_id"] = request_id
    return event_dict


def _timestamp_processor(logger: BoundLogger, method_name: str, event_dict: Dict[str, Any]) -> Dict[str, Any]:
    """Add timestamp to log event."""
    event_dict["timestamp"] = time.strftime("%Y-%m-%dT%H:%M:%S", time.localtime())
    return event_dict


def configure_logging() -> None:
    """Configure structlog with JSON output, log levels, and rotation."""
    # Configure structlog processors
    structlog.configure(
        processors=[
            add_log_level,
            _request_id_processor,
            _timestamp_processor,
            format_exc_info,
            TimeStamper(fmt="iso"),
            JSONRenderer(),
        ],
        wrapper_class=BoundLogger,
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(file=sys.stdout),
        cache_logger_on_first_use=True,
    )


def get_logger(name: Optional[str] = None) -> BoundLogger:
    """
    Get a structlog BoundLogger instance.

    Args:
        name: Optional logger name for contextual logging

    Returns:
        BoundLogger instance configured with JSON output
    """
    return structlog.get_logger(name)


class RotatingFileHandler:
    """Custom rotating file handler for log files."""

    def __init__(self, filename: str, max_bytes: int = MAX_LOG_SIZE, backup_count: int = BACKUP_COUNT):
        """
        Initialize the rotating file handler.

        Args:
            filename: Path to the log file
            max_bytes: Maximum size of log file before rotation
            backup_count: Number of backup files to keep
        """
        self.filename = filename
        self.max_bytes = max_bytes
        self.backup_count = backup_count
        self.current_size = 0
        self._ensure_directory_exists()

    def _ensure_directory_exists(self) -> None:
        """Ensure the log directory exists."""
        directory = os.path.dirname(self.filename)
        if directory and not os.path.exists(directory):
            os.makedirs(directory, exist_ok=True)

    def _rotate(self) -> None:
        """Rotate log files."""
        # Delete oldest backup if it exists
        oldest_backup = f"{self.filename}.{self.backup_count}"
        if os.path.exists(oldest_backup):
            os.remove(oldest_backup)

        # Shift existing backups
        for i in range(self.backup_count - 1, 0, -1):
            current_backup = f"{self.filename}.{i}"
            next_backup = f"{self.filename}.{i + 1}"
            if os.path.exists(current_backup):
                os.rename(current_backup, next_backup)

        # Rotate current log file
        if os.path.exists(self.filename):
            os.rename(self.filename, f"{self.filename}.1")

    def write(self, data: str) -> None:
        """
        Write log data to file, rotating if necessary.

        Args:
            data: Log data to write
        """
        try:
            # Check if rotation is needed
            if os.path.exists(self.filename):
                current_size = os.path.getsize(self.filename)
                if current_size + len(data) > self.max_bytes:
                    self._rotate()

            # Write to file
            with open(self.filename, "a", encoding="utf-8") as f:
                f.write(data)
                f.write("\n")
        except (OSError, IOError) as e:
            # Log error to stderr if file writing fails
            structlog.get_logger().error("file_write_error", error=str(e))


class StructuredLogger:
    """Structured logger with JSON output and request correlation."""

    def __init__(self):
        """Initialize the structured logger."""
        self.logger = get_logger()
        self._file_handler = RotatingFileHandler(LOG_FILE_PATH)

    def _log(self, level: str, message: str, **kwargs) -> None:
        """
        Log a message with the given level.

        Args:
            level: Log level (debug, info, warning, error, critical)
            message: Log message
            **kwargs: Additional context fields
        """
        # Extract request_id if provided
        request_id = kwargs.pop("request_id", None)
        
        # Log using structlog
        try:
            getattr(self.logger, level)(
                message=message,
                request_id=request_id,
                **kwargs
            )
            
            # Also write to file for persistence
            log_entry = {
                "level": level,
                "message": message,
                "request_id": request_id,
                **kwargs
            }
            self._file_handler.write(json.dumps(log_entry))
        except (TypeError, ValueError) as e:
            # Handle serialization errors
            getattr(self.logger, level)(
                message="log_serialization_error",
                error=str(e),
                original_message=message,
                original_level=level,
            )

    def debug(self, message: str, **kwargs) -> None:
        """
        Log a debug message.

        Args:
            message: Debug message
            **kwargs: Additional context fields
        """
        self._log("debug", message, **kwargs)

    def info(self, message: str, **kwargs) -> None:
        """
        Log an info message.

        Args:
            message: Info message
            **kwargs: Additional context fields
        """
        self._log("info", message, **kwargs)

    def warning(self, message: str, **kwargs) -> None:
        """
        Log a warning message.

        Args:
            message: Warning message
            **kwargs: Additional context fields
        """
        self._log("warning", message, **kwargs)

    def error(self, message: str, **kwargs) -> None:
        """
        Log an error message.

        Args:
            message: Error message
            **kwargs: Additional context fields
        """
        self._log("error", message, **kwargs)

    def critical(self, message: str, **kwargs) -> None:
        """
        Log a critical message.

        Args:
            message: Critical message
            **kwargs: Additional context fields
        """
        self._log("critical", message, **kwargs)


# Global logger instance
logger = StructuredLogger()
```
