Here's a structured logging system for a web application that complies with all your specified rules:

```python
import json
import os
import sys
from datetime import datetime
from typing import Any, Dict, Optional

import structlog
from pydantic import BaseModel
from structlog.processors import JSONRenderer
from structlog.stdlib import BoundLogger
from structlog.typing import EventDict, Processor


class LogConfig(BaseModel):
    """Configuration for the logging system."""
    log_level: str = "INFO"
    log_file: Optional[str] = None
    max_bytes: int = 10_485_760  # 10 MB
    backup_count: int = 5
    enable_correlation_id: bool = True


class CorrelationIDProcessor:
    """Processor to inject correlation ID into log events."""
    def __call__(self, logger: BoundLogger, method_name: str, event_dict: EventDict) -> EventDict:
        correlation_id = event_dict.get("correlation_id")
        if not correlation_id:
            # Generate a correlation ID if not provided (e.g., for internal logs)
            import uuid
            correlation_id = str(uuid.uuid4())
            event_dict["correlation_id"] = correlation_id
        return event_dict


class TimestampProcessor:
    """Processor to add ISO-formatted timestamp to log events."""
    def __call__(self, logger: BoundLogger, method_name: str, event_dict: EventDict) -> EventDict:
        event_dict["timestamp"] = datetime.utcnow().isoformat() + "Z"
        return event_dict


def _format_exception_for_log(event_dict: EventDict) -> EventDict:
    """Ensure exceptions are properly serialized for JSON output."""
    if "exception" in event_dict:
        exc = event_dict.pop("exception", None)
        if exc is not None:
            event_dict["exception"] = str(exc)
    return event_dict


def configure_logging(config: LogConfig) -> None:
    """Configure the logging system with JSON output and rotation."""
    # Determine log level
    level = getattr(structlog, config.log_level.upper(), None)
    if level is None:
        level = structlog.LOG_LEVELS["info"]
    else:
        level = structlog.LOG_LEVELS[config.log_level.lower()]

    # Configure processors
    processors = [
        structlog.contextvars.merge_contextvars,
        CorrelationIDProcessor(),
        TimestampProcessor(),
        structlog.stdlib.add_log_level,
        structlog.stdlib.add_logger_name,
        structlog.dev.set_exc_info,
        structlog.processors.format_exc_info,
        _format_exception_for_log,
        JSONRenderer(),
    ]

    # Set up output destination
    if config.log_file:
        from logging.handlers import RotatingFileHandler
        import logging

        # Create file handler with rotation
        handler = RotatingFileHandler(
            config.log_file,
            maxBytes=config.max_bytes,
            backupCount=config.backup_count,
            encoding="utf-8",
        )
        handler.setFormatter(structlog.stdlib.ProcessorFormatter(processor=JSONRenderer()))

        # Configure stdlib logging to use structlog
        logging.basicConfig(
            handlers=[handler],
            level=level,
            force=True,
        )
        structlog.configure(
            processors=processors,
            wrapper_class=structlog.stdlib.BoundLogger,
            context_class=dict,
            logger_factory=structlog.stdlib.LoggerFactory(),
            cache_logger_on_first_use=True,
        )
    else:
        # Console output (for containerized environments)
        structlog.configure(
            processors=processors,
            wrapper_class=structlog.stdlib.BoundLogger,
            context_class=dict,
            logger_factory=structlog.stdlib.LoggerFactory(),
            cache_logger_on_first_use=True,
        )


def get_logger(name: str = "app") -> BoundLogger:
    """Get a structured logger instance with correlation ID support."""
    return structlog.get_logger(name)


def bind_correlation_id(correlation_id: str) -> None:
    """Bind a correlation ID to the current context for request logging."""
    structlog.contextvars.bind_contextvars(correlation_id=correlation_id)


def clear_correlation_id() -> None:
    """Clear the correlation ID from the current context."""
    structlog.contextvars.clear_contextvars()
```

### Usage Example:

```python
# main.py
from pathlib import Path
from typing import Optional

from my_logging_module import LogConfig, configure_logging, get_logger, bind_correlation_id

# Configure logging on app startup
config = LogConfig(
    log_level="INFO",
    log_file=str(Path("logs/app.log")),
    max_bytes=5_000_000,
    backup_count=3,
)
configure_logging(config)

# In your request handler
def handle_request(request_id: str, data: dict) -> dict:
    # Bind correlation ID for this request
    bind_correlation_id(request_id)
    logger = get_logger("request_handler")
    
    logger.info("Processing request", request_id=request_id, endpoint="/api/data")
    
    try:
        # Process data
        result = {"status": "success", "processed_data": data}
        logger.info("Request completed successfully", result=result)
        return result
    except Exception as e:
        logger.exception("Request failed", error=str(e))
        return {"status": "error", "message": "Internal server error"}
```

- ✅ **Async-Ready**: Uses `structlog.contextvars` for thread-local correlation IDs
