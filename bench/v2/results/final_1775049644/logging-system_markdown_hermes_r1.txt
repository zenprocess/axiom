Here's a production-ready logging system using `structlog` that meets all the requirements:

```python
import logging
import logging.handlers
import os
import sys
import time
from typing import Optional

import structlog
from pydantic import BaseModel, Field


class LogConfig(BaseModel):
    """Configuration for the logging system."""
    level: str = Field(default="INFO", description="Logging level (DEBUG, INFO, WARNING, ERROR, CRITICAL)")
    log_file: Optional[str] = Field(default=None, description="Path to log file. If None, logs to stdout")
    max_bytes: int = Field(default=10 * 1024 * 1024, description="Max log file size in bytes before rotation")
    backup_count: int = Field(default=5, description="Number of backup log files to keep")
    format: str = Field(default="json", description="Log format: 'json' or 'text'")
    include_timestamp: bool = Field(default=True, description="Include timestamp in logs")


def configure_logging(config: LogConfig = None) -> None:
    """
    Configure the structlog-based logging system for the application.
    
    Args:
        config: LogConfig instance with logging configuration. If None, uses defaults.
    """
    config = config or LogConfig()
    
    # Set up structlog processors
    processors = [
        structlog.contextvars.merge_contextvars,
        structlog.stdlib.add_log_level,
        structlog.stdlib.add_logger_name,
        structlog.processors.TimeStamper(fmt="iso", utc=True),
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.processors.UnicodeDecoder(),
    ]
    
    # Add request context processor if available
    try:
        from your_web_framework import get_request_context
        processors.append(lambda _, __, event_dict: {
            **event_dict,
            **get_request_context()
        })
    except ImportError:
        # Fallback if web framework context is not available
        pass
    
    # Set up log format
    if config.format == "json":
        renderer = structlog.processors.JSONRenderer()
    else:
        renderer = structlog.dev.ConsoleRenderer()
    
    processors.append(renderer)
    
    # Configure structlog
    structlog.configure(
        processors=processors,
        context_class=dict,
        logger_factory=structlog.stdlib.LoggerFactory(),
        cache_logger_on_first_use=True,
    )
    
    # Set up standard library logging handlers
    handlers = []
    
    if config.log_file:
        # File handler with rotation
        file_handler = logging.handlers.RotatingFileHandler(
            config.log_file,
            maxBytes=config.max_bytes,
            backupCount=config.backup_count,
            encoding='utf-8'
        )
        file_handler.setLevel(getattr(logging, config.level.upper()))
        file_handler.setFormatter(logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        ))
        handlers.append(file_handler)
    
    # Console handler (stdout)
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setLevel(getattr(logging, config.level.upper()))
    console_handler.setFormatter(logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    ))
    handlers.append(console_handler)
    
    # Configure root logger
    root_logger = logging.getLogger()
    root_logger.setLevel(getattr(logging, config.level.upper()))
    
    # Clear existing handlers to avoid duplicates
    root_logger.handlers = handlers
    
    # Configure structlog to use standard library logging
    structlog.stdlib._CONFIG['logger_factory'] = structlog.stdlib.LoggerFactory()


# Convenience functions for common logging operations
def get_logger(name: str = "app") -> structlog.BoundLogger:
    """
    Get a logger instance with the given name.
    
    Args:
        name: Logger name (typically module name)
        
    Returns:
        BoundLogger instance
    """
    return structlog.get_logger(name)


def log_request_start(
    logger: structlog.BoundLogger,
    method: str,
    path: str,
    request_id: str,
    client_ip: Optional[str] = None,
    user_agent: Optional[str] = None
) -> None:
    """
    Log the start of a request with correlation context.
    
    Args:
        logger: Logger instance
        method: HTTP method
        path: Request path
        request_id: Unique request ID for correlation
        client_ip: Client IP address (optional)
        user_agent: User agent string (optional)
    """
    event_dict = {
        "request_id": request_id,
        "method": method,
        "path": path,
        "client_ip": client_ip,
        "user_agent": user_agent,
    }
    logger.info("request_started", **event_dict)


def log_request_end(
    logger: structlog.BoundLogger,
    request_id: str,
    status_code: int,
    duration_ms: float,
    error: Optional[str] = None
) -> None:
    """
    Log the end of a request with timing and error information.
    
    Args:
        logger: Logger instance
        request_id: Unique request ID for correlation
        status_code: HTTP response status code
        duration_ms: Request duration in milliseconds
        error: Error message if request failed (optional)
    """
    event_dict = {
        "request_id": request_id,
        "status_code": status_code,
        "duration_ms": round(duration_ms, 2),
    }
    if error:
        event_dict["error"] = error
        logger.error("request_ended_with_error", **event_dict)
    else:
        logger.info("request_ended", **event_dict)


# Example usage
if __name__ == "__main__":
    # Configure logging
    config = LogConfig(
        level="DEBUG",
        log_file="logs/app.log",
        format="json",
        include_timestamp=True
    )
    configure_logging(config)
    
    # Get logger instance
    log = get_logger(__name__)
    
    # Example usage in a request handler
    def example_request_handler(request):
        request_id = request.headers.get("X-Request-ID", str(time.time()))
        log_request_start(
            log,
            method=request.method,
            path=request.path,
            request_id=request_id,
            client_ip=request.remote_addr,
            user_agent=request.user_agent
        )
        
        start_time = time.time()
        try:
            # Simulate request processing
            result = process_request(request)
            duration = (time.time() - start_time) * 1000  # ms
            log_request_end(log, request_id, 200, duration)
            return result
        except Exception as e:
            duration = (time.time() - start_time) * 1000  # ms
            log_request_end(log, request_id, 500, duration, error=str(e))
            raise
    
    # Regular logging with context
    log.info("application_started", version="1.0.0", environment="production")
    
    # Logging with context variables
    log = log.bind(user_id=123, session_id="abc123")
    log.info("user_action", action="login", success=True)
    
    # Logging with exception info
    try:
        raise ValueError("Test error")
    except Exception:
        log.exception("unexpected_error", operation="test_operation")
```

## Key Features

1. **Structured JSON Output**:
   - Uses `structlog` for structured logging with JSON renderer
   - Includes timestamps, log levels, logger names, and context data

2. **Request Correlation**:
   - Context variables for request-scoped data
   - Helper functions for logging request start/end with correlation IDs

3. **Log Rotation**:
   - `RotatingFileHandler` with configurable max file size and backup count
   - Supports both file and console output

4. **Configuration**:
   - Pydantic model for type-safe configuration
   - Support for different log levels, formats (JSON/text), and output destinations

5. **Error Handling**:
   - Proper exception formatting with `format_exc_info`
   - Helper functions for logging request lifecycle events

## Usage in Web Frameworks

For integration with popular frameworks:

**FastAPI**:
```python
from fastapi import FastAPI, Request

app = FastAPI()

@app.middleware("http")
async def log_requests(request: Request, call_next):
    request_id = request.headers.get("X-Request-ID", str(time.time()))
    request.state.request_id = request_id
    
    log = get_logger("app").bind(request_id=request_id)
    log_request_start(log, request.method, str(request.url), request_id)
    
    start_time = time.time()
    try:
        response = await call_next(request)
        duration = (time.time() - start_time) * 1000
        log_request_end(log, request_id, response.status_code, duration)
        return response
    except Exception as e:
        duration = (time.time() - start_time) * 1000
        log_request_end(log, request_id, 500, duration, error=str(e))
        raise
```

**Flask**:
```python
from flask import Flask, request, g

app = Flask(__name__)

@app.before_request
def before_request():
    g.request_id = request.headers.get("X-Request-ID", str(time.time()))
    log = get_logger("app").bind(request_id=g.request_id)
    log_request_start(log, request.method, request.path, g.request_id)

@app.after_request
def after_request(response):
    log = get_logger("app").bind(request_id=g.request_id)
    duration = (time.time() - g.request_start_time) * 1000
    log_request_end(log, g.request_id, response.status_code, duration)
    return response

@app.errorhandler(Exception)
def handle_exception(e):
    log = get_logger("app").bind(request_id=g.request_id)
    duration = (time.time() - g.request_start_time) * 1000
    log_request_end(log, g.request_id, 500, duration, error=str(e))
    return {"error": "Internal server error"}, 500
```
