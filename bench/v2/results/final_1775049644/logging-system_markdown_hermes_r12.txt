Here's a production-ready Python logging system for web applications using `structlog`, with support for structured JSON output, log levels, request correlation IDs, and log rotation.

```python
import logging
import os
import sys
from datetime import datetime
from typing import Optional
from pathlib import Path

import structlog
from structlog.processors import (
    TimeStamper,
    JSONRenderer,
    add_log_level,
    StackInfoRenderer,
)
from structlog.stdlib import (
    filter_by_level,
    add_logger_name,
    BoundLogger,
)
from structlog.threadlocal import wrap_dict

# Configure structlog processors
LOG_LEVEL = os.getenv("LOG_LEVEL", "INFO").upper()

# Initialize thread-local context storage for correlation IDs
structlog.configure(
    context_class=wrap_dict(dict),
    processors=[
        add_log_level,
        add_logger_name,
        TimeStamper(fmt="iso"),
        StackInfoRenderer(),
        filter_by_level,
        JSONRenderer(),
    ],
    wrapper_class=BoundLogger,
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=True,
)

# Global log instance
log = structlog.get_logger()


def get_correlation_id() -> Optional[str]:
    """Get the current request correlation ID from thread-local context."""
    context = structlog.get_context(log)
    return context.get("correlation_id")


def set_correlation_id(correlation_id: str) -> None:
    """Set the correlation ID for the current request/thread."""
    structlog.get_context(log).update(correlation_id=correlation_id)


def clear_correlation_id() -> None:
    """Clear the correlation ID from the current request/thread."""
    structlog.get_context(log).pop("correlation_id", None)


class RequestLoggingMiddleware:
    """
    ASGI/WSGI middleware to add request correlation IDs and context to logs.
    
    Usage:
        # For FastAPI/Starlette:
        app.add_middleware(RequestLoggingMiddleware)
        
        # For Flask:
        @app.before_request
        def before_request():
            RequestLoggingMiddleware.start_request()
            
        @app.after_request
        def after_request(response):
            RequestLoggingMiddleware.end_request(response)
            return response
    """

    def __init__(self, app):
        self.app = app

    async def __call__(self, scope, receive, send):
        correlation_id = self._generate_correlation_id(scope)
        set_correlation_id(correlation_id)
        
        try:
            await self.app(scope, receive, send)
        finally:
            clear_correlation_id()

    @staticmethod
    def _generate_correlation_id(scope) -> str:
        """Generate or extract correlation ID from request headers."""
        headers = dict(scope.get("headers", []))
        return headers.get(b"x-request-id", b"").decode() or RequestLoggingMiddleware._new_id()

    @staticmethod
    def _new_id() -> str:
        """Generate a new correlation ID."""
        return f"req-{datetime.now().strftime('%Y%m%d%H%M%S')}-{os.urandom(4).hex()}"

    @staticmethod
    def start_request():
        """Flask/Django-style entry point for request logging."""
        correlation_id = RequestLoggingMiddleware._new_id()
        set_correlation_id(correlation_id)
        log.info("request_started", correlation_id=correlation_id)

    @staticmethod
    def end_request(response):
        """Flask/Django-style exit point for request logging."""
        correlation_id = get_correlation_id()
        log.info(
            "request_completed",
            correlation_id=correlation_id,
            status_code=getattr(response, "status_code", 200),
        )
        clear_correlation_id()


def setup_logging(
    log_file: Optional[str] = None,
    log_level: Optional[str] = None,
    max_bytes: int = 10 * 1024 * 1024,  # 10MB
    backup_count: int = 5,
) -> None:
    """
    Configure logging system with rotation and structured output.
    
    Args:
        log_file: Path to log file. If None, logs to stdout only.
        log_level: Logging level (DEBUG, INFO, WARNING, ERROR, CRITICAL)
        max_bytes: Maximum size of log file before rotation (in bytes)
        backup_count: Number of backup files to keep
    """
    # Update log level if provided
    if log_level:
        os.environ["LOG_LEVEL"] = log_level.upper()
        structlog.configure(
            processors=[
                add_log_level,
                add_logger_name,
                TimeStamper(fmt="iso"),
                StackInfoRenderer(),
                filter_by_level,
                JSONRenderer(),
            ],
            wrapper_class=BoundLogger,
            context_class=dict,
            logger_factory=structlog.PrintLoggerFactory(),
            cache_logger_on_first_use=True,
        )

    # Configure file handler for log rotation
    if log_file:
        log_path = Path(log_file)
        log_path.parent.mkdir(parents=True, exist_ok=True)
        
        file_handler = logging.handlers.RotatingFileHandler(
            log_file,
            maxBytes=max_bytes,
            backupCount=backup_count,
            encoding="utf-8",
        )
        
        # Format for file handler (JSON for structured logging)
        file_handler.setFormatter(
            structlog.stdlib.ProcessorFormatter(
                processor=JSONRenderer(),
                foreign_pre_chain=[
                    add_log_level,
                    add_logger_name,
                    TimeStamper(fmt="iso"),
                    StackInfoRenderer(),
                ],
            )
        )
        
        # Get structlog's logger and add file handler
        structlog_logger = logging.getLogger("structlog")
        structlog_logger.addHandler(file_handler)
        structlog_logger.setLevel(getattr(logging, os.environ["LOG_LEVEL"]))

    # Ensure stdout handler exists for development
    if not log_file or os.getenv("LOG_TO_STDOUT", "true").lower() == "true":
        console_handler = logging.StreamHandler(sys.stdout)
        console_handler.setFormatter(
            structlog.stdlib.ProcessorFormatter(
                processor=JSONRenderer(),
                foreign_pre_chain=[
                    add_log_level,
                    add_logger_name,
                    TimeStamper(fmt="iso"),
                    StackInfoRenderer(),
                ],
            )
        )
        
        structlog_logger = logging.getLogger("structlog")
        structlog_logger.addHandler(console_handler)
        structlog_logger.setLevel(getattr(logging, os.environ["LOG_LEVEL"]))
```

## Key Features

1. **Structured JSON Output**: All logs are formatted as JSON using `structlog`

2. **Request Correlation IDs**:
   - Automatic generation for new requests
   - Support for extracting from `X-Request-ID` header
   - Thread-local storage for context isolation

3. **Log Rotation**:
   - Configurable file size limits
   - Automatic backup file rotation
   - UTF-8 encoding support

4. **Flexible Configuration**:
   - Environment variable-based configuration
   - Runtime configuration via `setup_logging()`
   - Support for both file and console logging

5. **Middleware Support**:
   - ASGI/WSGI middleware for modern frameworks
   - Flask/Django-style hooks for traditional frameworks

## Usage Examples

### Basic Setup
```python
from logging_system import log, setup_logging

# Configure logging with file rotation
setup_logging(
    log_file="logs/app.log",
    log_level="INFO",
    max_bytes=5_000_000,
    backup_count=3
)

# Log with structured data
log.info("user_created", user_id=123, email="user@example.com")
# Output: {"event": "user_created", "user_id": 123, "email": "user@example.com", "level": "info", ...}
```

### With FastAPI
```python
from fastapi import FastAPI
from logging_system import RequestLoggingMiddleware, log, set_correlation_id

app = FastAPI()
app.add_middleware(RequestLoggingMiddleware)

@app.get("/users/{user_id}")
async def get_user(user_id: int):
    log.info("fetching_user", user_id=user_id)
    return {"id": user_id, "name": "John Doe"}
```

### With Flask
```python
from flask import Flask, g
from logging_system import RequestLoggingMiddleware, log

app = Flask(__name__)

@app.before_request
def before_request():
    RequestLoggingMiddleware.start_request()

@app.after_request
def after_request(response):
    RequestLoggingMiddleware.end_request(response)
    return response

@app.route("/users/<int:user_id>")
def get_user(user_id):
    log.info("fetching_user", user_id=user_id)
    return {"id": user_id, "name": "John Doe"}
```

### Correlation ID in Context
```python
from logging_system import set_correlation_id, log

# In your request handler
set_correlation_id("custom-correlation-id-123")
log.info("operation_started", operation="database_backup")
# Output includes correlation_id: "custom-correlation-id-123"
```
