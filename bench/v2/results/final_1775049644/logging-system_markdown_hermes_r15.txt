Here's a robust, production-ready logging system for a web application that follows your development rules:

```python
import os
import sys
import logging
import structlog
from typing import Optional
from pathlib import Path
from datetime import datetime
from logging.handlers import RotatingFileHandler
from structlog.processors import JSONRenderer
from structlog.stdlib import (
    filter_by_level,
    add_log_level,
    add_logger_name,
    add_stack_info,
    add_logger_context,
)
from structlog.contextvars import clear_contextvars, bind_contextvars


# Initialize structlog
structlog.configure(
    processors=[
        filter_by_level,
        add_log_level,
        add_logger_name,
        add_stack_info,
        add_logger_context,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.format_exc_info,
        JSONRenderer(),
    ],
    wrapper_class=structlog.make_filtering_bound_logger(logging.NOTSET),
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=True,
)

log = structlog.get_logger()


def setup_logging(
    level: str = "INFO",
    log_dir: Optional[str] = None,
    max_bytes: int = 10 * 1024 * 1024,  # 10MB
    backup_count: int = 5,
    service_name: str = "webapp",
) -> None:
    """
    Configure application logging with structured JSON output and file rotation.

    Args:
        level: Minimum log level (DEBUG, INFO, WARNING, ERROR, CRITICAL)
        log_dir: Directory to write log files. If None, logs only to stdout.
        max_bytes: Maximum size of each log file before rotation
        backup_count: Number of rotated log files to keep
        service_name: Name of the service for log identification
    """
    # Set log level
    log_level = getattr(logging, level.upper(), logging.INFO)
    
    # Configure console handler
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setLevel(log_level)
    
    # Create formatter that outputs JSON
    json_formatter = structlog.stdlib.ProcessorFormatter(
        processor=JSONRenderer(),
        foreign_pre_chain=[
            add_log_level,
            add_logger_name,
            add_stack_info,
            add_logger_context,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.format_exc_info,
        ],
    )
    console_handler.setFormatter(json_formatter)
    
    # Get root logger
    root_logger = logging.getLogger()
    root_logger.setLevel(log_level)
    root_logger.addHandler(console_handler)
    
    # Add file handler if log_dir is specified
    if log_dir:
        Path(log_dir).mkdir(parents=True, exist_ok=True)
        
        # Create rotating file handler
        file_path = os.path.join(log_dir, f"{service_name}.log")
        file_handler = RotatingFileHandler(
            file_path,
            maxBytes=max_bytes,
            backupCount=backup_count,
            encoding="utf-8",
        )
        file_handler.setLevel(log_level)
        file_handler.setFormatter(json_formatter)
        root_logger.addHandler(file_handler)
    
    # Configure structlog to use the same log level
    structlog.configure(
        processors=[
            filter_by_level,
            add_log_level,
            add_logger_name,
            add_stack_info,
            add_logger_context,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.format_exc_info,
            JSONRenderer(),
        ],
        wrapper_class=structlog.make_filtering_bound_logger(log_level),
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=True,
    )
    
    log.info("logging_configured", level=level, log_dir=log_dir)


def bind_request_context(request_id: str, user_id: Optional[str] = None) -> None:
    """
    Bind request-specific context to logs for correlation across the request lifecycle.
    
    Args:
        request_id: Unique identifier for the current request
        user_id: Optional user identifier for user-specific logging
    """
    clear_contextvars()
    bind_contextvars(request_id=request_id, user_id=user_id or "anonymous")


def get_logger(name: str) -> structlog.BoundLogger:
    """
    Get a namespaced logger for a specific module or component.
    
    Args:
        name: Logger name, typically __name__ from the calling module
        
    Returns:
        A structlog BoundLogger instance
    """
    return structlog.get_logger(name)
```

Key features:

1. **Structured JSON output**: Uses `structlog` with `JSONRenderer` for machine-readable logs
2. **Log levels**: Supports all standard Python log levels (DEBUG, INFO, WARNING, ERROR, CRITICAL)
3. **Request correlation**: `bind_request_context()` adds request ID and user ID for tracing across services
4. **Log rotation**: Uses `RotatingFileHandler` to prevent log files from growing unbounded
5. **Type hints**: All functions include comprehensive type annotations
6. **Error handling**: Logs exceptions with full stack traces via `format_exc_info`
7. **Context binding**: Uses `structlog.contextvars` for thread-safe request-scoped context

Usage example in a web application:

```python
# In your web framework's request middleware
async def logging_middleware(request, handler):
    request_id = request.headers.get("X-Request-ID", str(uuid.uuid4()))
    user_id = request.user.id if hasattr(request, "user") else None
    
    bind_request_context(request_id=request_id, user_id=user_id)
    log.info("request_started", method=request.method, path=request.path)
    
    try:
        response = await handler(request)
        log.info("request_completed", status_code=response.status_code)
        return response
    except Exception as e:
        log.error("request_failed", error=str(e), exc_info=True)
        raise

# In your application code
logger = get_logger(__name__)

async def process_order(order_id: str) -> dict:
    logger.info("processing_order", order_id=order_id)
    # ... processing logic
    logger.info("order_processed", order_id=order_id, duration_ms=123)
    return {"status": "success"}
```

To use in your application:

```python
# At application startup
setup_logging(
    level="INFO",
    log_dir="/var/log/webapp",
    service_name="order-service"
)

# In your middleware or request handler
bind_request_context(request_id="req-123", user_id="user-456")
log.info("request_received", path="/api/orders", method="POST")
```
