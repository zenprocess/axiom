Here's a production-ready structured logging system for a web application using `structlog`, with request correlation IDs, log levels, and log rotation.

```python
import sys
import os
import logging
from datetime import datetime
from typing import Optional
from pathlib import Path
import structlog
from structlog.types import Processor
from structlog.dev import ConsoleRenderer
from structlog.processors import (
    TimeStamper,
    add_log_level,
    format_exc_info,
    StackInfoRenderer,
    JSONRenderer,
)
from structlog.stdlib import (
    add_logger_name,
    filter_by_level,
    get_logger,
    set_logger,
)
from logging.handlers import RotatingFileHandler


# Configuration for log rotation
LOG_FILE_PATH = os.environ.get("LOG_FILE_PATH", "app.log")
MAX_LOG_SIZE = int(os.environ.get("MAX_LOG_SIZE", 10_000_000))  # 10MB
BACKUP_COUNT = int(os.environ.get("BACKUP_COUNT", 5))


def get_processors() -> list[Processor]:
    """Return the list of processors for structlog."""
    return [
        # Add timestamp in ISO format
        TimeStamper(fmt="iso"),
        # Add log level
        add_log_level,
        # Add logger name
        add_logger_name,
        # Add stack info
        StackInfoRenderer(),
        # Format exception info
        format_exc_info,
        # Processors for console or file output
        JSONRenderer(),
    ]


def setup_logging(
    *,
    log_level: str = "INFO",
    log_file: Optional[str] = None,
    debug: bool = False,
) -> None:
    """
    Set up structured logging for the application.

    Args:
        log_level: Minimum log level to output (DEBUG, INFO, WARNING, ERROR, CRITICAL)
        log_file: Optional path to write logs to file. If None, only stdout.
        debug: If True, use pretty console renderer instead of JSON.
    """
    # Ensure log directory exists
    if log_file:
        log_path = Path(log_file)
        log_path.parent.mkdir(parents=True, exist_ok=True)

    # Configure structlog
    structlog.configure(
        processors=get_processors(),
        wrapper_class=structlog.make_filtering_bound_logger(
            getattr(logging, log_level.upper())
        ),
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=True,
    )

    # Configure standard logging for compatibility
    logging.basicConfig(
        format="%(message)s",
        level=getattr(logging, log_level.upper()),
        handlers=[],
        force=True,
    )

    # Add file handler if log_file is specified
    if log_file:
        file_handler = RotatingFileHandler(
            log_file,
            maxBytes=MAX_LOG_SIZE,
            backupCount=BACKUP_COUNT,
            encoding="utf-8",
        )
        file_handler.setLevel(getattr(logging, log_level.upper()))
        file_handler.setFormatter(logging.Formatter("%(message)s"))
        logging.getLogger().addHandler(file_handler)

    # Ensure structlog uses the same logger as stdlib
    set_logger(get_logger())


class RequestLogger:
    """
    Context manager for request-scoped logging with correlation ID.
    """

    def __init__(self, correlation_id: Optional[str] = None):
        self.correlation_id = correlation_id or self._generate_correlation_id()
        self._bound_logger = None

    def _generate_correlation_id(self) -> str:
        """Generate a correlation ID (e.g., UUID4)."""
        import uuid
        return str(uuid.uuid4())

    def bind(self, **kwargs) -> None:
        """Bind additional context to the request logger."""
        if self._bound_logger:
            self._bound_logger = self._bound_logger.bind(**kwargs)

    def __enter__(self) -> "RequestLogger":
        # Create a bound logger with correlation ID
        self._bound_logger = get_logger().bind(request_id=self.correlation_id)
        # Make it available globally (e.g., for middlewares)
        structlog.contextvars.bind_contextvars(request_id=self.correlation_id)
        return self

    def __exit__(self, exc_type, exc_val, exc_tb) -> None:
        # Clear contextvars after request
        structlog.contextvars.unbind_contextvars("request_id")

    def info(self, event: str, **kwargs) -> None:
        """Log an INFO message with request context."""
        if self._bound_logger:
            self._bound_logger.info(event, **kwargs)
        else:
            get_logger().info(event, request_id=self.correlation_id, **kwargs)

    def debug(self, event: str, **kwargs) -> None:
        """Log a DEBUG message with request context."""
        if self._bound_logger:
            self._bound_logger.debug(event, **kwargs)
        else:
            get_logger().debug(event, request_id=self.correlation_id, **kwargs)

    def warning(self, event: str, **kwargs) -> None:
        """Log a WARNING message with request context."""
        if self._bound_logger:
            self._bound_logger.warning(event, **kwargs)
        else:
            get_logger().warning(event, request_id=self.correlation_id, **kwargs)

    def error(self, event: str, **kwargs) -> None:
        """Log an ERROR message with request context."""
        if self._bound_logger:
            self._bound_logger.error(event, **kwargs)
        else:
            get_logger().error(event, request_id=self.correlation_id, **kwargs)

    def critical(self, event: str, **kwargs) -> None:
        """Log a CRITICAL message with request context."""
        if self._bound_logger:
            self._bound_logger.critical(event, **kwargs)
        else:
            get_logger().critical(event, request_id=self.correlation_id, **kwargs)


# Global logger instance for easy access
log = get_logger()


# Example usage
if __name__ == "__main__":
    # Initialize logging
    setup_logging(log_level="DEBUG", log_file=LOG_FILE_PATH)

    # Example 1: Basic logging
    log.info("app_started", version="1.0.0", environment="production")

    # Example 2: Request-scoped logging
    with RequestLogger() as req_logger:
        req_logger.info("request_started", method="GET", path="/api/users")
        
        # Simulate some work
        try:
            # Your request handling logic here
            user = {"id": 123, "name": "Alice"}
            req_logger.info("user_fetched", user_id=user["id"])
            
            # Example with error handling
            raise ValueError("Simulated error")
        except Exception as e:
            req_logger.error("request_failed", error=str(e), exc_info=True)
```

## Key Features

1. **Structured JSON Output**: Uses `structlog` with `JSONRenderer` for machine-readable logs.

2. **Request Correlation IDs**: The `RequestLogger` class provides a context manager that automatically assigns and manages a correlation ID per request, making it easy to trace requests across services.

3. **Log Rotation**: Uses `RotatingFileHandler` with configurable size and backup count.

4. **Flexible Configuration**: Supports environment variables for log path, size, and backup count.

5. **Context Binding**: Uses `structlog.contextvars` to maintain request-scoped context across async boundaries.

6. **Standard Library Compatibility**: Works alongside `logging` for third-party libraries that may use it.

7. **Type Hints**: Fully typed with proper Pydantic-style type annotations.

8. **Error Handling**: Includes examples of structured error logging with `exc_info=True`.

## Usage in Web Frameworks

For FastAPI, Flask, or Django, integrate like this:

```python
# In your request middleware
async def logging_middleware(request: Request, call_next):
    correlation_id = request.headers.get("X-Request-ID") or str(uuid.uuid4())
    
    with RequestLogger(correlation_id) as req_logger:
        req_logger.info("incoming_request", path=str(request.url.path))
        response = await call_next(request)
        req_logger.info("request_completed", status_code=response.status_code)
        return response
```

## Environment Variables

- `LOG_FILE_PATH`: Path to log file (default: `app.log`)
- `MAX_LOG_SIZE`: Max size of log file before rotation (default: 10MB)
- `BACKUP_COUNT`: Number of backup files to keep (default: 5)
