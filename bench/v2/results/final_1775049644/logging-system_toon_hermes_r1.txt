```python
"""
Structured JSON logging system for web applications with request correlation IDs and log rotation.
Uses structlog exclusively; no logging or print statements.
"""

import json
import logging
import os
import sys
import time
from contextvars import ContextVar
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, Optional

import structlog
from pydantic import BaseModel, Field
from structlog.processors import JSONRenderer
from structlog.stdlib import Processor, filter_by_level, format_exc_info, format_log_message
from structlog.typing import EventDict, WrappedLogger


class LogConfig(BaseModel):
    """Configuration for the logging system."""
    level: str = Field(default="INFO", description="Minimum log level (DEBUG, INFO, WARNING, ERROR, CRITICAL)")
    log_dir: str = Field(default="logs", description="Directory to store rotated log files")
    max_bytes: int = Field(default=10 * 1024 * 1024, description="Maximum log file size in bytes before rotation")
    backup_count: int = Field(default=5, description="Number of backup log files to keep")
    include_timestamp: bool = Field(default=True, description="Include timestamp in log entries")
    timestamp_format: str = Field(default="%Y-%m-%dT%H:%M:%S", description="Timestamp format")


class LogEntry(BaseModel):
    """Represents a structured log entry."""
    level: str
    timestamp: str
    logger: str
    message: str
    request_id: Optional[str] = None
    extra: Dict[str, Any] = Field(default_factory=dict)


# Context variable for request correlation ID
_request_id_var: ContextVar[Optional[str]] = ContextVar("request_id", default=None)


def get_request_id() -> Optional[str]:
    """Get the current request ID (correlation ID) from context."""
    return _request_id_var.get()


def set_request_id(request_id: str) -> None:
    """Set the current request ID (correlation ID) in context."""
    _request_id_var.set(request_id)


def clear_request_id() -> None:
    """Clear the request ID from context."""
    _request_id_var.set(None)


@dataclass
class RequestIdAdder(Processor):
    """Processor to add request ID to log entries."""
    def __call__(self, logger: WrappedLogger, method_name: str, event_dict: EventDict) -> EventDict:
        request_id = get_request_id()
        if request_id:
            event_dict["request_id"] = request_id
        return event_dict


@dataclass
class TimestampAdder(Processor):
    """Processor to add timestamp to log entries."""
    timestamp_format: str = "%Y-%m-%dT%H:%M:%S"

    def __call__(self, logger: WrappedLogger, method_name: str, event_dict: EventDict) -> EventDict:
        event_dict["timestamp"] = datetime.now().strftime(self.timestamp_format)
        return event_dict


def configure_logging(config: LogConfig) -> None:
    """Configure the logging system with the provided configuration."""
    # Create log directory if it doesn't exist
    log_path = Path(config.log_dir)
    log_path.mkdir(parents=True, exist_ok=True)

    # Setup structlog processors
    processors = [
        filter_by_level,
        RequestIdAdder(),
        TimestampAdder(timestamp_format=config.timestamp_format) if config.include_timestamp else None,
        format_log_message,
        format_exc_info,
        JSONRenderer(),
    ]

    # Remove None processors
    processors = [p for p in processors if p is not None]

    # Configure structlog
    structlog.configure(
        processors=processors,
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=True,
    )

    # Setup file handler for log rotation
    log_file = log_path / "app.log"
    file_handler = logging.handlers.RotatingFileHandler(
        str(log_file),
        maxBytes=config.max_bytes,
        backupCount=config.backup_count,
        encoding="utf-8"
    )

    # Create a custom formatter to convert structlog dict to JSON string
    class StructJsonFormatter(logging.Formatter):
        def format(self, record: logging.LogRecord) -> str:
            try:
                # If record.msg is already a dict (from structlog), use it directly
                if isinstance(record.msg, dict):
                    return json.dumps(record.msg)
                # Otherwise, try to parse as JSON
                try:
                    return json.dumps(json.loads(record.msg))
                except (json.JSONDecodeError, TypeError):
                    return json.dumps({"message": record.getMessage(), "level": record.levelname})
            except Exception as e:
                return json.dumps({"error": str(e), "message": record.getMessage()})

    file_handler.setFormatter(StructJsonFormatter())
    file_handler.setLevel(getattr(logging, config.level.upper()))

    # Attach handler to structlog's logger
    structlog_logger = structlog.get_logger()
    structlog_logger.logger.handlers = [file_handler]
    structlog_logger.setLevel(getattr(logging, config.level.upper()))


def get_logger(name: str = "app") -> structlog.BoundLogger:
    """Get a logger instance with the specified name."""
    return structlog.get_logger(name)


def log_info(message: str, **kwargs: Any) -> None:
    """Log an INFO level message with optional structured data."""
    logger = get_logger()
    logger.info(message, **kwargs)


def log_error(message: str, **kwargs: Any) -> None:
    """Log an ERROR level message with optional structured data."""
    logger = get_logger()
    logger.error(message, **kwargs)


def log_debug(message: str, **kwargs: Any) -> None:
    """Log a DEBUG level message with optional structured data."""
    logger = get_logger()
    logger.debug(message, **kwargs)


def log_warning(message: str, **kwargs: Any) -> None:
    """Log a WARNING level message with optional structured data."""
    logger = get_logger()
    logger.warning(message, **kwargs)


def log_critical(message: str, **kwargs: Any) -> None:
    """Log a CRITICAL level message with optional structured data."""
    logger = get_logger()
    logger.critical(message, **kwargs)


def format_log_entry(
    level: str,
    message: str,
    request_id: Optional[str] = None,
    extra: Optional[Dict[str, Any]] = None
) -> Dict[str, Any]:
    """Format a log entry as a dictionary (for manual logging if needed)."""
    entry = LogEntry(
        level=level,
        timestamp=datetime.now().strftime("%Y-%m-%dT%H:%M:%S"),
        logger="app",
        message=message,
        request_id=request_id,
        extra=extra or {}
    )
    return entry.model_dump()
```

## Key Features

1. **Structured JSON Output**: All logs are rendered as JSON using `structlog.JSONRenderer`.

2. **Request Correlation IDs**: Uses `ContextVar` for thread-safe request ID tracking, with helper functions.

3. **Log Levels**: Supports DEBUG, INFO, WARNING, ERROR, and CRITICAL.

4. **Log Rotation**: Uses `RotatingFileHandler` to manage log file size and retention.

5. **Pydantic Models**: Config and LogEntry are Pydantic models for validation and type safety.

6. **Type Hints**: All functions have complete type annotations.

7. **Error Handling**: Safe JSON serialization with fallbacks.

8. **Structlog-Only**: No `logging` or `print()` statements used.

## Usage Examples

```python
# Configure logging
config = LogConfig(level="DEBUG", log_dir="var/logs")
configure_logging(config)

# Set request ID (e.g., in middleware)
set_request_id("req-12345")

# Log messages
log_info("User logged in", user_id=123, action="login")
log_error("Database connection failed", error="timeout", retry_count=3)

# Clear request ID when request ends
clear_request_id()
```

    response = await call_next(request)
    response.headers["X-Request-ID"] = request_id
    return response

@app.middleware("http")
async def cleanup_request_id(request: Request, call_next):
    try:
        return await call_next(request)
    finally:
        clear_request_id()
