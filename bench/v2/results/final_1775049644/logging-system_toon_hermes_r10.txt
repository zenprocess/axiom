### ✅ `logging_config.py`

```python
"""Structured logging configuration for web applications using structlog."""

from __future__ import annotations

import json
import logging
import sys
from datetime import datetime
from pathlib import Path
from typing import Any, Callable, Dict, Optional

import structlog
from pydantic import BaseModel, Field
from structlog.processors import JSONRenderer
from structlog.stdlib import ProcessorFormatter


class LogConfig(BaseModel):
    """Configuration for the structured logger."""

    level: str = Field(default="INFO", description="Logging level (DEBUG, INFO, WARNING, ERROR, CRITICAL)")
    log_file: Optional[Path] = Field(default=None, description="Path to log file. If None, logs to stdout.")
    rotate_when: str = Field(default="midnight", description="When to rotate logs (see TimedRotatingFileHandler)")
    rotate_interval: int = Field(default=1, description="Interval for log rotation")
    backup_count: int = Field(default=7, description="Number of backup files to keep")


def setup_logging(config: LogConfig) -> None:
    """
    Set up structlog with JSON output and file rotation.

    Args:
        config: LogConfig with logging parameters.
    """
    # Ensure log directory exists if logging to file
    if config.log_file:
        config.log_file.parent.mkdir(parents=True, exist_ok=True)

    # Configure structlog processors
    shared_processors = [
        structlog.contextvars.merge_contextvars,
        structlog.stdlib.add_logger_name,
        structlog.stdlib.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
    ]

    # Define renderer (JSON for production)
    renderer = JSONRenderer()

    # Configure structlog to use the renderer
    structlog.configure(
        processors=shared_processors + [ProcessorFormatter.wrap_for_formatter],
        wrapper_class=structlog.stdlib.BoundLogger,
        context_class=dict,
        logger_factory=structlog.stdlib.LoggerFactory(),
        cache_logger_on_first_use=True,
    )

    # Configure Python logging (used by ProcessorFormatter)
    logging.basicConfig(
        format="%(message)s",
        level=getattr(logging, config.level.upper(), logging.INFO),
        handlers=[],
        force=True,
    )

    # Main handler: JSON to stdout or file
    if config.log_file:
        handler = _RotatingFileHandler(
            filename=str(config.log_file),
            when=config.rotate_when,
            interval=config.rotate_interval,
            backupCount=config.backup_count,
        )
    else:
        handler = logging.StreamHandler(sys.stdout)

    formatter = ProcessorFormatter(
        processor=renderer,
        foreign_pre_chain=shared_processors,
    )
    handler.setFormatter(formatter)

    # Attach handler to root logger
    root_logger = logging.getLogger()
    root_logger.handlers.clear()
    root_logger.addHandler(handler)
    root_logger.setLevel(getattr(logging, config.level.upper(), logging.INFO))


class _RotatingFileHandler(logging.handlers.TimedRotatingFileHandler):
    """Custom file handler with JSON line formatting."""

    def __init__(
        self,
        filename: str,
        when: str = "midnight",
        interval: int = 1,
        backupCount: int = 0,
        encoding: Optional[str] = None,
        delay: bool = False,
        utc: bool = False,
        atTime: Optional[datetime] = None,
        errors: Optional[str] = None,
    ) -> None:
        super().__init__(
            filename=filename,
            when=when,
            interval=interval,
            backupCount=backupCount,
            encoding=encoding,
            delay=delay,
            utc=utc,
            atTime=atTime,
            errors=errors,
        )

    def emit(self, record: logging.LogRecord) -> None:
        """Emit a record as a JSON line."""
        try:
            msg = self.format(record)
            if not msg.endswith("\n"):
                msg += "\n"
            self.stream.write(msg)
            self.flush()
        except Exception:
            self.handleError(record)


def get_correlation_id() -> Optional[str]:
    """Get the current request correlation ID from contextvars."""
    return structlog.contextvars.get_contextvars().get("correlation_id")


def bind_correlation_id(correlation_id: str) -> None:
    """
    Bind a correlation ID to the current request context.

    Args:
        correlation_id: Unique ID for request tracing.
    """
    structlog.contextvars.bind_contextvars(correlation_id=correlation_id)


def clear_correlation_id() -> None:
    """Clear the correlation ID from the current context."""
    structlog.contextvars.unbind_contextvars("correlation_id")


def bind_extra(**kwargs: Any) -> None:
    """
    Bind additional key-value pairs to the current log context.

    Args:
        **kwargs: Key-value pairs to bind.
    """
    structlog.contextvars.bind_contextvars(**kwargs)


def clear_extra() -> None:
    """Clear all extra context variables."""
    structlog.contextvars.clear_contextvars()


def log_debug(msg: str, **kwargs: Any) -> None:
    """
    Log a DEBUG message with optional extra context.

    Args:
        msg: Log message.
        **kwargs: Additional structured fields.
    """
    structlog.get_logger("app").debug(msg, **kwargs)


def log_info(msg: str, **kwargs: Any) -> None:
    """
    Log an INFO message with optional extra context.

    Args:
        msg: Log message.
        **kwargs: Additional structured fields.
    """
    structlog.get_logger("app").info(msg, **kwargs)


def log_warning(msg: str, **kwargs: Any) -> None:
    """
    Log a WARNING message with optional extra context.

    Args:
        msg: Log message.
        **kwargs: Additional structured fields.
    """
    structlog.get_logger("app").warning(msg, **kwargs)


def log_error(msg: str, exc_info: bool = False, **kwargs: Any) -> None:
    """
    Log an ERROR message with optional traceback and extra context.

    Args:
        msg: Log message.
        exc_info: Whether to include exception info.
        **kwargs: Additional structured fields.
    """
    structlog.get_logger("app").error(msg, exc_info=exc_info, **kwargs)


def log_critical(msg: str, exc_info: bool = False, **kwargs: Any) -> None:
    """
    Log a CRITICAL message with optional traceback and extra context.

    Args:
        msg: Log message.
        exc_info: Whether to include exception info.
        **kwargs: Additional structured fields.
    """
    structlog.get_logger("app").critical(msg, exc_info=exc_info, **kwargs)
```

### ✅ Example usage (e.g., in a FastAPI middleware or request handler)

```python
# Example: FastAPI middleware
from fastapi import FastAPI, Request
from typing import Optional
import uuid

from logging_config import (
    setup_logging,
    LogConfig,
    bind_correlation_id,
    clear_correlation_id,
    log_info,
)

app = FastAPI()

# Initialize logging (e.g., on startup)
setup_logging(LogConfig(level="INFO", log_file=Path("logs/app.log")))

@app.middleware("http")
async def add_correlation_id(request: Request, call_next):
    correlation_id: str = request.headers.get("X-Request-ID", str(uuid.uuid4()))
    bind_correlation_id(correlation_id)
    log_info("Request started", method=request.method, path=request.url.path)

    try:
        response = await call_next(request)
        log_info("Request completed", status_code=response.status_code)
        return response
    finally:
        clear_correlation_id()

@app.get("/")
async def root():
    log_info("Root endpoint accessed")
    return {"message": "Hello, world!"}
```
