### `logging_config.py`

```python
import os
import sys
from typing import Any, Dict, Optional
from structlog import configure, get_logger, BoundLogger
from structlog.processors import (
    TimeStamper,
    add_log_level,
    JSONRenderer,
    StackInfoRenderer,
    format_exc_info,
)
from structlog.stdlib import ProcessorFormatter
from structlog.threadlocal import wrap_dict
from structlog.dev import ConsoleRenderer
from logging.handlers import RotatingFileHandler
import logging

# Ensure structlog is configured once at startup
configure(
    processors=[
        add_log_level,
        TimeStamper(fmt="iso"),
        StackInfoRenderer(),
        format_exc_info,
        ProcessorFormatter(
            processor=JSONRenderer(),
            foreign_pre_chain=[
                add_log_level,
                TimeStamper(fmt="iso"),
                StackInfoRenderer(),
                format_exc_info,
            ],
        ),
    ],
    wrapper_class=BoundLogger,
    context_class=wrap_dict(dict),
    logger_factory=structlog.stdlib.LoggerFactory(),
    cache_logger_on_first_use=True,
)


class RequestContext:
    """Thread-local context to store per-request correlation ID."""

    def __init__(self):
        self._storage = {}

    def set(self, request_id: str):
        self._storage["request_id"] = request_id

    def get(self) -> Optional[str]:
        return self._storage.get("request_id")

    def clear(self):
        self._storage.clear()


request_context = RequestContext()


def get_request_id() -> Optional[str]:
    """Get current request correlation ID."""
    return request_context.get()


# Pydantic model for structured log fields (optional but recommended)
from pydantic import BaseModel

class LogContext(BaseModel):
    request_id: Optional[str] = None
    user_id: Optional[str] = None
    path: Optional[str] = None
    method: Optional[str] = None
    status_code: Optional[int] = None


# Setup handlers manually for more control
def setup_logging(
    log_level: str = "INFO",
    log_file: str = "app.log",
    max_bytes: int = 10_000_000,
    backup_count: int = 5,
    console: bool = True,
):
    """
    Configure the logging system.

    Args:
        log_level: Logging level (DEBUG, INFO, WARNING, ERROR, CRITICAL)
        log_file: Path to the log file
        max_bytes: Max log file size before rotation
        backup_count: Number of backup files to keep
        console: Whether to log to stdout/stderr
    """
    level = getattr(logging, log_level.upper(), logging.INFO)

    # Root logger
    root_logger = logging.getLogger()
    root_logger.setLevel(level)

    # Clear existing handlers
    root_logger.handlers.clear()

    # File handler with rotation
    if log_file:
        os.makedirs(os.path.dirname(log_file) or ".", exist_ok=True)
        file_handler = RotatingFileHandler(
            log_file, maxBytes=max_bytes, backupCount=backup_count
        )
        file_handler.setLevel(level)
        file_handler.setFormatter(ProcessorFormatter(processor=JSONRenderer()))
        root_logger.addHandler(file_handler)

    # Console handler (for development)
    if console:
        console_handler = logging.StreamHandler(sys.stdout)
        console_handler.setLevel(level)
        console_handler.setFormatter(ProcessorFormatter(processor=ConsoleRenderer()))
        root_logger.addHandler(console_handler)


# Public API
logger = get_logger()

def bind(**kwargs):
    """Bind key-value pairs to the current log context."""
    return logger.bind(**kwargs)

def unbind(*keys):
    """Unbind keys from the current log context."""
    return logger.unbind(*keys)

def info(msg: str, **kwargs):
    logger.info(msg, **kwargs)

def error(msg: str, **kwargs):
    logger.error(msg, **kwargs)

def debug(msg: str, **kwargs):
    logger.debug(msg, **kwargs)

def warning(msg: str, **kwargs):
    logger.warning(msg, **kwargs)

def critical(msg: str, **kwargs):
    logger.critical(msg, **kwargs)

def log_with_context(log_func, msg: str, context: LogContext):
    """
    Log a message with structured context.

    Args:
        log_func: One of info/error/debug/warning/critical
        msg: Log message
        context: LogContext instance
    """
    try:
        log_func(
            msg,
            request_id=context.request_id,
            user_id=context.user_id,
            path=context.path,
            method=context.method,
            status_code=context.status_code,
        )
    except Exception as e:
        # Never swallow errors silently — at least log a fallback
        logger.error("Failed to log structured message", error=str(e))
```

### `middleware.py` (example integration for FastAPI)

```python
from fastapi import FastAPI, Request
from starlette.middleware.base import BaseHTTPMiddleware
from typing import Callable
import uuid
from logging_config import request_context, logger

class RequestLoggingMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next: Callable):
        request_id = request.headers.get("X-Request-ID") or str(uuid.uuid4())
        request_context.set(request_id)
        logger.bind(request_id=request_id)

        try:
            response = await call_next(request)
            # Optional: add response metadata
            logger.info(
                "Request completed",
                path=str(request.url.path),
                method=request.method,
                status_code=response.status_code,
            )
            response.headers["X-Request-ID"] = request_id
            return response
        except Exception as e:
            logger.error("Request failed", exc_info=True)
            raise
        finally:
            request_context.clear()
            logger.unbind("request_id")
```

### `main.py` (example usage)

```python
from fastapi import FastAPI
from logging_config import setup_logging, LogContext, info, error
from middleware import RequestLoggingMiddleware
from pydantic import BaseModel

app = FastAPI()
app.add_middleware(RequestLoggingMiddleware)

# Initialize logging at startup
setup_logging(log_level="INFO", log_file="logs/app.log", console=True)


class LoginRequest(BaseModel):
    username: str


@app.post("/login")
def login(request: LoginRequest):
    try:
        context = LogContext(
            request_id="test-123",
            user_id="user-456",
            path="/login",
            method="POST",
            status_code=200,
        )
        info("User logged in successfully", context=context)
        return {"status": "ok"}
    except Exception:
        error("Login failed", path="/login", user_id="user-456")
        raise
```
