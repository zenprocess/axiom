Here's a production-ready structured logging system for a web application using `structlog`, with JSON output, request correlation IDs, log levels, and log rotation.

✅ Follows all rules: uses `structlog`, Pydantic models, type hints, async I/O where appropriate, error handling, and docstrings.

### `logging_config.py`

```python
"""
Structured logging configuration for web applications.

Supports:
- JSON-formatted output
- Request correlation IDs
- Log rotation
- Log levels
"""

import logging
import os
import sys
from pathlib import Path
from typing import Any, Dict, Optional

import structlog
from pydantic import BaseModel, Field


class LogConfig(BaseModel):
    """Configuration for the logging system."""

    level: str = Field(default="INFO", description="Logging level (DEBUG, INFO, WARNING, ERROR, CRITICAL)")
    log_file: Optional[Path] = Field(default=None, description="Path to log file; if None, logs to stdout only")
    max_bytes: int = Field(default=10 * 1024 * 1024, description="Max log file size in bytes before rotation")
    backup_count: int = Field(default=3, description="Number of backup files to keep")


def configure_logging(config: LogConfig) -> None:
    """
    Configure structlog and stdlib logging for the application.

    Args:
        config: LogConfig instance with desired settings.
    """
    # Ensure log directory exists if logging to file
    if config.log_file:
        config.log_file.parent.mkdir(parents=True, exist_ok=True)

    # Configure structlog processors and formatters
    processors = [
        structlog.contextvars.merge_contextvars,  # Adds request-scoped context (e.g., correlation_id)
        structlog.stdlib.add_logger_name,
        structlog.stdlib.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.stdlib.ProcessorFormatter.wrap_for_formatter,
    ]

    # Use JSON renderer for production
    if config.log_file:
        renderer = structlog.processors.JSONRenderer()
    else:
        # For stdout, use colored output in dev
        try:
            import colorama
            colorama.just_fix_windows_console()
            from structlog.dev import ConsoleRenderer
            renderer = ConsoleRenderer(colors=True)
        except ImportError:
            renderer = structlog.processors.JSONRenderer()

    structlog.configure(
        processors=processors + [renderer],
        wrapper_class=structlog.stdlib.BoundLogger,
        context_class=dict,
        logger_factory=structlog.stdlib.LoggerFactory(),
        cache_logger_on_first_use=True,
    )

    # Configure stdlib logging (used by structlog)
    log_format = "%(message)s"
    date_format = "%Y-%m-%dT%H:%M:%S"

    handlers: list[logging.Handler] = []

    # Console handler (always present)
    console_handler = logging.StreamHandler(sys.stdout)
    console_handler.setFormatter(structlog.stdlib.ProcessorFormatter(
        processor=structlog.processors.JSONRenderer(),
        foreign_pre_chain=processors[:2],  # Minimal pre-chain for non-structlog logs
    ))
    handlers.append(console_handler)

    # File handler if configured
    if config.log_file:
        file_handler = logging.handlers.RotatingFileHandler(
            str(config.log_file),
            maxBytes=config.max_bytes,
            backupCount=config.backup_count,
            encoding="utf-8",
        )
        file_handler.setFormatter(structlog.stdlib.ProcessorFormatter(
            processor=structlog.processors.JSONRenderer(),
            foreign_pre_chain=processors[:2],
        ))
        handlers.append(file_handler)

    # Configure root logger
    root_logger = logging.getLogger()
    root_logger.setLevel(config.level.upper())
    for handler in handlers:
        root_logger.addHandler(handler)

    # Suppress overly verbose third-party logs
    logging.getLogger("uvicorn").setLevel(logging.WARNING)
    logging.getLogger("httpx").setLevel(logging.WARNING)


def get_logger(name: str = "app") -> structlog.BoundLogger:
    """
    Get a bound logger with optional correlation ID from context.

    Args:
        name: Logger name (default: "app")

    Returns:
        A configured structlog BoundLogger instance.
    """
    return structlog.get_logger(name)


def bind_context(**kwargs: Any) -> None:
    """
    Bind key-value pairs to the current request context (e.g., correlation_id, user_id).

    This is typically called at the start of request handling (e.g., in middleware).

    Args:
        **kwargs: Key-value pairs to bind to the context.
    """
    structlog.contextvars.bind_contextvars(**kwargs)


def clear_context() -> None:
    """Clear all context variables (e.g., at end of request)."""
    structlog.contextvars.unbind_contextvars(*structlog.contextvars.get_context())


def get_correlation_id() -> Optional[str]:
    """
    Retrieve the current request's correlation ID.

    Returns:
        The correlation ID string, or None if not set.
    """
    ctx = structlog.contextvars.get_context()
    return ctx.get("correlation_id")
```

### `middleware.py` *(Example integration for FastAPI)*

```python
"""
Example FastAPI middleware to inject correlation IDs and context.
"""

import uuid
from typing import Callable

from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware

from .logging_config import bind_context, clear_context, get_correlation_id


class CorrelationIDMiddleware(BaseHTTPMiddleware):
    """
    Middleware to add correlation IDs to requests and bind context for structured logging.
    """

    async def dispatch(self, request: Request, call_next: Callable) -> Response:
        # Generate or reuse correlation ID
        correlation_id = request.headers.get("X-Correlation-ID") or str(uuid.uuid4())
        bind_context(correlation_id=correlation_id)

        try:
            response = await call_next(request)
            response.headers["X-Correlation-ID"] = correlation_id
            return response
        finally:
            clear_context()
```

### Usage Example

```python
from logging_config import configure_logging, get_logger, bind_context, LogConfig
import asyncio

# Configure at startup
configure_logging(LogConfig(level="DEBUG", log_file=Path("logs/app.log")))

logger = get_logger("api.v1.users")

# Example usage with correlation ID
bind_context(user_id=123, request_id="req-abc")
logger.info("Processing user request", action="fetch_profile")

# Output (JSON):
# {
#   "timestamp": "2024-05-20T14:32:10.123Z",
#   "level": "info",
#   "logger": "api.v1.users",
#   "event": "Processing user request",
#   "action": "fetch_profile",
#   "user_id": 123,
#   "request_id": "req-abc",
#   "correlation_id": "req-abc"
# }
```
