It looks like file writes are being blocked. Here's the complete script — you can create it yourself with `cat > deploy.sh << 'EOF'`:

```bash
#!/usr/bin/env bash
# deploy.sh — build, push, tag, deploy with rollback support
# Usage:
#   ./deploy.sh [VERSION]            # deploy (default: version from pyproject.toml)
#   ./deploy.sh --rollback [VERSION] # rollback (default: previous git tag)
#
# Env vars:
#   REGISTRY      Docker registry prefix    (default: ghcr.io/zenprocess)
#   IMAGE_NAME    Docker image name         (default: axiom)
#   DEPLOY_HOST   SSH production host       (default: $AXIOM_DEPLOY_HOST)
#   DEPLOY_USER   SSH user                  (default: deploy)
#   DEPLOY_PATH   Remote app directory      (default: /opt/axiom)
#   DOCKER_FILE   Path to Dockerfile        (default: Dockerfile)
#   GIT_REMOTE    Git remote name           (default: origin)

set -euo pipefail

REGISTRY="${REGISTRY:-ghcr.io/zenprocess}"
IMAGE_NAME="${IMAGE_NAME:-axiom}"
DEPLOY_HOST="${DEPLOY_HOST:-${AXIOM_DEPLOY_HOST:-}}"
DEPLOY_USER="${DEPLOY_USER:-deploy}"
DEPLOY_PATH="${DEPLOY_PATH:-/opt/axiom}"
DOCKER_FILE="${DOCKER_FILE:-Dockerfile}"
GIT_REMOTE="${GIT_REMOTE:-origin}"

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
TAG_PREFIX="v"
LOG_FILE="${SCRIPT_DIR}/.deploy.log"

# ── Helpers ───────────────────────────────────────────────────────────────────

log() { echo "[$(date -u '+%Y-%m-%dT%H:%M:%SZ')] $*" | tee -a "$LOG_FILE"; }
err() { log "ERROR: $*" >&2; exit 1; }

require_cmd() {
  for cmd in "$@"; do
    command -v "$cmd" &>/dev/null || err "Required command not found: $cmd"
  done
}

get_project_version() {
  grep -m1 '^version' "${SCRIPT_DIR}/pyproject.toml" \
    | sed 's/.*= *"\(.*\)"/\1/'
}

get_previous_tag() {
  git -C "$SCRIPT_DIR" tag --sort=-version:refname \
    | grep "^${TAG_PREFIX}" \
    | sed -n '2p'
}

full_image() { echo "${REGISTRY}/${IMAGE_NAME}:${1}"; }

ssh_prod() {
  [[ -z "$DEPLOY_HOST" ]] && err "DEPLOY_HOST is not set"
  ssh -o StrictHostKeyChecking=no "${DEPLOY_USER}@${DEPLOY_HOST}" "$@"
}

# ── Steps ─────────────────────────────────────────────────────────────────────

step_build() {
  local version="$1"
  local image; image="$(full_image "$version")"
  local latest; latest="$(full_image latest)"

  log "Building $image"
  [[ -f "${SCRIPT_DIR}/${DOCKER_FILE}" ]] \
    || err "Dockerfile not found at ${SCRIPT_DIR}/${DOCKER_FILE}"

  docker build \
    --file   "${SCRIPT_DIR}/${DOCKER_FILE}" \
    --build-arg VERSION="$version" \
    --tag    "$image" \
    --tag    "$latest" \
    --label  "org.opencontainers.image.version=$version" \
    --label  "org.opencontainers.image.created=$(date -u +%Y-%m-%dT%H:%M:%SZ)" \
    --label  "org.opencontainers.image.revision=$(git -C "$SCRIPT_DIR" rev-parse HEAD)" \
    "$SCRIPT_DIR"
  log "Build OK: $image"
}

step_push() {
  local version="$1"
  log "Pushing $(full_image "$version")"
  docker push "$(full_image "$version")"
  docker push "$(full_image latest)"
  log "Push OK"
}

step_git_tag() {
  local version="$1"
  local tag="${TAG_PREFIX}${version}"

  if git -C "$SCRIPT_DIR" tag --list | grep -qx "$tag"; then
    log "Tag $tag already exists — skipping"
    return
  fi

  log "Tagging $tag"
  git -C "$SCRIPT_DIR" tag --annotate "$tag" \
    --message "Release ${tag} — $(date -u '+%Y-%m-%d')"
  git -C "$SCRIPT_DIR" push "$GIT_REMOTE" "$tag"
  log "Tag pushed: $tag"
}

step_deploy() {
  local version="$1"
  local image; image="$(full_image "$version")"

  log "Deploying $image → ${DEPLOY_USER}@${DEPLOY_HOST}:${DEPLOY_PATH}"

  # Save rollback pointer before touching anything
  local prev
  prev="$(ssh_prod "cat ${DEPLOY_PATH}/.version 2>/dev/null || echo ''")"
  [[ -n "$prev" ]] && ssh_prod "echo '${prev}' > ${DEPLOY_PATH}/.rollback" \
    && log "Rollback pointer saved: $prev"

  ssh_prod bash -s <<EOF
set -euo pipefail
cd "${DEPLOY_PATH}"
docker pull "${image}"
docker stop axiom 2>/dev/null || true
docker rm   axiom 2>/dev/null || true
docker run -d \
  --name axiom \
  --restart unless-stopped \
  --env-file .env \
  -p 8000:8000 \
  "${image}"
echo "${version}" > .version
docker ps --filter name=axiom --format 'table {{.ID}}\t{{.Status}}\t{{.Ports}}'
EOF
  log "Deploy OK: $version is live"
}

step_rollback() {
  local version="$1"
  local image; image="$(full_image "$version")"

  log "ROLLBACK → $image"

  ssh_prod bash -s <<EOF
set -euo pipefail
cd "${DEPLOY_PATH}"
docker pull "${image}"
docker stop axiom 2>/dev/null || true
docker rm   axiom 2>/dev/null || true
docker run -d \
  --name axiom \
  --restart unless-stopped \
  --env-file .env \
  -p 8000:8000 \
  "${image}"
echo "${version}" > .version
docker ps --filter name=axiom --format 'table {{.ID}}\t{{.Status}}\t{{.Ports}}'
EOF
  log "Rollback OK: running $version"
}

# ── Modes ─────────────────────────────────────────────────────────────────────

do_rollback() {
  local target="${1:-}"

  if [[ -z "$target" ]]; then
    target="$(ssh_prod "cat ${DEPLOY_PATH}/.rollback 2>/dev/null || echo ''")"
    [[ -z "$target" ]] && target="$(get_previous_tag | sed "s/^${TAG_PREFIX}//")"
    [[ -z "$target" ]] && err "Cannot determine rollback target — pass a version explicitly"
  fi

  require_cmd docker git ssh
  log "=== ROLLBACK to $target ==="
  step_rollback "$target"
  log "=== ROLLBACK COMPLETE ==="
}

do_deploy() {
  local version="${1:-$(get_project_version)}"
  [[ -z "$version" ]] && err "Could not determine version from pyproject.toml"

  require_cmd docker git ssh
  log "=== DEPLOY $version ==="
  step_build   "$version"
  step_push    "$version"
  step_git_tag "$version"
  step_deploy  "$version"
  log "=== DEPLOY COMPLETE: $version ==="
}

# ── Entrypoint ────────────────────────────────────────────────────────────────

case "${1:-}" in
  --rollback|-r) do_rollback "${2:-}" ;;
  --help|-h)     sed -n '2,9p' "$0" | sed 's/^# //' ;;
  *)             do_deploy "${1:-}" ;;
esac
```

After saving: `chmod +x deploy.sh`

---

**Key design decisions:**

| Feature | How it works |
|---|---|
| **Version source** | Reads from `pyproject.toml` automatically; override with `./deploy.sh 0.7.0` |
| **Rollback target** | Saved to `.rollback` on the remote *before* each deploy; `--rollback` reads it automatically, or accepts an explicit version |
| **Git tag** | Annotated tag, skipped if already exists (idempotent re-runs) |
| **Registry** | Pushes both `vX.Y.Z` and `latest` tags |
| **Deploy** | SSH + `docker run` — swap the `step_deploy` body for `kubectl set image` if you're on Kubernetes |
| **Logs** | Appended to `.deploy.log` in the project root |

**What you need to set up first:**
1. `Dockerfile` in the project root
2. `AXIOM_DEPLOY_HOST` env var (or `DEPLOY_HOST`)
3. `/opt/axiom/.env` on the production host
4. SSH key access for the deploy user
