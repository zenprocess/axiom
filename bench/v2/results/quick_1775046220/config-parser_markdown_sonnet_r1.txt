```python
# examples/config_loader.py
"""Priority-merging config loader: YAML < env vars < CLI flags."""

import argparse
import os
from pathlib import Path
from typing import Any

import structlog
import yaml
from pydantic import BaseModel, ValidationError

log = structlog.get_logger()


# ---------------------------------------------------------------------------
# Schema
# ---------------------------------------------------------------------------

class AppConfig(BaseModel):
    """Top-level application configuration."""

    host: str = "127.0.0.1"
    port: int = 8080
    debug: bool = False
    log_level: str = "info"
    database_url: str = "sqlite:///app.db"


# ---------------------------------------------------------------------------
# Loaders
# ---------------------------------------------------------------------------

async def load_yaml(path: Path) -> dict[str, Any]:
    """Load a YAML config file, returning {} if absent.

    Raises:
        yaml.YAMLError: If the file exists but cannot be parsed.
    """
    if not path.exists():
        log.debug("yaml_config_not_found", path=str(path))
        return {}

    try:
        text = path.read_text(encoding="utf-8")
        data: dict[str, Any] = yaml.safe_load(text) or {}
        log.info("yaml_config_loaded", path=str(path), keys=list(data.keys()))
        return data
    except yaml.YAMLError as exc:
        log.error("yaml_parse_error", path=str(path), error=str(exc))
        raise


def load_env(prefix: str = "APP_") -> dict[str, Any]:
    """Read prefixed env vars (e.g. APP_PORT=9090 → {"port": 9090})."""
    data: dict[str, Any] = {}
    for key, value in os.environ.items():
        if key.startswith(prefix):
            config_key = key[len(prefix):].lower()
            data[config_key] = _coerce(value)
    if data:
        log.info("env_config_loaded", keys=list(data.keys()))
    return data


def load_cli(argv: list[str] | None = None) -> tuple[dict[str, Any], str]:
    """Parse CLI flags; only explicitly-provided flags are returned.

    Returns:
        Tuple of (overrides dict, yaml config path string).
    """
    parser = argparse.ArgumentParser(description="App config overrides")
    parser.add_argument("--config", default="config.yaml")
    parser.add_argument("--host", default=None)
    parser.add_argument("--port", type=int, default=None)
    parser.add_argument("--debug", action="store_true", default=None)
    parser.add_argument("--log-level", dest="log_level", default=None)
    parser.add_argument("--database-url", dest="database_url", default=None)

    ns = parser.parse_args(argv)
    overrides: dict[str, Any] = {
        k: v for k, v in vars(ns).items() if k != "config" and v is not None
    }
    if overrides:
        log.info("cli_config_loaded", keys=list(overrides.keys()))
    return overrides, ns.config


# ---------------------------------------------------------------------------
# Merge & validate
# ---------------------------------------------------------------------------

async def load_config(argv: list[str] | None = None) -> AppConfig:
    """Merge all sources with priority: CLI > env vars > YAML > defaults.

    Raises:
        pydantic.ValidationError: If merged values fail schema validation.
    """
    cli_overrides, yaml_path = load_cli(argv)
    yaml_data = await load_yaml(Path(yaml_path))
    env_data = load_env()

    merged: dict[str, Any] = {**yaml_data, **env_data, **cli_overrides}
    log.debug("config_merged_raw", merged=merged)

    try:
        config = AppConfig(**merged)
    except ValidationError as exc:
        log.error("config_validation_failed", errors=exc.errors())
        raise

    log.info("config_ready", **config.model_dump())
    return config


# ---------------------------------------------------------------------------
# Helpers
# ---------------------------------------------------------------------------

def _coerce(value: str) -> Any:
    """Coerce env var strings to bool/int/float where unambiguous."""
    if value.lower() in {"true", "1", "yes"}:
        return True
    if value.lower() in {"false", "0", "no"}:
        return False
    try:
        return int(value)
    except ValueError:
        pass
    try:
        return float(value)
    except ValueError:
        pass
    return value


# ---------------------------------------------------------------------------
# Entrypoint
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    import asyncio

    async def _main() -> None:
        config = await load_config()
        log.info("final_config", **config.model_dump())  # structured, not print()

    asyncio.run(_main())
```

---

**Key design decisions:**

| Concern | Choice |
|---|---|
| Debugging output | `log.info("final_config", ...)` via structlog — no `print()` |
| File I/O | `async def load_yaml` — non-blocking |
| Data validation | `AppConfig(BaseModel)` — Pydantic rejects bad types at merge time |
| Priority order | defaults → YAML → `APP_*` env vars → `--flag` CLI args |
| Type coercion | `_coerce()` handles `"true"/"false"/"1"/"0"` from env strings |

**Usage examples:**
```bash
# YAML only
python config_loader.py --config prod.yaml

# env override (port becomes 9090)
APP_PORT=9090 python config_loader.py

# CLI wins over everything
APP_PORT=9090 python config_loader.py --port 3000  # → port=3000
```

**Dependencies needed:** `pyyaml`, `pydantic`, `structlog`
