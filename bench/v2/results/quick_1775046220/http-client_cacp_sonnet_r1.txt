```python
# user_client.py
"""Client module for fetching user data from the example REST API."""

import asyncio
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel, HttpUrl

log = structlog.get_logger(__name__)

BASE_URL = "https://api.example.com"


class Address(BaseModel):
    street: str
    city: str
    state: str
    zip_code: str


class User(BaseModel):
    id: int
    name: str
    email: str
    phone: Optional[str] = None
    address: Optional[Address] = None
    website: Optional[HttpUrl] = None


class UserNotFoundError(Exception):
    """Raised when a user ID does not exist."""


class UserAPIError(Exception):
    """Raised when the API returns an unexpected error."""


class UserClient:
    """Async HTTP client for the user REST API."""

    def __init__(self, base_url: str = BASE_URL, timeout: float = 10.0) -> None:
        self._base_url = base_url
        self._timeout = timeout

    async def get_user(self, user_id: int) -> User:
        """Fetch a single user by ID."""
        url = f"{self._base_url}/users/{user_id}"
        log.info("fetching_user", user_id=user_id, url=url)

        try:
            async with httpx.AsyncClient(timeout=self._timeout) as client:
                response = await client.get(url)

            if response.status_code == 404:
                raise UserNotFoundError(f"User {user_id} not found")

            response.raise_for_status()
            user = User.model_validate(response.json())
            log.info("user_fetched", user_id=user_id, name=user.name)
            return user

        except UserNotFoundError:
            raise
        except httpx.HTTPStatusError as exc:
            log.error("api_http_error", user_id=user_id,
                      status_code=exc.response.status_code)
            raise UserAPIError(
                f"API returned {exc.response.status_code} for user {user_id}"
            ) from exc
        except httpx.RequestError as exc:
            log.error("api_request_error", user_id=user_id, detail=str(exc))
            raise UserAPIError(f"Network error fetching user {user_id}") from exc


async def main() -> None:
    """Fetch user #42 and log the result."""
    client = UserClient()
    try:
        user = await client.get_user(42)
        log.info("result", id=user.id, name=user.name, email=user.email)
    except UserNotFoundError:
        log.warning("user_not_found", user_id=42)
    except UserAPIError as exc:
        log.error("failed_to_fetch_user", user_id=42, detail=str(exc))


if __name__ == "__main__":
    asyncio.run(main())
```

Key design decisions:

- **`User` / `Address`** — Pydantic `BaseModel` with typed fields; `Optional` fields default to `None` so partial API responses don't break validation.
- **`UserClient.get_user`** — `async def` using `httpx.AsyncClient`; the outer `try/except` separates 404 (`UserNotFoundError`) from other HTTP errors and network failures (`UserAPIError`), re-raising each with context.
- **Logging** — `structlog` throughout; structured key-value pairs (`user_id=`, `status_code=`) instead of interpolated strings.
- **`main`** — thin `async def` that handles all three outcomes (success, not-found, API error) without letting exceptions propagate unhandled.

Dependencies needed: `httpx`, `structlog`, `pydantic`.
