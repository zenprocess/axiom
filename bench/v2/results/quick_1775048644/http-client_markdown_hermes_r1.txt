```python
"""Module for fetching user data from the REST API."""

import os
from typing import Optional

import httpx
import structlog
from pydantic import BaseModel

log = structlog.get_logger()


class User(BaseModel):
    """User data model representing a user from the API."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class APIClientError(Exception):
    """Custom exception for API client errors."""
    pass


class APIClient:
    """Client for interacting with the REST API."""

    def __init__(self, base_url: str, timeout: float = 30.0):
        self.base_url = base_url.rstrip('/')
        self.timeout = timeout
        self._client: Optional[httpx.AsyncClient] = None

    async def __aenter__(self):
        self._client = httpx.AsyncClient(timeout=httpx.Timeout(self.timeout))
        return self

    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self._client:
            await self._client.aclose()

    async def fetch_user(self, user_id: int) -> User:
        """
        Fetch a user by ID from the API.

        Args:
            user_id: The ID of the user to fetch.

        Returns:
            User: The fetched user data.

        Raises:
            APIClientError: If the API request fails or returns an unexpected response.
        """
        url = f"{self.base_url}/users/{user_id}"

        try:
            response = await self._client.get(url)
            response.raise_for_status()
        except httpx.HTTPStatusError as e:
            log.error(
                "api_request_failed",
                url=url,
                status_code=e.response.status_code,
                error=str(e),
            )
            raise APIClientError(
                f"API returned status {e.response.status_code} for user {user_id}"
            ) from e
        except httpx.RequestError as e:
            log.error("api_request_error", url=url, error=str(e))
            raise APIClientError(f"Failed to connect to API for user {user_id}") from e

        try:
            data = response.json()
            user = User(**data)
        except (ValueError, KeyError) as e:
            log.error("api_response_parse_error", url=url, error=str(e), response_text=response.text)
            raise APIClientError(f"Failed to parse user data for user {user_id}") from e

        log.info("user_fetched", user_id=user.id, name=user.name)
        return user


async def main():
    """Main function that fetches user #42 from the API."""
    api_key = os.environ.get("API_KEY", "")
    if not api_key:
        log.warning("api_key_not_set", message="API key not set in environment")

    base_url = os.environ.get("API_BASE_URL", "https://api.example.com")

    async with APIClient(base_url=base_url) as client:
        try:
            user = await client.fetch_user(user_id=42)
            log.info("user_retrieved_successfully", user_id=user.id, name=user.name)
            # In a real application, you might return or process the user data here
            return user
        except APIClientError as e:
            log.error("failed_to_fetch_user", user_id=42, error=str(e))
            raise


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```
