```python
"""User data fetcher module for REST API interaction."""

import sys
from typing import Optional

import httpx
import structlog

# Configure structlog
structlog.configure(
    processors=[
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.processors.add_log_level,
        structlog.processors.StackInfoRenderer(),
        structlog.processors.format_exc_info,
        structlog.dev.ConsoleRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=False,
)

logger = structlog.get_logger()


class User(BaseModel):
    """User data model representing a user from the API."""
    id: int
    name: str
    email: str
    username: Optional[str] = None
    phone: Optional[str] = None
    website: Optional[str] = None


class APIError(Exception):
    """Custom exception for API-related errors."""
    pass


class UserFetcher:
    """Fetches user data from the REST API."""

    def __init__(self, base_url: str = "https://api.example.com"):
        """Initialize the fetcher with a base URL.

        Args:
            base_url: The base URL for the API.
        """
        self.base_url = base_url.rstrip("/")

    async def fetch_user(self, user_id: int) -> User:
        """Fetch user data by ID from the API.

        Args:
            user_id: The ID of the user to fetch.

        Returns:
            A User model instance.

        Raises:
            APIError: If the API request fails or returns an unexpected response.
        """
        url = f"{self.base_url}/users/{user_id}"

        try:
            async with httpx.AsyncClient() as client:
                response = await client.get(url, timeout=10.0)
                response.raise_for_status()
                data = response.json()
                return User(**data)

        except httpx.HTTPStatusError as exc:
            logger.error("HTTP error fetching user", user_id=user_id, status_code=exc.response.status_code)
            raise APIError(f"HTTP {exc.response.status_code} while fetching user {user_id}: {exc.response.text}") from exc
        except httpx.RequestError as exc:
            logger.error("Request error fetching user", user_id=user_id, error=str(exc))
            raise APIError(f"Request error while fetching user {user_id}: {exc}") from exc
        except Exception as exc:
            logger.error("Unexpected error fetching user", user_id=user_id, error=str(exc))
            raise APIError(f"Unexpected error while fetching user {user_id}") from exc


async def main() -> None:
    """Main entry point: fetches and prints user #42."""
    fetcher = UserFetcher()
    try:
        user = await fetcher.fetch_user(42)
        logger.info("User fetched successfully", user_id=user.id, name=user.name)
        print(user.model_dump_json(indent=2))  # For CLI output only — not used for logging
    except APIError as exc:
        logger.error("Failed to fetch user", user_id=42, error=str(exc))
        sys.exit(1)


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
```

### Key features:
- Uses `structlog` for structured logging (no `print()` in core logic; only final JSON output in `main()` for visibility).
- `User` Pydantic model with optional fields.
- `UserFetcher` class with async `fetch_user()` method.
- Comprehensive error handling with `try/except` for `httpx` exceptions.
- Type hints throughout.
- Docstrings for public methods/classes.
- Uses `httpx.AsyncClient` for async I/O.
