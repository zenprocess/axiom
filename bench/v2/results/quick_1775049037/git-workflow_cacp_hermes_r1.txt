```python
#!/usr/bin/env python3
"""
Automated Git Workflow Script

Creates feature branches, commits changes, pushes safely, and creates merge requests.
Uses structlog for logging and Pydantic for data models.
"""

import asyncio
import os
import subprocess
from pathlib import Path
from typing import Optional

import structlog
from pydantic import BaseModel, Field

structlog.configure(
    processors=[
        structlog.processors.add_log_level,
        structlog.processors.TimeStamper(fmt="iso"),
        structlog.dev.ConsoleRenderer(),
    ],
    context_class=dict,
    logger_factory=structlog.PrintLoggerFactory(),
    cache_logger_on_first_use=True,
)

logger = structlog.get_logger()


class GitConfig(BaseModel):
    """Configuration for Git repository."""
    repo_path: Path = Field(default_factory=Path.cwd)
    remote: str = "origin"
    default_branch: str = "main"


class MRConfig(BaseModel):
    """Configuration for Merge Request creation."""
    title: str
    description: str
    target_branch: str = "main"
    source_branch: str = "feature/my-feature"
    api_url: str = "https://api.github.com"
    token: str = Field(..., env="GITHUB_TOKEN")


class GitWorkflow:
    """Handles automated Git operations safely."""

    def __init__(self, config: GitConfig):
        self.config = config
        logger.info("GitWorkflow initialized", config=self.config.dict())

    async def _run_git_command(self, *args: str) -> str:
        """Run a Git command asynchronously."""
        try:
            result = await asyncio.to_thread(
                subprocess.run,
                ["git", *args],
                cwd=str(self.config.repo_path),
                capture_output=True,
                text=True,
                check=True,
            )
            return result.stdout.strip()
        except subprocess.CalledProcessError as e:
            logger.error(
                "Git command failed",
                command=" ".join(args),
                stderr=e.stderr,
                returncode=e.returncode,
            )
            raise

    async def create_feature_branch(self, branch_name: str) -> str:
        """
        Create a new feature branch from current HEAD.

        Args:
            branch_name: Name of the feature branch to create.

        Returns:
            Name of the created branch.
        """
        logger.info("Creating feature branch", branch=branch_name)
        await self._run_git_command("checkout", "-b", branch_name)
        return branch_name

    async def commit_changes(
        self, message: str, files: Optional[list[str]] = None
    ) -> str:
        """
        Commit staged changes.

        Args:
            message: Commit message.
            files: Optional list of files to commit (stages them first).

        Returns:
            Commit hash.
        """
        if files:
            await self._run_git_command("add", *files)
        
        result = await self._run_git_command("commit", "-m", message)
        # Extract commit hash from output
        commit_hash = result.split()[-1] if result else ""
        logger.info("Changes committed", commit_hash=commit_hash)
        return commit_hash

    async def push_branch(self, branch_name: str) -> None:
        """
        Push branch to remote repository.

        Args:
            branch_name: Name of the branch to push.
        """
        try:
            await self._run_git_command(
                "push", "--set-upstream", self.config.remote, branch_name
            )
            logger.info("Branch pushed successfully", branch=branch_name)
        except subprocess.CalledProcessError as e:
            logger.error(
                "Push failed (branch might already exist on remote)",
                branch=branch_name,
                error=str(e),
            )
            raise

    async def get_current_branch(self) -> str:
        """Get the current Git branch name."""
        return await self._run_git_command("rev-parse", "--abbrev-ref", "HEAD")

    async def ensure_clean_workspace(self) -> None:
        """Verify workspace is clean before proceeding."""
        status = await self._run_git_command("status", "--porcelain")
        if status:
            logger.warning("Workspace not clean", status=status)
            raise RuntimeError("Working directory has uncommitted changes")

    async def run_workflow(
        self,
        feature_branch: str,
        commit_message: str,
        files_to_commit: list[str],
        mr_config: MRConfig,
    ) -> str:
        """
        Run the complete git workflow.

        Args:
            feature_branch: Name of the feature branch to create.
            commit_message: Message for the commit.
            files_to_commit: List of files to commit.
            mr_config: Configuration for creating the merge request.

        Returns:
            URL of the created merge request.
        """
        await self.ensure_clean_workspace()
        current_branch = await self.get_current_branch()
        
        # Create feature branch
        await self.create_feature_branch(feature_branch)
        
        # Commit changes
        await self.commit_changes(commit_message, files_to_commit)
        
        # Push branch (safely)
        await self.push_branch(feature_branch)
        
        # Create merge request
        mr_url = await self.create_merge_request(mr_config)
        
        # Return to original branch
        await self._run_git_command("checkout", current_branch)
        
        return mr_url

    async def create_merge_request(self, config: MRConfig) -> str:
        """
        Create a merge request via GitHub API.

        Args:
            config: Configuration for the merge request.

        Returns:
            URL of the created merge request.
        """
        import aiohttp

        repo_path = self.config.repo_path
        repo_name = repo_path.name
        owner = "your-org"  # Replace with dynamic extraction or config

        payload = {
            "title": config.title,
            "body": config.description,
            "head": config.source_branch,
            "base": config.target_branch,
        }

        headers = {
            "Authorization": f"token {config.token}",
            "Accept": "application/vnd.github.v3+json",
        }

        url = f"{config.api_url}/repos/{owner}/{repo_name}/pulls"

        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(url, json=payload, headers=headers) as response:
                    if response.status == 201:
                        data = await response.json()
                        mr_url = data.get("html_url", "")
                        logger.info("Merge request created", mr_url=mr_url)
                        return mr_url
                    else:
                        error_text = await response.text()
                        logger.error(
                            "Failed to create MR",
                            status=response.status,
                            error=error_text,
                        )
                        raise RuntimeError(
                            f"MR creation failed: {response.status}"
                        )
        except aiohttp.ClientError as e:
            logger.error("HTTP client error", error=str(e))
            raise


async def main():
    """Main entry point."""
    # Example usage
    git_config = GitConfig()
    mr_config = MRConfig(
        title="Add new feature",
        description="Implements the new feature as per requirements",
        source_branch="feature/my-new-feature",
    )

    workflow = GitWorkflow(git_config)
    
    try:
        mr_url = await workflow.run_workflow(
            feature_branch="feature/my-new-feature",
            commit_message="feat: add new feature",
            files_to_commit=["README.md"],  # Example file
            mr_config=mr_config,
        )
        logger.info("Workflow completed successfully", mr_url=mr_url)
    except Exception as e:
        logger.error("Workflow failed", error=str(e))
        raise


if __name__ == "__main__":
    asyncio.run(main())
```
