```python
# test_framework.py
import inspect
import os
import sys
import traceback
from dataclasses import dataclass
from typing import Callable, Dict, List, Any, Optional, Tuple
from enum import Enum

from pydantic import BaseModel


class TestResultType(str, Enum):
    PASS = "PASS"
    FAIL = "FAIL"
    ERROR = "ERROR"


class TestResult(BaseModel):
    name: str
    result: TestResultType
    message: Optional[str] = None
    traceback: Optional[str] = None


@dataclass
class FixtureContext:
    setup: Any = None
    teardown: Any = None


class TestCase:
    def __init__(self, name: str, func: Callable, fixture: Optional[FixtureContext] = None):
        self.name = name
        self.func = func
        self.fixture = fixture or FixtureContext()

    def run(self) -> TestResult:
        result = TestResult(name=self.name, result=TestResultType.PASS)

        try:
            # Run setup if present
            if self.fixture.setup:
                self.fixture.setup()

            # Run test
            self.func()

            # Run teardown if present (even if test passed)
            if self.fixture.teardown:
                self.fixture.teardown()

        except AssertionError as e:
            result.result = TestResultType.FAIL
            result.message = str(e)
        except Exception as e:
            result.result = TestResultType.ERROR
            result.message = str(e)
            result.traceback = traceback.format_exc()
        finally:
            # Ensure teardown runs even on failure/error
            if self.fixture.teardown and result.result != TestResultType.PASS:
                try:
                    self.fixture.teardown()
                except Exception:
                    pass  # ignore teardown failure in error reporting

        return result


def assert_equal(actual: Any, expected: Any, msg: str = "") -> None:
    if actual != expected:
        raise AssertionError(f"Assertion failed: {msg or f'{actual!r} != {expected!r}'}")


def assert_true(condition: bool, msg: str = "") -> None:
    if not condition:
        raise AssertionError(f"Assertion failed: {msg or 'Expected True, got False'}")


def assert_false(condition: bool, msg: str = "") -> None:
    if condition:
        raise AssertionError(f"Assertion failed: {msg or 'Expected False, got True'}")


def _discover_tests_in_module(module, fixture_map: Dict[str, FixtureContext]) -> List[TestCase]:
    test_cases = []
    for name, obj in inspect.getmembers(module):
        if name.startswith("test_") and inspect.isfunction(obj):
            fixture = fixture_map.get(name)
            test_cases.append(TestCase(name=name, func=obj, fixture=fixture))
    return test_cases


def discover_tests(path: str = ".") -> List[TestCase]:
    """Recursively discover test functions in files matching `test_*.py`."""
    test_cases: List[TestCase] = []
    fixture_map: Dict[str, FixtureContext] = {}

    for root, _, files in os.walk(path):
        for fname in files:
            if not fname.startswith("test_") or not fname.endswith(".py"):
                continue

            fpath = os.path.join(root, fname)
            module_name = os.path.splitext(os.path.basename(fpath))[0]
            sys.path.insert(0, root)
            try:
                module = __import__(module_name)
                # Discover fixtures in the same module
                for name, obj in inspect.getmembers(module):
                    if name.startswith("setup_") and inspect.isfunction(obj):
                        test_name = name[6:]  # strip "setup_"
                        fixture_map.setdefault(test_name, FixtureContext()).setup = obj
                    elif name.startswith("teardown_") and inspect.isfunction(obj):
                        test_name = name[9:]  # strip "teardown_"
                        fixture_map.setdefault(test_name, FixtureContext()).teardown = obj

                # Collect tests
                test_cases.extend(_discover_tests_in_module(module, fixture_map))
            except Exception:
                # Skip modules that fail to import
                continue
            finally:
                sys.path.pop(0)

    return test_cases


class TestRunner:
    def __init__(self, verbosity: int = 1):
        self.verbosity = verbosity

    def run(self, tests: List[TestCase]) -> List[TestResult]:
        results: List[TestResult] = []

        for test in tests:
            result = test.run()
            results.append(result)

            # User-facing output (NOT debug logging — this is the test summary)
            if self.verbosity >= 2:
                status = "PASS" if result.result == TestResultType.PASS else "FAIL" if result.result == TestResultType.FAIL else "ERROR"
                print(f"{status}: {test.name}")
            else:
                if result.result == TestResultType.PASS:
                    print(".", end="")
                elif result.result == TestResultType.FAIL:
                    print("F", end="")
                elif result.result == TestResultType.ERROR:
                    print("E", end="")

        if self.verbosity < 2:
            print()  # newline after dots

        return results

    def print_summary(self, results: List[TestResult]) -> None:
        passed = sum(1 for r in results if r.result == TestResultType.PASS)
        failed = sum(1 for r in results if r.result == TestResultType.FAIL)
        errors = sum(1 for r in results if r.result == TestResultType.ERROR)
        total = len(results)

        print(f"\nRan {total} test(s)")
        print(f"  PASS: {passed}")
        print(f"  FAIL: {failed}")
        print(f"  ERROR: {errors}")

        if failed or errors:
            print("\nFailures/Errors:")
            for r in results:
                if r.result != TestResultType.PASS:
                    print(f"\n  {r.name} [{r.result.value}]")
                    if r.message:
                        print(f"    {r.message}")
                    if r.traceback:
                        for line in r.traceback.splitlines():
                            print(f"      {line}")


def main():
    tests = discover_tests(".")
    runner = TestRunner(verbosity=1)
    results = runner.run(tests)
    runner.print_summary(results)

    # Exit code: 0 if all pass, 1 otherwise
    sys.exit(1 if any(r.result != TestResultType.PASS for r in results) else 0)


if __name__ == "__main__":
    main()
```
